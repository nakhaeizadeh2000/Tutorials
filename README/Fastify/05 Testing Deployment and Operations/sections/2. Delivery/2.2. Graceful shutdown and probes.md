```javascript
// GOOD: probes splitting — live cheaping, ready deeping, traffic gating (executed live below).
const fastify = require('fastify')();
let storeUp = true;
fastify.get('/live', async () => ({ ok: true })); // live cheaping — process answering, sockets open
fastify.get('/ready', async (req, reply) => { // ready deeping — dependencies checking, stores pinging
  if (!storeUp) return reply.code(503).send({ ready: false });
  return { ready: true };
});
(async () => {
  const live = await fastify.inject({ method: 'GET', url: '/live' });
  storeUp = false;
  const ready = await fastify.inject({ method: 'GET', url: '/ready' });
  console.log(live.statusCode === 200 && ready.statusCode === 503 ? 'verdicts split' : 'probes merged');
})();
```

// Shared live+ready (readiness failing — orchestrators killing — boots crashing (recovery looping — availability dipping). Split verdicts; deepen ready.)
