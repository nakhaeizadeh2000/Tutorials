```javascript
// GOOD: boot gating — schemas declaring, boot crashing, deploys blocking (executed live below).
function gateConfig(env) { // schemas declaring — fields requiring, types constraining
  const port = Number(env.PORT ?? 3000);
  if (!env.DATABASE_URL) throw new Error('boot refused: DATABASE_URL missing'); // boot crashing — serving refusing, bad running never
  if (!Number.isInteger(port) || port < 1) throw new Error('boot refused: PORT invalid');
  return Object.freeze({ port, db: env.DATABASE_URL }); // deploys blocking — bad config shipping never
}
console.log(gateConfig({ DATABASE_URL: 'postgres://db/app' }).port); // 3000 (env proving — crashes earlying — deploys blocking)
try { gateConfig({}); } catch (e) { console.log(e.message); } // boot refused: DATABASE_URL missing
```

**Performance note** — gate economics: declaring costing schema lines (fields requiring — types constraining (contracts bounding — surprises preventing); crashing costing one throw (serving refusing — bad running never (failures earlying — damage preventing); blocking saving rollbacks (bad shipping never — rollouts protecting (correctness starting — chaos preventing); hoped equivalents costing incidents (staging discovering — users finding).
