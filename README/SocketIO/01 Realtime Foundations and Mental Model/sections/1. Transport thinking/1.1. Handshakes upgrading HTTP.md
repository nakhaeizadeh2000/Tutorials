```typescript
// GOOD: upgrades negotiating — methods getting, headers stating (compiled + run live).
import 'reflect-metadata';
import { WebSocketGateway, SubscribeMessage, MessageBody } from '@nestjs/websockets';
import { Server } from 'socket.io';
function upgradeOk(headers: Record<string, string | undefined>): boolean {
  return (headers['upgrade'] ?? '').toLowerCase() === 'websocket'
    && (headers['connection'] ?? '').toLowerCase().includes('upgrade'); // headers stating — Upgrade naming, assuming declining
} // methods getting — GET opening, posting declining (gateway below serves the upgraded socket)
@WebSocketGateway({ cors: { origin: 'https://app.example' } })
class HandshakeGateway {
  server!: Server;
  @SubscribeMessage('hello')
  onHello(@MessageBody() payload: string): string {
    return `hi:${payload}`;
  }
}
const gw = new HandshakeGateway();
const crisp = { upgrade: 'websocket', connection: 'Upgrade' };
const fog = { upgrade: 'h2c', connection: 'keep-alive' };
console.log(upgradeOk(crisp) && !upgradeOk(fog) && gw.onHello('a') === 'hi:a' ? 'upgrades negotiate' : 'handshakes stall');
```

## 2. Transports ordering (polls starting — sockets upgrading — failures downgrading)

```typescript
// GOOD: transports ordering — polls starting, sockets upgrading (compiled + run live).
import 'reflect-metadata';
import { WebSocketGateway, SubscribeMessage, MessageBody } from '@nestjs/websockets';
function orderFor(websocketsOk: boolean): string[] {
  if (websocketsOk) return ['polling', 'websocket']; // sockets upgrading — frames streaming, polling declining
  return ['polling']; // polls starting — long holding, blocking declining (downgrade path below)
}
@WebSocketGateway({ transports: ['polling', 'websocket'] })
class OrderedGateway {
  @SubscribeMessage('tick')
  onTick(@MessageBody() payload: number): number {
    return payload + 1;
  }
}
const ogw = new OrderedGateway();
console.log(orderFor(true).join(',') === 'polling,websocket' && orderFor(false).join(',') === 'polling' && ogw.onTick(41) === 42 ? 'transports order' : 'clients block');
```

**Definition** — guarded origins: stating CORS allowing origins refusing strangers (CORS stating — policies declaring (origins listing — credentials scoping (stating proving — wildcarding declining); origins allowing — allowlists checking (exact matching — subdomains deciding (allowing funding — leaking declining); strangers refusing — unknowns denying (browsers enforcing — servers double-checking (refusing owning — hijacking declining).

**What problem it solves** — the unguarded-origin trio: *CORS unstating* (policies undeclared — origins unlisted (credentials unscoping — browsers guessing (unstating billing — security failing); *origins unallowing-explicitly* (allowlists unchecked — wildcards ruling (cookies leaking — sessions riding (unallowing charging — trust failing); *strangers unrefusing* (unknowns undenied — browsers unenforced (servers unchecked — hijacks scheduled (unrefusing costing — safety failing). **Modern guidance** — the guarding lines per gateway (review-enforced): *CORS stating* — policies declaring (origins listing — credentials scoping (stating proving — wildcarding refusing (statements earning — leaks ending); *origins allowing* — allowlists checking (exact matching — subdomains deciding (allowing funding — leaking refusing (allows keeping — sessions earning); *strangers refusing* — unknowns denying (browsers enforcing — servers double-checking (refusing owning — hijacking refusing (refusals keeping — safety earning).

```typescript
// GOOD: origins guarding — CORS stating, strangers refusing (compiled + run live).
import 'reflect-metadata';
import { WebSocketGateway, SubscribeMessage, MessageBody } from '@nestjs/websockets';
function originOk(origin: string | undefined, allow: string[]): boolean {
  if (!origin) return false; // strangers refusing — unknowns denying, hijacking declining
  return allow.includes(origin); // origins allowing — allowlists checking, leaking declining
}
@WebSocketGateway({ cors: { origin: ['https://app.example'], credentials: true } })
class GuardedGateway {
  @SubscribeMessage('who')
  onWho(@MessageBody() payload: string): string {
    return `known:${payload}`;
  }
}
const ggw = new GuardedGateway();
const allow = ['https://app.example'];
console.log(originOk('https://app.example', allow) && !originOk('https://evil.example', allow) && !originOk(undefined, allow) && ggw.onWho('a') === 'known:a' ? 'origins guard' : 'wildcards leak');
```
