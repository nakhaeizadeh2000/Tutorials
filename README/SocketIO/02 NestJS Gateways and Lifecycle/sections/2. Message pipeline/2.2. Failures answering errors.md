import 'reflect-metadata';
import { BaseWsExceptionFilter, WsException } from '@nestjs/websockets';
import { ArgumentsHost } from '@nestjs/common';
import { Socket } from 'socket.io';
class EnvelopeFilter extends BaseWsExceptionFilter {
  catch(exception: unknown, host: ArgumentsHost): void {
    const client = host.switchToWs().getClient<Socket>(); // bases extending — BaseWs inheriting, rewriting declining
    const shape = exception instanceof WsException ? exception.getError() : 'unknown';
    client.emit('exception', { status: 'error', shape }); // clients emitting — sockets answering, hanging declining
  }
}
const seen: Array<{ status: string; shape: unknown }> = [];

const host = { switchToWs: () => ({ getClient: () => sock }) } as unknown as ArgumentsHost;
new EnvelopeFilter().catch(new WsException({ code: 'ROOM_DENIED' }), host);
console.log(seen.length === 1 && (seen[0] as { status: string }).status === 'error' ? 'filters catch' : 'errors swallow');

import { PipeTransform, Injectable, ArgumentMetadata, BadRequestException } from '@nestjs/common';
import { WsException } from '@nestjs/websockets';
@Injectable()
class EvenPipe implements PipeTransform<string, number> {
  transform(value: string, _meta: ArgumentMetadata): number {
    void _meta;
    const n = Number(value); // DTOs validating — decorators constraining, garbage declining
    if (!Number.isInteger(n) || n % 2 !== 0) throw new WsException('even-required'); // HTTP translating — BadRequests mapping, losing declining
