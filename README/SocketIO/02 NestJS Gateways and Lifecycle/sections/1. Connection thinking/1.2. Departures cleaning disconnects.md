// GOOD: leaves announcing — quits detecting, rosters pruning (compiled + run live).
import 'reflect-metadata';
import { WebSocketGateway, OnGatewayDisconnect } from '@nestjs/websockets';
import { Socket } from 'socket.io';
@WebSocketGateway()
class LeaveGateway implements OnGatewayDisconnect {
  readonly roster = new Set<string>(); // rosters pruning — members deleting, ghosting declining

    const id = (client.data as { sid?: string }).sid ?? 'anon'; // quits detecting — disconnects handling, missing declining
    this.roster.delete(id);
    return `left:${id}=${this.roster.size}`; // peers notifying — departures broadcasting, diverging declining
  }
}

const mk = (s: string) => ({ data: { sid: s } }) as unknown as Socket;

```typescript
// GOOD: resources freeing — timers clearing, subs dropping (compiled + run live).
import 'reflect-metadata';
import { WebSocketGateway, OnGatewayDisconnect } from '@nestjs/websockets';
import { Socket } from 'socket.io';
@WebSocketGateway()
class FreeGateway implements OnGatewayDisconnect {
  private timers = new Map<string, ReturnType<typeof setTimeout>>(); // timers clearing — intervals stopping, ticking declining
  arm(id: string): void {
    this.timers.set(id, setTimeout(() => undefined, 50)); // timers arming — intervals starting, idling declining
  }
  handleDisconnect(client: Socket): string {
    const id = (client.data as { sid?: string }).sid ?? 'anon';
    const timer = this.timers.get(id);
    if (timer) { clearTimeout(timer); this.timers.delete(id); } // subs dropping — listeners removing, leaking declining
    return `freed:${id}=${this.timers.size}`; // memory earning — references releasing, bloating declining
  }
}
const fgw = new FreeGateway();
fgw.arm('u1');
const mk = (s: string) => ({ data: { sid: s } }) as unknown as Socket;
console.log(fgw.handleDisconnect(mk('u1')) === 'freed:u1=0' ? 'resources free' : 'gateways leak');
```
