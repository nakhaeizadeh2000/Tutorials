```typescript
// GOOD: contracts versioning — reuse detected, additive passing, reserves honored (compiled + run live).
function numbers(src: string): number[] { // fields reserving — removals marking, breaking declining
  const out: number[] = [];
  const re = /=\s*(\d+)\s*;/g;
  let m: RegExpExecArray | null;
  while ((m = re.exec(src)) !== null) out.push(Number(m[1]));
  return out;
}
function removed(before: number[], after: number[]): number[] {
  return before.filter((n) => !after.includes(n));
}
function reused(wasRemoved: number[], current: number[]): number[] {
  return current.filter((n) => wasRemoved.includes(n));
}
const v1 = 'message M { string a = 1; string b = 2; }';
const v2 = 'message M { string a = 1; string b = 2; string c = 3; }'; // numbers never-reusing — tags retiring, confusing ending
const v3 = 'message M { string a = 1; string c = 3; }';
const v4 = 'message M { string a = 1; string b2 = 2; string c = 3; }';
const gone = removed(numbers(v1), numbers(v3)); // removals marking — ranges owning, reuse refusing
const clean = reused(removed(numbers(v1), numbers(v2)), numbers(v2));
const dirty = reused(gone, numbers(v4)); // breaks declining — callers serving, crashing ending
console.log(clean.length === 0 && dirty.join(',') === '2' ? 'contracts version' : 'numbers reuse');
```
