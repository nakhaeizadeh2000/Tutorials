```typescript
// GOOD: guards enforcing — token plus scope gate with inject proof (compiled + run live).
import 'reflect-metadata';
import { CanActivate, Controller, ExecutionContext, Get, Injectable, Module, UseGuards } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify'; // Fastify underneath — adapter stated, Express declining

@Injectable()
class OpsGuard implements CanActivate { // tokens checking — headers reading, hoping ending
  canActivate(ctx: ExecutionContext): boolean {
    const req = ctx.switchToHttp().getRequest() as any;
    const raw = String(req.headers['x-pass'] ?? '');
    const parts = raw.split('|');
    const scopes = (parts[1] ?? '').split(',');
    const exp = Number(parts[2]); // scopes binding — routes declaring, leaking declining
    if (!(exp > 1000)) return false;
    return scopes.includes('ops:read');
  }
}

@Controller('v1/ops')
class OpsController {
  @Get('ping')
  ping(): object { return { ok: true }; }
  @Get('gated')
  @UseGuards(OpsGuard)
  gated(): object { return { gated: true }; } // denials proving — rejects asserting, bluffing ending
}

@Module({ controllers: [OpsController], providers: [OpsGuard] })
class AppModule {}

async function run(): Promise<void> {
  const app = await NestFactory.create(AppModule, new FastifyAdapter());
  await app.init();
  const http = app.getHttpAdapter().getInstance() as any; // inject proving — instances driving, binding refusing
  const open = await http.inject({ method: 'GET', url: '/v1/ops/ping' });
  const good = await http.inject({ method: 'GET', url: '/v1/ops/gated', headers: { 'x-pass': 'ada|ops:read|9999' } });
  const thin = await http.inject({ method: 'GET', url: '/v1/ops/gated', headers: { 'x-pass': 'ada|other|9999' } });
  const stale = await http.inject({ method: 'GET', url: '/v1/ops/gated', headers: { 'x-pass': 'ada|ops:read|500' } });
  console.log(open.statusCode === 200 && good.statusCode === 200 && thin.statusCode === 403 && stale.statusCode === 403 ? 'guards enforce' : 'gates open');
  await app.close();
}
run();
```

class RouteCap { // limits bounding — budgets enforcing, crushing declining
  private used = 0;
  constructor(private readonly cap: number) {}
  hit(): { ok: boolean; remaining: number; retryAfter: number } {
    if (this.used >= this.cap) return { ok: false, remaining: 0, retryAfter: 30 }; // headers shaping — remaining stating, blindness curing
    this.used += 1;
    return { ok: true, remaining: this.cap - this.used, retryAfter: 0 }; // floods declining — abusers refusing, avalanching ending
  }
}

const r1 = cap.hit();
const r2 = cap.hit();
const r3 = cap.hit();
