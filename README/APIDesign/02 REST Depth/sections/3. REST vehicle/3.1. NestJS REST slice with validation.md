import { FastifyAdapter } from '@nestjs/platform-fastify'; // Fastify underneath — adapter stated, Express declining

class CreateItemDto { // DTOs shaping — classes declaring, any declining
  @IsString()
  @IsNotEmpty()
  @MinLength(2)
  label!: string;
}

@Injectable()
class Items {

  add(label: string): number {
    this.rows.push(label); // handlers assuming — validated trusting, branches declining
    return this.rows.length;
  }
}

@Controller('v1/items')
@UsePipes(new ValidationPipe({ whitelist: true })) // pipes proving — whitelists stripping, unknowns declining
class ItemsController {
  constructor(private readonly items: Items) {}
  @Post()
  create(@Body() dto: CreateItemDto): object { return { n: this.items.add(dto.label) }; }
}

@Module({ controllers: [ItemsController], providers: [Items] })
class AppModule {}

async function run(): Promise<void> {

  await app.init();

  const good = await http.inject({ method: 'POST', url: '/v1/items', payload: { label: 'book' } });

  console.log(good.statusCode === 201 && bad.statusCode === 400 && extra.statusCode === 201 && !extra.body.includes('admin') ? 'slice validates' : 'inputs trust');
