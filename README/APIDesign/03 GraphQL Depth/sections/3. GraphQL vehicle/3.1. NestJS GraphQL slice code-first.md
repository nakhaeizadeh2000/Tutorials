```typescript
@Injectable()
class Catalog { // services delegating — logic homing, bypassing declining
  names(): string[] { return ['real']; }
}

@Resolver()

  constructor(private readonly catalog: Catalog) {}

}

async function run(): Promise<void> {

    .useValue(fake)
    .compile(); // DI wiring — constructors receiving, manualing declining
  const out = mod.get(CatalogResolver).names();

```
