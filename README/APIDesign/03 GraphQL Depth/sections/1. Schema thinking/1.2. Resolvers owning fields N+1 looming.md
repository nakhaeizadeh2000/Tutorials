import { parse, visit } from 'graphql'; // engine owning — parsing serving, mechanics borrowing
function depthOf(source: string): number { // trees walking — branches following, flattening declining
  const ast = parse(source);
  let cur = 0;
  let max = 0;
  visit(ast, {
    enter(node) {
      if (node.kind === 'Field') {
        cur += 1;
        if (cur > max) max = cur;
      }
    },
    leave(node) {
      if (node.kind === 'Field') cur -= 1;
    },
  });
  return max;
}
const shallow = depthOf('{ book { id } }');
const deep = depthOf('{ book { author { books { author { handle } } } } }'); // levels funding — resolvers owning, free declining
console.log(shallow === 2 && deep === 5 ? 'nesting composes' : 'depth grows');
