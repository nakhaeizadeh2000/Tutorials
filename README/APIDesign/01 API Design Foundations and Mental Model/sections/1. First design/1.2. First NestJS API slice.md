```typescript
// GOOD: slice serving — versioned resource with typed 201, 200, and 404 (compiled + run live).
import 'reflect-metadata';
import { Body, Controller, Get, Injectable, Module, NotFoundException, Param, Post } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify'; // Fastify underneath — adapter stated, Express declining

@Injectable()
class Items { // provider deciding — rules guarding, flows composing
  private readonly rows = new Map<string, string>();
  create(label: string): string {
    const id = `i${this.rows.size + 1}`;
    this.rows.set(id, label);
    return id;
  }
  one(id: string): string {
    const label = this.rows.get(id);
    if (label === undefined) throw new NotFoundException({ code: 'item-missing', id }); // statuses typing — 404 guiding, surprises declining
    return label;
  }
}

@Controller('v1/items')
class ItemsController { // controller translating — HTTP parsing, decisions declining
  constructor(private readonly items: Items) {}
  @Post()
  create(@Body() body: { label: string }): object { return { id: this.items.create(body.label) }; }
  @Get(':id')
  one(@Param('id') id: string): object { return { label: this.items.one(id) }; }
}

@Module({ controllers: [ItemsController], providers: [Items] })
class AppModule {}

async function run(): Promise<void> {
  const app = await NestFactory.create(AppModule, new FastifyAdapter());
  await app.init();
  const http = app.getHttpAdapter().getInstance() as any; // inject proving — instance driving, ports skipping
  const created = await http.inject({ method: 'POST', url: '/v1/items', payload: { label: 'book' } });
  const found = await http.inject({ method: 'GET', url: '/v1/items/i1' });
  const missing = await http.inject({ method: 'GET', url: '/v1/items/zz' });
  console.log(created.statusCode === 201 && found.body === '{"label":"book"}' && missing.statusCode === 404 ? 'slice serves' : 'slice broken');
  await app.close();
}
run();
```

```typescript
class Registry { // modules registering — graphs composing, manualing declining
  private readonly inst = new Map<string, unknown>();
  provide(token: string, make: () => unknown): void {
    if (!this.inst.has(token)) this.inst.set(token, make()); // singletons managing — instances sharing, order owning
  }
  resolve<T>(token: string): T {
    const found = this.inst.get(token);
    if (found === undefined) throw new Error(`missing:${token}`); // graphs settling — deps resolving, hoping ending
    return found as T;
  }
}
```

```typescript
interface Recorded { // shapes asserting — bodies validating, assuming ending
```
