```typescript
// GOOD: entries guarding — shop journey behind token plus scope gates (compiled + run live).
import 'reflect-metadata';
import { CanActivate, Controller, ExecutionContext, Get, Injectable, Module, UseGuards } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify'; // Fastify underneath — adapter stated, Express declining

@Injectable()
class ShopGuard implements CanActivate { // tokens checking — headers reading, hoping ending
  canActivate(ctx: ExecutionContext): boolean {
    const req = ctx.switchToHttp().getRequest() as any;
    const raw = String(req.headers['x-pass'] ?? '');
    const parts = raw.split('|');
    const scopes = (parts[1] ?? '').split(',');
    const exp = Number(parts[2]); // scopes binding — orders declaring, leaking declining
    if (!(exp > 1000)) return false;
    return scopes.includes('shop:buy');
  }
}

@Controller('v1/shop')
class ShopController {
  @Get('browse')
  browse(): object { return { items: ['a', 'b'] }; }
  @Get('order')
  @UseGuards(ShopGuard)
  order(): object { return { id: 'o1', total: 42 }; } // denials proving — rejects asserting, bluffing ending
}

@Module({ controllers: [ShopController], providers: [ShopGuard] })
class ShopModule {}

async function runShop(): Promise<void> {
  const app = await NestFactory.create(ShopModule, new FastifyAdapter());
  await app.init();
  const http = app.getHttpAdapter().getInstance() as any; // inject proving — instances driving, binding refusing
  const browse = await http.inject({ method: 'GET', url: '/v1/shop/browse' });
  const good = await http.inject({ method: 'GET', url: '/v1/shop/order', headers: { 'x-pass': 'ada|shop:buy|9999' } });
  const thin = await http.inject({ method: 'GET', url: '/v1/shop/order', headers: { 'x-pass': 'ada|browse|9999' } });
  const stale = await http.inject({ method: 'GET', url: '/v1/shop/order', headers: { 'x-pass': 'ada|shop:buy|500' } });
  console.log(browse.statusCode === 200 && good.statusCode === 200 && thin.statusCode === 403 && stale.statusCode === 403 ? 'entries guard' : 'shop opens');
  await app.close();
}
runShop();
```
