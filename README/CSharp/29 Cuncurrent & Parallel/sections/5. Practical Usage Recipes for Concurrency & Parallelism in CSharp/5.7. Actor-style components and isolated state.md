### 1. **Single-writer / mailbox pattern with Channels**

**Basic actor pattern**

- Use a `Channel<T>` as a mailbox with a single reader (actor loop).

```csharp
using System;
using System.Threading;
using System.Threading.Channels;
using System.Threading.Tasks;

public sealed class SimpleActor : IAsyncDisposable
{
    private readonly Channel<Message> _mailbox;
    private readonly CancellationTokenSource _cts = new();
    private readonly Task _runner;

    public SimpleActor()
    {
        _mailbox = Channel.CreateUnbounded<Message>(new UnboundedChannelOptions
        {
            SingleReader = true,
            SingleWriter = false
        });
        _runner = Task.Run(RunAsync);
    }

    public ValueTask SendAsync(Message message, CancellationToken cancellationToken = default)
        => _mailbox.Writer.WriteAsync(message, cancellationToken);

    private async Task RunAsync()
    {
        try
        {
            await foreach (var message in _mailbox.Reader.ReadAllAsync(_cts.Token).ConfigureAwait(false))
            {
                await ProcessMessageAsync(message);
            }
        }
        catch (OperationCanceledException)
        {
            // Expected on shutdown
        }
    }

    private async Task ProcessMessageAsync(Message message)
    {
        // Process message (single-threaded)
        await Task.Delay(10); // Simulate work
    }

    public async ValueTask DisposeAsync()
    {
        _cts.Cancel();
        _mailbox.Writer.Complete();
        await _runner.ConfigureAwait(false);
    }
}

public class Message { }
```

**Actor with state**

- Encapsulate mutable state inside the actor loop.

```csharp
public sealed class CounterActor : IAsyncDisposable
{
    private readonly Channel<CounterMessage> _mailbox;
    private readonly CancellationTokenSource _cts = new();
    private readonly Task _runner;
    private int _value; // Isolated state

    public CounterActor()
    {
        _mailbox = Channel.CreateUnbounded<CounterMessage>(new UnboundedChannelOptions
        {
            SingleReader = true,
            SingleWriter = false
        });
        _runner = Task.Run(RunAsync);
    }

    public ValueTask IncrementAsync(int amount, CancellationToken cancellationToken = default)
        => _mailbox.Writer.WriteAsync(new IncrementMessage(amount), cancellationToken);

    public Task<int> GetValueAsync(CancellationToken cancellationToken = default)
    {
        var message = new GetValueMessage();
        _mailbox.Writer.TryWrite(message);
        return message.Task;
    }

    private async Task RunAsync()
    {
        try
        {
            await foreach (var message in _mailbox.Reader.ReadAllAsync(_cts.Token).ConfigureAwait(false))
            {
                switch (message)
                {
                    case IncrementMessage inc:
                        _value += inc.Amount; // Safe: single-threaded access
                        break;
                    case GetValueMessage get:
                        get.SetResult(_value);
                        break;
                }
            }
        }
        catch (OperationCanceledException)
        {
            // Expected on shutdown
        }
    }

    public async ValueTask DisposeAsync()
    {
        _cts.Cancel();
        _mailbox.Writer.Complete();
        await _runner.ConfigureAwait(false);
    }
}

public abstract class CounterMessage { }

public class IncrementMessage : CounterMessage
{
    public int Amount { get; }
    public IncrementMessage(int amount) => Amount = amount;
}

public class GetValueMessage : CounterMessage
{
    private readonly TaskCompletionSource<int> _tcs = new();
    public Task<int> Task => _tcs.Task;
    public void SetResult(int value) => _tcs.SetResult(value);
}
```

**Mentor emphasis**

- **Use Channels** as mailboxes for actors.
- **Single reader** ensures sequential message processing.
- **Encapsulate state** inside the actor loop.
- **Use message types** to represent commands and queries.

### 2. **Encapsulating mutable state behind an actor loop**

**Complex state management**

- Actors can manage complex state structures safely.

```csharp
public sealed class OrderBookActor : IAsyncDisposable
{
    private readonly Channel<OrderBookMessage> _mailbox;
    private readonly CancellationTokenSource _cts = new();
    private readonly Task _runner;
    
    // Complex isolated state
    private readonly Dictionary<string, Order> _orders = new();
    private readonly List<Order> _pendingOrders = new();
    private decimal _totalValue = 0;

    public OrderBookActor()
    {
        _mailbox = Channel.CreateUnbounded<OrderBookMessage>(new UnboundedChannelOptions
        {
            SingleReader = true,
            SingleWriter = false
        });
        _runner = Task.Run(RunAsync);
    }

    public ValueTask AddOrderAsync(Order order, CancellationToken cancellationToken = default)
        => _mailbox.Writer.WriteAsync(new AddOrderMessage(order), cancellationToken);

    public Task<Order?> GetOrderAsync(string orderId, CancellationToken cancellationToken = default)
    {
        var message = new GetOrderMessage(orderId);
        _mailbox.Writer.TryWrite(message);
        return message.Task;
    }

    public Task<OrderBookSnapshot> GetSnapshotAsync(CancellationToken cancellationToken = default)
    {
        var message = new GetSnapshotMessage();
        _mailbox.Writer.TryWrite(message);
        return message.Task;
    }

    private async Task RunAsync()
    {
        try
        {
            await foreach (var message in _mailbox.Reader.ReadAllAsync(_cts.Token).ConfigureAwait(false))
            {
                switch (message)
                {
                    case AddOrderMessage add:
                        // All state mutations are safe: single-threaded
                        _orders[add.Order.Id] = add.Order;
                        _pendingOrders.Add(add.Order);
                        _totalValue += add.Order.Value;
                        break;
                    case GetOrderMessage get:
                        _orders.TryGetValue(get.OrderId, out var order);
                        get.SetResult(order);
                        break;
                    case GetSnapshotMessage snapshot:
                        snapshot.SetResult(new OrderBookSnapshot
                        {
                            OrderCount = _orders.Count,
                            PendingCount = _pendingOrders.Count,
                            TotalValue = _totalValue
                        });
                        break;
                }
            }
        }
        catch (OperationCanceledException)
        {
            // Expected on shutdown
        }
    }

    public async ValueTask DisposeAsync()
    {
        _cts.Cancel();
        _mailbox.Writer.Complete();
        await _runner.ConfigureAwait(false);
    }
}

public class Order
{
    public string Id { get; set; } = "";
    public decimal Value { get; set; }
}

public class OrderBookSnapshot
{
    public int OrderCount { get; set; }
    public int PendingCount { get; set; }
    public decimal TotalValue { get; set; }
}

public abstract class OrderBookMessage { }

public class AddOrderMessage : OrderBookMessage
{
    public Order Order { get; }
    public AddOrderMessage(Order order) => Order = order;
}

public class GetOrderMessage : OrderBookMessage
{
    public string OrderId { get; }
    private readonly TaskCompletionSource<Order?> _tcs = new();
    public Task<Order?> Task => _tcs.Task;
    public GetOrderMessage(string orderId) => OrderId = orderId;
    public void SetResult(Order? order) => _tcs.SetResult(order);
}

public class GetSnapshotMessage : OrderBookMessage
{
    private readonly TaskCompletionSource<OrderBookSnapshot> _tcs = new();
    public Task<OrderBookSnapshot> Task => _tcs.Task;
    public void SetResult(OrderBookSnapshot snapshot) => _tcs.SetResult(snapshot);
}
```

**State invariants**

- Actors can maintain complex invariants safely.

```csharp
public sealed class BankAccountActor : IAsyncDisposable
{
    private readonly Channel<AccountMessage> _mailbox;
    private readonly CancellationTokenSource _cts = new();
    private readonly Task _runner;
    
    // Invariant: Balance >= 0
    private decimal _balance = 0;
    private readonly List<Transaction> _transactions = new();

    public BankAccountActor()
    {
        _mailbox = Channel.CreateUnbounded<AccountMessage>(new UnboundedChannelOptions
        {
            SingleReader = true,
            SingleWriter = false
        });
        _runner = Task.Run(RunAsync);
    }

    public ValueTask DepositAsync(decimal amount, CancellationToken cancellationToken = default)
        => _mailbox.Writer.WriteAsync(new DepositMessage(amount), cancellationToken);

    public Task<bool> WithdrawAsync(decimal amount, CancellationToken cancellationToken = default)
    {
        var message = new WithdrawMessage(amount);
        _mailbox.Writer.TryWrite(message);
        return message.Task;
    }

    private async Task RunAsync()
    {
        try
        {
            await foreach (var message in _mailbox.Reader.ReadAllAsync(_cts.Token).ConfigureAwait(false))
            {
                switch (message)
                {
                    case DepositMessage deposit:
                        _balance += deposit.Amount;
                        _transactions.Add(new Transaction
                        {
                            Type = TransactionType.Deposit,
                            Amount = deposit.Amount,
                            Balance = _balance
                        });
                        // Invariant maintained: balance can only increase
                        break;
                    case WithdrawMessage withdraw:
                        if (_balance >= withdraw.Amount)
                        {
                            _balance -= withdraw.Amount;
                            _transactions.Add(new Transaction
                            {
                                Type = TransactionType.Withdraw,
                                Amount = withdraw.Amount,
                                Balance = _balance
                            });
                            withdraw.SetResult(true);
                        }
                        else
                        {
                            withdraw.SetResult(false);
                        }
                        // Invariant maintained: balance never goes negative
                        break;
                }
            }
        }
        catch (OperationCanceledException)
        {
            // Expected on shutdown
        }
    }

    public async ValueTask DisposeAsync()
    {
        _cts.Cancel();
        _mailbox.Writer.Complete();
        await _runner.ConfigureAwait(false);
    }
}

public abstract class AccountMessage { }

public class DepositMessage : AccountMessage
{
    public decimal Amount { get; }
    public DepositMessage(decimal amount) => Amount = amount;
}

public class WithdrawMessage : AccountMessage
{
    public decimal Amount { get; }
    private readonly TaskCompletionSource<bool> _tcs = new();
    public Task<bool> Task => _tcs.Task;
    public WithdrawMessage(decimal amount) => Amount = amount;
    public void SetResult(bool success) => _tcs.SetResult(success);
}

public class Transaction
{
    public TransactionType Type { get; set; }
    public decimal Amount { get; set; }
    public decimal Balance { get; set; }
}

public enum TransactionType
{
    Deposit,
    Withdraw
}
```

**Mentor emphasis**

- **Encapsulate all mutable state** inside the actor loop.
- **Maintain invariants** safely with single-threaded access.
- **Use message types** to represent state mutations.
- **Return results** via `TaskCompletionSource` for queries.

### 3. **Command-processing and workflow orchestration in actors**

**Command pattern**

- Actors process commands and return results.

```csharp
public sealed class WorkflowActor : IAsyncDisposable
{
    private readonly Channel<WorkflowCommand> _mailbox;
    private readonly CancellationTokenSource _cts = new();
    private readonly Task _runner;
    private WorkflowState _state = WorkflowState.Idle;

    public WorkflowActor()
    {
        _mailbox = Channel.CreateUnbounded<WorkflowCommand>(new UnboundedChannelOptions
        {
            SingleReader = true,
            SingleWriter = false
        });
        _runner = Task.Run(RunAsync);
    }

    public Task<TResult> ExecuteAsync<TResult>(WorkflowCommand command, CancellationToken cancellationToken = default)
    {
        if (command is ICommandWithResult<TResult> cmdWithResult)
        {
            _mailbox.Writer.TryWrite(command);
            return cmdWithResult.Task;
        }
        throw new ArgumentException("Command must implement ICommandWithResult<T>");
    }

    private async Task RunAsync()
    {
        try
        {
            await foreach (var command in _mailbox.Reader.ReadAllAsync(_cts.Token).ConfigureAwait(false))
            {
                switch (command)
                {
                    case StartWorkflowCommand start:
                        if (_state == WorkflowState.Idle)
                        {
                            _state = WorkflowState.Running;
                            await ExecuteWorkflowAsync();
                            _state = WorkflowState.Completed;
                            start.SetResult(true);
                        }
                        else
                        {
                            start.SetResult(false);
                        }
                        break;
                    case GetStateCommand getState:
                        getState.SetResult(_state);
                        break;
                }
            }
        }
        catch (OperationCanceledException)
        {
            // Expected on shutdown
        }
    }

    private async Task ExecuteWorkflowAsync()
    {
        // Execute workflow steps
        await Task.Delay(100);
    }

    public async ValueTask DisposeAsync()
    {
        _cts.Cancel();
        _mailbox.Writer.Complete();
        await _runner.ConfigureAwait(false);
    }
}

public abstract class WorkflowCommand { }

public interface ICommandWithResult<T>
{
    Task<T> Task { get; }
}

public class StartWorkflowCommand : WorkflowCommand, ICommandWithResult<bool>
{
    private readonly TaskCompletionSource<bool> _tcs = new();
    public Task<bool> Task => _tcs.Task;
    public void SetResult(bool result) => _tcs.SetResult(result);
}

public class GetStateCommand : WorkflowCommand, ICommandWithResult<WorkflowState>
{
    private readonly TaskCompletionSource<WorkflowState> _tcs = new();
    public Task<WorkflowState> Task => _tcs.Task;
    public void SetResult(WorkflowState state) => _tcs.SetResult(state);
}

public enum WorkflowState
{
    Idle,
    Running,
    Completed
}
```

**Workflow orchestration**

- Actors can orchestrate multi-step workflows.

```csharp
public sealed class OrderProcessingActor : IAsyncDisposable
{
    private readonly Channel<OrderCommand> _mailbox;
    private readonly CancellationTokenSource _cts = new();
    private readonly Task _runner;
    private readonly Dictionary<string, OrderProcess> _processes = new();

    public OrderProcessingActor()
    {
        _mailbox = Channel.CreateUnbounded<OrderCommand>(new UnboundedChannelOptions
        {
            SingleReader = true,
            SingleWriter = false
        });
        _runner = Task.Run(RunAsync);
    }

    public ValueTask ProcessOrderAsync(Order order, CancellationToken cancellationToken = default)
        => _mailbox.Writer.WriteAsync(new ProcessOrderCommand(order), cancellationToken);

    private async Task RunAsync()
    {
        try
        {
            await foreach (var command in _mailbox.Reader.ReadAllAsync(_cts.Token).ConfigureAwait(false))
            {
                switch (command)
                {
                    case ProcessOrderCommand process:
                        await ProcessOrderWorkflowAsync(process.Order);
                        break;
                }
            }
        }
        catch (OperationCanceledException)
        {
            // Expected on shutdown
        }
    }

    private async Task ProcessOrderWorkflowAsync(Order order)
    {
        var process = new OrderProcess { OrderId = order.Id, State = OrderState.Validating };
        _processes[order.Id] = process;

        // Step 1: Validate
        process.State = OrderState.Validating;
        await ValidateOrderAsync(order);
        
        // Step 2: Reserve inventory
        process.State = OrderState.Reserving;
        await ReserveInventoryAsync(order);
        
        // Step 3: Process payment
        process.State = OrderState.Paying;
        await ProcessPaymentAsync(order);
        
        // Step 4: Fulfill
        process.State = OrderState.Fulfilling;
        await FulfillOrderAsync(order);
        
        process.State = OrderState.Completed;
    }

    private async Task ValidateOrderAsync(Order order) => await Task.Delay(10);
    private async Task ReserveInventoryAsync(Order order) => await Task.Delay(10);
    private async Task ProcessPaymentAsync(Order order) => await Task.Delay(10);
    private async Task FulfillOrderAsync(Order order) => await Task.Delay(10);

    public async ValueTask DisposeAsync()
    {
        _cts.Cancel();
        _mailbox.Writer.Complete();
        await _runner.ConfigureAwait(false);
    }
}

public class OrderProcess
{
    public string OrderId { get; set; } = "";
    public OrderState State { get; set; }
}

public enum OrderState
{
    Validating,
    Reserving,
    Paying,
    Fulfilling,
    Completed
}

public abstract class OrderCommand { }

public class ProcessOrderCommand : OrderCommand
{
    public Order Order { get; }
    public ProcessOrderCommand(Order order) => Order = order;
}
```

**Mentor emphasis**

- **Use commands** to represent operations.
- **Orchestrate workflows** within actors for sequential processing.
- **Maintain workflow state** inside the actor.
- **Handle errors** appropriately in workflow steps.

### 4. **Scaling out actors: sharding and routing messages**

**Actor sharding**

- Distribute actors across multiple instances for scalability.

```csharp
public sealed class ShardedActorSystem
{
    private readonly Actor[] _actors;
    private readonly int _shardCount;

    public ShardedActorSystem(int shardCount = 10)
    {
        _shardCount = shardCount;
        _actors = new Actor[shardCount];
        for (int i = 0; i < shardCount; i++)
        {
            _actors[i] = new Actor($"Actor-{i}");
        }
    }

    public ValueTask SendAsync(string key, Message message, CancellationToken cancellationToken = default)
    {
        var shardIndex = GetShardIndex(key);
        return _actors[shardIndex].SendAsync(message, cancellationToken);
    }

    private int GetShardIndex(string key)
    {
        // Simple hash-based sharding
        return Math.Abs(key.GetHashCode()) % _shardCount;
    }

    public async Task DisposeAsync()
    {
        foreach (var actor in _actors)
        {
            await actor.DisposeAsync();
        }
    }
}

public sealed class Actor : IAsyncDisposable
{
    private readonly Channel<Message> _mailbox;
    private readonly CancellationTokenSource _cts = new();
    private readonly Task _runner;
    private readonly string _name;

    public Actor(string name)
    {
        _name = name;
        _mailbox = Channel.CreateUnbounded<Message>(new UnboundedChannelOptions
        {
            SingleReader = true,
            SingleWriter = false
        });
        _runner = Task.Run(RunAsync);
    }

    public ValueTask SendAsync(Message message, CancellationToken cancellationToken = default)
        => _mailbox.Writer.WriteAsync(message, cancellationToken);

    private async Task RunAsync()
    {
        try
        {
            await foreach (var message in _mailbox.Reader.ReadAllAsync(_cts.Token).ConfigureAwait(false))
            {
                await ProcessMessageAsync(message);
            }
        }
        catch (OperationCanceledException)
        {
            // Expected on shutdown
        }
    }

    private async Task ProcessMessageAsync(Message message)
    {
        Console.WriteLine($"{_name} processing message");
        await Task.Delay(10);
    }

    public async ValueTask DisposeAsync()
    {
        _cts.Cancel();
        _mailbox.Writer.Complete();
        await _runner.ConfigureAwait(false);
    }
}
```

**Message routing**

- Route messages to specific actors based on criteria.

```csharp
public sealed class RoutedActorSystem
{
    private readonly Dictionary<string, Actor> _actors = new();
    private readonly object _lock = new();

    public async Task<Actor> GetOrCreateActorAsync(string actorId)
    {
        lock (_lock)
        {
            if (!_actors.TryGetValue(actorId, out var actor))
            {
                actor = new Actor(actorId);
                _actors[actorId] = actor;
            }
            return actor;
        }
    }

    public async ValueTask SendAsync(string actorId, Message message, CancellationToken cancellationToken = default)
    {
        var actor = await GetOrCreateActorAsync(actorId);
        await actor.SendAsync(message, cancellationToken);
    }

    public async Task DisposeAsync()
    {
        foreach (var actor in _actors.Values)
        {
            await actor.DisposeAsync();
        }
    }
}
```

**Mentor emphasis**

- **Shard actors** for horizontal scalability.
- **Route messages** to specific actors based on keys.
- **Use consistent hashing** for shard selection.
- **Manage actor lifecycle** (create, dispose) appropriately.

### 5. **When to choose actors over locks or transactional structures**

**When to use actors**

- **High contention**: Many threads competing for the same resource.
- **Complex invariants**: Multiple related pieces of state that must be consistent.
- **Sequential processing**: Operations must be processed in order.
- **Isolation**: Need to isolate state from other components.

**Example: high contention scenario**

```csharp
// BAD: High contention with locks
public class HighContentionCounter
{
    private int _value;
    private readonly object _lock = new();

    public void Increment()
    {
        lock (_lock) // Contention hotspot
        {
            _value++;
        }
    }
}

// GOOD: Actor eliminates contention
public class ActorCounter
{
    private readonly CounterActor _actor = new();
    
    public ValueTask IncrementAsync() => _actor.IncrementAsync(1);
}
```

**When NOT to use actors**

- **Simple operations**: `Interlocked` is sufficient for simple counters.
- **Low contention**: Locks are fine when contention is rare.
- **Read-heavy workloads**: `ReaderWriterLockSlim` may be better.
- **Performance-critical**: Actors have message-passing overhead.

**Example: simple operation**

```csharp
// GOOD: Simple operation, no need for actor
public class SimpleCounter
{
    private int _value;
    
    public void Increment() => Interlocked.Increment(ref _value);
    public int Value => _value;
}

// Overkill: Actor for simple counter
public class OverkillCounter
{
    private readonly CounterActor _actor = new();
    
    public ValueTask IncrementAsync() => _actor.IncrementAsync(1);
}
```

**Trade-offs**

- **Actors**: Higher overhead, better isolation, sequential processing.
- **Locks**: Lower overhead, shared state, concurrent processing.
- **Lock-free**: Lowest overhead, complex to implement, limited operations.

**Decision matrix**

| Scenario | Recommendation |
|----------|---------------|
| Simple counter, low contention | `Interlocked` |
| Dictionary with moderate contention | `ConcurrentDictionary` |
| Complex state, high contention | Actor |
| Sequential processing required | Actor |
| Read-heavy workload | `ReaderWriterLockSlim` |
| Performance-critical, simple operations | Lock-free or `Interlocked` |

**Mentor emphasis**

- **Use actors** for high contention and complex invariants.
- **Use locks** for simple operations with low contention.
- **Consider overhead** when choosing between actors and locks.
- **Measure** to make informed decisions.

---

**Summary**

- **Use Channels** as mailboxes for actors with single readers.
- **Encapsulate state** inside actor loops for thread safety.
- **Process commands** sequentially within actors.
- **Shard actors** for horizontal scalability.
- **Choose actors** when you have high contention or complex invariants.
