### 1. **First result wins with Task.WhenAny and cancellation of losers**

**Basic pattern**

- Use `Task.WhenAny` to get the first completed task, then cancel the others.

```csharp
using System;
using System.Collections.Generic;
using System.Linq;
using System.Net.Http;
using System.Threading;
using System.Threading.Tasks;

public static class FirstResultWins
{
    public static async Task<string> GetFirstResultAsync(
        IEnumerable<string> urls,
        CancellationToken cancellationToken = default)
    {
        using var httpClient = new HttpClient();
        using var cts = CancellationTokenSource.CreateLinkedTokenSource(cancellationToken);
        
        var tasks = urls.Select(url => 
            httpClient.GetStringAsync(url, cts.Token)).ToList();
        
        var completedTask = await Task.WhenAny(tasks).ConfigureAwait(false);
        
        // Cancel remaining tasks
        cts.Cancel();
        
        try
        {
            return await completedTask.ConfigureAwait(false);
        }
        catch (Exception ex)
        {
            // Handle error if first result failed
            throw new InvalidOperationException("All requests failed", ex);
        }
    }
}
```

**With timeout**

- Combine `WhenAny` with a timeout task.

```csharp
public static class FirstResultWithTimeout
{
    public static async Task<string?> GetFirstResultWithTimeoutAsync(
        IEnumerable<string> urls,
        TimeSpan timeout,
        CancellationToken cancellationToken = default)
    {
        using var httpClient = new HttpClient();
        using var cts = CancellationTokenSource.CreateLinkedTokenSource(cancellationToken);
        cts.CancelAfter(timeout);
        
        var tasks = urls.Select(url => 
            httpClient.GetStringAsync(url, cts.Token)).ToList();
        
        var timeoutTask = Task.Delay(timeout, cts.Token);
        var allTasks = tasks.Concat(new[] { timeoutTask }).ToList();
        
        var completedTask = await Task.WhenAny(allTasks).ConfigureAwait(false);
        
        if (completedTask == timeoutTask)
        {
            cts.Cancel();
            return null; // Timeout
        }
        
        cts.Cancel(); // Cancel remaining tasks
        
        try
        {
            return await ((Task<string>)completedTask).ConfigureAwait(false);
        }
        catch (Exception ex)
        {
            throw new InvalidOperationException("Request failed", ex);
        }
    }
}
```

**With result validation**

- Validate the first result before accepting it.

```csharp
public static class FirstValidResult
{
    public static async Task<string> GetFirstValidResultAsync(
        IEnumerable<string> urls,
        Func<string, bool> validator,
        CancellationToken cancellationToken = default)
    {
        using var httpClient = new HttpClient();
        using var cts = CancellationTokenSource.CreateLinkedTokenSource(cancellationToken);
        
        var tasks = urls.Select(url => 
            httpClient.GetStringAsync(url, cts.Token)).ToList();
        
        var remainingTasks = new List<Task<string>>(tasks);
        
        while (remainingTasks.Count > 0)
        {
            var completedTask = await Task.WhenAny(remainingTasks).ConfigureAwait(false);
            remainingTasks.Remove(completedTask);
            
            try
            {
                var result = await completedTask.ConfigureAwait(false);
                if (validator(result))
                {
                    cts.Cancel(); // Cancel remaining tasks
                    return result;
                }
            }
            catch
            {
                // Continue to next task
            }
        }
        
        throw new InvalidOperationException("No valid result found");
    }
}
```

**Mentor emphasis**

- **Cancel remaining tasks** after getting the first result.
- **Handle exceptions** from the first completed task.
- **Use timeout tasks** to enforce deadlines.
- **Validate results** before accepting them.

### 2. **Fan-out/fan-in patterns with Task.WhenAll**

**Basic fan-out/fan-in**

- Start multiple operations, wait for all to complete, then aggregate results.

```csharp
public static class FanOutFanIn
{
    public static async Task<int[]> ProcessAllAsync(
        IEnumerable<int> items,
        Func<int, Task<int>> processor,
        CancellationToken cancellationToken = default)
    {
        var tasks = items.Select(item => processor(item)).ToArray();
        return await Task.WhenAll(tasks).ConfigureAwait(false);
    }

    public static async Task ProcessAllAsync(
        IEnumerable<int> items,
        Func<int, Task> processor,
        CancellationToken cancellationToken = default)
    {
        var tasks = items.Select(item => processor(item)).ToArray();
        await Task.WhenAll(tasks).ConfigureAwait(false);
    }
}

// Usage
public static class FanOutFanInExamples
{
    public static async Task<int[]> DownloadSizesAsync(string[] urls)
    {
        using var httpClient = new HttpClient();
        
        return await FanOutFanIn.ProcessAllAsync(
            urls,
            async url =>
            {
                var response = await httpClient.GetAsync(url);
                return (int)response.Content.Headers.ContentLength ?? 0;
            });
    }
}
```

**With concurrency limit**

- Limit the number of concurrent operations.

```csharp
public static class BoundedFanOutFanIn
{
    public static async Task<TResult[]> ProcessAllAsync<T, TResult>(
        IEnumerable<T> items,
        Func<T, Task<TResult>> processor,
        int maxConcurrency,
        CancellationToken cancellationToken = default)
    {
        var semaphore = new SemaphoreSlim(maxConcurrency);
        var tasks = items.Select(async item =>
        {
            await semaphore.WaitAsync(cancellationToken).ConfigureAwait(false);
            try
            {
                return await processor(item).ConfigureAwait(false);
            }
            finally
            {
                semaphore.Release();
            }
        }).ToArray();
        
        return await Task.WhenAll(tasks).ConfigureAwait(false);
    }
}

// Usage
public static class BoundedExamples
{
    public static async Task<string[]> DownloadWithLimitAsync(string[] urls)
    {
        using var httpClient = new HttpClient();
        
        return await BoundedFanOutFanIn.ProcessAllAsync(
            urls,
            async url => await httpClient.GetStringAsync(url),
            maxConcurrency: 10); // Max 10 concurrent downloads
    }
}
```

**With progress tracking**

- Track progress of fan-out operations.

```csharp
public static class FanOutWithProgress
{
    public static async Task<TResult[]> ProcessAllWithProgressAsync<T, TResult>(
        IEnumerable<T> items,
        Func<T, Task<TResult>> processor,
        IProgress<int> progress,
        CancellationToken cancellationToken = default)
    {
        var itemsList = items.ToList();
        var results = new TResult[itemsList.Count];
        var completedCount = 0;
        var lockObj = new object();
        
        var tasks = itemsList.Select((item, index) => 
            processor(item).ContinueWith(task =>
            {
                lock (lockObj)
                {
                    results[index] = task.Result;
                    completedCount++;
                    progress.Report(completedCount);
                }
                return task.Result;
            }, cancellationToken)).ToArray();
        
        return await Task.WhenAll(tasks).ConfigureAwait(false);
    }
}
```

**Mentor emphasis**

- **Use `Task.WhenAll`** to wait for all operations.
- **Limit concurrency** with `SemaphoreSlim` for resource management.
- **Track progress** for long-running fan-out operations.
- **Handle exceptions** appropriately (see section on partial failures).

### 3. **Batching many small operations for throughput**

**Batching pattern**

- Group small operations into batches for better throughput.

```csharp
using System;
using System.Collections.Generic;
using System.Linq;
using System.Threading;
using System.Threading.Tasks;

public static class Batching
{
    public static async Task ProcessBatchesAsync<T>(
        IEnumerable<T> items,
        Func<IEnumerable<T>, Task> batchProcessor,
        int batchSize,
        CancellationToken cancellationToken = default)
    {
        var batch = new List<T>(batchSize);
        
        foreach (var item in items)
        {
            batch.Add(item);
            
            if (batch.Count >= batchSize)
            {
                await batchProcessor(batch).ConfigureAwait(false);
                batch.Clear();
            }
        }
        
        if (batch.Count > 0)
        {
            await batchProcessor(batch).ConfigureAwait(false);
        }
    }

    public static async Task<TResult[]> ProcessBatchesAsync<T, TResult>(
        IEnumerable<T> items,
        Func<IEnumerable<T>, Task<TResult[]>> batchProcessor,
        int batchSize,
        CancellationToken cancellationToken = default)
    {
        var results = new List<TResult>();
        var batch = new List<T>(batchSize);
        
        foreach (var item in items)
        {
            batch.Add(item);
            
            if (batch.Count >= batchSize)
            {
                var batchResults = await batchProcessor(batch).ConfigureAwait(false);
                results.AddRange(batchResults);
                batch.Clear();
            }
        }
        
        if (batch.Count > 0)
        {
            var batchResults = await batchProcessor(batch).ConfigureAwait(false);
            results.AddRange(batchResults);
        }
        
        return results.ToArray();
    }
}

// Usage
public static class BatchingExamples
{
    public static async Task SaveRecordsAsync(IEnumerable<Record> records)
    {
        await Batching.ProcessBatchesAsync(
            records,
            async batch => await SaveBatchAsync(batch),
            batchSize: 100);
    }

    private static async Task SaveBatchAsync(IEnumerable<Record> batch)
    {
        // Batch database insert, API call, etc.
        await Task.Delay(10);
    }
}
```

**Parallel batching**

- Process batches in parallel for even better throughput.

```csharp
public static class ParallelBatching
{
    public static async Task ProcessBatchesParallelAsync<T>(
        IEnumerable<T> items,
        Func<IEnumerable<T>, Task> batchProcessor,
        int batchSize,
        int maxConcurrentBatches,
        CancellationToken cancellationToken = default)
    {
        var batches = items
            .Select((item, index) => new { item, index })
            .GroupBy(x => x.index / batchSize)
            .Select(g => g.Select(x => x.item).ToList())
            .ToList();
        
        var semaphore = new SemaphoreSlim(maxConcurrentBatches);
        var tasks = batches.Select(async batch =>
        {
            await semaphore.WaitAsync(cancellationToken).ConfigureAwait(false);
            try
            {
                await batchProcessor(batch).ConfigureAwait(false);
            }
            finally
            {
                semaphore.Release();
            }
        }).ToArray();
        
        await Task.WhenAll(tasks).ConfigureAwait(false);
    }
}
```

**Time-based batching**

- Batch items by time instead of count.

```csharp
public static class TimeBasedBatching
{
    public static async Task ProcessTimeBatchesAsync<T>(
        IAsyncEnumerable<T> items,
        Func<IEnumerable<T>, Task> batchProcessor,
        TimeSpan batchWindow,
        CancellationToken cancellationToken = default)
    {
        var batch = new List<T>();
        var lastFlush = DateTime.UtcNow;
        
        await foreach (var item in items.WithCancellation(cancellationToken).ConfigureAwait(false))
        {
            batch.Add(item);
            
            if (DateTime.UtcNow - lastFlush >= batchWindow)
            {
                if (batch.Count > 0)
                {
                    await batchProcessor(batch).ConfigureAwait(false);
                    batch.Clear();
                }
                lastFlush = DateTime.UtcNow;
            }
        }
        
        if (batch.Count > 0)
        {
            await batchProcessor(batch).ConfigureAwait(false);
        }
    }
}
```

**Mentor emphasis**

- **Batch operations** to reduce overhead and improve throughput.
- **Process batches in parallel** when possible.
- **Use time-based batching** for real-time scenarios.
- **Choose batch size** based on latency vs throughput trade-offs.

### 4. **Handling partial failures and compensating actions**

**Partial failure handling**

- Handle cases where some operations succeed and others fail.

```csharp
public static class PartialFailureHandling
{
    public static async Task<Result<T>[]> ProcessWithPartialFailuresAsync<T>(
        IEnumerable<T> items,
        Func<T, Task<Result<T>>> processor,
        CancellationToken cancellationToken = default)
    {
        var tasks = items.Select(item => processor(item)).ToArray();
        var results = await Task.WhenAll(tasks).ConfigureAwait(false);
        return results;
    }

    public static async Task<(TResult[] Successes, Exception[] Failures)> ProcessWithSeparationAsync<T, TResult>(
        IEnumerable<T> items,
        Func<T, Task<TResult>> processor,
        CancellationToken cancellationToken = default)
    {
        var tasks = items.Select(async item =>
        {
            try
            {
                var result = await processor(item).ConfigureAwait(false);
                return (Success: true, Result: result, Exception: (Exception?)null);
            }
            catch (Exception ex)
            {
                return (Success: false, Result: default(TResult)!, Exception: ex);
            }
        }).ToArray();
        
        var results = await Task.WhenAll(tasks).ConfigureAwait(false);
        
        var successes = results.Where(r => r.Success).Select(r => r.Result).ToArray();
        var failures = results.Where(r => !r.Success).Select(r => r.Exception!).ToArray();
        
        return (successes, failures);
    }
}

public class Result<T>
{
    public bool Success { get; set; }
    public T? Value { get; set; }
    public Exception? Error { get; set; }
}
```

**Compensating actions**

- Execute compensating actions for failed operations.

```csharp
public static class CompensatingActions
{
    public static async Task<bool> ProcessWithCompensationAsync<T>(
        T item,
        Func<T, Task> operation,
        Func<T, Task> compensation,
        CancellationToken cancellationToken = default)
    {
        try
        {
            await operation(item).ConfigureAwait(false);
            return true;
        }
        catch (Exception ex)
        {
            try
            {
                await compensation(item).ConfigureAwait(false);
            }
            catch (Exception compEx)
            {
                // Log compensation failure
                throw new AggregateException(
                    "Operation and compensation both failed",
                    ex, compEx);
            }
            return false;
        }
    }

    public static async Task ProcessBatchWithCompensationAsync<T>(
        IEnumerable<T> items,
        Func<T, Task> operation,
        Func<T, Task> compensation,
        CancellationToken cancellationToken = default)
    {
        var completed = new List<T>();
        
        foreach (var item in items)
        {
            try
            {
                await operation(item).ConfigureAwait(false);
                completed.Add(item);
            }
            catch (Exception ex)
            {
                // Compensate all completed items
                foreach (var completedItem in completed)
                {
                    try
                    {
                        await compensation(completedItem).ConfigureAwait(false);
                    }
                    catch
                    {
                        // Log but continue
                    }
                }
                throw; // Re-throw original exception
            }
        }
    }
}
```

**Retry with exponential backoff**

- Retry failed operations with exponential backoff.

```csharp
public static class RetryWithBackoff
{
    public static async Task<T> RetryAsync<T>(
        Func<Task<T>> operation,
        int maxRetries = 3,
        TimeSpan? initialDelay = null,
        CancellationToken cancellationToken = default)
    {
        var delay = initialDelay ?? TimeSpan.FromSeconds(1);
        Exception? lastException = null;
        
        for (int attempt = 0; attempt <= maxRetries; attempt++)
        {
            try
            {
                return await operation().ConfigureAwait(false);
            }
            catch (Exception ex)
            {
                lastException = ex;
                
                if (attempt < maxRetries)
                {
                    await Task.Delay(delay, cancellationToken).ConfigureAwait(false);
                    delay = TimeSpan.FromMilliseconds(delay.TotalMilliseconds * 2); // Exponential backoff
                }
            }
        }
        
        throw new InvalidOperationException("Max retries exceeded", lastException);
    }
}
```

**Mentor emphasis**

- **Handle partial failures** gracefully.
- **Separate successes from failures** for analysis.
- **Implement compensating actions** for transactional scenarios.
- **Use retry with backoff** for transient failures.

### 5. **Coordinating multiple services / APIs concurrently in a single request**

**Multiple API calls**

- Call multiple APIs concurrently and aggregate results.

```csharp
using System;
using System.Collections.Generic;
using System.Linq;
using System.Net.Http;
using System.Threading;
using System.Threading.Tasks;

public static class MultiServiceCoordination
{
    public static async Task<CombinedResult> GetCombinedDataAsync(
        string userId,
        CancellationToken cancellationToken = default)
    {
        using var httpClient = new HttpClient();
        
        // Fan-out: Start all requests concurrently
        var userTask = httpClient.GetStringAsync($"/api/users/{userId}", cancellationToken);
        var ordersTask = httpClient.GetStringAsync($"/api/orders?userId={userId}", cancellationToken);
        var preferencesTask = httpClient.GetStringAsync($"/api/preferences/{userId}", cancellationToken);
        
        // Wait for all to complete
        await Task.WhenAll(userTask, ordersTask, preferencesTask).ConfigureAwait(false);
        
        // Aggregate results
        return new CombinedResult
        {
            User = await userTask.ConfigureAwait(false),
            Orders = await ordersTask.ConfigureAwait(false),
            Preferences = await preferencesTask.ConfigureAwait(false)
        };
    }
}

public class CombinedResult
{
    public string User { get; set; } = "";
    public string Orders { get; set; } = "";
    public string Preferences { get; set; } = "";
}
```

**Dependent operations**

- Handle operations that depend on previous results.

```csharp
public static class DependentOperations
{
    public static async Task<FinalResult> ProcessDependentAsync(
        string initialData,
        CancellationToken cancellationToken = default)
    {
        using var httpClient = new HttpClient();
        
        // Step 1: Get initial data
        var step1Result = await httpClient.GetStringAsync($"/api/step1/{initialData}", cancellationToken)
            .ConfigureAwait(false);
        
        // Step 2 & 3: Can run in parallel after step 1
        var step2Task = httpClient.GetStringAsync($"/api/step2/{step1Result}", cancellationToken);
        var step3Task = httpClient.GetStringAsync($"/api/step3/{step1Result}", cancellationToken);
        
        await Task.WhenAll(step2Task, step3Task).ConfigureAwait(false);
        
        // Step 4: Depends on step 2 and 3
        var step2Result = await step2Task.ConfigureAwait(false);
        var step3Result = await step3Task.ConfigureAwait(false);
        
        var step4Result = await httpClient.GetStringAsync(
            $"/api/step4?step2={step2Result}&step3={step3Result}", cancellationToken)
            .ConfigureAwait(false);
        
        return new FinalResult { Result = step4Result };
    }
}

public class FinalResult
{
    public string Result { get; set; } = "";
}
```

**With timeout and fallback**

- Use fallback values when services timeout.

```csharp
public static class ServiceWithFallback
{
    public static async Task<string> GetDataWithFallbackAsync(
        CancellationToken cancellationToken = default)
    {
        using var httpClient = new HttpClient();
        using var cts = CancellationTokenSource.CreateLinkedTokenSource(cancellationToken);
        cts.CancelAfter(TimeSpan.FromSeconds(5)); // 5 second timeout
        
        try
        {
            return await httpClient.GetStringAsync("/api/data", cts.Token).ConfigureAwait(false);
        }
        catch (TaskCanceledException)
        {
            // Timeout: return fallback
            return "fallback-data";
        }
        catch (Exception ex)
        {
            // Other error: return fallback
            return "fallback-data";
        }
    }
}
```

**Circuit breaker pattern**

- Prevent cascading failures with a circuit breaker.

```csharp
public class CircuitBreaker
{
    private int _failureCount = 0;
    private DateTime? _lastFailureTime = null;
    private bool _isOpen = false;
    private readonly object _lock = new();
    private readonly int _failureThreshold = 5;
    private readonly TimeSpan _timeout = TimeSpan.FromSeconds(30);

    public async Task<T> ExecuteAsync<T>(Func<Task<T>> operation)
    {
        lock (_lock)
        {
            if (_isOpen)
            {
                if (_lastFailureTime.HasValue && 
                    DateTime.UtcNow - _lastFailureTime.Value > _timeout)
                {
                    _isOpen = false;
                    _failureCount = 0;
                }
                else
                {
                    throw new InvalidOperationException("Circuit breaker is open");
                }
            }
        }
        
        try
        {
            var result = await operation().ConfigureAwait(false);
            
            lock (_lock)
            {
                _failureCount = 0;
            }
            
            return result;
        }
        catch (Exception ex)
        {
            lock (_lock)
            {
                _failureCount++;
                _lastFailureTime = DateTime.UtcNow;
                
                if (_failureCount >= _failureThreshold)
                {
                    _isOpen = true;
                }
            }
            throw;
        }
    }
}
```

**Mentor emphasis**

- **Call services concurrently** when possible for better performance.
- **Handle dependencies** between operations appropriately.
- **Use timeouts and fallbacks** for resilience.
- **Implement circuit breakers** to prevent cascading failures.
