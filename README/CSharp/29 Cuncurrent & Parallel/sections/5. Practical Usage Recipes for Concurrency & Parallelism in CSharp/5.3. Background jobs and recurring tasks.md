### 1. **Fire-and-forget patterns that are actually safe in ASP.NET Core**

**The problem with naive fire-and-forget**

- **Task.Run without awaiting**: The request completes, but the background work may be cancelled when the request context is disposed.
- **Unhandled exceptions**: Exceptions in fire-and-forget tasks can crash the process or go unnoticed.
- **No cancellation support**: Background work can't be cancelled during application shutdown.

**Pattern 1: Background task queue (recommended)**

```csharp
using System;
using System.Threading;
using System.Threading.Channels;
using System.Threading.Tasks;
using Microsoft.Extensions.Hosting;

// Register as singleton
public class BackgroundTaskQueue : IBackgroundTaskQueue
{
    private readonly Channel<Func<CancellationToken, ValueTask>> _queue;
    
    public BackgroundTaskQueue(int capacity = 100)
    {
        var options = new BoundedChannelOptions(capacity)
        {
            FullMode = BoundedChannelFullMode.Wait
        };
        _queue = Channel.CreateBounded<Func<CancellationToken, ValueTask>>(options);
    }
    
    public async ValueTask QueueBackgroundWorkItemAsync(
        Func<CancellationToken, ValueTask> workItem,
        CancellationToken ct = default)
    {
        if (workItem == null)
            throw new ArgumentNullException(nameof(workItem));
        
        await _queue.Writer.WriteAsync(workItem, ct).ConfigureAwait(false);
    }
    
    public async ValueTask<Func<CancellationToken, ValueTask>> DequeueAsync(
        CancellationToken ct)
    {
        return await _queue.Reader.ReadAsync(ct).ConfigureAwait(false);
    }
}

public interface IBackgroundTaskQueue
{
    ValueTask QueueBackgroundWorkItemAsync(
        Func<CancellationToken, ValueTask> workItem,
        CancellationToken ct = default);
    
    ValueTask<Func<CancellationToken, ValueTask>> DequeueAsync(CancellationToken ct);
}

// Background service to process the queue
public class QueuedHostedService : BackgroundService
{
    private readonly IBackgroundTaskQueue _taskQueue;
    private readonly ILogger<QueuedHostedService> _logger;
    
    public QueuedHostedService(
        IBackgroundTaskQueue taskQueue,
        ILogger<QueuedHostedService> logger)
    {
        _taskQueue = taskQueue;
        _logger = logger;
    }
    
    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        while (!stoppingToken.IsCancellationRequested)
        {
            try
            {
                var workItem = await _taskQueue.DequeueAsync(stoppingToken)
                    .ConfigureAwait(false);
                
                try
                {
                    await workItem(stoppingToken).ConfigureAwait(false);
                }
                catch (Exception ex)
                {
                    _logger.LogError(ex, "Error executing background work item");
                }
            }
            catch (OperationCanceledException)
            {
                // Shutdown requested
                break;
            }
        }
    }
}

// Usage in a controller or service:
public class MyController : ControllerBase
{
    private readonly IBackgroundTaskQueue _taskQueue;
    
    public MyController(IBackgroundTaskQueue taskQueue)
    {
        _taskQueue = taskQueue;
    }
    
    [HttpPost("process")]
    public IActionResult Process([FromBody] ProcessRequest request)
    {
        // Queue background work
        _ = _taskQueue.QueueBackgroundWorkItemAsync(async ct =>
        {
            await ProcessAsync(request, ct).ConfigureAwait(false);
        });
        
        return Accepted(); // Return immediately
    }
    
    private async ValueTask ProcessAsync(ProcessRequest request, CancellationToken ct)
    {
        // Long-running background work
        await Task.Delay(5000, ct).ConfigureAwait(false);
    }
}

// Registration in Program.cs or Startup.cs:
// builder.Services.AddSingleton<IBackgroundTaskQueue, BackgroundTaskQueue>();
// builder.Services.AddHostedService<QueuedHostedService>();
```

**Pattern 2: Task.Run with proper exception handling (simple cases)**

```csharp
using System;
using System.Threading;
using System.Threading.Tasks;
using Microsoft.Extensions.Hosting;

public static class SafeFireAndForget
{
    // ✅ GOOD: Proper exception handling and cancellation support
    public static void FireAndForgetSafe(
        Func<CancellationToken, Task> work,
        IHostApplicationLifetime lifetime,
        ILogger logger)
    {
        _ = Task.Run(async () =>
        {
            try
            {
                await work(lifetime.ApplicationStopping).ConfigureAwait(false);
            }
            catch (OperationCanceledException)
            {
                // Expected during shutdown
                logger.LogInformation("Background work cancelled during shutdown");
            }
            catch (Exception ex)
            {
                logger.LogError(ex, "Unhandled exception in background work");
            }
        }, lifetime.ApplicationStopping);
    }
}

// Usage:
public class MyService
{
    private readonly IHostApplicationLifetime _lifetime;
    private readonly ILogger<MyService> _logger;
    
    public MyService(IHostApplicationLifetime lifetime, ILogger<MyService> logger)
    {
        _lifetime = lifetime;
        _logger = logger;
    }
    
    public void StartBackgroundWork()
    {
        SafeFireAndForget.FireAndForgetSafe(
            async ct =>
            {
                while (!ct.IsCancellationRequested)
                {
                    await DoWorkAsync(ct).ConfigureAwait(false);
                    await Task.Delay(TimeSpan.FromMinutes(1), ct).ConfigureAwait(false);
                }
            },
            _lifetime,
            _logger
        );
    }
    
    private async Task DoWorkAsync(CancellationToken ct)
    {
        // Background work
    }
}
```

```csharp
using System;

using Microsoft.Extensions.Hosting;

public class LongRunningBackgroundService : BackgroundService

        }
        
        _logger.LogInformation("Background service stopped");
    }

    {
        // Background work
        await Task.Delay(1000, ct).ConfigureAwait(false);
    }
}

```

**Common mistakes**

1. **Fire-and-forget without exception handling**
   ```csharp
   // ❌ WRONG: Unhandled exceptions can crash the process
   _ = Task.Run(async () => await DoWorkAsync());
   
   // ✅ CORRECT: Handle exceptions
   _ = Task.Run(async () =>
   {
       try { await DoWorkAsync(); }
       catch (Exception ex) { logger.LogError(ex, "Error"); }
   });
   ```

2. **Not respecting cancellation**
   ```csharp
   // ❌ WRONG: Work continues after shutdown
   _ = Task.Run(async () => await DoWorkAsync());
   
   // ✅ CORRECT: Use cancellation token
   _ = Task.Run(async () => await DoWorkAsync(stoppingToken), stoppingToken);
   ```

3. **Using Task.Run in ASP.NET Core without proper context**
   ```csharp
   // ❌ WRONG: May capture request context unnecessarily
   _ = Task.Run(async () => await DoWorkAsync());
   
   // ✅ CORRECT: Use ConfigureAwait(false) or BackgroundService
   ```

**Performance notes**

- **Task.Run overhead**: Each fire-and-forget `Task.Run` allocates a Task and uses a Thread Pool thread. For high-frequency scenarios, use a queue-based approach.
- **Queue-based approach**: More efficient for high-throughput scenarios, provides backpressure, and ensures proper shutdown.

**Good to know**

- In ASP.NET Core, prefer `IHostedService` / `BackgroundService` for long-running background work.
- For one-off fire-and-forget tasks, use a queue-based approach to ensure proper lifecycle management.
- Never use `async void` except in UI event handlers (see **1.3**).

### 2. **Implementing recurring work with PeriodicTimer**

**What PeriodicTimer is**

- `PeriodicTimer` (introduced in .NET 6) is designed for async loops with periodic execution.
- Unlike `System.Threading.Timer`, it prevents overlapping executions and integrates naturally with `async`/`await`.

**Basic pattern**

```csharp
using System;
using System.Threading;
using System.Threading.Tasks;
using Microsoft.Extensions.Hosting;

public class PeriodicBackgroundService : BackgroundService
{
    private readonly ILogger<PeriodicBackgroundService> _logger;
    private readonly TimeSpan _period;
    
    public PeriodicBackgroundService(
        ILogger<PeriodicBackgroundService> logger,
        TimeSpan period)
    {
        _logger = logger;
        _period = period;
    }
    
    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        using var timer = new PeriodicTimer(_period);
        
        try
        {
            while (!stoppingToken.IsCancellationRequested)
            {
                // Wait for next tick
                bool hasNextTick = await timer.WaitForNextTickAsync(stoppingToken)
                    .ConfigureAwait(false);
                
                if (!hasNextTick)
                {
                    // Timer was disposed or cancelled
                    break;
                }
                
                // Do periodic work
                await DoPeriodicWorkAsync(stoppingToken).ConfigureAwait(false);
            }
        }
        catch (OperationCanceledException)
        {
            _logger.LogInformation("Periodic work cancelled");
        }
    }
    
    private async Task DoPeriodicWorkAsync(CancellationToken ct)
    {
        _logger.LogInformation("Executing periodic work");
        // Your periodic work here
        await Task.Delay(100, ct).ConfigureAwait(false);
    }
}
```

**Pattern with error handling and drift prevention**

```csharp
using System;
using System.Diagnostics;
using System.Threading;
using System.Threading.Tasks;
using Microsoft.Extensions.Hosting;

public class AccuratePeriodicService : BackgroundService
{
    private readonly ILogger<AccuratePeriodicService> _logger;
    private readonly TimeSpan _period;
    
    public AccuratePeriodicService(
        ILogger<AccuratePeriodicService> logger,
        TimeSpan period)
    {
        _logger = logger;
        _period = period;
    }
    
    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        using var timer = new PeriodicTimer(_period);
        
        while (!stoppingToken.IsCancellationRequested)
        {
            var sw = Stopwatch.StartNew();
            
            bool hasNextTick = await timer.WaitForNextTickAsync(stoppingToken)
                .ConfigureAwait(false);
            
            if (!hasNextTick)
                break;
            
            try
            {
                await DoPeriodicWorkAsync(stoppingToken).ConfigureAwait(false);
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Error in periodic work");
                // Continue to next iteration (or break if critical)
            }
            
            sw.Stop();
            
            // Log if work took longer than period (drift)
            if (sw.Elapsed > _period)
            {
                _logger.LogWarning(
                    "Periodic work took {Duration}ms, longer than period {Period}ms",
                    sw.ElapsedMilliseconds,
                    _period.TotalMilliseconds);
            }
        }
    }
    
    private async Task DoPeriodicWorkAsync(CancellationToken ct)
    {
        // Periodic work
        await Task.Delay(100, ct).ConfigureAwait(false);
    }
}
```

**Common mistakes**

1. **Not disposing PeriodicTimer**
   ```csharp
   // ❌ WRONG: Timer not disposed
   var timer = new PeriodicTimer(TimeSpan.FromSeconds(1));
   // Timer leaks
   
   // ✅ CORRECT: Use using
   using var timer = new PeriodicTimer(TimeSpan.FromSeconds(1));
   ```

2. **Ignoring return value of WaitForNextTickAsync**
   ```csharp
   // ⚠️ CAUTION: Return value indicates if timer was cancelled/disposed
   bool hasNextTick = await timer.WaitForNextTickAsync(ct);
   if (!hasNextTick)
       break; // Exit loop
   ```

3. **Long-running work causing drift**
   ```csharp
   // ⚠️ CAUTION: If DoWorkAsync takes longer than period, next tick is delayed
   await timer.WaitForNextTickAsync(ct);
   await DoWorkAsync(ct); // If this takes 10 minutes and period is 5 minutes...
   
   // ✅ CONSIDER: Run work in background, don't block the timer loop
   _ = Task.Run(async () => await DoWorkAsync(ct), ct);
   await timer.WaitForNextTickAsync(ct);
   ```

**Performance notes**

- **PeriodicTimer vs System.Threading.Timer**: `PeriodicTimer` is async-friendly and prevents overlapping executions. `System.Threading.Timer` can have overlapping callbacks. See **3.6** for detailed comparison.
- **Drift**: If periodic work takes longer than the period, the next execution is delayed. Consider running long work in the background or using a different scheduling strategy.

**Good to know**

- `PeriodicTimer` is lightweight and efficient for async periodic work.
- For more complex scheduling (cron-like), consider libraries like `Quartz.NET` or `Hangfire`.
- `WaitForNextTickAsync` returns `false` when the timer is disposed or cancellation is requested.

### 3. **Scheduling background jobs with IHostedService / BackgroundService**

**What IHostedService and BackgroundService are**

- `IHostedService` is the interface for long-running background services in ASP.NET Core.
- `BackgroundService` is a base class that implements `IHostedService` with a convenient `ExecuteAsync` pattern.

**Basic BackgroundService pattern**

```csharp
using System;
using System.Threading;
using System.Threading.Tasks;
using Microsoft.Extensions.Hosting;

public class MyBackgroundService : BackgroundService
{
    private readonly ILogger<MyBackgroundService> _logger;
    
    public MyBackgroundService(ILogger<MyBackgroundService> logger)
    {
        _logger = logger;
    }
    
    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        _logger.LogInformation("Background service starting");
        
        try
        {
            await RunAsync(stoppingToken).ConfigureAwait(false);
        }
        catch (OperationCanceledException)
        {
            _logger.LogInformation("Background service cancelled");
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Fatal error in background service");
            throw; // Re-throw to signal failure
        }
        
        _logger.LogInformation("Background service stopped");
    }
    
    private async Task RunAsync(CancellationToken ct)
    {
        while (!ct.IsCancellationRequested)
        {
            await DoWorkAsync(ct).ConfigureAwait(false);
            await Task.Delay(TimeSpan.FromMinutes(1), ct).ConfigureAwait(false);
        }
    }
    
    private async Task DoWorkAsync(CancellationToken ct)
    {
        _logger.LogInformation("Doing background work");
        // Your work here
        await Task.Delay(100, ct).ConfigureAwait(false);
    }
}

// Registration in Program.cs:
// builder.Services.AddHostedService<MyBackgroundService>();
```

**Pattern with dependency injection**

```csharp
using System;
using System.Threading;
using System.Threading.Tasks;
using Microsoft.Extensions.Hosting;

public class DataProcessingService : BackgroundService
{
    private readonly IServiceProvider _serviceProvider;
    private readonly ILogger<DataProcessingService> _logger;

            await Task.Delay(TimeSpan.FromMinutes(5), stoppingToken).ConfigureAwait(false);
        }
    }
}
```

**Common mistakes**

1. **Not handling exceptions**
   ```csharp
   // ❌ WRONG: Unhandled exception stops the service
   protected override async Task ExecuteAsync(CancellationToken stoppingToken)
   {
       while (!stoppingToken.IsCancellationRequested)
       {
           await DoWorkAsync(stoppingToken); // If this throws, service stops
       }
   }
   
   // ✅ CORRECT: Handle exceptions
   protected override async Task ExecuteAsync(CancellationToken stoppingToken)
   {
       while (!stoppingToken.IsCancellationRequested)
       {
           try
           {
               await DoWorkAsync(stoppingToken);
           }
           catch (Exception ex)
           {
               logger.LogError(ex, "Error in background work");
               // Decide: continue or break
           }
       }
   }
   ```

2. **Not respecting cancellation token**
   ```csharp
   // ❌ WRONG: Work continues after shutdown request
   protected override async Task ExecuteAsync(CancellationToken stoppingToken)
   {
       while (true) // Never checks cancellation
       {
           await DoWorkAsync();
       }
   }
   
   // ✅ CORRECT: Check cancellation
   while (!stoppingToken.IsCancellationRequested)
   {
       await DoWorkAsync(stoppingToken);
   }
   ```

3. **Using scoped services without creating a scope**
   ```csharp
   // ❌ WRONG: Scoped service may not work correctly
   private readonly IMyScopedService _service; // Injected as singleton
   
   // ✅ CORRECT: Create scope when needed
   using var scope = _serviceProvider.CreateScope();
   var service = scope.ServiceProvider.GetRequiredService<IMyScopedService>();
   ```

**Performance notes**

- **BackgroundService lifecycle**: The service starts when the host starts and stops when the host stops. Ensure `ExecuteAsync` completes promptly when cancellation is requested.
- **Multiple services**: Multiple `BackgroundService` instances run concurrently. Be mindful of resource usage (CPU, memory, I/O).

**Good to know**

- `BackgroundService` is the recommended way to implement long-running background work in ASP.NET Core.
- The host waits for all `IHostedService` instances to stop during shutdown (with a timeout).
- Use `IHostApplicationLifetime` to be notified of application lifecycle events if needed.

### 4. **Queue-based background processing with Channels**

**Why use a queue**

- **Decoupling**: Producers (HTTP requests) don't wait for consumers (background workers).
- **Backpressure**: Bounded queues prevent unbounded memory growth.
- **Scalability**: Multiple workers can process items concurrently.
- **Resilience**: Items can be retried or persisted.

**Pattern 1: Single producer, single consumer**

```csharp
using System;
using System.Threading;
using System.Threading.Channels;
using System.Threading.Tasks;
using Microsoft.Extensions.Hosting;

public class QueueBasedBackgroundService : BackgroundService
{
    private readonly Channel<WorkItem> _channel;
    private readonly ILogger<QueueBasedBackgroundService> _logger;
    
    public QueueBasedBackgroundService(
        Channel<WorkItem> channel,
        ILogger<QueueBasedBackgroundService> logger)
    {
        _channel = channel;
        _logger = logger;
    }
    
    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        await foreach (var item in _channel.Reader.ReadAllAsync(stoppingToken)
            .ConfigureAwait(false))
        {
            try
            {
                await ProcessItemAsync(item, stoppingToken).ConfigureAwait(false);
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Error processing item {ItemId}", item.Id);
            }
        }
    }
    
    private async Task ProcessItemAsync(WorkItem item, CancellationToken ct)
    {
        _logger.LogInformation("Processing item {ItemId}", item.Id);
        // Process the item
        await Task.Delay(100, ct).ConfigureAwait(false);
    }
}

public record WorkItem(int Id, string Data);

// Producer (e.g., in a controller):
public class WorkController : ControllerBase
{
    private readonly ChannelWriter<WorkItem> _writer;
    
    public WorkController(Channel<WorkItem> channel)
    {
        _writer = channel.Writer;
    }
    
    [HttpPost("queue")]
    public async Task<IActionResult> QueueWork([FromBody] WorkItem item)
    {
        await _writer.WriteAsync(item).ConfigureAwait(false);
        return Accepted();
    }
}
```

**Pattern 2: Multiple workers (fan-out)**

```csharp
using System;
using System.Linq;
using System.Threading;
using System.Threading.Channels;
using System.Threading.Tasks;
using Microsoft.Extensions.Hosting;

public class MultiWorkerBackgroundService : BackgroundService
{
    private readonly ChannelReader<WorkItem> _reader;
    private readonly ILogger<MultiWorkerBackgroundService> _logger;
    private readonly int _workerCount;
    
    public MultiWorkerBackgroundService(
        ChannelReader<WorkItem> reader,
        ILogger<MultiWorkerBackgroundService> logger,
        int workerCount = 4)
    {
        _reader = reader;
        _logger = logger;
        _workerCount = workerCount;
    }
    
    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        // Start multiple workers
        var workers = Enumerable.Range(0, _workerCount)
            .Select(i => RunWorkerAsync(i, stoppingToken))
            .ToArray();
        
        // Wait for all workers to complete
        await Task.WhenAll(workers).ConfigureAwait(false);
    }
    
    private async Task RunWorkerAsync(int workerId, CancellationToken stoppingToken)
    {
        _logger.LogInformation("Worker {WorkerId} started", workerId);
        
        await foreach (var item in _reader.ReadAllAsync(stoppingToken)
            .ConfigureAwait(false))
        {
            try
            {
                await ProcessItemAsync(item, workerId, stoppingToken)
                    .ConfigureAwait(false);
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Worker {WorkerId} error processing item {ItemId}",
                    workerId, item.Id);
            }
        }
        
        _logger.LogInformation("Worker {WorkerId} stopped", workerId);
    }
    
    private async Task ProcessItemAsync(
        WorkItem item,
        int workerId,
        CancellationToken ct)
    {
        _logger.LogInformation("Worker {WorkerId} processing item {ItemId}",
            workerId, item.Id);
        await Task.Delay(100, ct).ConfigureAwait(false);
    }
}
```

**Pattern 3: Bounded queue with backpressure**

```csharp
using System;
using System.Threading;
using System.Threading.Channels;
using System.Threading.Tasks;

public class BoundedQueueService
{
    private readonly Channel<WorkItem> _channel;
    
    public BoundedQueueService(int capacity = 1000)
    {
        var options = new BoundedChannelOptions(capacity)
        {
            FullMode = BoundedChannelFullMode.Wait, // Producer waits when full
            SingleReader = false,
            SingleWriter = false
        };
        _channel = Channel.CreateBounded<WorkItem>(options);
    }
    
    public ChannelWriter<WorkItem> Writer => _channel.Writer;
    public ChannelReader<WorkItem> Reader => _channel.Reader;
    
    // Producer: waits if queue is full
    public async ValueTask EnqueueAsync(WorkItem item, CancellationToken ct = default)
    {
        await _channel.Writer.WriteAsync(item, ct).ConfigureAwait(false);
    }
    
    // Alternative: Drop oldest if full (for telemetry/logging)
    public static Channel<WorkItem> CreateDropOldestChannel(int capacity)
    {
        var options = new BoundedChannelOptions(capacity)
        {
            FullMode = BoundedChannelFullMode.DropOldest,
            SingleReader = false,
            SingleWriter = false
        };
        return Channel.CreateBounded<WorkItem>(options);
    }
}
```

**Common mistakes**

1. **Not completing the writer**
   ```csharp
   // ⚠️ CAUTION: If writer is never completed, ReadAllAsync never ends
   // Complete the writer when done:
   channel.Writer.Complete();
   ```

2. **Unbounded queue causing memory issues**
   ```csharp
   // ❌ WRONG: Unbounded growth
   var channel = Channel.CreateUnbounded<WorkItem>();
   
   // ✅ CORRECT: Use bounded channel
   var channel = Channel.CreateBounded<WorkItem>(1000);
   ```

3. **Not handling channel completion**
   ```csharp
   // ⚠️ CAUTION: ReadAllAsync completes when writer is completed
   await foreach (var item in channel.Reader.ReadAllAsync(ct))
   {
       // Process items
   }
   // Loop exits when writer is completed
   ```

**Performance notes**

- **Channel performance**: `Channel<T>` is highly optimized for async producer-consumer scenarios. It's more efficient than `BlockingCollection<T>` for async code.
- **Bounded vs unbounded**: Bounded channels provide backpressure but may cause producers to wait. Unbounded channels can grow indefinitely.
- **SingleReader/SingleWriter**: Setting these to `true` enables optimizations but restricts usage to one reader/writer.

**Good to know**

- Use `Channel<T>` for async producer-consumer patterns in modern .NET.
- For CPU-bound work queues, consider `BlockingCollection<T>` (see **5.4**).
- Channels integrate well with `IAsyncEnumerable<T>` via `ReadAllAsync`.

### 5. **Graceful shutdown: draining queues and cancelling work**

**Why graceful shutdown matters**

- **Data loss prevention**: Complete in-flight work before shutting down.
- **Resource cleanup**: Close connections, flush buffers, save state.
- **User experience**: Finish current requests before stopping.

**Pattern 1: Draining a queue during shutdown**

```csharp
using System;
using System.Threading;
using System.Threading.Channels;
using System.Threading.Tasks;
using Microsoft.Extensions.Hosting;

public class GracefulQueueService : BackgroundService
{
    private readonly ChannelReader<WorkItem> _reader;
    private readonly ILogger<GracefulQueueService> _logger;
    private readonly IHostApplicationLifetime _lifetime;
    
    public GracefulQueueService(
        ChannelReader<WorkItem> reader,
        ILogger<GracefulQueueService> logger,
        IHostApplicationLifetime lifetime)
    {
        _reader = reader;
        _logger = logger;
        _lifetime = lifetime;
    }
    
    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        _logger.LogInformation("Queue service started");
        
        try
        {
            await ProcessQueueAsync(stoppingToken).ConfigureAwait(false);
        }
        catch (OperationCanceledException)
        {
            _logger.LogInformation("Queue service cancelled");
        }
        
        _logger.LogInformation("Queue service stopped");
    }
    
    private async Task ProcessQueueAsync(CancellationToken stoppingToken)
    {
        // Process items until cancellation is requested
        await foreach (var item in _reader.ReadAllAsync(stoppingToken)
            .ConfigureAwait(false))
        {
            // Use a linked token with a timeout for each item
            using var itemCts = CancellationTokenSource.CreateLinkedTokenSource(stoppingToken);
            itemCts.CancelAfter(TimeSpan.FromMinutes(5)); // Max time per item
            
            try
            {
                await ProcessItemAsync(item, itemCts.Token).ConfigureAwait(false);
            }
            catch (OperationCanceledException) when (itemCts.Token.IsCancellationRequested)
            {
                if (stoppingToken.IsCancellationRequested)
                {
                    _logger.LogWarning("Item {ItemId} cancelled during shutdown", item.Id);
                    // Optionally: re-queue item or save for later processing
                }
                else
                {
                    _logger.LogWarning("Item {ItemId} timed out", item.Id);
                }
            }
        }
        
        // After ReadAllAsync completes, drain any remaining items
        await DrainRemainingItemsAsync(stoppingToken).ConfigureAwait(false);
    }
    
    private async Task DrainRemainingItemsAsync(CancellationToken stoppingToken)
    {
        _logger.LogInformation("Draining remaining items");
        
        // Try to read remaining items with a timeout
        using var drainCts = CancellationTokenSource.CreateLinkedTokenSource(stoppingToken);
        drainCts.CancelAfter(TimeSpan.FromSeconds(30)); // Max drain time
        
        while (!drainCts.Token.IsCancellationRequested)
        {
            if (_reader.TryRead(out var item))
            {
                try
                {
                    await ProcessItemAsync(item, drainCts.Token).ConfigureAwait(false);
                }
                catch (OperationCanceledException)
                {
                    _logger.LogWarning("Could not drain item {ItemId} before shutdown", item.Id);
                    break;
                }
            }
            else
            {
                // No more items
                break;
            }
        }
        
        _logger.LogInformation("Finished draining items");
    }
    
    private async Task ProcessItemAsync(WorkItem item, CancellationToken ct)
    {
        _logger.LogInformation("Processing item {ItemId}", item.Id);
        await Task.Delay(100, ct).ConfigureAwait(false);
    }
}
```

**Pattern 2: Completing channel writer on shutdown**

```csharp
using System;
using System.Threading;
using System.Threading.Channels;
using System.Threading.Tasks;
using Microsoft.Extensions.Hosting;

public class ChannelLifecycleManager : IHostedService
{
    private readonly ChannelWriter<WorkItem> _writer;
    private readonly IHostApplicationLifetime _lifetime;
    private readonly ILogger<ChannelLifecycleManager> _logger;

        IHostApplicationLifetime lifetime,

        _writer = writer;
        _lifetime = lifetime;
        _logger = logger;
    }

    {
        // Register shutdown handler
        _lifetime.ApplicationStopping.Register(() =>
        {
            _logger.LogInformation("Completing channel writer");
            _writer.Complete();
        });
        
        return Task.CompletedTask;
    }
    
    public Task StopAsync(CancellationToken cancellationToken)
    {

    }
}
```

**Pattern 3: Shutdown with timeout**

```csharp
using System;
using System.Threading;
using System.Threading.Tasks;
using Microsoft.Extensions.Hosting;

public class ShutdownWithTimeoutService : BackgroundService
{
    private readonly ILogger<ShutdownWithTimeoutService> _logger;
    private readonly TimeSpan _shutdownTimeout;
    
    public ShutdownWithTimeoutService(
        ILogger<ShutdownWithTimeoutService> logger,
        TimeSpan? shutdownTimeout = null)
    {
        _logger = logger;
        _shutdownTimeout = shutdownTimeout ?? TimeSpan.FromSeconds(30);
    }
    
    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        while (!stoppingToken.IsCancellationRequested)
        {
            await DoWorkAsync(stoppingToken).ConfigureAwait(false);
            await Task.Delay(TimeSpan.FromSeconds(1), stoppingToken).ConfigureAwait(false);
        }
    }
    
    public override async Task StopAsync(CancellationToken cancellationToken)
    {
        _logger.LogInformation("Stopping service with timeout {Timeout}s", 
            _shutdownTimeout.TotalSeconds);
        
        // Create a linked token with timeout
        using var cts = CancellationTokenSource.CreateLinkedTokenSource(cancellationToken);
        cts.CancelAfter(_shutdownTimeout);
        
        try
        {
            // Signal cancellation to ExecuteAsync
            await base.StopAsync(cts.Token).ConfigureAwait(false);
        }
        catch (OperationCanceledException) when (cts.Token.IsCancellationRequested && !cancellationToken.IsCancellationRequested)
        {
            _logger.LogWarning("Service stop timed out after {Timeout}s", 
                _shutdownTimeout.TotalSeconds);
            throw;
        }
    }
    
    private async Task DoWorkAsync(CancellationToken ct)
    {
        _logger.LogInformation("Doing work");
        await Task.Delay(100, ct).ConfigureAwait(false);
    }
}
```

**Common mistakes**

1. **Not draining queues**
   ```csharp
   // ❌ WRONG: Items lost on shutdown
   protected override async Task ExecuteAsync(CancellationToken stoppingToken)
   {
       await foreach (var item in _reader.ReadAllAsync(stoppingToken))
       {
           await ProcessItemAsync(item, stoppingToken);
       }
       // If shutdown happens, remaining items are lost
   }
   
   // ✅ CORRECT: Drain remaining items
   ```

2. **Infinite shutdown wait**
   ```csharp
   // ❌ WRONG: May wait forever
   await base.StopAsync(CancellationToken.None);
   
   // ✅ CORRECT: Use timeout
   using var cts = CancellationTokenSource.CreateLinkedTokenSource(cancellationToken);
   cts.CancelAfter(TimeSpan.FromSeconds(30));
   await base.StopAsync(cts.Token);
   ```

3. **Not completing channel writer**
   ```csharp
   // ⚠️ CAUTION: If writer is never completed, ReadAllAsync never ends
   // Complete on shutdown:
   _lifetime.ApplicationStopping.Register(() => _writer.Complete());
   ```

**Performance notes**

- **Shutdown timeout**: Set reasonable timeouts (e.g., 30 seconds) to balance between completing work and responding to shutdown requests promptly.
- **Draining overhead**: Draining queues adds latency to shutdown. Consider whether all items must be processed or if some can be dropped/saved.

**Good to know**

- ASP.NET Core host waits for all `IHostedService` instances to stop (with a default timeout of 5 seconds, configurable).
- Use `IHostApplicationLifetime.ApplicationStopping` to be notified when shutdown is requested.
- For critical work, consider persisting queue items to durable storage (database, message queue) for recovery after restart.
