### 1. **Simple producer-consumer with BlockingCollection**

**When to use BlockingCollection**

- **CPU-bound work**: Simple producer-consumer scenarios where blocking threads is acceptable.
- **Legacy code**: Existing codebases already using `BlockingCollection`.
- **Minimal dependencies**: When you want to avoid additional NuGet packages.

**Basic pattern**

```csharp
using System;
using System.Collections.Concurrent;
using System.Threading;
using System.Threading.Tasks;

public class SimpleProducerConsumer
{
    private readonly BlockingCollection<WorkItem> _queue;
    private readonly CancellationTokenSource _cts;
    
    public SimpleProducerConsumer(int capacity = 100)
    {
        _queue = new BlockingCollection<WorkItem>(capacity);
        _cts = new CancellationTokenSource();
    }
    
    // Producer
    public void Produce(WorkItem item)
    {
        _queue.Add(item, _cts.Token);
    }
    
    // Consumer (run on background thread)
    public void StartConsuming()
    {
        Task.Run(() =>
        {
            foreach (var item in _queue.GetConsumingEnumerable(_cts.Token))
            {
                ProcessItem(item);
            }
        }, _cts.Token);
    }
    
    public void Stop()
    {
        _queue.CompleteAdding();
        _cts.Cancel();
    }
    
    private void ProcessItem(WorkItem item)
    {
        Console.WriteLine($"Processing item {item.Id}");
        // CPU-bound work
        Thread.Sleep(100);
    }
}

public record WorkItem(int Id, string Data);
```

**Pattern with multiple consumers**

```csharp
using System;
using System.Collections.Concurrent;
using System.Linq;
using System.Threading;
using System.Threading.Tasks;

public class MultiConsumerProducerConsumer
{
    private readonly BlockingCollection<WorkItem> _queue;
    private readonly CancellationTokenSource _cts;
    private readonly int _consumerCount;
    
    public MultiConsumerProducerConsumer(int capacity = 100, int consumerCount = 4)
    {
        _queue = new BlockingCollection<WorkItem>(capacity);
        _cts = new CancellationTokenSource();
        _consumerCount = consumerCount;
    }
    
    public void Produce(WorkItem item)
    {
        _queue.Add(item, _cts.Token);
    }
    
    public void StartConsuming()
    {
        var consumers = Enumerable.Range(0, _consumerCount)
            .Select(i => Task.Run(() => Consume(i), _cts.Token))
            .ToArray();
        
        // Wait for all consumers (optional)
        Task.WaitAll(consumers);
    }
    
    private void Consume(int consumerId)
    {
        foreach (var item in _queue.GetConsumingEnumerable(_cts.Token))
        {
            try
            {
                ProcessItem(item, consumerId);
            }
            catch (Exception ex)
            {
                Console.WriteLine($"Consumer {consumerId} error: {ex.Message}");
            }
        }
        
        Console.WriteLine($"Consumer {consumerId} stopped");
    }
    
    private void ProcessItem(WorkItem item, int consumerId)
    {
        Console.WriteLine($"Consumer {consumerId} processing item {item.Id}");
        Thread.Sleep(100);
    }
    
    public void Stop()
    {
        _queue.CompleteAdding();
        _cts.Cancel();
    }
}
```

**Pattern with error handling**

```csharp
using System;
using System.Collections.Concurrent;
using System.Threading;
using System.Threading.Tasks;

public class RobustProducerConsumer
{
    private readonly BlockingCollection<WorkItem> _queue;
    private readonly BlockingCollection<WorkItem> _deadLetterQueue;
    private readonly CancellationTokenSource _cts;
    private readonly int _maxRetries;
    
    public RobustProducerConsumer(int capacity = 100, int maxRetries = 3)
    {
        _queue = new BlockingCollection<WorkItem>(capacity);
        _deadLetterQueue = new BlockingCollection<WorkItem>(capacity);
        _cts = new CancellationTokenSource();
        _maxRetries = maxRetries;
    }
    
    public void Produce(WorkItem item)
    {
        _queue.Add(item, _cts.Token);
    }
    
    public void StartConsuming()
    {
        Task.Run(() =>
        {
            foreach (var item in _queue.GetConsumingEnumerable(_cts.Token))
            {
                int retryCount = 0;
                bool success = false;
                
                while (retryCount < _maxRetries && !success)
                {
                    try
                    {
                        ProcessItem(item);
                        success = true;
                    }
                    catch (Exception ex)
                    {
                        retryCount++;
                        Console.WriteLine($"Error processing item {item.Id}, retry {retryCount}: {ex.Message}");
                        
                        if (retryCount >= _maxRetries)
                        {
                            // Move to dead letter queue
                            _deadLetterQueue.Add(item, _cts.Token);
                        }
                        else
                        {
                            Thread.Sleep(1000 * retryCount); // Exponential backoff
                        }
                    }
                }
            }
        }, _cts.Token);
    }
    
    private void ProcessItem(WorkItem item)
    {
        // May throw exceptions
        if (item.Id % 10 == 0)
            throw new InvalidOperationException("Simulated error");
        
        Console.WriteLine($"Processing item {item.Id}");
        Thread.Sleep(100);
    }
    
    public void Stop()
    {
        _queue.CompleteAdding();
        _deadLetterQueue.CompleteAdding();
        _cts.Cancel();
    }
}
```

**Common mistakes**

1. **Not calling CompleteAdding**
   ```csharp
   // ❌ WRONG: GetConsumingEnumerable never completes
   foreach (var item in _queue.GetConsumingEnumerable())
   {
       // Infinite loop if CompleteAdding is never called
   }
   
   // ✅ CORRECT: Call CompleteAdding when done
   _queue.CompleteAdding();
   ```

2. **Blocking on UI thread**
   ```csharp
   // ❌ WRONG: Blocks UI thread
   foreach (var item in _queue.GetConsumingEnumerable())
   {
       ProcessItem(item);
   }
   
   // ✅ CORRECT: Run on background thread
   Task.Run(() =>
   {
       foreach (var item in _queue.GetConsumingEnumerable())
       {
           ProcessItem(item);
       }
   });
   ```

3. **Not handling cancellation**
   ```csharp
   // ❌ WRONG: No cancellation support
   _queue.Add(item);
   
   // ✅ CORRECT: Pass cancellation token
   _queue.Add(item, cancellationToken);
   ```

**Performance notes**

- **BlockingCollection overhead**: Minimal overhead, but blocks threads. For async code, prefer `Channel<T>` (see next section).
- **Bounded capacity**: Set a capacity to prevent unbounded memory growth. Producers will block when the queue is full.
- **Thread blocking**: `BlockingCollection` blocks threads. This is fine for CPU-bound work but can contribute to Thread Pool starvation if misused.

**Good to know**

- `BlockingCollection` is thread-safe and designed for producer-consumer scenarios.
- For async code, prefer `Channel<T>` (see **3.3** and next section).
- `GetConsumingEnumerable` automatically handles `CompleteAdding` and cancellation.

### 2. **Bounded channels for backpressure and load shedding**

**Why bounded channels**

- **Backpressure**: Producers slow down when consumers can't keep up.
- **Memory protection**: Prevents unbounded queue growth.
- **Async-friendly**: Non-blocking async operations.

**Basic bounded channel pattern**

```csharp
using System;
using System.Threading;
using System.Threading.Channels;
using System.Threading.Tasks;

public class BoundedChannelProducerConsumer
{
    private readonly Channel<WorkItem> _channel;
    private readonly CancellationTokenSource _cts;
    
    public BoundedChannelProducerConsumer(int capacity = 100)
    {
        var options = new BoundedChannelOptions(capacity)
        {
            FullMode = BoundedChannelFullMode.Wait, // Producer waits when full
            SingleReader = false,
            SingleWriter = false
        };
        _channel = Channel.CreateBounded<WorkItem>(options);
        _cts = new CancellationTokenSource();
    }
    
    // Producer
    public async ValueTask ProduceAsync(WorkItem item, CancellationToken ct = default)
    {
        await _channel.Writer.WriteAsync(item, ct).ConfigureAwait(false);
    }
    
    // Consumer
    public async Task StartConsumingAsync()
    {
        await foreach (var item in _channel.Reader.ReadAllAsync(_cts.Token)
            .ConfigureAwait(false))
        {
            await ProcessItemAsync(item, _cts.Token).ConfigureAwait(false);
        }
    }
    
    public void Complete()
    {
        _channel.Writer.Complete();
        _cts.Cancel();
    }
    
    private async Task ProcessItemAsync(WorkItem item, CancellationToken ct)
    {
        Console.WriteLine($"Processing item {item.Id}");
        await Task.Delay(100, ct).ConfigureAwait(false);
    }
}
```

**Pattern with load shedding (drop oldest)**

```csharp
using System;
using System.Threading;
using System.Threading.Channels;
using System.Threading.Tasks;

public class LoadSheddingChannel
{
    private readonly Channel<WorkItem> _channel;
    
    public LoadSheddingChannel(int capacity = 1000)
    {
        // Drop oldest item when full (useful for telemetry/logging)
        var options = new BoundedChannelOptions(capacity)
        {
            FullMode = BoundedChannelFullMode.DropOldest,
            SingleReader = false,
            SingleWriter = false
        };
        _channel = Channel.CreateBounded<WorkItem>(options);
    }
    
    public ChannelWriter<WorkItem> Writer => _channel.Writer;
    public ChannelReader<WorkItem> Reader => _channel.Reader;
    
    // Producer: never blocks, but may drop items
    public async ValueTask TryProduceAsync(WorkItem item, CancellationToken ct = default)
    {
        if (_channel.Writer.TryWrite(item))
        {
            return; // Successfully written
        }
        
        // Channel is full, item was dropped (DropOldest mode)
        // Optionally log or handle the drop
        Console.WriteLine($"Dropped item {item.Id} (channel full)");
    }
}
```

**Pattern with multiple consumers**

```csharp
using System;
using System.Linq;
using System.Threading;
using System.Threading.Channels;
using System.Threading.Tasks;

public class MultiConsumerChannel
{
    private readonly ChannelReader<WorkItem> _reader;
    private readonly int _consumerCount;
    private readonly CancellationTokenSource _cts;
    
    public MultiConsumerChannel(
        ChannelReader<WorkItem> reader,
        int consumerCount = 4)
    {
        _reader = reader;
        _consumerCount = consumerCount;
        _cts = new CancellationTokenSource();
    }
    
    public async Task StartConsumingAsync()
    {
        var consumers = Enumerable.Range(0, _consumerCount)
            .Select(i => ConsumeAsync(i, _cts.Token))
            .ToArray();
        
        await Task.WhenAll(consumers).ConfigureAwait(false);
    }
    
    private async Task ConsumeAsync(int consumerId, CancellationToken ct)
    {
        await foreach (var item in _reader.ReadAllAsync(ct).ConfigureAwait(false))
        {
            try
            {
                await ProcessItemAsync(item, consumerId, ct).ConfigureAwait(false);
            }
            catch (Exception ex)
            {
                Console.WriteLine($"Consumer {consumerId} error: {ex.Message}");
            }
        }
        
        Console.WriteLine($"Consumer {consumerId} stopped");
    }
    
    private async Task ProcessItemAsync(
        WorkItem item,
        int consumerId,
        CancellationToken ct)
    {
        Console.WriteLine($"Consumer {consumerId} processing item {item.Id}");
        await Task.Delay(100, ct).ConfigureAwait(false);
    }
    
    public void Stop()
    {
        _cts.Cancel();
    }
}

// Usage:
var channel = Channel.CreateBounded<WorkItem>(100);
var multiConsumer = new MultiConsumerChannel(channel.Reader, consumerCount: 4);
_ = multiConsumer.StartConsumingAsync();

// Producers write to channel.Writer
await channel.Writer.WriteAsync(new WorkItem(1, "data"));
```

**Pattern with backpressure monitoring**

```csharp
using System;
using System.Threading;
using System.Threading.Channels;
using System.Threading.Tasks;

public class MonitoredChannel
{
    private readonly Channel<WorkItem> _channel;
    private long _producedCount;
    private long _consumedCount;
    private long _droppedCount;
    
    public MonitoredChannel(int capacity = 100)
    {
        var options = new BoundedChannelOptions(capacity)
        {
            FullMode = BoundedChannelFullMode.DropOldest
        };
        _channel = Channel.CreateBounded<WorkItem>(options);
    }
    
    public async ValueTask ProduceAsync(WorkItem item, CancellationToken ct = default)
    {
        if (_channel.Writer.TryWrite(item))
        {
            Interlocked.Increment(ref _producedCount);
        }
        else
        {
            Interlocked.Increment(ref _droppedCount);
            // Wait for space (if using Wait mode)
            await _channel.Writer.WriteAsync(item, ct).ConfigureAwait(false);
            Interlocked.Increment(ref _producedCount);
        }
    }
    
    public async Task ConsumeAsync(CancellationToken ct)
    {
        await foreach (var item in _channel.Reader.ReadAllAsync(ct)
            .ConfigureAwait(false))
        {
            await ProcessItemAsync(item, ct).ConfigureAwait(false);
            Interlocked.Increment(ref _consumedCount);
        }
    }
    
    public (long Produced, long Consumed, long Dropped) GetStats()
    {
        return (
            Interlocked.Read(ref _producedCount),
            Interlocked.Read(ref _consumedCount),
            Interlocked.Read(ref _droppedCount)
        );
    }
    
    private async Task ProcessItemAsync(WorkItem item, CancellationToken ct)
    {
        await Task.Delay(100, ct).ConfigureAwait(false);
    }
}
```

**Common mistakes**

1. **Unbounded channel causing memory issues**
   ```csharp
   // ❌ WRONG: Can grow indefinitely
   var channel = Channel.CreateUnbounded<WorkItem>();
   
   // ✅ CORRECT: Use bounded channel
   var channel = Channel.CreateBounded<WorkItem>(1000);
   ```

2. **Not completing the writer**
   ```csharp
   // ⚠️ CAUTION: ReadAllAsync never completes if writer is not completed
   // Complete when done:
   channel.Writer.Complete();
   ```

3. **Wrong FullMode for the scenario**
   ```csharp
   // ⚠️ CAUTION: Wait mode may cause producers to block indefinitely
   // Use DropOldest for telemetry/logging, Wait for critical data
   var options = new BoundedChannelOptions(100)
   {
       FullMode = BoundedChannelFullMode.DropOldest // or Wait
   };
   ```

**Performance notes**

- **Channel performance**: `Channel<T>` is highly optimized for async scenarios. It's more efficient than `BlockingCollection<T>` for async code.
- **Backpressure**: Bounded channels with `Wait` mode provide natural backpressure. Producers automatically slow down when the queue is full.
- **Load shedding**: `DropOldest` / `DropNewest` modes prevent memory growth but lose data. Use only for non-critical data (telemetry, UI updates).

**Good to know**

- Use `SingleReader = true` / `SingleWriter = true` when applicable to enable optimizations.
- Channels integrate seamlessly with `IAsyncEnumerable<T>` via `ReadAllAsync`.
- For CPU-bound work, `BlockingCollection` may be simpler. For async I/O, prefer `Channel<T>`.

### 3. **Multi-stage pipeline with TPL Dataflow (TransformBlock, ActionBlock)**

**What TPL Dataflow provides**

- **Composable blocks**: Transform, filter, batch, aggregate.
- **Built-in parallelism**: Each block can process items in parallel.
- **Bounded capacity**: Automatic backpressure between stages.
- **Completion propagation**: Blocks complete when upstream completes.

**Basic pipeline pattern**

```csharp
// Requires NuGet: System.Threading.Tasks.Dataflow
using System;
using System.Threading.Tasks;
using System.Threading.Tasks.Dataflow;

public class BasicDataflowPipeline
{
    public static async Task RunAsync()
    {
        // Stage 1: Transform
        var transformBlock = new TransformBlock<int, string>(
            input => $"Processed: {input}",
            new ExecutionDataflowBlockOptions
            {
                MaxDegreeOfParallelism = 4 // Process 4 items concurrently
            });
        
        // Stage 2: Action (sink)
        var actionBlock = new ActionBlock<string>(
            result => Console.WriteLine(result),
            new ExecutionDataflowBlockOptions
            {
                MaxDegreeOfParallelism = 2
            });
        
        // Link stages
        transformBlock.LinkTo(actionBlock, new DataflowLinkOptions
        {
            PropagateCompletion = true // Complete actionBlock when transformBlock completes
        });
        
        // Send data
        for (int i = 0; i < 10; i++)
        {
            await transformBlock.SendAsync(i).ConfigureAwait(false);
        }
        
        // Signal completion
        transformBlock.Complete();
        
        // Wait for pipeline to finish
        await actionBlock.Completion.ConfigureAwait(false);
    }
}
```

**Multi-stage pipeline with filtering**

```csharp
using System;
using System.Threading.Tasks;
using System.Threading.Tasks.Dataflow;

public class MultiStagePipeline
{
    public static async Task RunAsync()
    {
        // Stage 1: Transform
        var transformBlock = new TransformBlock<int, ProcessedItem>(
            input => new ProcessedItem(input, input * 2),
            new ExecutionDataflowBlockOptions { MaxDegreeOfParallelism = 4 });
        
        // Stage 2: Filter
        var filterBlock = new TransformBlock<ProcessedItem, ProcessedItem>(
            item => item,
            new ExecutionDataflowBlockOptions { MaxDegreeOfParallelism = 2 });
        
        // Stage 3: Batch (group items)
        var batchBlock = new BatchBlock<ProcessedItem>(batchSize: 10);
        
        // Stage 4: Aggregate
        var aggregateBlock = new TransformBlock<ProcessedItem[], string>(
            batch =>
            {
                var sum = 0;
                foreach (var item in batch)
                {
                    if (item.Value > 10) // Filter condition
                        sum += item.Value;
                }
                return $"Sum: {sum}";
            },
            new ExecutionDataflowBlockOptions { MaxDegreeOfParallelism = 1 });
        
        // Stage 5: Output
        var outputBlock = new ActionBlock<string>(
            result => Console.WriteLine(result));
        
        // Link pipeline
        transformBlock.LinkTo(filterBlock);
        filterBlock.LinkTo(batchBlock);
        batchBlock.LinkTo(aggregateBlock);
        aggregateBlock.LinkTo(outputBlock);
        
        // Propagate completion
        transformBlock.LinkTo(DataflowBlock.NullTarget<ProcessedItem>()); // Discard filtered items
        filterBlock.LinkTo(batchBlock, new DataflowLinkOptions
        {
            PropagateCompletion = true
        });
        batchBlock.LinkTo(aggregateBlock, new DataflowLinkOptions
        {
            PropagateCompletion = true
        });
        aggregateBlock.LinkTo(outputBlock, new DataflowLinkOptions
        {
            PropagateCompletion = true
        });
        
        // Send data
        for (int i = 0; i < 100; i++)
        {
            await transformBlock.SendAsync(i).ConfigureAwait(false);
        }
        
        transformBlock.Complete();
        await outputBlock.Completion.ConfigureAwait(false);
    }
}

public record ProcessedItem(int Id, int Value);
```

**Fan-out/fan-in pattern**

```csharp
using System;
using System.Linq;
using System.Threading.Tasks;
using System.Threading.Tasks.Dataflow;

public class FanOutFanInPipeline
{
    public static async Task RunAsync()
    {
        // Source
        var sourceBlock = new BufferBlock<int>();
        
        // Fan-out: Multiple transform blocks
        var transform1 = new TransformBlock<int, string>(
            x => $"T1: {x}",
            new ExecutionDataflowBlockOptions { MaxDegreeOfParallelism = 2 });
        
        var transform2 = new TransformBlock<int, string>(
            x => $"T2: {x}",
            new ExecutionDataflowBlockOptions { MaxDegreeOfParallelism = 2 });
        
        // Fan-in: Join results
        var joinBlock = new JoinBlock<string, string>();
        
        var outputBlock = new ActionBlock<Tuple<string, string>>(
            tuple => Console.WriteLine($"{tuple.Item1}, {tuple.Item2}"));
        
        // Link: Fan-out
        sourceBlock.LinkTo(transform1);
        sourceBlock.LinkTo(transform2);
        
        // Link: Fan-in
        transform1.LinkTo(joinBlock.Target1);
        transform2.LinkTo(joinBlock.Target2);
        joinBlock.LinkTo(outputBlock);
        
        // Propagate completion
        transform1.LinkTo(DataflowBlock.NullTarget<string>());
        transform2.LinkTo(DataflowBlock.NullTarget<string>());
        
        // Send data
        for (int i = 0; i < 10; i++)
        {
            await sourceBlock.SendAsync(i).ConfigureAwait(false);
        }
        
        sourceBlock.Complete();
        await outputBlock.Completion.ConfigureAwait(false);
    }
}
```

**Common mistakes**

1. **Not propagating completion**
   ```csharp
   // ❌ WRONG: Downstream blocks never complete
   transformBlock.LinkTo(actionBlock);
   transformBlock.Complete();
   await actionBlock.Completion; // Waits forever
   
   // ✅ CORRECT: Propagate completion
   transformBlock.LinkTo(actionBlock, new DataflowLinkOptions
   {
       PropagateCompletion = true
   });
   ```

2. **Not handling completion of all blocks**
   ```csharp
   // ⚠️ CAUTION: If a block has multiple outputs, complete all paths
   transformBlock.LinkTo(actionBlock1);
   transformBlock.LinkTo(actionBlock2);
   transformBlock.Complete();
   
   // Wait for all consumers
   await Task.WhenAll(actionBlock1.Completion, actionBlock2.Completion);
   ```

3. **Unbounded capacity causing memory issues**
   ```csharp
   // ❌ WRONG: No capacity limit
   var block = new ActionBlock<int>(x => Process(x));
   
   // ✅ CORRECT: Set bounded capacity
   var block = new ActionBlock<int>(
       x => Process(x),
       new ExecutionDataflowBlockOptions
       {
           BoundedCapacity = 100
       });
   ```

**Performance notes**

- **Dataflow overhead**: TPL Dataflow has some overhead compared to manual pipelines. Use when the composability benefits outweigh the cost.
- **Parallelism**: Set `MaxDegreeOfParallelism` based on the nature of work (CPU-bound vs I/O-bound).
- **Bounded capacity**: Use `BoundedCapacity` to prevent memory growth and provide backpressure.

**Good to know**

- TPL Dataflow is excellent for complex multi-stage pipelines with transformations, filtering, and batching.
- For simpler scenarios, `Channel<T>` may be sufficient.
- Dataflow blocks are composable and reusable, making it easy to build complex processing pipelines.

### 4. **Combining CPU-bound and I/O-bound stages in a pipeline**

**The challenge**

- CPU-bound stages benefit from parallelism (`MaxDegreeOfParallelism`).
- I/O-bound stages benefit from high concurrency (many async operations).
- Mixing them requires careful coordination.

**Pattern: CPU-bound transform followed by I/O-bound action**

```csharp
using System;
using System.Net.Http;
using System.Threading.Tasks;
using System.Threading.Tasks.Dataflow;

public class MixedPipeline
{
    public static async Task RunAsync()
    {
        using var http = new HttpClient();
        
        // Stage 1: CPU-bound transform (parallel)
        var cpuTransformBlock = new TransformBlock<int, ProcessedData>(
            input =>
            {
                // CPU-bound work
                var result = HeavyComputation(input);
                return new ProcessedData(input, result);
            },
            new ExecutionDataflowBlockOptions
            {
                MaxDegreeOfParallelism = Environment.ProcessorCount // Use all cores
            });
        
        // Stage 2: I/O-bound action (high concurrency)
        var ioActionBlock = new ActionBlock<ProcessedData>(
            async data =>
            {
                // I/O-bound work
                await SendToApiAsync(http, data).ConfigureAwait(false);
            },
            new ExecutionDataflowBlockOptions
            {
                MaxDegreeOfParallelism = DataflowBlockOptions.Unbounded, // Unlimited concurrency
                BoundedCapacity = 1000 // But limit buffering
            });
        
        // Link stages
        cpuTransformBlock.LinkTo(ioActionBlock, new DataflowLinkOptions
        {
            PropagateCompletion = true
        });
        
        // Send data
        for (int i = 0; i < 1000; i++)
        {
            await cpuTransformBlock.SendAsync(i).ConfigureAwait(false);
        }
        
        cpuTransformBlock.Complete();
        await ioActionBlock.Completion.ConfigureAwait(false);
    }
    
    private static int HeavyComputation(int input)
    {
        // CPU-bound work
        int sum = 0;
        for (int i = 0; i < 1000; i++)
        {
            sum += input * i;
        }
        return sum;
    }
    
    private static async Task SendToApiAsync(HttpClient http, ProcessedData data)
    {
        // I/O-bound work
        await http.PostAsync("https://api.example.com/data", 
            new StringContent(data.ToString())).ConfigureAwait(false);
    }
}

public record ProcessedData(int Id, int Value);
```

**Pattern: I/O-bound fetch followed by CPU-bound processing**

```csharp
using System;
using System.Net.Http;
using System.Threading.Tasks;
using System.Threading.Tasks.Dataflow;

public class IOThenCPUPipeline
{
    public static async Task RunAsync()
    {
        using var http = new HttpClient();
        
        // Stage 1: I/O-bound fetch (high concurrency)
        var fetchBlock = new TransformBlock<int, string>(
            async id =>
            {
                // I/O-bound: Fetch data
                var response = await http.GetStringAsync($"https://api.example.com/{id}")
                    .ConfigureAwait(false);
                return response;
            },
            new ExecutionDataflowBlockOptions
            {
                MaxDegreeOfParallelism = DataflowBlockOptions.Unbounded,
                BoundedCapacity = 500
            });
        
        // Stage 2: CPU-bound processing (parallel)
        var processBlock = new TransformBlock<string, ProcessedResult>(
            data =>
            {
                // CPU-bound: Process data
                return ProcessData(data);
            },
            new ExecutionDataflowBlockOptions
            {
                MaxDegreeOfParallelism = Environment.ProcessorCount,
                BoundedCapacity = 100
            });
        
        // Stage 3: I/O-bound save (high concurrency)
        var saveBlock = new ActionBlock<ProcessedResult>(
            async result =>
            {
                // I/O-bound: Save result
                await SaveResultAsync(result).ConfigureAwait(false);
            },
            new ExecutionDataflowBlockOptions
            {
                MaxDegreeOfParallelism = DataflowBlockOptions.Unbounded,
                BoundedCapacity = 200
            });
        
        // Link pipeline
        fetchBlock.LinkTo(processBlock, new DataflowLinkOptions
        {
            PropagateCompletion = true
        });
        processBlock.LinkTo(saveBlock, new DataflowLinkOptions
        {
            PropagateCompletion = true
        });
        
        // Send IDs to process
        for (int i = 0; i < 1000; i++)
        {
            await fetchBlock.SendAsync(i).ConfigureAwait(false);
        }
        
        fetchBlock.Complete();
        await saveBlock.Completion.ConfigureAwait(false);
    }
    
    private static ProcessedResult ProcessData(string data)
    {
        // CPU-bound processing
        return new ProcessedResult(data.Length, data.GetHashCode());
    }
    
    private static async Task SaveResultAsync(ProcessedResult result)
    {
        // I/O-bound save
        await Task.Delay(10).ConfigureAwait(false);
    }
}

public record ProcessedResult(int Length, int Hash);
```

**Pattern: Throttling I/O stage**

```csharp
using System;
using System.Net.Http;
using System.Threading;
using System.Threading.Tasks;
using System.Threading.Tasks.Dataflow;

public class ThrottledIOPipeline
{
    public static async Task RunAsync()
    {
        using var http = new HttpClient();
        using var semaphore = new SemaphoreSlim(10); // Max 10 concurrent I/O operations
        
        // CPU-bound stage
        var cpuBlock = new TransformBlock<int, int>(
            input => HeavyComputation(input),
            new ExecutionDataflowBlockOptions
            {
                MaxDegreeOfParallelism = Environment.ProcessorCount
            });
        
        // I/O-bound stage with throttling
        var ioBlock = new ActionBlock<int>(
            async id =>
            {
                await semaphore.WaitAsync().ConfigureAwait(false);
                try
                {
                    await SendToApiAsync(http, id).ConfigureAwait(false);
                }
                finally
                {
                    semaphore.Release();
                }
            },
            new ExecutionDataflowBlockOptions
            {
                MaxDegreeOfParallelism = DataflowBlockOptions.Unbounded,
                BoundedCapacity = 100
            });
        
        cpuBlock.LinkTo(ioBlock, new DataflowLinkOptions
        {
            PropagateCompletion = true
        });
        
        for (int i = 0; i < 1000; i++)
        {
            await cpuBlock.SendAsync(i).ConfigureAwait(false);
        }
        
        cpuBlock.Complete();
        await ioBlock.Completion.ConfigureAwait(false);
    }
    
    private static int HeavyComputation(int input) => input * input;
    
    private static async Task SendToApiAsync(HttpClient http, int id)
    {
        await http.GetStringAsync($"https://api.example.com/{id}").ConfigureAwait(false);
    }
}
```

**Common mistakes**

1. **Using CPU parallelism for I/O**
   ```csharp
   // ❌ WRONG: MaxDegreeOfParallelism doesn't help with I/O
   var ioBlock = new ActionBlock<int>(
       async id => await FetchAsync(id),
       new ExecutionDataflowBlockOptions
       {
           MaxDegreeOfParallelism = 4 // Doesn't help with async I/O
       });
   
   // ✅ CORRECT: Use Unbounded for I/O, throttle externally if needed
   var ioBlock = new ActionBlock<int>(
       async id => await FetchAsync(id),
       new ExecutionDataflowBlockOptions
       {
           MaxDegreeOfParallelism = DataflowBlockOptions.Unbounded
       });
   ```

2. **Not setting BoundedCapacity**
   ```csharp
   // ❌ WRONG: Unbounded buffering between stages
   var block = new ActionBlock<int>(x => Process(x));
   
   // ✅ CORRECT: Set bounded capacity for backpressure
   var block = new ActionBlock<int>(
       x => Process(x),
       new ExecutionDataflowBlockOptions { BoundedCapacity = 100 });
   ```

**Performance notes**

- **CPU-bound stages**: Set `MaxDegreeOfParallelism` to `Environment.ProcessorCount` or lower.
- **I/O-bound stages**: Use `DataflowBlockOptions.Unbounded` for `MaxDegreeOfParallelism`, but throttle externally with `SemaphoreSlim` if needed.
- **BoundedCapacity**: Set appropriate capacity to prevent memory growth while allowing pipelining.

**Good to know**

- Mixing CPU-bound and I/O-bound stages is common in real-world pipelines.
- Use `BoundedCapacity` to control memory usage and provide backpressure between stages.
- Consider using `Channel<T>` for simpler scenarios or TPL Dataflow for complex multi-stage pipelines.

### 5. **Error handling and retries inside pipeline stages**

**Why error handling matters**

- **Resilience**: Pipeline should continue processing even if some items fail.
- **Observability**: Log errors for debugging and monitoring.
- **Data integrity**: Decide whether to retry, skip, or send to dead letter queue.

**Pattern: Error handling with dead letter queue**

```csharp
using System;
using System.Threading.Tasks;
using System.Threading.Tasks.Dataflow;

public class ErrorHandlingPipeline
{
    public static async Task RunAsync()
    {
        // Main processing block
        var processBlock = new TransformBlock<int, ProcessedItem>(
            input =>
            {
                if (input % 10 == 0)
                    throw new InvalidOperationException($"Error processing {input}");
                
                return new ProcessedItem(input, input * 2);
            },
            new ExecutionDataflowBlockOptions
            {
                MaxDegreeOfParallelism = 4
            });
        
        // Success path
        var successBlock = new ActionBlock<ProcessedItem>(
            item => Console.WriteLine($"Success: {item.Id}"));
        
        // Dead letter queue
        var deadLetterBlock = new ActionBlock<int>(
            item => Console.WriteLine($"Dead letter: {item}"));
        
        // Link: Success path
        processBlock.LinkTo(successBlock, new DataflowLinkOptions
        {
            PropagateCompletion = true
        });
        
        // Link: Error path (items that cause exceptions)
        // Note: Dataflow doesn't have built-in error routing, so we need to catch in the block
        var errorHandlingBlock = new TransformBlock<int, (bool Success, ProcessedItem? Item, int? FailedId)>(
            input =>
            {
                try
                {
                    if (input % 10 == 0)
                        throw new InvalidOperationException($"Error processing {input}");
                    
                    return (Success: true, Item: new ProcessedItem(input, input * 2), FailedId: (int?)null);
                }
                catch
                {
                    return (Success: false, Item: null, FailedId: input);
                }
            });
        
        var successRouter = new ActionBlock<(bool Success, ProcessedItem? Item, int? FailedId)>(
            tuple =>
            {
                if (tuple.Success && tuple.Item != null)
                {
                    Console.WriteLine($"Success: {tuple.Item.Id}");
                }
                else if (tuple.FailedId.HasValue)
                {
                    Console.WriteLine($"Dead letter: {tuple.FailedId.Value}");
                }
            });
        
        errorHandlingBlock.LinkTo(successRouter, new DataflowLinkOptions
        {
            PropagateCompletion = true
        });
        
        // Send data
        for (int i = 0; i < 100; i++)
        {
            await errorHandlingBlock.SendAsync(i).ConfigureAwait(false);
        }
        
        errorHandlingBlock.Complete();
        await successRouter.Completion.ConfigureAwait(false);
    }
}

public record ProcessedItem(int Id, int Value);
```

**Pattern: Retry with exponential backoff**

```csharp
using System;
using System.Threading;
using System.Threading.Tasks;
using System.Threading.Tasks.Dataflow;

public class RetryPipeline
{
    public static async Task RunAsync()
    {
        var processBlock = new TransformBlock<int, ProcessedItem>(
            async input =>
            {
                return await ProcessWithRetryAsync(input, maxRetries: 3)
                    .ConfigureAwait(false);
            },
            new ExecutionDataflowBlockOptions
            {
                MaxDegreeOfParallelism = DataflowBlockOptions.Unbounded
            });
        
        var outputBlock = new ActionBlock<ProcessedItem>(
            item => Console.WriteLine($"Processed: {item.Id}"));
        
        processBlock.LinkTo(outputBlock, new DataflowLinkOptions
        {
            PropagateCompletion = true
        });
        
        for (int i = 0; i < 100; i++)
        {
            await processBlock.SendAsync(i).ConfigureAwait(false);
        }
        
        processBlock.Complete();
        await outputBlock.Completion.ConfigureAwait(false);
    }
    
    private static async Task<ProcessedItem> ProcessWithRetryAsync(
        int input,
        int maxRetries)
    {
        Exception? lastException = null;
        
        for (int attempt = 0; attempt <= maxRetries; attempt++)
        {
            try
            {
                return await ProcessItemAsync(input).ConfigureAwait(false);
            }
            catch (Exception ex) when (IsTransientError(ex))
            {
                lastException = ex;
                if (attempt < maxRetries)
                {
                    var delay = TimeSpan.FromMilliseconds(100 * Math.Pow(2, attempt));
                    await Task.Delay(delay).ConfigureAwait(false);
                }
            }
        }
        
        throw new InvalidOperationException(
            $"Failed after {maxRetries + 1} attempts", lastException);
    }
    
    private static async Task<ProcessedItem> ProcessItemAsync(int input)
    {
        // Simulate transient failures
        if (input % 5 == 0 && Random.Shared.NextDouble() < 0.5)
        {
            throw new InvalidOperationException("Transient error");
        }
        
        await Task.Delay(10).ConfigureAwait(false);
        return new ProcessedItem(input, input * 2);
    }
    
    private static bool IsTransientError(Exception ex) => true;
}
```

**Pattern: Circuit breaker in pipeline**

```csharp
using System;
using System.Threading;
using System.Threading.Tasks;
using System.Threading.Tasks.Dataflow;

public class CircuitBreakerPipeline
{
    private class SimpleCircuitBreaker
    {
        private int _failureCount;
        private DateTime _lastFailureTime;
        private bool _isOpen;
        private readonly object _lock = new();
        private readonly int _threshold;
        private readonly TimeSpan _openDuration;
        
        public SimpleCircuitBreaker(int threshold = 5, TimeSpan? openDuration = null)
        {
            _threshold = threshold;
            _openDuration = openDuration ?? TimeSpan.FromSeconds(60);
        }
        
        public bool IsOpen()
        {
            lock (_lock)
            {
                if (!_isOpen)
                    return false;
                
                if (DateTime.UtcNow - _lastFailureTime > _openDuration)
                {
                    _isOpen = false;
                    _failureCount = 0;
                    return false;
                }
                
                return true;
            }
        }
        
        public void RecordSuccess()
        {
            lock (_lock)
            {
                _failureCount = 0;
                _isOpen = false;
            }
        }
        
        public void RecordFailure()
        {
            lock (_lock)
            {
                _failureCount++;
                _lastFailureTime = DateTime.UtcNow;
                if (_failureCount >= _threshold)
                {
                    _isOpen = true;
                }
            }
        }
    }
    
    public static async Task RunAsync()
    {
        var circuitBreaker = new SimpleCircuitBreaker(threshold: 5);
        
        var processBlock = new TransformBlock<int, ProcessedItem>(
            async input =>
            {
                if (circuitBreaker.IsOpen())
                {
                    throw new InvalidOperationException("Circuit breaker is open");
                }
                
                try
                {
                    var result = await ProcessItemAsync(input).ConfigureAwait(false);
                    circuitBreaker.RecordSuccess();
                    return result;
                }
                catch (Exception ex)
                {
                    circuitBreaker.RecordFailure();
                    throw;
                }
            },
            new ExecutionDataflowBlockOptions
            {
                MaxDegreeOfParallelism = DataflowBlockOptions.Unbounded
            });
        
        var outputBlock = new ActionBlock<ProcessedItem>(
            item => Console.WriteLine($"Processed: {item.Id}"));
        
        var errorBlock = new ActionBlock<int>(
            item => Console.WriteLine($"Error: {item}"));
        
        // Route success and errors
        processBlock.LinkTo(outputBlock);
        processBlock.LinkTo(DataflowBlock.NullTarget<ProcessedItem>()); // Discard on error
        
        for (int i = 0; i < 100; i++)
        {
            await processBlock.SendAsync(i).ConfigureAwait(false);
        }
        
        processBlock.Complete();
        await Task.WhenAll(outputBlock.Completion, errorBlock.Completion)
            .ConfigureAwait(false);
    }
    
    private static async Task<ProcessedItem> ProcessItemAsync(int input)
    {
        if (input % 10 == 0)
            throw new InvalidOperationException("Simulated error");
        
        await Task.Delay(10).ConfigureAwait(false);
        return new ProcessedItem(input, input * 2);
    }
}
```

**Common mistakes**

1. **Not handling exceptions**
   ```csharp
   // ❌ WRONG: Exceptions stop the pipeline
   var block = new TransformBlock<int, int>(input => Process(input));
   
   // ✅ CORRECT: Catch exceptions in the block
   var block = new TransformBlock<int, int>(input =>
   {
       try { return Process(input); }
       catch { return -1; } // Or handle appropriately
   });
   ```

2. **Retrying non-transient errors**
   ```csharp
   // ❌ WRONG: Retries everything
   catch (Exception ex)
   {
       // Retry logic
   }
   
   // ✅ CORRECT: Only retry transient errors
   catch (Exception ex) when (IsTransientError(ex))
   {
       // Retry logic
   }
   ```

**Performance notes**

- **Error handling overhead**: Try-catch has minimal overhead. The main cost is the retry delays.
- **Dead letter queues**: Consider using a separate channel or queue for failed items to avoid blocking the main pipeline.

**Good to know**

- Error handling in pipelines requires careful design. Decide whether to retry, skip, or route to a dead letter queue.
- Use circuit breakers to prevent cascading failures when downstream services are failing.
- Log errors for observability and debugging.
