### 1. **Consuming async streams (IAsyncEnumerable\<T\>) from I/O sources**

**Reading files line by line**

- Use `StreamReader.ReadLineAsync()` in an async iterator.

```csharp
using System;
using System.Collections.Generic;
using System.IO;
using System.Runtime.CompilerServices;
using System.Threading;
using System.Threading.Tasks;

public static class FileStreaming
{
    public static async IAsyncEnumerable<string> ReadLinesAsync(
        string filePath,
        [EnumeratorCancellation] CancellationToken cancellationToken = default)
    {
        using var reader = new StreamReader(filePath);
        string? line;
        
        while ((line = await reader.ReadLineAsync().ConfigureAwait(false)) != null
            && !cancellationToken.IsCancellationRequested)
        {
            yield return line;
        }
    }

    public static async Task ProcessLargeFileAsync(string filePath)
    {
        await foreach (var line in ReadLinesAsync(filePath).ConfigureAwait(false))
        {
            // Process each line without loading entire file into memory
            ProcessLine(line);
        }
    }

    private static void ProcessLine(string line)
    {
        Console.WriteLine($"Processed: {line}");
    }
}
```

**Reading from HTTP streams**

- Use `HttpClient` with `ReadAsStreamAsync()` and process chunks.

```csharp
using System;
using System.Collections.Generic;
using System.Net.Http;
using System.Runtime.CompilerServices;

            yield return line;
        }
    }

    public static async Task DownloadAndProcessAsync(string url)
    {
        await foreach (var line in ReadHttpLinesAsync(url).ConfigureAwait(false))
        {
            ProcessLine(line);
        }
    }

}
```

**Reading from database streams**

- Use `DbDataReader` to stream results without loading all into memory.

```csharp
using System;
using System.Collections.Generic;
using System.Data.Common;
using System.Runtime.CompilerServices;
using System.Threading;
using System.Threading.Tasks;

public static class DatabaseStreaming
{
    public static async IAsyncEnumerable<Record> ReadRecordsAsync(
        DbConnection connection,
        string query,
        [EnumeratorCancellation] CancellationToken cancellationToken = default)
    {
        await using var command = connection.CreateCommand();
        command.CommandText = query;
        
        await using var reader = await command.ExecuteReaderAsync(cancellationToken).ConfigureAwait(false);
        
        while (await reader.ReadAsync(cancellationToken).ConfigureAwait(false))
        {
            yield return new Record
            {
                Id = reader.GetInt32(0),
                Name = reader.GetString(1),
                Value = reader.GetDecimal(2)
            };
        }
    }

    public static async Task ProcessLargeQueryAsync(DbConnection connection, string query)
    {
        await foreach (var record in ReadRecordsAsync(connection, query).ConfigureAwait(false))
        {
            ProcessRecord(record);
        }
    }

    private static void ProcessRecord(Record record) { /* ... */ }
}

{

    public decimal Value { get; set; }
```

**Reading from sockets**

- Use `NetworkStream` with async read operations.

```csharp
using System;
using System.Collections.Generic;
using System.Net.Sockets;
using System.Runtime.CompilerServices;
using System.Text;
using System.Threading;
using System.Threading.Tasks;

public static class SocketStreaming
{
    public static async IAsyncEnumerable<string> ReadLinesAsync(
        NetworkStream stream,
        [EnumeratorCancellation] CancellationToken cancellationToken = default)
    {
        var buffer = new byte[4096];
        var lineBuilder = new StringBuilder();
        
        while (!cancellationToken.IsCancellationRequested)
        {
            var bytesRead = await stream.ReadAsync(buffer, 0, buffer.Length, cancellationToken)
                .ConfigureAwait(false);
            
            if (bytesRead == 0) break; // Connection closed
            
            var text = Encoding.UTF8.GetString(buffer, 0, bytesRead);
            
            foreach (var ch in text)
            {
                if (ch == '\n')
                {
                    yield return lineBuilder.ToString();
                    lineBuilder.Clear();
                }
                else if (ch != '\r')
                {
                    lineBuilder.Append(ch);
                }
            }
        }
        
        // Yield remaining line if any
        if (lineBuilder.Length > 0)
        {
            yield return lineBuilder.ToString();
        }
    }
}
```

**Mentor emphasis**

- **Use `ConfigureAwait(false)`** in library code to avoid context capture.
- **Handle cancellation** with `CancellationToken`.
- **Dispose resources** properly (use `using` statements).
- **Process incrementally** to avoid loading everything into memory.

### 2. **Transforming, filtering, and aggregating streaming data**

**Transforming streams**

- Create extension methods for common transformations.

```csharp
using System;
using System.Collections.Generic;
using System.Runtime.CompilerServices;
using System.Threading;
using System.Threading.Tasks;

public static class StreamTransformations
{
    public static async IAsyncEnumerable<TOut> SelectAsync<TIn, TOut>(
        this IAsyncEnumerable<TIn> source,
        Func<TIn, TOut> selector,
        [EnumeratorCancellation] CancellationToken cancellationToken = default)
    {
        await foreach (var item in source.WithCancellation(cancellationToken).ConfigureAwait(false))
        {
            yield return selector(item);
        }
    }

    public static async IAsyncEnumerable<T> WhereAsync<T>(
        this IAsyncEnumerable<T> source,
        Func<T, bool> predicate,
        [EnumeratorCancellation] CancellationToken cancellationToken = default)
    {
        await foreach (var item in source.WithCancellation(cancellationToken).ConfigureAwait(false))
        {
            if (predicate(item))
            {
                yield return item;
            }
        }
    }

    public static async IAsyncEnumerable<TOut> SelectManyAsync<TIn, TOut>(
        this IAsyncEnumerable<TIn> source,
        Func<TIn, IAsyncEnumerable<TOut>> selector,
        [EnumeratorCancellation] CancellationToken cancellationToken = default)
    {
        await foreach (var item in source.WithCancellation(cancellationToken).ConfigureAwait(false))
        {
            await foreach (var subItem in selector(item).WithCancellation(cancellationToken).ConfigureAwait(false))
            {
                yield return subItem;
            }
        }
    }
}

// Usage
public static class TransformationExamples
{
    public static async Task TransformExample()
    {
        var numbers = GetNumbersAsync();
        
        // Transform: multiply by 2
        var doubled = numbers.SelectAsync(x => x * 2);
        
        // Filter: only even numbers
        var evens = doubled.WhereAsync(x => x % 2 == 0);
        
        await foreach (var value in evens.ConfigureAwait(false))
        {
            Console.WriteLine(value);
        }
    }

    private static async IAsyncEnumerable<int> GetNumbersAsync()
    {
        for (int i = 0; i < 10; i++)
        {
            await Task.Delay(100);
            yield return i;
        }
    }
}
```

**Aggregating streams**

- Use `AggregateAsync` for reductions.

```csharp
public static class StreamAggregations
{
    public static async Task<TAccumulate> AggregateAsync<T, TAccumulate>(
        this IAsyncEnumerable<T> source,
        TAccumulate seed,
        Func<TAccumulate, T, TAccumulate> accumulator,
        CancellationToken cancellationToken = default)
    {
        var result = seed;
        await foreach (var item in source.WithCancellation(cancellationToken).ConfigureAwait(false))
        {
            result = accumulator(result, item);
        }
        return result;
    }

    public static async Task<int> SumAsync(
        this IAsyncEnumerable<int> source,
        CancellationToken cancellationToken = default)
    {
        return await source.AggregateAsync(0, (sum, item) => sum + item, cancellationToken);
    }

    public static async Task<List<T>> ToListAsync<T>(
        this IAsyncEnumerable<T> source,
        CancellationToken cancellationToken = default)
    {
        var list = new List<T>();
        await foreach (var item in source.WithCancellation(cancellationToken).ConfigureAwait(false))
        {
            list.Add(item);
        }
        return list;
    }
}

// Usage
public static class AggregationExamples
{
    public static async Task<int> SumNumbersAsync()
    {
        var numbers = GetNumbersAsync();
        return await numbers.SumAsync();
    }
}
```

**Batching streams**

- Group items into batches for batch processing.

```csharp
public static class StreamBatching
{
    public static async IAsyncEnumerable<List<T>> BatchAsync<T>(
        this IAsyncEnumerable<T> source,
        int batchSize,
        [EnumeratorCancellation] CancellationToken cancellationToken = default)
    {
        var batch = new List<T>(batchSize);
        
        await foreach (var item in source.WithCancellation(cancellationToken).ConfigureAwait(false))
        {
            batch.Add(item);
            
            if (batch.Count >= batchSize)
            {
                yield return batch;
                batch = new List<T>(batchSize);
            }
        }
        
        if (batch.Count > 0)
        {
            yield return batch;
        }
    }
}

// Usage
public static class BatchingExamples
{
    public static async Task ProcessBatchesAsync()
    {
        var items = GetItemsAsync();
        
        await foreach (var batch in items.BatchAsync(100).ConfigureAwait(false))
        {
            await ProcessBatchAsync(batch);
        }
    }

    private static async Task ProcessBatchAsync(List<Item> batch)
    {
        // Process batch
    }
}
```

**Mentor emphasis**

- **Create reusable extension methods** for common transformations.
- **Use `WithCancellation`** to propagate cancellation tokens.
- **Batch operations** for better throughput when processing many items.
- **Keep transformations pure** (no side effects) when possible.

### 3. **Bridging event sources to async streams and/or IObservable\<T\>**

**Converting events to IAsyncEnumerable**

- Use `TaskCompletionSource` to bridge events to async streams.

```csharp
using System;
using System.Collections.Generic;
using System.Runtime.CompilerServices;
using System.Threading;
using System.Threading.Tasks;

public static class EventToAsyncEnumerable
{
    public static IAsyncEnumerable<T> FromEvent<T>(
        Action<EventHandler<T>> subscribe,
        Action<EventHandler<T>> unsubscribe,
        CancellationToken cancellationToken = default)
    {
        return new EventAsyncEnumerable<T>(subscribe, unsubscribe, cancellationToken);
    }

    private class EventAsyncEnumerable<T> : IAsyncEnumerable<T>
    {
        private readonly Action<EventHandler<T>> _subscribe;
        private readonly Action<EventHandler<T>> _unsubscribe;
        private readonly CancellationToken _cancellationToken;

        public EventAsyncEnumerable(
            Action<EventHandler<T>> subscribe,
            Action<EventHandler<T>> unsubscribe,
            CancellationToken cancellationToken)
        {
            _subscribe = subscribe;
            _unsubscribe = unsubscribe;
            _cancellationToken = cancellationToken;
        }

        public async IAsyncEnumerator<T> GetAsyncEnumerator(CancellationToken cancellationToken = default)
        {
            using var cts = CancellationTokenSource.CreateLinkedTokenSource(_cancellationToken, cancellationToken);
            var channel = System.Threading.Channels.Channel.CreateUnbounded<T>();
            
            EventHandler<T> handler = (sender, args) =>
            {
                channel.Writer.TryWrite(args);
            };
            
            _subscribe(handler);
            
            try
            {
                await foreach (var item in channel.Reader.ReadAllAsync(cts.Token).ConfigureAwait(false))
                {
                    yield return item;
                }
            }
            finally
            {
                _unsubscribe(handler);
                channel.Writer.Complete();
            }
        }
    }
}

// Usage
public static class EventBridgeExamples
{
    public static async Task ConsumeEventsAsync()
    {
        var eventSource = new EventSource();
        
        var events = EventToAsyncEnumerable.FromEvent<EventArgs>(
            handler => eventSource.Event += handler,
            handler => eventSource.Event -= handler);
        
        await foreach (var evt in events.ConfigureAwait(false))
        {
            Console.WriteLine("Event received");
        }
    }
}

public class EventSource
{
    public event EventHandler<EventArgs>? Event;
    
    public void RaiseEvent()
    {
        Event?.Invoke(this, EventArgs.Empty);
    }
}
```

**Converting IAsyncEnumerable to IObservable**

- Use `System.Reactive` or implement a simple bridge.

```csharp
using System;
using System.Collections.Generic;
using System.Reactive;
using System.Reactive.Linq;
using System.Threading;
using System.Threading.Tasks;

public static class AsyncEnumerableToObservable
{
    public static IObservable<T> ToObservable<T>(
        this IAsyncEnumerable<T> source,
        CancellationToken cancellationToken = default)
    {
        return Observable.Create<T>(observer =>
        {
            var cts = CancellationTokenSource.CreateLinkedTokenSource(cancellationToken);
            var task = Task.Run(async () =>
            {
                try
                {
                    await foreach (var item in source.WithCancellation(cts.Token).ConfigureAwait(false))
                    {
                        observer.OnNext(item);
                    }
                    observer.OnCompleted();
                }
                catch (Exception ex)
                {
                    observer.OnError(ex);
                }
            }, cts.Token);
            
            return () =>
            {
                cts.Cancel();
                task.Dispose();
            };
        });
    }
}
```

**Converting IObservable to IAsyncEnumerable**

- Use `ToAsyncEnumerable` extension method.

```csharp
public static class ObservableToAsyncEnumerable
{
    public static IAsyncEnumerable<T> ToAsyncEnumerable<T>(
        this IObservable<T> source,
        CancellationToken cancellationToken = default)
    {
        return new ObservableAsyncEnumerable<T>(source, cancellationToken);
    }

    private class ObservableAsyncEnumerable<T> : IAsyncEnumerable<T>
    {
        private readonly IObservable<T> _source;
        private readonly CancellationToken _cancellationToken;

        public ObservableAsyncEnumerable(IObservable<T> source, CancellationToken cancellationToken)
        {
            _source = source;
            _cancellationToken = cancellationToken;
        }

        public async IAsyncEnumerator<T> GetAsyncEnumerator(CancellationToken cancellationToken = default)
        {
            using var cts = CancellationTokenSource.CreateLinkedTokenSource(_cancellationToken, cancellationToken);
            var channel = System.Threading.Channels.Channel.CreateUnbounded<T>();
            
            var subscription = _source.Subscribe(
                item => channel.Writer.TryWrite(item),
                error => channel.Writer.Complete(error),
                () => channel.Writer.Complete());
            
            try
            {
                await foreach (var item in channel.Reader.ReadAllAsync(cts.Token).ConfigureAwait(false))
                {
                    yield return item;
                }
            }
            finally
            {
                subscription.Dispose();
            }
        }
    }
}
```

**Mentor emphasis**

- **Use Channels** as a bridge between push and pull models.
- **Handle cancellation** properly when bridging models.
- **Dispose subscriptions** to avoid memory leaks.
- **Consider using System.Reactive** for complex reactive scenarios.

### 4. **Applying backpressure and rate limiting in streaming pipelines**

**What is backpressure?**

- **Backpressure** occurs when a producer is faster than a consumer.
- Need to **slow down** the producer or **buffer** items.
- Prevents memory exhaustion.

**Using bounded Channels for backpressure**

- Bounded channels automatically apply backpressure when full.

```csharp
using System;
using System.Threading.Channels;
using System.Threading.Tasks;

public static class BackpressureExamples
{
    public static async Task ProcessWithBackpressureAsync()
    {
        // Bounded channel applies backpressure when full
        var channel = Channel.CreateBounded<int>(new BoundedChannelOptions(100)
        {
            FullMode = BoundedChannelFullMode.Wait // Wait when full
        });
        
        // Producer
        var producer = Task.Run(async () =>
        {
            for (int i = 0; i < 10000; i++)
            {
                // WriteAsync will wait if channel is full
                await channel.Writer.WriteAsync(i);
            }
            channel.Writer.Complete();
        });
        
        // Consumer (slower than producer)
        await foreach (var item in channel.Reader.ReadAllAsync())
        {
            await ProcessItemAsync(item); // Slow processing
        }
        
        await producer;
    }

    private static async Task ProcessItemAsync(int item)
    {
        await Task.Delay(10); // Simulate slow processing
    }
}
```

**Rate limiting**

- Limit the rate at which items are produced or consumed.

```csharp
using System;
using System.Collections.Generic;
using System.Runtime.CompilerServices;
using System.Threading;
using System.Threading.Tasks;

public static class RateLimiting
{
    public static async IAsyncEnumerable<T> RateLimitAsync<T>(
        this IAsyncEnumerable<T> source,
        int itemsPerSecond,
        [EnumeratorCancellation] CancellationToken cancellationToken = default)
    {
        var delay = TimeSpan.FromSeconds(1.0 / itemsPerSecond);
        var lastYield = DateTime.UtcNow;
        
        await foreach (var item in source.WithCancellation(cancellationToken).ConfigureAwait(false))
        {
            var now = DateTime.UtcNow;
            var timeSinceLastYield = now - lastYield;
            
            if (timeSinceLastYield < delay)
            {
                await Task.Delay(delay - timeSinceLastYield, cancellationToken).ConfigureAwait(false);
            }
            
            yield return item;
            lastYield = DateTime.UtcNow;
        }
    }
}

// Usage
public static class RateLimitingExamples
{
    public static async Task ProcessWithRateLimitAsync()
    {
        var items = GetItemsAsync();
        
        // Limit to 10 items per second
        var rateLimited = items.RateLimitAsync(10);
        
        await foreach (var item in rateLimited.ConfigureAwait(false))
        {
            ProcessItem(item);
        }
    }
}
```

**Semaphore-based throttling**

- Use `SemaphoreSlim` to limit concurrent processing.

```csharp
public static class ThrottlingExamples
{
    public static async Task ProcessWithThrottlingAsync()
    {
        var semaphore = new SemaphoreSlim(10); // Max 10 concurrent
        
        await foreach (var item in GetItemsAsync().ConfigureAwait(false))
        {
            await semaphore.WaitAsync();
            
            // Process item (don't await in the loop body)
            _ = Task.Run(async () =>
            {
                try
                {
                    await ProcessItemAsync(item);
                }
                finally
                {
                    semaphore.Release();
                }
            });
        }
    }
}
```

**Mentor emphasis**

- **Use bounded Channels** for automatic backpressure.
- **Rate limit** when needed to avoid overwhelming consumers.
- **Throttle** concurrent operations with `SemaphoreSlim`.
- **Monitor memory usage** to detect backpressure issues.

### 5. **Long-lived streaming scenarios (telemetry, log ingestion, real-time feeds)**

**Telemetry streaming**

- Stream telemetry data continuously.

```csharp
using System;
using System.Collections.Generic;
using System.Runtime.CompilerServices;
using System.Threading;
using System.Threading.Tasks;

public class TelemetryStreamer
{
    private readonly IAsyncEnumerable<TelemetryEvent> _events;

    public TelemetryStreamer()
    {
        _events = GenerateTelemetryAsync();
    }

    public async Task StartStreamingAsync(CancellationToken cancellationToken = default)
    {
        await foreach (var evt in _events.WithCancellation(cancellationToken).ConfigureAwait(false))
        {
            await SendTelemetryAsync(evt);
        }
    }

    private async IAsyncEnumerable<TelemetryEvent> GenerateTelemetryAsync(
        [EnumeratorCancellation] CancellationToken cancellationToken = default)
    {
        while (!cancellationToken.IsCancellationRequested)
        {
            yield return new TelemetryEvent
            {
                Timestamp = DateTime.UtcNow,
                Value = GetCurrentValue()
            };
            
            await Task.Delay(TimeSpan.FromSeconds(1), cancellationToken).ConfigureAwait(false);
        }
    }

    private async Task SendTelemetryAsync(TelemetryEvent evt)
    {
        // Send to telemetry service
        await Task.Delay(10);
    }

    private double GetCurrentValue() => new Random().NextDouble();
}

public class TelemetryEvent
{
    public DateTime Timestamp { get; set; }
    public double Value { get; set; }
}
```
