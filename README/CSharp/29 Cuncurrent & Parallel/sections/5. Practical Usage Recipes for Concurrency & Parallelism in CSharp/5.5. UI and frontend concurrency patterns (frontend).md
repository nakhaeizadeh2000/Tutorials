### 1. **Offloading CPU work from the UI thread with Task.Run (frontend)**

**Why offload CPU work**

- **UI responsiveness**: CPU-bound work blocks the UI thread, making the application unresponsive.
- **User experience**: Users expect the UI to remain interactive during long operations.
- **Best practice**: Keep the UI thread free for rendering and user input.

**Basic pattern: Offload CPU work**

```csharp
using System;
using System.Threading.Tasks;
using System.Windows.Forms; // WinForms example

public partial class MainForm : Form
{
    private Button _processButton;
    private Label _statusLabel;
    
    public MainForm()
    {
        InitializeComponent();
        _processButton.Click += ProcessButton_Click;
    }
    
    // ✅ GOOD: Offload CPU work to background thread
    private async void ProcessButton_Click(object sender, EventArgs e)
    {
        _processButton.Enabled = false;
        _statusLabel.Text = "Processing...";
        
        try
        {
            // Offload CPU-bound work
            var result = await Task.Run(() => HeavyComputation()).ConfigureAwait(true);
            
            // Back on UI thread (ConfigureAwait(true) ensures UI context)
            _statusLabel.Text = $"Result: {result}";
        }
        catch (Exception ex)
        {
            _statusLabel.Text = $"Error: {ex.Message}";
        }
        finally
        {
            _processButton.Enabled = true;
        }
    }
    
    private int HeavyComputation()
    {
        // CPU-bound work that would block UI thread
        int sum = 0;
        for (int i = 0; i < 1_000_000_000; i++)
        {
            sum += i;
        }
        return sum;
    }
}
```

**Pattern: Progress reporting**

```csharp
using System;
using System.Threading;
using System.Threading.Tasks;
using System.Windows.Forms;

public partial class ProgressForm : Form
{
    private Button _processButton;
    private ProgressBar _progressBar;
    private Label _statusLabel;

    {
        InitializeComponent();
        _processButton.Click += ProcessButton_Click;
    }
    
    private async void ProcessButton_Click(object sender, EventArgs e)
    {
        _processButton.Enabled = false;
        _progressBar.Value = 0;
        
        var progress = new Progress<int>(value =>
        {
            // This runs on the UI thread (captured SynchronizationContext)
            _progressBar.Value = value;
            _statusLabel.Text = $"{value}%";
        });
        
        try
        {
            await Task.Run(() => HeavyComputationWithProgress(progress))
                .ConfigureAwait(true);
            
            _statusLabel.Text = "Complete!";
        }
        catch (Exception ex)
        {
            _statusLabel.Text = $"Error: {ex.Message}";
        }
        finally
        {
            _processButton.Enabled = true;
        }
    }
    
    private void HeavyComputationWithProgress(IProgress<int> progress)
    {
        for (int i = 0; i <= 100; i++)
        {
            // Simulate work
            Thread.Sleep(50);
            
            // Report progress (automatically marshaled to UI thread)
            progress.Report(i);
        }
    }
}
```

**Pattern: WPF with async commands**

```csharp
using System;
using System.Threading.Tasks;
using System.Windows.Input;
using System.Windows;

public class MainViewModel : ViewModelBase
{
    private string _status = "Ready";
    private bool _isProcessing;
    
    public string Status
    {
        get => _status;
        set => SetProperty(ref _status, value);
    }
    
    public bool IsProcessing
    {
        get => _isProcessing;
        set => SetProperty(ref _isProcessing, value);
    }
    
    public ICommand ProcessCommand { get; }
    
    public MainViewModel()
    {
        ProcessCommand = new AsyncRelayCommand(ProcessAsync, () => !IsProcessing);
    }
    
    private async Task ProcessAsync()
    {
        IsProcessing = true;
        Status = "Processing...";
        
        try
        {
            // Offload CPU work
            var result = await Task.Run(() => HeavyComputation())
                .ConfigureAwait(true); // Return to UI thread
            
            Status = $"Result: {result}";
        }
        catch (Exception ex)
        {
            Status = $"Error: {ex.Message}";
        }
        finally
        {
            IsProcessing = false;
        }
    }
    
    private int HeavyComputation()
    {
        int sum = 0;
        for (int i = 0; i < 1_000_000_000; i++)
        {
            sum += i;
        }
        return sum;
    }
}

// Simple async command implementation
public class AsyncRelayCommand : ICommand
{
    private readonly Func<Task> _execute;
    private readonly Func<bool> _canExecute;
    private bool _isExecuting;
    
    public AsyncRelayCommand(Func<Task> execute, Func<bool>? canExecute = null)
    {
        _execute = execute ?? throw new ArgumentNullException(nameof(execute));
        _canExecute = canExecute ?? (() => true);
    }
    
    public event EventHandler? CanExecuteChanged;
    
    public bool CanExecute(object? parameter)
    {
        return !_isExecuting && _canExecute();
    }
    
    public async void Execute(object? parameter)
    {
        if (CanExecute(parameter))
        {
            _isExecuting = true;
            CanExecuteChanged?.Invoke(this, EventArgs.Empty);
            
            try
            {
                await _execute().ConfigureAwait(true);
            }
            finally
            {
                _isExecuting = false;
                CanExecuteChanged?.Invoke(this, EventArgs.Empty);
            }
        }
    }
}
```

**Common mistakes**

1. **Blocking the UI thread**
   ```csharp
   // ❌ WRONG: Blocks UI thread
   private void ProcessButton_Click(object sender, EventArgs e)
   {
       var result = HeavyComputation(); // UI freezes!
       _statusLabel.Text = $"Result: {result}";
   }
   
   // ✅ CORRECT: Offload to background
   private async void ProcessButton_Click(object sender, EventArgs e)
   {
       var result = await Task.Run(() => HeavyComputation());
       _statusLabel.Text = $"Result: {result}";
   }
   ```

2. **Not using ConfigureAwait(true) when updating UI**
   ```csharp
   // ⚠️ CAUTION: May not return to UI thread
   var result = await Task.Run(() => HeavyComputation()).ConfigureAwait(false);
   _statusLabel.Text = $"Result: {result}"; // May throw if not on UI thread
   
   // ✅ CORRECT: Use ConfigureAwait(true) to return to UI thread
   var result = await Task.Run(() => HeavyComputation()).ConfigureAwait(true);
   _statusLabel.Text = $"Result: {result}"; // Safe: on UI thread
   ```

3. **Using Task.Run for I/O-bound work**
   ```csharp
   // ❌ WRONG: Task.Run doesn't help with async I/O
   var data = await Task.Run(() => httpClient.GetStringAsync(url));
   
   // ✅ CORRECT: Just await the async I/O directly
   var data = await httpClient.GetStringAsync(url);
   ```

**Performance notes**

- **Task.Run overhead**: `Task.Run` allocates a Task and uses a Thread Pool thread. Only use it for CPU-bound work.
- **UI thread**: Keep the UI thread free for rendering. Any work > 16ms (60 FPS) should be offloaded.
- **Progress reporting**: `IProgress<T>` automatically marshals to the captured `SynchronizationContext`, making it safe for UI updates.

**Good to know**

- Use `Task.Run` only for CPU-bound work. For I/O-bound work, use `async`/`await` directly.
- `ConfigureAwait(true)` ensures continuation runs on the captured `SynchronizationContext` (UI thread).
- `IProgress<T>` is thread-safe and automatically posts to the UI thread when created on the UI thread.

### 2. **Responsive UI with async commands and cancellation (frontend)**

**Why async commands matter**

- **User control**: Users can cancel long-running operations.
- **Responsiveness**: UI remains interactive during operations.
- **Error handling**: Proper exception handling without blocking.

**Pattern: Async command with cancellation**

```csharp
using System;
using System.Threading;
using System.Threading.Tasks;
using System.Windows.Input;
using System.Windows;

public class CancellableViewModel : ViewModelBase
{
    private CancellationTokenSource? _cts;
    private string _status = "Ready";
    private bool _isProcessing;
    
    public string Status
    {
        get => _status;
        set => SetProperty(ref _status, value);
    }
    
    public bool IsProcessing
    {
        get => _isProcessing;
        set => SetProperty(ref _isProcessing, value);
    }
    
    public ICommand StartCommand { get; }
    public ICommand CancelCommand { get; }
    
    public CancellableViewModel()
    {
        StartCommand = new AsyncRelayCommand(StartAsync, () => !IsProcessing);
        CancelCommand = new RelayCommand(Cancel, () => IsProcessing);
    }
    
    private async Task StartAsync()
    {
        _cts = new CancellationTokenSource();
        IsProcessing = true;
        Status = "Processing...";
        
        try
        {
            await Task.Run(() => LongRunningWork(_cts.Token), _cts.Token)
                .ConfigureAwait(true);
            
            Status = "Complete!";
        }
        catch (OperationCanceledException)
        {
            Status = "Cancelled";
        }
        catch (Exception ex)
        {
            Status = $"Error: {ex.Message}";
        }
        finally
        {
            IsProcessing = false;
            _cts?.Dispose();
            _cts = null;
        }
    }
    
    private void Cancel()
    {
        _cts?.Cancel();
    }
    
    private void LongRunningWork(CancellationToken ct)
    {
        for (int i = 0; i < 100; i++)
        {
            ct.ThrowIfCancellationRequested();
            Thread.Sleep(100); // Simulate work
        }
    }
}
```

**Pattern: WPF with cancellation token**

```csharp
using System;
using System.Threading;
using System.Threading.Tasks;
using System.Windows.Input;

public class AsyncCancellableCommand : ICommand
{
    private readonly Func<CancellationToken, Task> _execute;
    private readonly Func<bool> _canExecute;
    private CancellationTokenSource? _cts;
    private bool _isExecuting;
    
    public AsyncCancellableCommand(
        Func<CancellationToken, Task> execute,
        Func<bool>? canExecute = null)
    {
        _execute = execute ?? throw new ArgumentNullException(nameof(execute));
        _canExecute = canExecute ?? (() => true);
    }
    
    public event EventHandler? CanExecuteChanged;
    
    public bool CanExecute(object? parameter)
    {
        return !_isExecuting && _canExecute();
    }
    
    public async void Execute(object? parameter)
    {
        if (CanExecute(parameter))
        {
            _isExecuting = true;
            _cts = new CancellationTokenSource();
            CanExecuteChanged?.Invoke(this, EventArgs.Empty);
            
            try
            {
                await _execute(_cts.Token).ConfigureAwait(true);
            }
            catch (OperationCanceledException)
            {
                // Expected on cancellation
            }
            finally
            {
                _isExecuting = false;
                _cts?.Dispose();
                _cts = null;
                CanExecuteChanged?.Invoke(this, EventArgs.Empty);
            }
        }
    }
    
    public void Cancel()
    {
        _cts?.Cancel();
    }
}

// Usage:
public class ViewModel
{
    private AsyncCancellableCommand? _processCommand;
    
    public ICommand ProcessCommand => _processCommand ??= new AsyncCancellableCommand(
        async ct =>
        {
            await Task.Run(() => LongWork(ct), ct).ConfigureAwait(true);
        }
    );
    
    private void LongWork(CancellationToken ct)
    {
        for (int i = 0; i < 100; i++)
        {
            ct.ThrowIfCancellationRequested();
            Thread.Sleep(100);
        }
    }
}
```

**Pattern: WinForms with cancellation**

```csharp
using System;
using System.Threading;
using System.Threading.Tasks;
using System.Windows.Forms;

public partial class CancellableForm : Form
{
    private Button _startButton;
    private Button _cancelButton;
    private Label _statusLabel;
    private CancellationTokenSource? _cts;
    
    public CancellableForm()
    {
        InitializeComponent();
        _startButton.Click += StartButton_Click;
        _cancelButton.Click += CancelButton_Click;
        _cancelButton.Enabled = false;
    }
    
    private async void StartButton_Click(object sender, EventArgs e)

        _cts = new CancellationTokenSource();
        _startButton.Enabled = false;
        _cancelButton.Enabled = true;
        _statusLabel.Text = "Processing...";
        
        try
        {
            await Task.Run(() => LongWork(_cts.Token), _cts.Token)
                .ConfigureAwait(true);
            
            _statusLabel.Text = "Complete!";
        }
        catch (OperationCanceledException)
        {
            _statusLabel.Text = "Cancelled";
        }
        catch (Exception ex)
        {
            _statusLabel.Text = $"Error: {ex.Message}";
        }
        finally
        {
            _startButton.Enabled = true;
            _cancelButton.Enabled = false;
            _cts?.Dispose();
            _cts = null;
        }
    }
    
    private void CancelButton_Click(object sender, EventArgs e)
    {
        _cts?.Cancel();
    }
    
    private void LongWork(CancellationToken ct)
    {
        for (int i = 0; i < 100; i++)
        {
            ct.ThrowIfCancellationRequested();
            Thread.Sleep(100);
        }
    }
}
```

**Common mistakes**

1. **Not disposing CancellationTokenSource**
   ```csharp
   // ❌ WRONG: Resource leak
   _cts = new CancellationTokenSource();
   // Never disposed
   
   // ✅ CORRECT: Dispose in finally
   try { /* work */ }
   finally { _cts?.Dispose(); }
   ```

2. **Not checking cancellation in loops**
   ```csharp
   // ❌ WRONG: Can't cancel
   for (int i = 0; i < 100; i++)
   {
       Thread.Sleep(100); // No cancellation check
   }
   
   // ✅ CORRECT: Check cancellation
   for (int i = 0; i < 100; i++)
   {
       ct.ThrowIfCancellationRequested();
       Thread.Sleep(100);
   }
   ```

3. **Swallowing cancellation exceptions**
   ```csharp
   // ⚠️ CAUTION: May hide cancellation
   catch (Exception ex)
   {
       // Swallows OperationCanceledException
   }
   
   // ✅ CORRECT: Handle cancellation separately
   catch (OperationCanceledException)
   {
       // Handle cancellation
   }
   catch (Exception ex)
   {
       // Handle other errors
   }
   ```

**Performance notes**

- **Cancellation overhead**: Checking `CancellationToken` has negligible overhead. Check frequently in loops.
- **UI responsiveness**: Cancellation allows users to stop long operations, improving perceived performance.

**Good to know**

- Always dispose `CancellationTokenSource` to avoid resource leaks.
- Use `CancellationToken.ThrowIfCancellationRequested()` for clean cancellation in loops.
- Handle `OperationCanceledException` separately from other exceptions.

### 3. **Throttling / debouncing UI events with async patterns (frontend)**

**Why throttle/debounce**

- **Performance**: Prevent excessive event handling (e.g., text input, scroll events).
- **User experience**: Reduce flickering and improve responsiveness.
- **Resource usage**: Limit API calls or expensive operations.

**Pattern: Debouncing text input**

```csharp
using System;
using System.Threading;
using System.Threading.Tasks;
using System.Windows.Forms;

public partial class DebouncedForm : Form
{
    private TextBox _searchBox;
    private Label _resultsLabel;
    private CancellationTokenSource? _debounceCts;
    
    public DebouncedForm()
    {
        InitializeComponent();
        _searchBox.TextChanged += SearchBox_TextChanged;
    }
    
    private async void SearchBox_TextChanged(object sender, EventArgs e)
    {
        // Cancel previous debounce
        _debounceCts?.Cancel();
        _debounceCts?.Dispose();
        _debounceCts = new CancellationTokenSource();
        
        var searchText = _searchBox.Text;
        var token = _debounceCts.Token;
        
        try
        {
            // Wait for debounce period
            await Task.Delay(TimeSpan.FromMilliseconds(300), token)
                .ConfigureAwait(true);
            
            // Perform search (only if not cancelled)
            var results = await SearchAsync(searchText, token).ConfigureAwait(true);
            _resultsLabel.Text = $"Found: {results.Count} results";
        }
        catch (OperationCanceledException)
        {
            // Debounce cancelled, ignore
        }
    }
    
    private async Task<List<string>> SearchAsync(string query, CancellationToken ct)
    {
        // Simulate API call
        await Task.Delay(100, ct).ConfigureAwait(false);
        return new List<string> { "Result 1", "Result 2" };
    }
}
```

**Pattern: Throttling scroll events**

```csharp
using System;
using System.Threading;
using System.Threading.Tasks;
using System.Windows.Forms;

public partial class ThrottledForm : Form
{
    private ScrollBar _scrollBar;
    private Label _positionLabel;
    private DateTime _lastUpdate = DateTime.MinValue;
    private readonly TimeSpan _throttleInterval = TimeSpan.FromMilliseconds(100);
    private readonly SemaphoreSlim _throttleSemaphore = new(1, 1);
    
    public ThrottledForm()
    {
        InitializeComponent();
        _scrollBar.Scroll += ScrollBar_Scroll;
    }
    
    private async void ScrollBar_Scroll(object sender, ScrollEventArgs e)
    {
        // Throttle: Only process if enough time has passed
        var now = DateTime.UtcNow;
        if (now - _lastUpdate < _throttleInterval)
        {
            return; // Skip this event
        }
        
        _lastUpdate = now;
        
        // Process scroll event
        await ProcessScrollAsync(e.NewValue).ConfigureAwait(true);
    }
    
    // Alternative: Using semaphore for throttling
    private async void ScrollBar_ScrollWithSemaphore(object sender, ScrollEventArgs e)
    {
        if (!await _throttleSemaphore.WaitAsync(0).ConfigureAwait(false))
        {
            return; // Already processing, skip
        }
        
        try
        {
            await ProcessScrollAsync(e.NewValue).ConfigureAwait(true);
            await Task.Delay(_throttleInterval).ConfigureAwait(false);
        }
        finally
        {
            _throttleSemaphore.Release();
        }
    }
    
    private async Task ProcessScrollAsync(int position)
    {
        _positionLabel.Text = $"Position: {position}";
        // Simulate work
        await Task.Delay(10).ConfigureAwait(true);
    }
}
```

**Pattern: Reusable debouncer**

```csharp
using System;
using System.Threading;
using System.Threading.Tasks;

public class Debouncer
{
    private CancellationTokenSource? _cts;
    private readonly TimeSpan _delay;
    
    public Debouncer(TimeSpan delay)
    {
        _delay = delay;
    }
    
    public async Task DebounceAsync(Func<Task> action, CancellationToken ct = default)
    {
        _cts?.Cancel();
        _cts?.Dispose();
        _cts = CancellationTokenSource.CreateLinkedTokenSource(ct);
        
        var token = _cts.Token;
        
        try
        {
            await Task.Delay(_delay, token).ConfigureAwait(false);
            await action().ConfigureAwait(false);
        }
        catch (OperationCanceledException)
        {
            // Debounce cancelled, ignore
        }
    }
    
    public void Dispose()
    {
        _cts?.Cancel();
        _cts?.Dispose();
    }
}

// Usage:
public partial class FormWithDebouncer : Form
{
    private TextBox _inputBox;
    private Debouncer _debouncer = new(TimeSpan.FromMilliseconds(300));
    
    public FormWithDebouncer()
    {
        InitializeComponent();
        _inputBox.TextChanged += InputBox_TextChanged;
    }
    
    private async void InputBox_TextChanged(object sender, EventArgs e)
    {
        var text = _inputBox.Text;
        await _debouncer.DebounceAsync(async () =>
        {
            await ProcessInputAsync(text).ConfigureAwait(true);
        }).ConfigureAwait(false);
    }
    
    private async Task ProcessInputAsync(string text)
    {
        // Process input
        await Task.Delay(100).ConfigureAwait(false);
    }
}
```

**Pattern: WPF with reactive extensions (Rx)**

```csharp
using System;
using System.Reactive.Linq;
using System.Windows.Controls;

public partial class ReactiveForm : Window
{
    private TextBox _searchBox;
    
    public ReactiveForm()
    {
        InitializeComponent();
        
        // Debounce text changes using Rx
        Observable.FromEventPattern<TextChangedEventHandler, TextChangedEventArgs>(
                h => _searchBox.TextChanged += h,
                h => _searchBox.TextChanged -= h)
            .Select(_ => _searchBox.Text)
            .Throttle(TimeSpan.FromMilliseconds(300))
            .DistinctUntilChanged()
            .ObserveOnDispatcher() // Ensure UI thread
            .Subscribe(async text =>
            {
                var results = await SearchAsync(text).ConfigureAwait(true);
                UpdateResults(results);
            });
    }
    
    private async Task<List<string>> SearchAsync(string query)
    {
        await Task.Delay(100).ConfigureAwait(false);
        return new List<string>();
    }
    
    private void UpdateResults(List<string> results)
    {
        // Update UI
    }
}
```

**Common mistakes**

1. **Not cancelling previous debounce**
   ```csharp
   // ❌ WRONG: Multiple debounces run simultaneously
   private async void TextChanged(object sender, EventArgs e)
   {
       await Task.Delay(300);
       Process();
   }
   
   // ✅ CORRECT: Cancel previous debounce
   _debounceCts?.Cancel();
   _debounceCts = new CancellationTokenSource();
   await Task.Delay(300, _debounceCts.Token);
   ```

2. **Not disposing CancellationTokenSource**
   ```csharp
   // ❌ WRONG: Resource leak
   _debounceCts = new CancellationTokenSource();
   
   // ✅ CORRECT: Dispose when done
   _debounceCts?.Dispose();
   ```

**Performance notes**

- **Debounce delay**: Typical delays are 200-500ms for text input, 50-100ms for scroll events.
- **Throttle interval**: Set based on how frequently you want to process events. Too short defeats the purpose; too long feels laggy.

**Good to know**

- Debouncing delays execution until events stop. Throttling limits execution frequency.
- Use debouncing for text input, search boxes. Use throttling for scroll events, resize events.
- Consider using reactive extensions (Rx) for complex event handling scenarios.

### 4. **Using async streams to consume event streams (frontend)**

**What async streams provide**

- **Pull-based consumption**: Process events as they arrive without blocking.
- **Cancellation support**: Natural integration with `CancellationToken`.
- **Composability**: Can be filtered, transformed, and aggregated.

**Pattern: Converting events to async stream**

```csharp
using System;
using System.Collections.Generic;
using System.Runtime.CompilerServices;
using System.Threading;
using System.Threading.Tasks;
using System.Windows.Forms;

public static class EventToAsyncStream
{
    public static async IAsyncEnumerable<string> ToAsyncStream(
        TextBox textBox,
        [EnumeratorCancellation] CancellationToken cancellationToken = default)
    {
        var tcs = new TaskCompletionSource<string>();
        string? lastValue = null;
        
        void Handler(object? sender, EventArgs e)
        {
            var value = textBox.Text;
            if (value != lastValue)
            {
                lastValue = value;
                var currentTcs = tcs;
                tcs = new TaskCompletionSource<string>();
                currentTcs.SetResult(value);
            }
        }
        
        textBox.TextChanged += Handler;
        
        try
        {
            while (!cancellationToken.IsCancellationRequested)
            {
                var value = await tcs.Task.ConfigureAwait(true);
                yield return value;
            }
        }
        finally
        {
            textBox.TextChanged -= Handler;
        }
    }
}

// Usage:
public partial class AsyncStreamForm : Form
{
    private TextBox _inputBox;
    private Label _outputLabel;
    private CancellationTokenSource? _cts;
    
    public AsyncStreamForm()
    {
        InitializeComponent();
        _inputBox.TextChanged += InputBox_TextChanged;
    }
    
    private async void InputBox_TextChanged(object sender, EventArgs e)
    {
        _cts?.Cancel();
        _cts?.Dispose();
        _cts = new CancellationTokenSource();
        
        _ = ProcessStreamAsync(_cts.Token);
    }
    
    private async Task ProcessStreamAsync(CancellationToken ct)
    {
        await foreach (var value in EventToAsyncStream.ToAsyncStream(_inputBox, ct)
            .ConfigureAwait(true))
        {
            // Process each text change
            var processed = await ProcessValueAsync(value, ct).ConfigureAwait(true);
            _outputLabel.Text = processed;
        }
    }
    
    private async Task<string> ProcessValueAsync(string value, CancellationToken ct)
    {
        await Task.Delay(100, ct).ConfigureAwait(false);
        return $"Processed: {value}";
    }
}
```

**Pattern: Mouse move events as async stream**

```csharp
using System;
using System.Collections.Generic;
using System.Drawing;
using System.Runtime.CompilerServices;
using System.Threading;
using System.Threading.Tasks;
using System.Windows.Forms;

public static class MouseMoveStream
{
    public static async IAsyncEnumerable<Point> ToAsyncStream(
        Control control,
        [EnumeratorCancellation] CancellationToken cancellationToken = default)
    {
        var tcs = new TaskCompletionSource<Point>();
        
        void Handler(object? sender, MouseEventArgs e)
        {
            tcs.SetResult(e.Location);
            tcs = new TaskCompletionSource<Point>();
        }
        
        control.MouseMove += Handler;
        
        try
        {
            while (!cancellationToken.IsCancellationRequested)
            {
                var point = await tcs.Task.ConfigureAwait(true);
                yield return point;
            }
        }
        finally
        {
            control.MouseMove -= Handler;
        }
    }
}

// Usage:
public partial class MouseTrackingForm : Form
{
    public MouseTrackingForm()
    {
        InitializeComponent();
        _ = TrackMouseAsync();
    }
    
    private async Task TrackMouseAsync()
    {
        await foreach (var point in MouseMoveStream.ToAsyncStream(this)
            .ConfigureAwait(true))
        {
            // Process mouse movements
            Console.WriteLine($"Mouse at: {point}");
        }
    }
}
```

**Pattern: Filtering and transforming event streams**

```csharp
using System;
using System.Collections.Generic;
using System.Linq;
using System.Runtime.CompilerServices;
using System.Threading;
using System.Threading.Tasks;

public static class StreamExtensions
{
    public static async IAsyncEnumerable<TResult> SelectAsync<TSource, TResult>(
        this IAsyncEnumerable<TSource> source,
        Func<TSource, Task<TResult>> selector,
        [EnumeratorCancellation] CancellationToken cancellationToken = default)
    {
        await foreach (var item in source.WithCancellation(cancellationToken)
            .ConfigureAwait(false))
        {
            yield return await selector(item).ConfigureAwait(false);
        }
    }
    
    public static async IAsyncEnumerable<T> WhereAsync<T>(
        this IAsyncEnumerable<T> source,
        Func<T, bool> predicate,
        [EnumeratorCancellation] CancellationToken cancellationToken = default)
    {
        await foreach (var item in source.WithCancellation(cancellationToken)
            .ConfigureAwait(false))
        {
            if (predicate(item))
            {
                yield return item;
            }
        }
    }
}

// Usage:
await foreach (var value in EventToAsyncStream.ToAsyncStream(textBox)
    .WhereAsync(v => v.Length > 3)
    .SelectAsync(async v => await ProcessAsync(v))
    .ConfigureAwait(true))
{
    UpdateUI(value);
}
```

**Common mistakes**

1. **Not unsubscribing from events**
   ```csharp
   // ❌ WRONG: Memory leak
   control.Event += Handler;
   // Never unsubscribes
   
   // ✅ CORRECT: Unsubscribe in finally
   try { /* consume stream */ }
   finally { control.Event -= Handler; }
   ```

2. **Not handling cancellation**
   ```csharp
   // ❌ WRONG: Can't cancel
   await foreach (var item in stream)
   {
       Process(item);
   }
   
   // ✅ CORRECT: Pass cancellation token
   await foreach (var item in stream.WithCancellation(ct))
   {
       Process(item);
   }
   ```

**Performance notes**

- **Event overhead**: Converting events to async streams has minimal overhead. The main cost is the event handler invocation.
- **Backpressure**: Async streams are pull-based, so backpressure is handled naturally by the consumer.

**Good to know**

- Async streams are excellent for consuming UI events in a composable way.
- Use `WithCancellation` to support cancellation in async streams.
- Always unsubscribe from events in `finally` blocks to prevent memory leaks.

### 5. **Avoiding deadlocks with ConfigureAwait and sync context (frontend)**

**Why deadlocks happen**

- **Sync-over-async**: Calling `.Result` or `.Wait()` on async code that tries to return to the UI thread.
- **Context capture**: Async code captures `SynchronizationContext` and tries to return to it, but the UI thread is blocked waiting.

**The deadlock pattern**

```csharp
// ❌ DEADLOCK: This will deadlock in UI applications
private void Button_Click(object sender, EventArgs e)
{
    var result = GetDataAsync().Result; // Blocks UI thread
    // GetDataAsync tries to return to UI thread, but it's blocked → DEADLOCK
}

private async Task<string> GetDataAsync()
{
    await Task.Delay(1000); // Tries to return to UI thread
    return "data";
}
```

**Solution 1: Use async/await end-to-end**

```csharp
// ✅ CORRECT: Use async/await
private async void Button_Click(object sender, EventArgs e)
{
    var result = await GetDataAsync().ConfigureAwait(true);
    // Update UI
}

private async Task<string> GetDataAsync()
{
    await Task.Delay(1000).ConfigureAwait(false); // Library code: don't capture context
    return "data";
}
```

**Solution 2: Use ConfigureAwait(false) in library code**

```csharp
// ✅ CORRECT: Library code doesn't capture UI context
public static async Task<string> LibraryMethodAsync()
{
    // Don't capture context in library code
    var data = await FetchDataAsync().ConfigureAwait(false);
    return ProcessData(data);
}

// Application code can await and update UI
private async void Button_Click(object sender, EventArgs e)
{
    var result = await LibraryMethodAsync().ConfigureAwait(true);
    _label.Text = result; // Safe: on UI thread
}
```

**Solution 3: Task.Run for CPU-bound work**

```csharp
// ✅ CORRECT: Offload CPU work, then return to UI
private async void Button_Click(object sender, EventArgs e)
{
    var result = await Task.Run(() => HeavyComputation()).ConfigureAwait(true);
    _label.Text = result; // Safe: on UI thread
}
```

**Pattern: Safe fire-and-forget**

```csharp
using System;
using System.Threading.Tasks;
using System.Windows.Forms;

public static class SafeFireAndForget
{
    // ✅ GOOD: Fire-and-forget with proper exception handling
    public static void FireAndForget(Task task, Action<Exception>? onError = null)
    {
        _ = task.ContinueWith(t =>
        {
            if (t.IsFaulted && onError != null)
            {
                // Handle on UI thread if needed
                Application.BeginInvoke(new Action(() => onError(t.Exception!.InnerException!)));
            }
        }, TaskScheduler.FromCurrentSynchronizationContext());
    }
}

// Usage:
private void Button_Click(object sender, EventArgs e)
{
    SafeFireAndForget.FireAndForget(
        DoWorkAsync(),
        ex => MessageBox.Show($"Error: {ex.Message}")
    );
}

private async Task DoWorkAsync()
{
    await Task.Delay(1000).ConfigureAwait(false);
    throw new Exception("Test error");
}
```

**Pattern: Avoiding deadlocks in libraries**

```csharp
// ✅ GOOD: Library code that doesn't deadlock
public class DataService
{
    public async Task<string> GetDataAsync()
    {
        // Use ConfigureAwait(false) to avoid capturing context
        using var http = new HttpClient();
        var response = await http.GetStringAsync("https://api.example.com/data")
            .ConfigureAwait(false);
        
        // Process data (CPU-bound, no context needed)
        var processed = ProcessData(response);
        
        return processed;
    }
    
    private string ProcessData(string data)
    {
        // CPU-bound work
        return data.ToUpper();
    }
}

// Application code:
private async void Button_Click(object sender, EventArgs e)
{
    var service = new DataService();
    var result = await service.GetDataAsync().ConfigureAwait(true); // Return to UI
    _label.Text = result; // Safe: on UI thread
}
```

**Common mistakes**

1. **Sync-over-async**
   ```csharp
   // ❌ WRONG: Deadlock risk
   private void Button_Click(object sender, EventArgs e)
   {
       var result = GetDataAsync().Result; // Blocks UI thread
   }
   
   // ✅ CORRECT: Use async/await
   private async void Button_Click(object sender, EventArgs e)
   {
       var result = await GetDataAsync();
   }
   ```

2. **Not using ConfigureAwait(false) in library code**
   ```csharp
   // ❌ WRONG: Unnecessary context capture
   public async Task<string> LibraryMethod()
   {
       return await FetchAsync(); // Captures context unnecessarily
   }
   
   // ✅ CORRECT: Don't capture context in library code
   public async Task<string> LibraryMethod()
   {
       return await FetchAsync().ConfigureAwait(false);
   }
   ```

3. **Mixing sync and async**
   ```csharp
   // ❌ WRONG: Can cause deadlocks
   private void SyncMethod()
   {
       var result = AsyncMethod().Result;
   }
   
   // ✅ CORRECT: Make the whole chain async
   private async Task AsyncMethod()
   {
       var result = await OtherAsyncMethod();
   }
   ```

**Performance notes**

- **Context capture**: Capturing `SynchronizationContext` has a small overhead. Use `ConfigureAwait(false)` in library code to avoid it.
- **Deadlock prevention**: Using `async`/`await` end-to-end prevents deadlocks. Never use `.Result` or `.Wait()` on async code.

**Good to know**

- **Golden rule**: Use `async`/`await` end-to-end. Never use `.Result` or `.Wait()` on async code.
- **Library code**: Always use `ConfigureAwait(false)` in library code to avoid unnecessary context capture.
- **Application code**: Use `ConfigureAwait(true)` (or omit it) when you need to return to the UI thread.
