### 1. **Using async I/O APIs with async/await end-to-end**

**The golden rule**

- **Never block on I/O**. Use `async`/`await` from the entry point (controller, handler, main) all the way down to the I/O call.
- **Avoid sync-over-async**: Don't call `.Result`, `.Wait()`, or `.GetAwaiter().GetResult()` on async methods.

**File I/O patterns**

**Pattern 1: Reading files asynchronously**

```csharp
using System;
using System.IO;
using System.Text;
using System.Threading;
using System.Threading.Tasks;

public static class FileIO
{
    // ✅ GOOD: End-to-end async
    public static async Task<string> ReadFileAsync(string filePath, CancellationToken ct = default)
    {
        using var reader = new StreamReader(filePath, Encoding.UTF8);
        return await reader.ReadToEndAsync(ct).ConfigureAwait(false);
    }
    
    // ✅ GOOD: Reading line by line
    public static async Task<string[]> ReadLinesAsync(string filePath, CancellationToken ct = default)
    {
        var lines = new List<string>();
        using var reader = new StreamReader(filePath, Encoding.UTF8);
        
        string? line;
        while ((line = await reader.ReadLineAsync(ct).ConfigureAwait(false)) != null)
        {
            lines.Add(line);
        }
        
        return lines.ToArray();
    }
    
    // ✅ GOOD: Writing files
    public static async Task WriteFileAsync(string filePath, string content, CancellationToken ct = default)
    {
        await File.WriteAllTextAsync(filePath, content, ct).ConfigureAwait(false);
    }
}
```

**Pattern 2: HTTP requests**

```csharp
using System;
using System.Net.Http;
using System.Text.Json;
using System.Threading;
using System.Threading.Tasks;

public static class HttpIO
{
    private static readonly HttpClient SharedClient = new()
    {
        Timeout = TimeSpan.FromSeconds(30)
    };
    
    // ✅ GOOD: End-to-end async HTTP
    public static async Task<string> GetStringAsync(string url, CancellationToken ct = default)
    {
        using var response = await SharedClient.GetAsync(url, ct).ConfigureAwait(false);
        response.EnsureSuccessStatusCode();
        return await response.Content.ReadAsStringAsync(ct).ConfigureAwait(false);
    }
    
    // ✅ GOOD: POST with JSON
    public static async Task<TResponse> PostJsonAsync<TRequest, TResponse>(
        string url,
        TRequest request,
        CancellationToken ct = default)
    {
        var json = JsonSerializer.Serialize(request);
        var content = new StringContent(json, System.Text.Encoding.UTF8, "application/json");
        
        using var response = await SharedClient.PostAsync(url, content, ct).ConfigureAwait(false);
        response.EnsureSuccessStatusCode();
        
        var responseJson = await response.Content.ReadAsStringAsync(ct).ConfigureAwait(false);
        return JsonSerializer.Deserialize<TResponse>(responseJson)!;
    }
}
```

**Pattern 3: Database operations**

```csharp
using System;
using System.Data.Common;
using System.Threading;
using System.Threading.Tasks;

public static class DatabaseIO
{
    // ✅ GOOD: Async database query
    public static async Task<List<T>> QueryAsync<T>(
        DbConnection connection,
        string sql,
        Func<DbDataReader, T> mapper,
        CancellationToken ct = default)
    {
        var results = new List<T>();
        
        await using var command = connection.CreateCommand();
        command.CommandText = sql;
        
        await using var reader = await command.ExecuteReaderAsync(ct).ConfigureAwait(false);
        
        while (await reader.ReadAsync(ct).ConfigureAwait(false))
        {
            results.Add(mapper(reader));
        }
        
        return results;
    }
}
```

**Pattern 4: Socket operations**

```csharp
using System;
using System.Net.Sockets;
using System.Text;
using System.Threading;
using System.Threading.Tasks;

public static class SocketIO
{
    // ✅ GOOD: Async socket send/receive
    public static async Task<string> ReceiveAsync(
        Socket socket,
        int bufferSize = 4096,
        CancellationToken ct = default)
    {
        var buffer = new byte[bufferSize];
        var received = await socket.ReceiveAsync(
            new ArraySegment<byte>(buffer),
            SocketFlags.None,
            ct
        ).ConfigureAwait(false);
        
        return Encoding.UTF8.GetString(buffer, 0, received);
    }
    
    public static async Task SendAsync(
        Socket socket,
        string message,
        CancellationToken ct = default)
    {
        var bytes = Encoding.UTF8.GetBytes(message);
        await socket.SendAsync(
            new ArraySegment<byte>(bytes),
            SocketFlags.None,
            ct
        ).ConfigureAwait(false);
    }
}
```

**Common mistakes**

1. **Sync-over-async**
   ```csharp
   // ❌ WRONG: Blocks thread, can cause deadlocks
   var content = httpClient.GetStringAsync(url).Result;
   
   // ✅ CORRECT: Use await
   var content = await httpClient.GetStringAsync(url);
   ```

2. **Forgetting ConfigureAwait(false) in library code**
   ```csharp
   // ⚠️ CAUTION: In library code, use ConfigureAwait(false) to avoid context capture
   public static async Task<string> LibraryMethod()
   {
       return await httpClient.GetStringAsync(url).ConfigureAwait(false);
   }
   
   // ✅ In application code (ASP.NET Core, UI), ConfigureAwait(false) is optional
   // but still recommended for performance
   ```

3. **Not disposing resources**
   ```csharp
   // ❌ WRONG: Resource leak
   var reader = new StreamReader(filePath);
   var content = await reader.ReadToEndAsync();
   // reader not disposed!
   
   // ✅ CORRECT: Use using or await using
   using var reader = new StreamReader(filePath);
   var content = await reader.ReadToEndAsync();
   ```

4. **Mixing sync and async I/O**
   ```csharp
   // ❌ WRONG: Don't mix sync and async
   File.WriteAllText(path, content); // Sync
   await File.WriteAllTextAsync(path, content); // Async - pick one!
   ```

**Performance notes**

- **I/O completion ports**: Modern async I/O uses kernel-level I/O completion ports (Windows) or epoll/kqueue (Linux), avoiding thread blocking. See **2.4**.
- **Thread Pool efficiency**: Async I/O doesn't consume Thread Pool threads during waits. This enables handling thousands of concurrent operations with a small thread pool.
- **ConfigureAwait(false)**: In library code, always use `ConfigureAwait(false)` to avoid unnecessary context capture and improve performance.

**Good to know**

- Use `IAsyncDisposable` for async cleanup:
  ```csharp
  await using var resource = new AsyncResource();
  // Automatic disposal when scope exits
  ```
- For file I/O, prefer `File.ReadAllTextAsync` / `File.WriteAllTextAsync` over manual `StreamReader`/`StreamWriter` when reading/writing entire files.

### 2. **Coordinating many concurrent I/O operations with Task.WhenAll**

**The pattern**

- Create multiple I/O tasks.
- Use `Task.WhenAll` to wait for all to complete.
- Handle errors appropriately (aggregate exceptions).

**Basic pattern**

```csharp
using System;
using System.Collections.Generic;
using System.Linq;
using System.Net.Http;
using System.Threading;
using System.Threading.Tasks;

public static class ConcurrentIO
{
    // ✅ GOOD: Fan-out/fan-in with WhenAll
    public static async Task<string[]> DownloadAllAsync(
        IEnumerable<string> urls,
        CancellationToken ct = default)
    {
        using var http = new HttpClient();
        
        // Create all tasks
        var tasks = urls.Select(url => DownloadAsync(http, url, ct)).ToArray();
        
        // Wait for all to complete
        return await Task.WhenAll(tasks).ConfigureAwait(false);
    }
    
    private static async Task<string> DownloadAsync(
        HttpClient http,
        string url,
        CancellationToken ct)
    {
        var response = await http.GetAsync(url, ct).ConfigureAwait(false);
        response.EnsureSuccessStatusCode();
        return await response.Content.ReadAsStringAsync(ct).ConfigureAwait(false);
    }
}
```

**Error handling**

```csharp
using System;
using System.Collections.Generic;
using System.Linq;
using System.Net.Http;
using System.Threading;
using System.Threading.Tasks;

public static class ConcurrentIOWithErrors
{
    // Pattern 1: Fail fast (throw on first error)
    public static async Task<string[]> DownloadAllOrFailAsync(
        IEnumerable<string> urls,
        CancellationToken ct = default)
    {
        using var http = new HttpClient();
        var tasks = urls.Select(url => DownloadAsync(http, url, ct)).ToArray();
        
        // Throws AggregateException if any task fails
        return await Task.WhenAll(tasks).ConfigureAwait(false);
    }
    
    // Pattern 2: Continue on error, collect results
    public static async Task<(string? Result, Exception? Error)[]> DownloadAllWithErrorsAsync(
        IEnumerable<string> urls,
        CancellationToken ct = default)
    {
        using var http = new HttpClient();
        
        var tasks = urls.Select(async url =>
        {
            try
            {
                var result = await DownloadAsync(http, url, ct).ConfigureAwait(false);
                return (Result: result, Error: (Exception?)null);
            }
            catch (Exception ex)
            {
                return (Result: (string?)null, Error: ex);
            }
        }).ToArray();
        
        return await Task.WhenAll(tasks).ConfigureAwait(false);
    }
    
    // Pattern 3: Partial results (successful ones)
    public static async Task<List<string>> DownloadSuccessfulAsync(
        IEnumerable<string> urls,
        CancellationToken ct = default)
    {
        using var http = new HttpClient();
        var tasks = urls.Select(url => DownloadAsync(http, url, ct)).ToArray();
        
        var results = new List<string>();
        var completedTasks = await Task.WhenAll(tasks.Select(async task =>
        {
            try
            {
                return await task;
            }
            catch
            {
                return null; // Ignore failures
            }
        })).ConfigureAwait(false);
        
        results.AddRange(completedTasks.Where(r => r != null)!);
        return results;
    }
    
    private static async Task<string> DownloadAsync(
        HttpClient http,
        string url,
        CancellationToken ct)
    {
        var response = await http.GetAsync(url, ct).ConfigureAwait(false);
        response.EnsureSuccessStatusCode();
        return await response.Content.ReadAsStringAsync(ct).ConfigureAwait(false);
    }
}
```

**Batching for very large collections**

```csharp
using System;
using System.Collections.Generic;
using System.Linq;
using System.Net.Http;
using System.Threading;
using System.Threading.Tasks;

public static class BatchedConcurrentIO
{
    // Process in batches to avoid creating too many tasks
    public static async Task<string[]> DownloadInBatchesAsync(
        IEnumerable<string> urls,
        int batchSize = 100,
        CancellationToken ct = default)
    {
        using var http = new HttpClient();
        var allResults = new List<string>();
        var urlList = urls.ToList();
        
        for (int i = 0; i < urlList.Count; i += batchSize)
        {
            var batch = urlList.Skip(i).Take(batchSize);
            var tasks = batch.Select(url => DownloadAsync(http, url, ct)).ToArray();
            var batchResults = await Task.WhenAll(tasks).ConfigureAwait(false);
            allResults.AddRange(batchResults);
        }
        
        return allResults.ToArray();
    }
    
    private static async Task<string> DownloadAsync(
        HttpClient http,
        string url,
        CancellationToken ct)
    {
        var response = await http.GetAsync(url, ct).ConfigureAwait(false);
        response.EnsureSuccessStatusCode();
        return await response.Content.ReadAsStringAsync(ct).ConfigureAwait(false);
    }
}
```

**Common mistakes**

1. **Creating tasks but not awaiting them**
   ```csharp
   // ❌ WRONG: Fire-and-forget, no error handling
   urls.Select(url => DownloadAsync(url)).ToArray();
   
   // ✅ CORRECT: Await WhenAll
   await Task.WhenAll(urls.Select(url => DownloadAsync(url)));
   ```

2. **Not handling AggregateException**
   ```csharp
   // ⚠️ CAUTION: WhenAll throws AggregateException
   try
   {
       await Task.WhenAll(tasks);
   }
   catch (AggregateException ex)
   {
       // Handle multiple exceptions
       foreach (var inner in ex.InnerExceptions)
       {
           // Log or handle each error
       }
   }
   ```

3. **Creating too many concurrent tasks**
   ```csharp
   // ❌ WRONG: May overwhelm the server or exhaust resources
   await Task.WhenAll(Enumerable.Range(0, 10000).Select(i => DownloadAsync(i)));
   
   // ✅ CORRECT: Use throttling (see next section)
   ```

**Performance notes**

- **Scalability**: `Task.WhenAll` with async I/O can handle thousands of concurrent operations efficiently because I/O doesn't block threads.
- **Memory**: Each task has a small allocation overhead. For very large collections, consider batching.
- **Error aggregation**: `Task.WhenAll` aggregates all exceptions into an `AggregateException`. Use `Task.WhenAll` with individual try-catch if you need per-task error handling.

**Good to know**

- `Task.WhenAll` returns an array of results in the same order as the input tasks.
- For "first wins" scenarios, use `Task.WhenAny` (see **5.8**).

### 3. **Throttling I/O concurrency with SemaphoreSlim**

**Why throttle**

- **Server limits**: APIs/databases have rate limits and connection limits.
- **Resource exhaustion**: Too many concurrent connections can exhaust memory, file handles, or network buffers.
- **Fairness**: Prevent a few clients from monopolizing resources.

**The pattern**

- Use `SemaphoreSlim` to limit the number of concurrent I/O operations.
- Acquire the semaphore before starting I/O, release it after completion (use `try/finally` or `using`).

**Basic throttling**

```csharp
using System;
using System.Collections.Generic;
using System.Linq;
using System.Net.Http;
using System.Threading;
using System.Threading.Tasks;

public static class ThrottledIO
{
    // ✅ GOOD: Limit to 10 concurrent downloads
    public static async Task<string[]> DownloadWithThrottlingAsync(
        IEnumerable<string> urls,
        int maxConcurrency = 10,
        CancellationToken ct = default)
    {
        using var http = new HttpClient();
        using var semaphore = new SemaphoreSlim(maxConcurrency);
        
        var tasks = urls.Select(async url =>
        {
            await semaphore.WaitAsync(ct).ConfigureAwait(false);
            try
            {
                return await DownloadAsync(http, url, ct).ConfigureAwait(false);
            }
            finally
            {
                semaphore.Release();
            }
        }).ToArray();
        
        return await Task.WhenAll(tasks).ConfigureAwait(false);
    }
    
    private static async Task<string> DownloadAsync(
        HttpClient http,
        string url,
        CancellationToken ct)
    {
        var response = await http.GetAsync(url, ct).ConfigureAwait(false);
        response.EnsureSuccessStatusCode();
        return await response.Content.ReadAsStringAsync(ct).ConfigureAwait(false);
    }
}
```

**Reusable throttled executor**

```csharp
using System;
using System.Collections.Generic;
using System.Threading;
using System.Threading.Tasks;

public static class ThrottledExecutor
{
    public static async Task<TResult[]> ExecuteAsync<TItem, TResult>(
        IEnumerable<TItem> items,
        Func<TItem, CancellationToken, Task<TResult>> processor,
        int maxConcurrency,
        CancellationToken ct = default)
    {
        using var semaphore = new SemaphoreSlim(maxConcurrency);
        
        var tasks = items.Select(async item =>
        {
            await semaphore.WaitAsync(ct).ConfigureAwait(false);
            try
            {
                return await processor(item, ct).ConfigureAwait(false);
            }
            finally
            {
                semaphore.Release();
            }
        }).ToArray();
        
        return await Task.WhenAll(tasks).ConfigureAwait(false);
    }
}

// Usage:
var results = await ThrottledExecutor.ExecuteAsync(
    urls,
    async (url, ct) => await httpClient.GetStringAsync(url, ct),
    maxConcurrency: 10,
    ct
);
```

**Dynamic throttling based on load**

```csharp
using System;
using System.Collections.Generic;
using System.Linq;
using System.Net.Http;
using System.Threading;
using System.Threading.Tasks;

public static class AdaptiveThrottling
{
    public static async Task<string[]> DownloadWithAdaptiveThrottlingAsync(
        IEnumerable<string> urls,
        int initialConcurrency = 10,
        CancellationToken ct = default)
    {
        using var http = new HttpClient();
        int currentLimit = initialConcurrency;
        using var semaphore = new SemaphoreSlim(currentLimit);
        
        var tasks = urls.Select(async url =>
        {
            await semaphore.WaitAsync(ct).ConfigureAwait(false);
            try
            {
                var startTime = DateTime.UtcNow;
                var result = await DownloadAsync(http, url, ct).ConfigureAwait(false);
                var duration = DateTime.UtcNow - startTime;
                
                // Adjust concurrency based on response time
                if (duration.TotalSeconds > 5 && currentLimit > 1)
                {
                    Interlocked.CompareExchange(ref currentLimit, currentLimit - 1, currentLimit);
                }
                
                return result;
            }
            finally
            {
                semaphore.Release();
            }
        }).ToArray();
        
        return await Task.WhenAll(tasks).ConfigureAwait(false);
    }
    
    private static async Task<string> DownloadAsync(
        HttpClient http,
        string url,
        CancellationToken ct)
    {
        var response = await http.GetAsync(url, ct).ConfigureAwait(false);
        response.EnsureSuccessStatusCode();
        return await response.Content.ReadAsStringAsync(ct).ConfigureAwait(false);
    }
}
```

**Common mistakes**

1. **Not releasing the semaphore**
   ```csharp
   // ❌ WRONG: Semaphore leak
   await semaphore.WaitAsync();
   await DownloadAsync(url); // If this throws, semaphore never released!
   
   // ✅ CORRECT: Use try/finally
   await semaphore.WaitAsync();
   try
   {
       await DownloadAsync(url);
   }
   finally
   {
       semaphore.Release();
   }
   ```

2. **Blocking on semaphore wait**
   ```csharp
   // ❌ WRONG: Blocks thread
   semaphore.Wait(); // Synchronous wait
   
   // ✅ CORRECT: Use async wait
   await semaphore.WaitAsync(ct);
   ```

3. **Too high or too low concurrency**
   ```csharp
   // ⚠️ CAUTION: Tune based on:
   // - Server rate limits
   // - Network bandwidth
   // - Memory constraints
   // - Measured throughput
   ```

**Performance notes**

- **SemaphoreSlim vs Semaphore**: `SemaphoreSlim` is lighter-weight and async-friendly. Use `Semaphore` only for cross-process scenarios.
- **Optimal concurrency**: Start with `Environment.ProcessorCount * 2` for I/O-bound work, then tune based on measurements. Too high can cause contention; too low wastes resources.
- **Memory**: `SemaphoreSlim` has minimal overhead. The main cost is the queued tasks waiting for the semaphore.

**Good to know**

- For database connections, use connection pooling (e.g., `SqlConnection` pooling) in addition to semaphore throttling.
- Consider using `System.Threading.Channels` with bounded capacity for producer-consumer throttling (see **5.4**).

### 4. **Per-request timeouts and cancellation patterns**

**Why timeouts matter**

- **Hanging requests**: Network issues, slow servers, or bugs can cause requests to hang indefinitely.
- **Resource cleanup**: Timeouts ensure resources (connections, memory) are released promptly.
- **User experience**: Users expect responses within a reasonable time.

**Pattern 1: Per-request timeout with CancellationTokenSource**

```csharp
using System;
using System.Net.Http;
using System.Threading;
using System.Threading.Tasks;

public static class TimeoutPatterns
{
    // ✅ GOOD: Per-request timeout
    public static async Task<string> GetWithTimeoutAsync(
        HttpClient http,
        string url,
        TimeSpan timeout,
        CancellationToken externalCt = default)
    {
        using var cts = CancellationTokenSource.CreateLinkedTokenSource(externalCt);
        cts.CancelAfter(timeout);
        
        try
        {
            return await http.GetStringAsync(url, cts.Token).ConfigureAwait(false);
        }
        catch (OperationCanceledException) when (cts.Token.IsCancellationRequested && !externalCt.IsCancellationRequested)
        {
            throw new TimeoutException($"Request to {url} timed out after {timeout}");
        }
    }
}
```

**Pattern 2: Timeout with fallback**

```csharp
using System;
using System.Net.Http;
using System.Threading;
using System.Threading.Tasks;

public static class TimeoutWithFallback
{
    public static async Task<string> GetWithFallbackAsync(
        HttpClient http,
        string primaryUrl,
        string fallbackUrl,
        TimeSpan timeout,
        CancellationToken ct = default)
    {
        using var cts = CancellationTokenSource.CreateLinkedTokenSource(ct);
        cts.CancelAfter(timeout);
        
        try
        {
            return await http.GetStringAsync(primaryUrl, cts.Token).ConfigureAwait(false);
        }
        catch (OperationCanceledException) when (cts.Token.IsCancellationRequested)
        {
            // Primary timed out, try fallback
            cts.CancelAfter(timeout); // Reset timeout for fallback
            return await http.GetStringAsync(fallbackUrl, cts.Token).ConfigureAwait(false);
        }
    }
}
```

**Pattern 3: Timeout per operation in a batch**

```csharp
using System;
using System.Collections.Generic;
using System.Linq;
using System.Net.Http;
using System.Threading;
using System.Threading.Tasks;

public static class BatchWithTimeouts
{
    public static async Task<(string? Result, bool TimedOut)[]> DownloadWithIndividualTimeoutsAsync(
        IEnumerable<string> urls,
        TimeSpan perRequestTimeout,
        CancellationToken ct = default)
    {
        using var http = new HttpClient();
        
        var tasks = urls.Select(async url =>
        {
            using var cts = CancellationTokenSource.CreateLinkedTokenSource(ct);
            cts.CancelAfter(perRequestTimeout);
            
            try
            {
                var result = await http.GetStringAsync(url, cts.Token).ConfigureAwait(false);
                return (Result: result, TimedOut: false);
            }
            catch (OperationCanceledException) when (cts.Token.IsCancellationRequested)
            {
                return (Result: (string?)null, TimedOut: true);
            }
        }).ToArray();
        
        return await Task.WhenAll(tasks).ConfigureAwait(false);
    }
}
```

**Pattern 4: HttpClient timeout vs per-request timeout**

```csharp
using System;
using System.Net.Http;
using System.Threading;
using System.Threading.Tasks;

public static class HttpClientTimeouts
{
    // HttpClient.Timeout applies to all requests
    // Per-request timeout overrides it
    public static async Task<string> GetWithBothTimeoutsAsync(
        HttpClient http, // Assume http.Timeout is set
        string url,
        TimeSpan? perRequestTimeout = null,
        CancellationToken ct = default)
    {
        if (perRequestTimeout.HasValue)
        {
            using var cts = CancellationTokenSource.CreateLinkedTokenSource(ct);
            cts.CancelAfter(perRequestTimeout.Value);
            return await http.GetStringAsync(url, cts.Token).ConfigureAwait(false);
        }
        
        return await http.GetStringAsync(url, ct).ConfigureAwait(false);
    }
}
```

**Common mistakes**

1. **Not linking cancellation tokens**
   ```csharp
   // ❌ WRONG: Ignores external cancellation
   using var cts = new CancellationTokenSource();
   cts.CancelAfter(timeout);
   await http.GetStringAsync(url, cts.Token); // Ignores ct parameter!
   
   // ✅ CORRECT: Link tokens
   using var cts = CancellationTokenSource.CreateLinkedTokenSource(externalCt);
   cts.CancelAfter(timeout);
   await http.GetStringAsync(url, cts.Token);
   ```

2. **Not disposing CancellationTokenSource**
   ```csharp
   // ❌ WRONG: Resource leak
   var cts = new CancellationTokenSource();
   cts.CancelAfter(timeout);
   
   // ✅ CORRECT: Use using
   using var cts = CancellationTokenSource.CreateLinkedTokenSource(ct);
   ```

3. **Swallowing timeout exceptions**
   ```csharp
   // ⚠️ CAUTION: Distinguish timeout from user cancellation
   catch (OperationCanceledException ex)
   {
       if (cts.Token.IsCancellationRequested && !externalCt.IsCancellationRequested)
       {
           // This was a timeout
       }
       else
       {
           // This was user cancellation, rethrow
           throw;
       }
   }
   ```

**Performance notes**

- **CancellationTokenSource allocation**: Creating a `CancellationTokenSource` per request has minimal overhead. For high-throughput scenarios, consider reusing timeouts or using `HttpClient.Timeout`.
- **Cancellation propagation**: Cancellation tokens propagate through async calls efficiently. The cost is negligible compared to I/O latency.

**Good to know**

- `HttpClient.Timeout` applies to the entire request (including connection establishment). Per-request timeouts give more control.
- For database operations, use `CommandTimeout` in addition to `CancellationToken`:
  ```csharp
  command.CommandTimeout = (int)timeout.TotalSeconds;
  await command.ExecuteReaderAsync(ct);
  ```

### 5. **Retries, backoff, and circuit breakers in concurrent I/O flows**

**Why retries matter**

- **Transient failures**: Network glitches, temporary server errors (5xx), timeouts.
- **Resilience**: Automatic retries improve success rates without user intervention.
- **Backoff**: Prevents overwhelming failing services.

**Pattern 1: Simple retry with exponential backoff**

```csharp
using System;
using System.Net.Http;
using System.Threading;
using System.Threading.Tasks;

public static class RetryPatterns
{
    public static async Task<string> GetWithRetryAsync(
        HttpClient http,
        string url,
        int maxRetries = 3,
        TimeSpan? initialDelay = null,
        CancellationToken ct = default)
    {
        initialDelay ??= TimeSpan.FromSeconds(1);
        Exception? lastException = null;
        
        for (int attempt = 0; attempt <= maxRetries; attempt++)
        {
            try
            {
                return await http.GetStringAsync(url, ct).ConfigureAwait(false);
            }
            catch (HttpRequestException ex) when (IsTransientError(ex))
            {
                lastException = ex;
                if (attempt < maxRetries)
                {
                    var delay = TimeSpan.FromMilliseconds(
                        initialDelay.Value.TotalMilliseconds * Math.Pow(2, attempt)
                    );
                    await Task.Delay(delay, ct).ConfigureAwait(false);
                }
            }
        }
        
        throw new HttpRequestException($"Failed after {maxRetries + 1} attempts", lastException);
    }
    
    private static bool IsTransientError(HttpRequestException ex)
    {
        // Retry on network errors, timeouts, 5xx errors
        // Don't retry on 4xx (client errors) except 429 (rate limit)
        return true; // Simplified; check status code in real implementation
    }
}
```

**Pattern 2: Retry with jitter (prevent thundering herd)**

```csharp
using System;
using System.Net.Http;
using System.Threading;
using System.Threading.Tasks;

public static class RetryWithJitter
{
    private static readonly Random Random = new();
    
    public static async Task<string> GetWithRetryAndJitterAsync(
        HttpClient http,
        string url,
        int maxRetries = 3,
        TimeSpan initialDelay = default,
        CancellationToken ct = default)
    {
        if (initialDelay == default)
            initialDelay = TimeSpan.FromSeconds(1);
        
        Exception? lastException = null;
        
        for (int attempt = 0; attempt <= maxRetries; attempt++)
        {
            try
            {
                return await http.GetStringAsync(url, ct).ConfigureAwait(false);
            }
            catch (HttpRequestException ex) when (IsTransientError(ex))
            {
                lastException = ex;
                if (attempt < maxRetries)
                {
                    // Exponential backoff with jitter
                    var baseDelay = initialDelay.TotalMilliseconds * Math.Pow(2, attempt);
                    var jitter = Random.Next(0, (int)(baseDelay * 0.1)); // 10% jitter
                    var delay = TimeSpan.FromMilliseconds(baseDelay + jitter);
                    
                    await Task.Delay(delay, ct).ConfigureAwait(false);
                }
            }
        }
        
        throw new HttpRequestException($"Failed after {maxRetries + 1} attempts", lastException);
    }
    
    private static bool IsTransientError(HttpRequestException ex) => true;
}
```

**Pattern 3: Circuit breaker**

```csharp
using System;
using System.Net.Http;
using System.Threading;
using System.Threading.Tasks;

public class CircuitBreaker
{
    private readonly int _failureThreshold;
    private readonly TimeSpan _openDuration;
    private int _failureCount;
    private DateTime _lastFailureTime;
    private bool _isOpen;
    private readonly object _lock = new();
    
    public CircuitBreaker(int failureThreshold = 5, TimeSpan? openDuration = null)
    {
        _failureThreshold = failureThreshold;
        _openDuration = openDuration ?? TimeSpan.FromSeconds(60);
    }
    
    public async Task<T> ExecuteAsync<T>(Func<Task<T>> operation)
    {
        if (IsOpen())
        {
            throw new InvalidOperationException("Circuit breaker is open");
        }
        
        try
        {
            var result = await operation().ConfigureAwait(false);
            OnSuccess();
            return result;
        }
        catch (Exception ex)
        {
            OnFailure();
            throw;
        }
    }
    
    private bool IsOpen()
    {
        lock (_lock)
        {
            if (!_isOpen)
                return false;
            
            // Check if we should attempt to close (half-open state)
            if (DateTime.UtcNow - _lastFailureTime > _openDuration)
            {
                _isOpen = false;
                _failureCount = 0;
                return false;
            }
            
            return true;
        }
    }
    
    private void OnSuccess()
    {
        lock (_lock)
        {
            _failureCount = 0;
            _isOpen = false;
        }
    }
    
    private void OnFailure()
    {
        lock (_lock)
        {
            _failureCount++;
            _lastFailureTime = DateTime.UtcNow;
            
            if (_failureCount >= _failureThreshold)
            {
                _isOpen = true;
            }
        }
    }
}

// Usage:
var circuitBreaker = new CircuitBreaker(failureThreshold: 5, openDuration: TimeSpan.FromSeconds(60));

try
{
    var result = await circuitBreaker.ExecuteAsync(() => httpClient.GetStringAsync(url));
}
catch (InvalidOperationException)
{
    // Circuit breaker is open, skip the call
}
```

**Pattern 4: Retry with circuit breaker**

```csharp
using System;
using System.Net.Http;
using System.Threading;
using System.Threading.Tasks;

public static class RetryWithCircuitBreaker
{
    public static async Task<string> GetWithRetryAndCircuitBreakerAsync(
        HttpClient http,
        CircuitBreaker circuitBreaker,
        string url,
        int maxRetries = 3,
        CancellationToken ct = default)
    {
        Exception? lastException = null;
        
        for (int attempt = 0; attempt <= maxRetries; attempt++)
        {
            try
            {
                return await circuitBreaker.ExecuteAsync(
                    () => httpClient.GetStringAsync(url, ct)
                ).ConfigureAwait(false);
            }
            catch (InvalidOperationException)
            {
                // Circuit breaker is open, fail fast
                throw new HttpRequestException("Circuit breaker is open");
            }
            catch (HttpRequestException ex) when (IsTransientError(ex))
            {
                lastException = ex;
                if (attempt < maxRetries)
                {
                    await Task.Delay(TimeSpan.FromSeconds(Math.Pow(2, attempt)), ct)
                        .ConfigureAwait(false);
                }
            }
        }
        
        throw new HttpRequestException($"Failed after {maxRetries + 1} attempts", lastException);
    }
    
    private static bool IsTransientError(HttpRequestException ex) => true;
}
```

**Common mistakes**

1. **Retrying non-transient errors**
   ```csharp
   // ❌ WRONG: Don't retry 404 (Not Found) or 400 (Bad Request)
   catch (HttpRequestException ex)
   {
       // Retries everything, including client errors
   }
   
   // ✅ CORRECT: Only retry transient errors
   catch (HttpRequestException ex) when (IsTransientError(ex))
   {
       // Retry logic
   }
   ```

2. **No backoff (thundering herd)**
   ```csharp
   // ❌ WRONG: All retries happen immediately
   catch (Exception)
   {
       await Task.Delay(TimeSpan.FromSeconds(1)); // Same delay every time
   }
   
   // ✅ CORRECT: Exponential backoff with jitter
   ```

3. **Not respecting cancellation**
   ```csharp
   // ❌ WRONG: Ignores cancellation during retry delay
   await Task.Delay(delay); // No ct parameter
   
   // ✅ CORRECT: Respect cancellation
   await Task.Delay(delay, ct);
   ```

**Performance notes**

- **Retry overhead**: Retries add latency. Use retries only for transient failures, not for bugs or configuration errors.
- **Circuit breaker**: Prevents cascading failures and reduces load on failing services. The overhead is minimal (a few lock operations).
- **Jitter**: Prevents synchronized retries from multiple clients, which can cause "thundering herd" problems.

**Good to know**

- Consider using **Polly** (a popular .NET resilience library) for production retry, circuit breaker, and timeout policies:
  ```csharp
  var policy = Policy
      .Handle<HttpRequestException>()
      .WaitAndRetryAsync(3, retryAttempt => 
          TimeSpan.FromSeconds(Math.Pow(2, retryAttempt)));
  
  await policy.ExecuteAsync(() => httpClient.GetStringAsync(url));
  ```
- For concurrent I/O, ensure retry logic is thread-safe if shared across requests.
- Log retry attempts for observability and debugging.
