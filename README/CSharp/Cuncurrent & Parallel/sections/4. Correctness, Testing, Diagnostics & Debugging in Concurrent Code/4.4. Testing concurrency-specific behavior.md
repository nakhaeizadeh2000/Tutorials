### 1. **Stress testing and randomization of schedules**

**What is stress testing?**

- **Stress testing** runs code with **high concurrency** and **many iterations** to expose race conditions and timing bugs.
- The goal is to **amplify** the probability of seeing bugs that only occur under specific interleavings.

**Basic stress test pattern**

```csharp
using System;
using System.Collections.Concurrent;
using System.Linq;
using System.Threading;
using System.Threading.Tasks;
using Xunit;

public class StressTests
{
    [Fact]
    public void Counter_StressTest()
    {
        const int iterations = 10000;
        const int threads = 10;
        var counter = new Counter();
        var exceptions = new ConcurrentBag<Exception>();

        Parallel.For(0, threads, threadId =>
        {
            try
            {
                for (int i = 0; i < iterations; i++)
                {
                    counter.Increment();
                    counter.Decrement();
                }
            }
            catch (Exception ex)
            {
                exceptions.Add(ex);
            }
        });

        Assert.Empty(exceptions);
        Assert.Equal(0, counter.Value); // Should be 0 after equal increments/decrements
    }
}

public class Counter
{
    private int _value;

    public void Increment() => Interlocked.Increment(ref _value);
    public void Decrement() => Interlocked.Decrement(ref _value);
    public int Value => _value;
}
```

- Run the test with **many threads** and **many iterations**.
- Verify that invariants hold (e.g., counter should be 0 after equal increments/decrements).

**Randomizing schedules**

- **Random delays** can expose different interleavings.
- Use `Random` with controlled seeds for reproducibility.

```csharp
[Fact]
public void RaceCondition_StressTestWithRandomDelays()
{
    const int iterations = 1000;
    var random = new Random(42); // Fixed seed for reproducibility
    var sharedState = new SharedState();
    var errors = new ConcurrentBag<string>();

    Parallel.For(0, iterations, i =>
    {
        try
        {
            // Random delay to vary interleavings
            Thread.Sleep(random.Next(0, 10));
            
            sharedState.Update(i);
            var value = sharedState.Read();
            
            // Verify invariant
            if (value < 0)
            {
                errors.Add($"Invalid value: {value}");
            }
        }
        catch (Exception ex)
        {
            errors.Add(ex.Message);
        }
    });

    Assert.Empty(errors);
}

public class SharedState
{
    private int _value;
    private readonly object _lock = new();

    public void Update(int newValue)
    {
        lock (_lock)
        {
            _value = newValue;
        }
    }

    public int Read()
    {
        lock (_lock)
        {
            return _value;
        }
    }
}
```

**Testing with varying thread counts**

```csharp
[Theory]
[InlineData(1)]
[InlineData(2)]
[InlineData(4)]
[InlineData(8)]
[InlineData(16)]
public void ConcurrentDictionary_StressTestWithVaryingThreads(int threadCount)
{
    const int iterationsPerThread = 1000;
    var dict = new ConcurrentDictionary<int, int>();
    var exceptions = new ConcurrentBag<Exception>();

    Parallel.For(0, threadCount, new ParallelOptions
    {
        MaxDegreeOfParallelism = threadCount
    }, threadId =>
    {
        try
        {
            for (int i = 0; i < iterationsPerThread; i++)
            {
                var key = threadId * iterationsPerThread + i;
                dict.TryAdd(key, key * 2);
            }
        }
        catch (Exception ex)
        {
            exceptions.Add(ex);
        }
    });

    Assert.Empty(exceptions);
    Assert.Equal(threadCount * iterationsPerThread, dict.Count);
}
```

- Test with **different thread counts** to expose bugs that only occur with specific parallelism levels.

**Testing for memory leaks**

```csharp
[Fact]
public void EventHandler_StressTestForLeaks()
{
    const int iterations = 10000;
    var leaks = new List<WeakReference>();

    for (int i = 0; i < iterations; i++)
    {
        var publisher = new EventPublisher();
        var subscriber = new EventSubscriber();
        
        publisher.Event += subscriber.Handle;
        leaks.Add(new WeakReference(subscriber));
        
        publisher.Event -= subscriber.Handle;
        publisher = null;
    }

    GC.Collect();
    GC.WaitForPendingFinalizers();
    GC.Collect();

    var aliveCount = leaks.Count(wr => wr.IsAlive);
    Assert.True(aliveCount < iterations * 0.01, 
        $"Memory leak detected: {aliveCount} objects still alive");
}
```

- Use `WeakReference` to detect memory leaks in long-running stress tests.

**Mentor emphasis**

- **Run stress tests** with high concurrency and many iterations.
- **Use fixed random seeds** for reproducibility.
- **Test with varying thread counts** to expose parallelism-specific bugs.
- **Verify invariants** hold under stress (not just that no exceptions occur).

### 2. **Injecting delays, yields, and context switches to expose races**

**Why inject delays?**

- **Delays** can force specific interleavings that expose race conditions.
- **Yields** can cause context switches at critical points.
- **Controlled delays** make tests deterministic while exposing timing bugs.

**Injecting delays at critical points**

```csharp
public interface IDelayInjector
{
    void InjectDelay();
    Task InjectDelayAsync();
}

public class TestDelayInjector : IDelayInjector
{
    private readonly List<int> _delayPoints = new();
    private int _callCount = 0;

    public void InjectDelay()
    {
        _callCount++;
        if (_delayPoints.Contains(_callCount))
        {
            Thread.Sleep(10); // Force delay at specific points
        }
    }

    public Task InjectDelayAsync()
    {
        InjectDelay();
        return Task.CompletedTask;
    }

    public void AddDelayPoint(int point) => _delayPoints.Add(point);
    public void Clear() => _delayPoints.Clear();
}

public class RaceConditionExample
{
    private readonly IDelayInjector _delayInjector;
    private int _value;
    private bool _initialized;

    public RaceConditionExample(IDelayInjector delayInjector)
    {
        _delayInjector = delayInjector;
    }

    public void Initialize(int value)
    {
        _delayInjector.InjectDelay(); // Point 1
        _value = value;
        _delayInjector.InjectDelay(); // Point 2
        _initialized = true;
    }

    public int Read()
    {
        if (!_initialized)
            throw new InvalidOperationException("Not initialized");
        
        return _value;
    }
}
```

**Testing with injected delays**

```csharp
[Fact]
public void RaceCondition_ExposedByDelays()
{
    var delayInjector = new TestDelayInjector();
    var example = new RaceConditionExample(delayInjector);
    var errors = new ConcurrentBag<Exception>();

    // Thread 1: Initialize
    var t1 = Task.Run(() =>
    {
        delayInjector.AddDelayPoint(2); // Delay after setting _value, before _initialized
        example.Initialize(42);
    });

    // Thread 2: Read (might see _initialized=true but _value=0)
    var t2 = Task.Run(() =>
    {
        Thread.Sleep(5); // Give thread 1 a head start
        try
        {
            var value = example.Read();
            if (value != 42)
            {
                errors.Add(new Exception($"Read wrong value: {value}"));
            }
        }
        catch (Exception ex)
        {
            errors.Add(ex);
        }
    });

    Task.WaitAll(t1, t2);
    Assert.Empty(errors); // Should catch the race condition
}
```

**Using Task.Yield to force context switches**

```csharp
public class YieldInjector
{
    private readonly List<int> _yieldPoints = new();
    private int _callCount = 0;

    public async Task YieldIfNeeded()
    {
        _callCount++;
        if (_yieldPoints.Contains(_callCount))
        {
            await Task.Yield(); // Force context switch
        }
    }

    public void AddYieldPoint(int point) => _yieldPoints.Add(point);
}

[Fact]
public async Task AsyncRaceCondition_ExposedByYields()
{
    var yieldInjector = new YieldInjector();
    var service = new AsyncService(yieldInjector);
    var results = new ConcurrentBag<int>();

    yieldInjector.AddYieldPoint(1); // Yield at critical point

    var tasks = Enumerable.Range(0, 10)
        .Select(async i =>
        {
            var result = await service.ProcessAsync(i);
            results.Add(result);
        });

    await Task.WhenAll(tasks);
    
    Assert.Equal(10, results.Count);
    // Verify no duplicates or missing values
    Assert.Equal(10, results.Distinct().Count());
}
```

**Injecting delays in async code**

```csharp
public interface IAsyncDelayInjector
{
    Task DelayAsync(TimeSpan delay, CancellationToken cancellationToken = default);
}

public class TestAsyncDelayInjector : IAsyncDelayInjector
{
    private readonly List<(int CallIndex, TimeSpan Delay)> _delays = new();
    private int _callCount = 0;

    public Task DelayAsync(TimeSpan delay, CancellationToken cancellationToken = default)
    {
        _callCount++;
        var actualDelay = _delays
            .Where(d => d.CallIndex == _callCount)
            .Select(d => d.Delay)
            .FirstOrDefault(delay);
        
        return Task.Delay(actualDelay, cancellationToken);
    }

    public void AddDelay(int callIndex, TimeSpan delay)
        => _delays.Add((callIndex, delay));
}

public class AsyncService
{
    private readonly IAsyncDelayInjector _delayInjector;
    private int _counter;

    public AsyncService(IAsyncDelayInjector delayInjector)
    {
        _delayInjector = delayInjector;
    }

    public async Task<int> ProcessAsync(int value)
    {
        await _delayInjector.DelayAsync(TimeSpan.FromMilliseconds(1));
        var result = Interlocked.Increment(ref _counter);
        await _delayInjector.DelayAsync(TimeSpan.FromMilliseconds(1));
        return result;
    }
}
```

**Mentor emphasis**

- **Inject delays** at critical points to force specific interleavings.
- **Use Task.Yield** to force context switches in async code.
- **Make delays controllable** so tests are deterministic.
- **Test different delay patterns** to expose various race conditions.

### 3. **Using custom schedulers/executors in tests**

**Why custom schedulers?**

- **Custom schedulers** allow you to control **when** and **where** tasks execute.
- Useful for:
  - **Deterministic testing** (control execution order).
  - **Testing synchronization contexts** (see 2.3).
  - **Injecting failures** at specific points.

**Creating a test scheduler**

```csharp
using System;
using System.Collections.Generic;
using System.Threading;
using System.Threading.Tasks;

public class TestTaskScheduler : TaskScheduler
{
    private readonly Queue<Task> _tasks = new();
    private readonly object _lock = new();

    protected override void QueueTask(Task task)
    {
        lock (_lock)
        {
            _tasks.Enqueue(task);
        }
    }

    protected override bool TryExecuteTaskInline(Task task, bool taskWasPreviouslyQueued)
    {
        return false; // Never inline
    }

    protected override IEnumerable<Task> GetScheduledTasks()
    {
        lock (_lock)
        {
            return _tasks.ToArray();
        }
    }

    public void ExecuteAll()
    {
        while (true)
        {
            Task? task;
            lock (_lock)
            {
                if (_tasks.Count == 0) break;
                task = _tasks.Dequeue();
            }
            
            TryExecuteTask(task);
        }
    }

    public void ExecuteOne()
    {
        Task? task;
        lock (_lock)
        {
            if (_tasks.Count == 0) return;
            task = _tasks.Dequeue();
        }
        
        TryExecuteTask(task);
    }
}
```

**Testing with custom scheduler**

```csharp
[Fact]
public void TaskScheduler_DeterministicExecution()
{
    var scheduler = new TestTaskScheduler();
    var results = new List<int>();

    var task1 = Task.Factory.StartNew(() => results.Add(1), 
        CancellationToken.None, TaskCreationOptions.None, scheduler);
    var task2 = Task.Factory.StartNew(() => results.Add(2), 
        CancellationToken.None, TaskCreationOptions.None, scheduler);
    var task3 = Task.Factory.StartNew(() => results.Add(3), 
        CancellationToken.None, TaskCreationOptions.None, scheduler);

    // Execute tasks in specific order
    scheduler.ExecuteOne(); // Execute task1
    scheduler.ExecuteOne(); // Execute task2
    scheduler.ExecuteOne(); // Execute task3

    Assert.Equal(new[] { 1, 2, 3 }, results);
}
```

**Testing SynchronizationContext**

```csharp
public class TestSynchronizationContext : SynchronizationContext
{
    private readonly Queue<Action> _pending = new();
    private readonly object _lock = new();

    public override void Post(SendOrPostCallback d, object? state)
    {
        lock (_lock)
        {
            _pending.Enqueue(() => d(state));
        }
    }

    public override void Send(SendOrPostCallback d, object? state)
    {
        d(state); // Execute synchronously
    }

    public void ExecutePending()
    {
        while (true)
        {
            Action? action;
            lock (_lock)
            {
                if (_pending.Count == 0) break;
                action = _pending.Dequeue();
            }
            action();
        }
    }
}

[Fact]
public async Task SynchronizationContext_ControlledExecution()
{
    var context = new TestSynchronizationContext();
    SynchronizationContext.SetSynchronizationContext(context);
    
    var results = new List<int>();

    async Task AsyncMethod()
    {
        await Task.Delay(1); // This will post to SynchronizationContext
        results.Add(1);
    }

    var task = AsyncMethod();
    
    // Execute pending work on the context
    context.ExecutePending();
    
    await task;
    Assert.Contains(1, results);
}
```

**Testing with custom TaskFactory**

```csharp
[Fact]
public void CustomTaskFactory_ControlledExecution()
{
    var scheduler = new TestTaskScheduler();
    var factory = new TaskFactory(scheduler);
    var executionOrder = new List<int>();

    var task1 = factory.StartNew(() => executionOrder.Add(1));
    var task2 = factory.StartNew(() => executionOrder.Add(2));
    var task3 = factory.StartNew(() => executionOrder.Add(3));

    // Control execution order
    scheduler.ExecuteOne(); // Execute 1
    scheduler.ExecuteOne(); // Execute 2
    scheduler.ExecuteOne(); // Execute 3

    Task.WaitAll(task1, task2, task3);
    Assert.Equal(new[] { 1, 2, 3 }, executionOrder);
}
```

**Mentor emphasis**

- **Custom schedulers** enable deterministic testing of concurrent code.
- **Control execution order** to test specific interleavings.
- **Test SynchronizationContext behavior** with custom contexts.
- **Use TaskFactory** with custom schedulers for fine-grained control.

### 4. **Simulating failures and cancellations in async flows**

**Why simulate failures?**

- **Failures** (exceptions, cancellations) are common in async code.
- Testing failure paths ensures **resilience** and **proper error handling**.
- Many bugs only appear when operations fail at specific points.

**Simulating exceptions**

```csharp
public interface IUnreliableService
{
    Task<string> GetDataAsync(CancellationToken cancellationToken = default);
    void SetFailureRate(double rate);
    void SetFailurePoint(int point);
}

public class UnreliableService : IUnreliableService
{
    private readonly IReliableService _reliableService;
    private double _failureRate = 0.0;
    private int _callCount = 0;
    private readonly HashSet<int> _failurePoints = new();
    private readonly Random _random = new(42);

    public UnreliableService(IReliableService reliableService)
    {
        _reliableService = reliableService;
    }

    public async Task<string> GetDataAsync(CancellationToken cancellationToken = default)
    {
        _callCount++;
        
        // Check for failure point
        if (_failurePoints.Contains(_callCount))
        {
            throw new InvalidOperationException($"Simulated failure at call {_callCount}");
        }

        // Check for random failure
        if (_random.NextDouble() < _failureRate)
        {
            throw new InvalidOperationException("Simulated random failure");
        }

        return await _reliableService.GetDataAsync(cancellationToken);
    }

    public void SetFailureRate(double rate) => _failureRate = rate;
    public void SetFailurePoint(int point) => _failurePoints.Add(point);
}

[Fact]
public async Task RetryLogic_HandlesFailures()
{
    var reliableService = new ReliableService();
    var unreliableService = new UnreliableService(reliableService);
    unreliableService.SetFailurePoint(1); // Fail on first call

    var service = new RetryService(unreliableService);
    
    // Should retry and succeed on second attempt
    var result = await service.GetDataWithRetryAsync(maxRetries: 3);
    
    Assert.NotNull(result);
}

public class RetryService
{
    private readonly IUnreliableService _service;

    public RetryService(IUnreliableService service)
    {
        _service = service;
    }

    public async Task<string> GetDataWithRetryAsync(int maxRetries)
    {
        for (int i = 0; i < maxRetries; i++)
        {
            try
            {
                return await _service.GetDataAsync();
            }
            catch (Exception)
            {
                if (i == maxRetries - 1) throw;
                await Task.Delay(10); // Brief delay before retry
            }
        }
        throw new InvalidOperationException("Max retries exceeded");
    }
}
```

### 5. **Property-based testing for concurrent invariants**

**What is property-based testing?**

- **Property-based testing** generates random inputs and verifies that **invariants** hold.
- Instead of testing specific cases, you test **properties** that should always be true.
- Libraries like **FsCheck** (F#) or **FsCheck.NET** (C#) enable property-based testing.

**Basic property-based test**

```csharp
using FsCheck;
using FsCheck.Xunit;

public class PropertyBasedTests
{
    [Property]
    public bool Counter_IncrementDecrement_IsIdempotent(int value)
    {
        var counter = new Counter();
        
        for (int i = 0; i < value; i++)
        {
            counter.Increment();
        }
        
        for (int i = 0; i < value; i++)
        {
            counter.Decrement();
        }
        
        return counter.Value == 0; // Property: equal increments/decrements result in 0
    }
}
```

**Testing concurrent invariants**

```csharp
[Property]
public bool ConcurrentDictionary_AddRemove_IsConsistent(int[] keys, int[] values)
{
    if (keys.Length != values.Length) return true; // Skip invalid inputs
    
    var dict = new ConcurrentDictionary<int, int>();
    var operations = new List<(string Op, int Key, int Value)>();

    // Add all key-value pairs
    for (int i = 0; i < keys.Length; i++)
    {
        dict.TryAdd(keys[i], values[i]);
        operations.Add(("Add", keys[i], values[i]));
    }

    // Remove all keys
    foreach (var key in keys)
    {
        dict.TryRemove(key, out _);
        operations.Add(("Remove", key, 0));
    }

    // Property: after removing all added keys, dictionary should be empty
    return dict.IsEmpty;
}
```

**Testing ordering invariants**

```csharp
[Property]
public bool PriorityQueue_MaintainsOrder(int[] values)
{
    var queue = new ConcurrentPriorityQueue<int>();
    
    // Add all values
    foreach (var value in values)
    {
        queue.Enqueue(value);
    }

    // Remove all values and verify they come out in order
    var removed = new List<int>();
    while (queue.TryDequeue(out var value))
    {
        removed.Add(value);
    }

    // Property: removed values should be sorted
    var sorted = values.OrderBy(x => x).ToArray();
    return removed.SequenceEqual(sorted);
}
```

**Testing with concurrent operations**

```csharp
[Property(MaxTest = 100)] // Limit tests for concurrent code
public bool ConcurrentCounter_StressTest(int threadCount, int iterationsPerThread)
{
    if (threadCount < 1 || threadCount > 100) return true; // Skip extreme values
    if (iterationsPerThread < 1 || iterationsPerThread > 1000) return true;

    var counter = new Counter();
    var exceptions = new ConcurrentBag<Exception>();

    Parallel.For(0, threadCount, threadId =>
    {
        try
        {
            for (int i = 0; i < iterationsPerThread; i++)
            {
                counter.Increment();
            }
        }
        catch (Exception ex)
        {
            exceptions.Add(ex);
        }
    });

    // Property: counter should equal total increments
    var expected = threadCount * iterationsPerThread;
    return exceptions.IsEmpty && counter.Value == expected;
}
```
