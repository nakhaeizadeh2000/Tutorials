## 7.1. Real production bugs caused by array mistakes

### 1) **Sparse pagination and `length` truncation bugs**

**Bug** — API paginated `GET /items?offset=…&limit=20` stitched via `arr[offset] = page` without compaction, leaving holes; later `arr.map(render)` skipped holes, rendering missing rows silently.

```js
// Buggy stitching — creates holes
function buggyStitch(pages) { // pages: [[a,b], [c,d]] at offsets 0,20 but naively at 0,1?
  const all = [];
  pages.forEach((page, pi) => { all[pi * 20] = page; }); // actually assigns array as element, but demo hole:
  return all;
}
// More canonical hole bug: pre-size then fill partially
const buggy = new Array(5); // 5 holes
[10, 20].forEach((v, i) => buggy[i] = v);
console.log(buggy); // [10,20, <3 empty>] — holes at 2,3,4
console.log(buggy.map(x => x * 2)); // [20,40, <3 empty>] — holes preserved, downstream length 5 but only 2 rendered
console.log(buggy.filter(x => x != null)); // [10,20] — filter densifies but length changes unexpectedly
console.log(JSON.stringify(buggy)); // [10,20,null,null,null] — JSON densifies holes to null, wire length 5 but last 3 are null!

// Fixed — dense stitching
function goodStitch(pages) {
  return pages.flat(); // or Array.from({length: total}, (_,i)=> fetch)
}
const good = [10, 20];
console.log(good.map(x => x * 2)); // [20,40] — dense, correct length

// Diagnostics: detect holes before shipping
function assertDense(arr, label) {
  for (let i = 0; i < arr.length; i++) if (!Object.hasOwn(arr, i)) throw new Error(`${label}: hole at ${i}`);
}
try { assertDense(buggy, "buggy"); } catch (e) { console.log(e.message); } // buggy: hole at 2
assertDense(good, "good");

// Normalizing sparse input — Array.from densifies
const normalized = Array.from(buggy, x => x ?? 0);
console.log(normalized); // [10,20,0,0,0] — explicit fill
```

### 2) **`sort()` without `compareFn` ships lexical order to prod**

**Bug** — price leaderboard `prices.sort()` without comparator; `["9","10","20"]` lexical sorts as `["10","20","9"]` — tests with single-digit fixtures pass, prod with double-digits fails.

```js
// Buggy — lexical
const prices = [9, 10, 20, 100];
console.log(prices.slice().sort()); // [10,100,20,9] — lexical!
console.log(prices.slice().sort((a, b) => a - b)); // [9,10,20,100] — numeric

// Real incident — leaderboard
const leaderboard = [{ name: "b", score: 20 }, { name: "a", score: 100 }, { name: "c", score: 9 }];
leaderboard.sort((a, b) => a.score - b.score > 0 ? 1 : -1); // boolean-ish comparator also wrong — need negative
console.log(leaderboard.map(p => p.score)); // [9,20,100] — but comparator above not stable; correct below
console.log([{ score: 9 }, { score: 100 }, { score: 20 }].toSorted((a, b) => a.score - b.score).map(p => p.score)); // [9,20,100]

// Guard — enforce compareFn via grep and unit test with multi-digit fixtures
function sortedNumeric(arr) { return arr.toSorted((a, b) => a - b); }
console.log(sortedNumeric([10, 9, 100, 20])); // [9,10,20,100]

// Object sort — lexical stringify is useless
const items = [{ v: 10 }, { v: 2 }];
console.log(items.slice().sort()); // [object Object] vs same → no order (stable keeps input)
console.log(items.toSorted((a, b) => a.v - b.v).map(i => i.v)); // [2,10]
```

### 3) **Mutating a shared array under state management**

**Bug** — Redux/React reducer does `state.items.sort()` or `state.items.splice()` → mutates previous state, React bails out (`nextState.items === prevState.items`), UI doesn't re-render; time-travel shows corrupted history.

```js
// Buggy reducer — mutates
function buggyReducer(state, action) {
  if (action.type === "SORT") {
    state.items.sort((a, b) => a - b); // mutates state.items alias
    return { ...state }; // shallow copy but items still same array reference
  }
  return state;
}
let state = { items: [3, 1, 2] };
const prevItems = state.items;
state = buggyReducer(state, { type: "SORT" });
console.log(prevItems); // [1,2,3] — corrupted! previous state's items mutated
console.log(prevItems === state.items); // true — same reference, React would bail

// Fixed — copy-era returns new array, reference changes
function goodReducer(state, action) {
  if (action.type === "SORT") {
    return { ...state, items: state.items.toSorted((a, b) => a - b) };
  }
  if (action.type === "ADD") {
    return { ...state, items: state.items.toSpliced(state.items.length, 0, action.item) }; // immutable append
  }
  return state;
}
let state2 = { items: [3, 1, 2] };
const prev2 = state2.items;
state2 = goodReducer(state2, { type: "SORT" });
console.log(prev2); // [3,1,2] — preserved
console.log(prev2 !== state2.items); // true — new reference triggers render
console.log(state2.items); // [1,2,3]

// Shallow still leaks nested — copy-era is shallow too
const nestedState = { rows: [{ x: 1 }, { x: 2 }] };
const nextNested = { ...nestedState, rows: nestedState.rows.toReversed() };
nextNested.rows[0].x = 99;
console.log(nestedState.rows[1].x); // 99 — nested alias leaked (see §3.2/Objects 5.1)
// Deep copy when mutating nested:
function deepCopyRows(state) { return { ...state, rows: state.rows.map(r => ({ ...r })) }; }
```

### 4) **Splice-inside-loop index-shift skip**

**Bug** — `for (i=0; i<arr.length; i++) if (pred(arr[i])) arr.splice(i,1)` skips the element that shifts into `i` after splice.

```js
// Buggy forward splice — skips 3
const buggyLoop = [1, 2, 3, 4];
for (let i = 0; i < buggyLoop.length; i++) {
  if (buggyLoop[i] % 2 === 0) buggyLoop.splice(i, 1);
}
console.log(buggyLoop); // [1,3,4] — 4 skipped (shifted to i=1 after removing 2)

// Fixes — reverse iteration or filter
const fixReverse = [1, 2, 3, 4];
for (let i = fixReverse.length - 1; i >= 0; i--) if (fixReverse[i] % 2 === 0) fixReverse.splice(i, 1);
console.log(fixReverse); // [1,3] — correct

const fixFilter = [1, 2, 3, 4].filter(x => x % 2 !== 0);
console.log(fixFilter); // [1,3] — idiomatic, non-mutating

// Also buggy with holes — delete creates hole, length unchanged, loop visits hole as undefined
const deleteBug = [1, 2, 3];
delete deleteBug[1];
console.log(deleteBug, deleteBug.length); // [1, <hole>, 3] 3
for (let i = 0; i < deleteBug.length; i++) console.log(i, deleteBug[i]); // visits hole as undefined
```

### 5) **Fill-aliased nested arrays**

**Bug** — `Array(3).fill([])` shares one inner array; pushing to one pushes to all — passes unit tests with read-only checks, fails on first mutation in prod.

```js
// Buggy
const buggyFill = Array(3).fill([]);
buggyFill[0].push(1);
console.log(buggyFill); // [[1],[1],[1]] — all share same array!

// Fixed
const goodFill = Array.from({ length: 3 }, () => []);
goodFill[0].push(1);
console.log(goodFill); // [[1], [], []] — distinct

// Same for fill with objects
const badObjs = Array(2).fill({ x: 1 });
badObjs[0].x = 99;
console.log(badObjs[1].x); // 99 — shared
const goodObjs = Array.from({ length: 2 }, () => ({ x: 1 }));
goodObjs[0].x = 99;
console.log(goodObjs[1].x); // 1 — distinct
```
