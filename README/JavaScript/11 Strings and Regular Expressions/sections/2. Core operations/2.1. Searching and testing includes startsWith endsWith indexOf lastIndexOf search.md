## 2.1. Searching and testing — includes startsWith endsWith indexOf lastIndexOf search

### 1) **Substring existence — `includes`, `startsWith`, `endsWith` with position**

**Definition** — `s.includes(searchString, position)` → boolean (found at/after position); `s.startsWith(searchString, position)` → boolean (at position); `s.endsWith(searchString, endPosition)` → boolean (ending before endPosition). All take a *string* search, not regex — passing a regex throws `TypeError`. `position`/`endPosition` clamp to `0..length`.

**Modern guidance** — use these for literal substring checks. They are intent-clear and locale-free, don't touch `lastIndex`, and read faster than regex or `indexOf >= 0`.

```js
const s = "JavaScript is great, JavaScript is fun";
console.log(s.includes("great")); // true
console.log(s.includes("Great")); // false — case sensitive
console.log(s.includes("great", 20)); // false — search starts at 20 (past great)
console.log(s.startsWith("Java")); // true
console.log(s.startsWith("Script", 4)); // true — at index 4
console.log(s.endsWith("fun")); // true
console.log(s.endsWith("Java", 10)); // true — first 10 chars "JavaScript" ends with "Java"? slice(0,10)="JavaScript" -> endsWith?
console.log("JavaScript".endsWith("Java", 4)); // true — first 4 chars "Java"

// Empty string edge — always found
console.log("abc".includes("")); // true
console.log("abc".startsWith("")); // true
console.log("abc".endsWith("")); // true
console.log("abc".includes("", 5)); // true — position beyond length still true for empty

// Regex passed throws — use indexOf/search for regex
try { console.log("hi".includes(/hi/)); } catch (e) { console.log(e.name); } // TypeError

// Position clamping
console.log("abc".includes("b", -100)); // true — clamped to 0
console.log("abc".includes("b", 100)); // false — beyond length

// Idiomatic guards
function hasExtension(path, ext) { return path.endsWith(ext); }
console.log(hasExtension("app.ts", ".ts")); // true
function isPrefixed(id) { return id.startsWith("user:"); }
console.log(isPrefixed("user:123")); // true
```

### 2) **Index search — `indexOf`/`lastIndexOf` forward/backward with position clamping**

**Definition** — `s.indexOf(search, pos)` scans forward from `pos` (clamped to `0..length`, default 0); `s.lastIndexOf(search, pos)` scans backward from `pos` (clamped, default `+Infinity` → length). Return index of first match or `-1`. Both do literal string search, no regex. Empty `search` returns `pos` clamped.

**What problem it solves / history** — before `includes`/`startsWith` (ES2015), all existence was `indexOf !== -1`. The `pos` argument let loops advance.

```js
const str = "ababa";
console.log(str.indexOf("aba")); // 0 — first
console.log(str.indexOf("aba", 1)); // 2 — next from 1
console.log(str.indexOf("x")); // -1
console.log(str.lastIndexOf("aba")); // 2 — last
console.log(str.lastIndexOf("aba", 1)); // 0 — backward from 1
console.log("aaaa".indexOf("aa", 2)); // 2 — overlapping not skipped beyond pos

// Empty search — returns clamped position
console.log("abc".indexOf("", 1)); // 1
console.log("abc".indexOf("", 10)); // 3 — clamped to length
console.log("abc".lastIndexOf("", 1)); // 1

// -1 vs 0 — the classic bug
const hay = "banana";
if (hay.indexOf("ana") ) console.log("found? truthy bug — index 1 is truthy but 0 is falsy");
console.log(hay.indexOf("ana") !== -1); // true — correct check
console.log(hay.indexOf("ban") !== -1 && hay.indexOf("ban") === 0); // startsWith via indexOf legacy

// Loop to find all occurrences (advance past match)
function allIndices(haystack, needle) {
  const out = [];
  if (needle === "") return [...haystack].map((_, i) => i).concat(haystack.length); // empty matches every position
  let pos = 0;
  while (true) {
    const i = haystack.indexOf(needle, pos);
    if (i === -1) break;
    out.push(i);
    pos = i + needle.length; // advance past match (use i+1 for overlapping)
  }
  return out;
}
console.log(allIndices("aaaa", "aa")); // [ 0, 2 ] — non-overlapping
console.log(allIndices("ababab", "ab")); // [ 0, 2, 4 ]

// lastIndexOf uses backward semantics — pos is rightmost start
console.log("ababa".lastIndexOf("ba", 3)); // 3 — "ba" at 3 is within bound; 1 also qualifies but 3 is rightmost
```

### 3) **`search` delegates to a RegExp**

**Definition** — `s.search(regexp)` converts `regexp` to `RegExp` if needed (string pattern creates `new RegExp(string)` without `g` reuse), then returns index of first match or `-1`. It ignores `g`/`y` and `lastIndex` — always searches from 0. If you pass a plain string without regex syntax, `indexOf`/`includes` are faster and safer.

```js
console.log("hello 123 world".search(/\d+/)); // 6 — first digits
console.log("hello 123 world".search(/\d+/g)); // 6 — g ignored, still 6
console.log("hello".search(/xyz/)); // -1

// String arg becomes RegExp — careful with metachars
console.log("a.b".search("a.b")); // 0 — "." treated as regex wildcard, not literal dot
console.log("a.b".indexOf("a.b")); // -1 — literal check correct
console.log("a.b".includes("a.b")); // false — literal
// Fix: escape if you came from user input (see 4.1) — RegExp.escape ES2025 (Node 22 may lack)
if (typeof RegExp.escape === "function") {
  console.log("a.b".search(RegExp(RegExp.escape("a.b")))); // -1? "a.b" literal not in "a.b"? actually search literal -> index 0 escaped
  console.log("a.b".search(RegExp(RegExp.escape("a.b"))) === 0); // true
} else {
  console.log("RegExp.escape not in this runtime — use polyfill or manual escape for now");
}

// lastIndex ignored proof
const re = /ab/g;
re.lastIndex = 2;
console.log("ab ab".search(re), re.lastIndex); // 0 2 — lastIndex unchanged (vs exec/test which mutate)

// When to choose who
// - Literal substring: includes / startsWith / endsWith / indexOf  (clear, fast, no escaping)
// - Regex pattern: search / match / exec (pattern power)
// search returns index only — if you need the match text/groups, use match/exec (see 4.2)
console.log("fee fi fo".search(/f\w+/)); // 0 — index only
console.log("fee fi fo".match(/f\w+/)); // [ 'fee' ] — match text
```

### 4) **Common confusion**

- "`indexOf` returns truthy when found" — no, `0` is falsy but is a valid index; use `!== -1` or `includes`.
- "`endsWith('x', n)` means start at `n`" — no, `endsWith` second arg is `endPosition` (slice `0..n`); `startsWith` second arg is `start position`.
- "`search` respects `lastIndex`" — no, `search` always starts at 0; only `exec`/`test` with `g`/`y` advance `lastIndex`.
- "Empty string is not found" — no, all three `includes`/`indexOf` treat `""` as found at clamped `pos`.
- "Passing regex to `includes` works like `search`" — no, it throws `TypeError` (intentional — forces `search`/`test` for regex).

### 5) **Performance note**

Literal search (`includes`/`indexOf`) uses a fast substring algorithm (Boyer-Moore/Two-Way per engine, no regex compilation); prefer it over `/literal/.test(s)` when you don't need pattern power — fewer allocations, no `RegExp` object. `search` with a regex compiles the pattern if you pass a string (implicit `new RegExp`); reuse a literal `/re/` outside loops.

### 6) **Mentor note**

Have juniors rewrite `if (str.indexOf(x))` to `if (str.includes(x))` and explain the `0-is-falsy` bug — then require `includes`/`startsWith`/`endsWith` in review for literal checks. Show `search("a.b")` matching `"axb"` to cement literal-vs-regex gap, and require `RegExp.escape` for any user-provided pattern that goes into `search`/`RegExp`. Flag any `search` where the result is ignored and the match text is needed (use `match`).
