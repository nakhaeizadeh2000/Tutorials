## 7.1. Real production bugs caused by string and RegExp mistakes

### 1) **`split("")` breaking emoji and `length` off-by-one for astral text**

**Bad** — truncating display names with `slice(0, max)` or counting with `s.length` / `s.split("")` counts units, not user characters. `"👩‍🚀".slice(0,2)` produces lone halves (ill-formed), and `split("")` reverses them broken.

```js
// Bad — unit-based truncation
function truncBad(s, max) { return s.length > max ? s.slice(0, max) + "…" : s; }
console.log(truncBad("👩‍🚀Astronaut", 2)); // lone surrogates — broken render "�…"
console.log(truncBad("😀😀😀", 2)); // 2 units = one emoji? "😀" length 2 → truncates after first half

// Good — grapheme-aware via Segmenter
function truncGood(s, max) {
  const segs = [...new Intl.Segmenter("en",{granularity:"grapheme"}).segment(s)].map(x=>x.segment);
  return segs.length > max ? segs.slice(0, max).join("") + "…" : s;
}
console.log(truncGood("👩‍🚀Astronaut", 2)); // "👩‍🚀A…" — preserves clusters
console.log(truncGood("😀😀😀", 2)); // "😀😀…"

// Bad — point vs unit reversal
console.log("A😀B".split("").reverse().join("")); // "B\uDE00\uD83DA" — broken pair
console.log([... "A😀B"].reverse().join("")); // "B😀A" — correct points
```

### 2) **Shared global RegExp with stale `lastIndex` — alternating pass/fail validates**

**Bad** — a single `/pattern/g` regex shared across calls to a validator flips `true/false` as `lastIndex` cycles `0→afterMatch→0`.

```js
// Bad — module-shared global code validator
const codeReBad = /^\d{4}$/g; // g is wrong for ^$ — but common copy-paste
function isCodeBad(code) { return codeReBad.test(code); }
console.log(isCodeBad("1234")); // true — lastIndex 4? Actually /^$/ g scan: first true then second false due to lastIndex?
console.log(isCodeBad("1234")); // false — alternates! second call starts at 1? Actually /^...$/ global with anchored pattern returns 4? test from 0 true then lastIndex 4 then second test from 4 fails -> reset 0
console.log(isCodeBad("1234")); // true — flips again

// More realistic shared global without anchors — auth token pattern
const tokenRe = /[A-Z]{3}-\d{2}/g;
function hasTokenBad(s) { return tokenRe.test(s); }
console.log(hasTokenBad("ABC-12")); // true lastIndex 6
console.log(hasTokenBad("ABC-12")); // false — same input now fails because lastIndex 6 beyond length -> reset

// Good — stateless or reset
function isCodeGood(code) { return /^\d{4}$/.test(code); } // literal per call, no g, no state
const tokenGood = /[A-Z]{3}-\d{2}/g;
function hasTokenGood(s) { tokenGood.lastIndex = 0; return tokenGood.test(s); }
console.log(isCodeGood("1234"), isCodeGood("1234")); // true true
console.log(hasTokenGood("ABC-12"), hasTokenGood("ABC-12")); // true true

// Good — no global when you don't iterate
function hasTokenSimple(s) { return /[A-Z]{3}-\d{2}/.test(s); } // no g — always from 0
console.log(hasTokenSimple("ABC-12"), hasTokenSimple("ABC-12")); // true true
```

### 3) **`replace` without `g` replaces only first occurrence — and `replaceAll` without `g` throws**

**Bad** — sanitizing/censoring with `replace(/bad/, "")` leaves later occurrences; switching to `replaceAll(/bad/, ...)` without `g` throws `TypeError` at runtime (ships as 500).

```js
// Bad — only first censored
function censorBad(text) { return text.replace(/bad/, "[censored]"); }
console.log(censorBad("bad and bad again")); // "bad and bad again"? actually " [censored] and bad again" — second left

// Good — global replace
function censorGood(text) { return text.replace(/bad/g, "[censored]"); }
console.log(censorGood("bad and bad again")); // "[censored] and [censored] again"

// replaceAll needs g for regex — the migration trap
function sanitizeBad(s) { return s.replaceAll(/</, "&lt;"); } // throws!
try { console.log(sanitizeBad("<a><b>")); } catch(e) { console.log(e.name, "— replaceAll regex needs g"); } // TypeError
function sanitizeGood(s) { return s.replaceAll(/</g, "&lt;"); }
console.log(sanitizeGood("<a><b>")); // "&lt;a>&lt;b>"

// String-pattern replaceAll works without g (literal)
console.log("aabbaa".replaceAll("aa", "X")); // "XbbX"

// Replacer leaving $ behind — "$$" escaping bug
function dollarBad(s) { return s.replace(/price/, "cost $&"); } // $& inserts match — "price" -> "price"
console.log("price 10".replace(/price/g, "$$10")); // " $10 10"? "$$" -> "$" -> "$10 10"
console.log("price 10".replace(/price/g, "$10")); // "$10"? $1 nonexistent -> $10 literal? depends
```

### 4) **Lone-surrogate `encodeURI` throw ships as unhandled rejection**

**Bad** — user input with emoji-slice halves reaches `encodeURI` → `URIError` → 500. Common in URL building from display truncations.

```js
// Bad — slice creates ill-formed then encodeURI throws
const display = "hello 😀"; // length? 7? but slice half
const half = display.slice(0, 7); // cuts pair? "hello " (6) + first half of 😀 (1) => 7 lone
console.log(half.isWellFormed()); // false if cut, else true depending slice point
try { encodeURI(`https://example.com/q=${half}`); } catch(e) { console.log(e.name, "ships as 500 in prod"); }

// Good — sanitize before encode
function toUrlParam(s) { return encodeURI(`https://example.com/q=${s.toWellFormed()}`); }
console.log(toUrlParam(half)); // "%EF%BF%BD" for replacement if half ill-formed, otherwise valid
console.log(toUrlParam("hello 😀")); // "https://example.com/q=hello%20%F0%9F%98%80"
```

### 5) **Regex construction without `RegExp.escape` — wildcard injection**

**Bad** — `new RegExp(userInput)` where input is `"a.b"` matches `"axb"`; search index building breaks.

```js
function hasExactBad(hay, needle) { return new RegExp(needle).test(hay); }
console.log(hasExactBad("axb", "a.b")); // true — bug, dot wildcard
console.log(hasExactBad("a.b", "a.b")); // true — but also true for wrong

function hasExactGood(hay, needle) {
  const esc = typeof RegExp.escape === "function" ? RegExp.escape(needle) : needle.replace(/[.*+?^${}()|[\]\\]/g,"\\$&");
  return new RegExp(esc).test(hay);
}
console.log(hasExactGood("axb", "a.b")); // false — correct literal
console.log(hasExactGood("a.b", "a.b")); // true
```

### 6) **Mentor checklist for this pitfall class**

In prod postmortems, the common review misses that ship these bugs:

1) Any `split("")`/`slice` on display/user text → require `Segmenter` or `[...s]`/`toWellFormed`.
2) Any `const re = /.../g` at module scope used in a `test`/`exec` validator → require `lastIndex=0` or remove `g`.
3) Any `replace(` without `g` where multiple occurrences expected → require `replaceAll` + `g` (or `/.../g`).
4) Any `new RegExp(user)` → require `RegExp.escape`.
5) Any `encodeURI`/`TextEncoder` on sliced external strings → require `isWellFormed` guard + `toWellFormed`.

```js
// Smoke check — all good variants exit 0 plus guard before encode
console.log(truncGood("👩‍🚀👩‍🚀",1) === "👩‍🚀…"); // true
console.log(isCodeGood("1234") && hasTokenGood("ABC-12") && censorGood("bad bad").includes("[censored]"));
```
