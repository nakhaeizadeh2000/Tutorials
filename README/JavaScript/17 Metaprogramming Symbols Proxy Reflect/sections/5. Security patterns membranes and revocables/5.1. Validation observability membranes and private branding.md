## 5.1. Validation observability membranes and private branding

### 1) **Validation proxies — intercept `set`/`defineProperty` to enforce schemas**

**Definition** — a validation proxy wraps the target and rejects invalid writes via `set`/`defineProperty` returning `false` (or throwing). Because direct assignment `proxy[key]=val` goes through `[[Set]]`, every mutation site is covered without rewriting callers.

```js
// Validation proxy — range checks and required fields
function validatedUser(data){
  return new Proxy(data, {
    set(t,k,v,r){
      if (k === "age" && (typeof v !== "number" || v < 0 || v > 150)) return false; // soft failure → strict throws
      if (k === "name" && typeof v !== "string") throw new TypeError("name must be string");
      return Reflect.set(t,k,v,r);
    },
    defineProperty(t,k,desc){
      if (k === "age" && "value" in desc && (desc.value < 0 || desc.value > 150)) return false;
      return Reflect.defineProperty(t,k,desc);
    }
  });
}
const u = validatedUser({ name: "Ari", age: 30 });
console.log(Reflect.set(u, "age", 200)); // false — blocked (ownKeys invariant not triggered)
try { u.age = 200; } catch (e){ console.log("strict:", e.name); } // TypeError in module-strict
u.age = 31; console.log(u.age); // 31
console.log(Reflect.defineProperty(u, "age", { value: 200, configurable:true })); // false

// Bad vs good — incomplete trap coverage leaks via defineProperty
const leaky = new Proxy({ age: 30 }, {
  set(t,k,v,r){ if (k==="age" && v>150) return false; return Reflect.set(t,k,v,r); }
  // no defineProperty trap → Object.defineProperty bypasses validation
});
Object.defineProperty(leaky, "age", { value: 999 }); // succeeds — leak
console.log(leaky.age); // 999 — should have been blocked; fix by adding defineProperty trap
```

- **Modern guidance** — always guard both `set` and `defineProperty`; `set` alone is bypassed by `Object.defineProperty`/`Reflect.defineProperty`.

### 2) **Observability — `get`/`set`/`has`/`apply` as non-invasive logging**

**Definition** — an observability proxy forwards everything via `Reflect.*` while logging or counting. It must still honor invariants (e.g., `get` must return frozen values, `ownKeys` must include non-configurable keys — see [2.2](<../2. Proxy core/2.2. Proxy invariants non-configurable non-extensible and exotic constraints.md>)).

```js
// Observability — count reads without changing shape
function observable(obj){
  let reads = 0;
  const proxy = new Proxy(obj, {
    get(t,k,r){ reads++; return Reflect.get(t,k,r); },
    has(t,k){ reads++; return Reflect.has(t,k); }
  });
  return { proxy, get reads(){ return reads; } };
}
const obs = observable({ a: 1, b: 2 });
console.log(obs.proxy.a + obs.proxy.b); // reads incremented per get
console.log("reads", obs.reads); // 2

const obs2 = observable({ x: 1 });
console.log("x" in obs2.proxy, "y" in obs2.proxy); // has trap counts
console.log(obs2.proxy.x); // get counts
console.log("reads2", obs2.reads); // 3

// Lazy materialization — virtual property computed on first access then defined
const lazy = new Proxy({ _real: null }, {
  get(t,k,r){
    if (k === "expensive" && t._real === null){ t._real = { value: 42 }; console.log("materialized"); }
    if (k === "expensive") return t._real.value;
    return Reflect.get(t,k,r);
  }
});
console.log(lazy.expensive); // materialized 42
console.log(lazy.expensive); // 42 — cached, no re-materialize

// Pitfall — get must still satisfy invariant for frozen target
const frozen = Object.freeze({ pin: 7 });
const obsFrozen = new Proxy(frozen, {
  get(t,k,r){ console.log(`get ${String(k)}`); return Reflect.get(t,k,r); } // must return 7 for pin, not 999
});
console.log(obsFrozen.pin); // 7 — correct; returning 999 would throw TypeError
```

### 3) **Membranes — wrapping an entire object graph**

**Definition** — a membrane ensures every object crossing the boundary is wrapped. `get`/`getOwnPropertyDescriptor`/`ownKeys` must wrap returned objects; `set`/`defineProperty` must unwrap proxies before storing. Cache via `WeakMap` target→proxy to preserve identity.

```js
// Membrane — identity-preserving wrapping of object graph
function membrane(target){
  const cache = new WeakMap();
  const handler = {
    get(t,k,r){
      const v = Reflect.get(t,k,r);
      if (v !== null && typeof v === "object") return wrap(v);
      return v;
    },
    set(t,k,v,r){
      // unwrap proxy if caller passed a wrapped value
      const raw = v && typeof v === "object" && cache.has(v) ? v : v; // simplified: check if v is proxy wrapping?
      // better: track reverse map proxy→target — omitted for brevity; real membranes keep two WeakMaps
      return Reflect.set(t,k,raw,r);
    },
    getOwnPropertyDescriptor(t,k){
      const d = Reflect.getOwnPropertyDescriptor(t,k);
      if (d && "value" in d && d.value !== null && typeof d.value === "object") d.value = wrap(d.value);
      return d;
    },
    ownKeys(t){ return Reflect.ownKeys(t); },
    getPrototypeOf(t){ const p = Reflect.getPrototypeOf(t); return p ? wrap(p) : p; }
  };
  function wrap(obj){
    if (obj === null || typeof obj !== "object") return obj;
    if (cache.has(obj)) return cache.get(obj);
    const proxy = new Proxy(obj, handler);
    cache.set(obj, proxy);
    return proxy;
  }
  return wrap(target);
}
const orig = { nested: { x: 1 }, arr: [1,2] };
const m = membrane(orig);
console.log(m.nested.x); // 1 — nested auto-wrapped
console.log(m.nested === m.nested); // true — identity preserved via cache
console.log(m.nested !== orig.nested); // true — wrapped distinct
console.log(m.arr === m.arr); // true — cached

// Correct forwarding preserves receiver — membrane's get uses Reflect.get with receiver (see 3.2)
const withGetter = { _v: 1, get v(){ return this._v; } };
const mGetter = membrane(withGetter);
mGetter._v = 99; // data prop writes through to target (see 3.2 note) — WeakMap private state (next) avoids this
console.log(withGetter._v); // 99 — data props still target-anchored; use WeakMap for per-membrane privacy (see §4)

// Missing step — unwrap on set is simplified; production membranes keep reverse WeakMap proxy→target to avoid double-wrapping
```

- **Modern guidance** — membrane `set` must unwrap, `get` must re-wrap; forgetting either creates identity splits or double-wrapping leaks. Always pass `receiver` through `Reflect.get`.

### 4) **Private branding without `#private` — `WeakMap`/`WeakSet` vs `Symbol.hasInstance` trap**

**Definition** — class `#private` fields (see [Prototypes 3.3](<../../../07 Prototypes and Classes/sections/3. Class syntax/3.3. Private fields and methods brand checks and encapsulation.md>)) enforce branding via `BrandCheck`. Proxy-based branding achieves similar isolation without syntax: store private state in `WeakMap<proxyOrTarget, secret>` or `WeakSet` brand set; reveal via custom `hasInstance` or `get` trap.

```js
// WeakMap private state tied to proxy identity — works with membranes
const priv = new WeakMap();
class Counter {
  constructor(){ priv.set(this, { n: 0 }); }
  inc(){ priv.get(this).n++; }
  get value(){ return priv.get(this).n; }
}
const c = new Counter(); c.inc(); console.log(c.value); // 1
console.log(priv.get(c)); // {n:1} — anyone with priv can read; hide priv in closure for true privacy

// Proxy-based brand via closure WeakSet + hasInstance hook
function makeBrand(){
  const brand = new WeakSet();
  class Branded {
    constructor(){ brand.add(this); }
    static [Symbol.hasInstance](x){ return brand.has(x); }
    method(){ if (!brand.has(this)) throw new TypeError("not branded"); return "ok"; }
  }
  return Branded;
}
const B = makeBrand();
const b = new B();
console.log(b instanceof B); // true — via @@hasInstance WeakSet
console.log({} instanceof B); // false
console.log(b.method()); // ok
try { B.prototype.method.call({}); } catch(e){ console.log("unbranded:", e.name); }

// Why not just Symbol("priv")? Symbol keys are enumerable via getOwnPropertySymbols/Reflect.ownKeys (see 1.2) — not private, just hidden
const SYM = Symbol("secret");
const o = { [SYM]: 123 };
console.log(Object.getOwnPropertySymbols(o)[0] === SYM); // true — recoverable; WeakMap is not
```

- **Modern guidance** — prefer `#private` for class privacy (ergonomic + engine-optimized). Use `WeakMap` branding when you need per-proxy privacy across membranes or for plain objects without class syntax. `Symbol("priv")` is obfuscation, not secrecy.

### 5) **Common confusion**

- "`Proxy` with `set` validation blocks `Object.defineProperty`" — only if you trap `defineProperty`; `set` alone does not intercept descriptor-defined writes.
- "`Membrane wraps target once at creation`" — no; it must intercept every `get` that returns an object, recursively — one-time wrapping misses lazily added nested objects.
- "`Symbol("priv")` is private" — no; `Reflect.ownKeys` reveals it. True privacy is `#` or `WeakMap` closure.

### 6) **Performance note**

Each `get` through a membrane does one `WeakMap.get` + one `Reflect.get`. On a hot path with thousands of property reads, this doubles cost. Cache the unwrapped target when looping internally; expose the membrane only at API boundaries.

### 7) **Mentor note**

In review, ask: "Does this handler forward `receiver`? Does `set` guard `defineProperty`? Does `get` re-wrap?" Any "no" is a bug. Require the two `WeakMap`s (target→proxy, proxy→target) in any membrane design doc.
