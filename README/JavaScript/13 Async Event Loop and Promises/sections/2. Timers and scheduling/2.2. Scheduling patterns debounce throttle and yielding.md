## 2.2. Scheduling patterns debounce throttle and yielding

### 1) **Debounce vs throttle — coalescing bursts vs rate-limiting**

**Definition** — *debounce* waits for silence (reset timer on each call, fire once after quiet period — trailing; or on first call then silence — leading). *Throttle* allows at most one call per interval (leads to periodic reports while scrolling/resizing). Both are timer-backed and need `cancel`/`flush`.

**What problem it solves / history** — input and scroll events fire dozens per frame; calling an API or layout per event freezes the app. Debounce/throttle turn O(n) handlers into O(1) scheduled macrotasks.

**Modern guidance** — add `AbortSignal` to both; expose `cancel` and `flush`; prefer `signal` abort over manual `clearTimeout` at call sites.

```js
// Trailing debounce — coalesces rapid calls into one after quiet
function debounce(fn, delay, { signal } = {}) {
  let id = null, lastArgs, lastThis;
  function flush() {
    if (id !== null) { clearTimeout(id); id = null; const a = lastArgs, t = lastThis; lastArgs = lastThis = null; fn.apply(t, a); }
  }
  function debounced(...args) {
    lastArgs = args; lastThis = this;
    if (id !== null) clearTimeout(id);
    id = setTimeout(flush, delay);
  }
  debounced.cancel = () => { if (id !== null) clearTimeout(id); id = null; };
  debounced.flush = flush;
  signal?.addEventListener("abort", debounced.cancel, { once: true });
  return debounced;
}

let calls = 0;
const d = debounce(() => calls++, 10);
d(); d(); d(); // last wins
await new Promise((r) => setTimeout(r, 25));
console.log(calls); // 1 — coalesced

// Leading + trailing variant
function debounceLeading(fn, delay) {
  let id = null, pending = false;
  return function (...args) {
    const doit = id === null;
    if (id !== null) clearTimeout(id);
    id = setTimeout(() => { id = null; if (pending) { pending = false; fn.apply(this, args); } }, delay);
    if (doit) fn.apply(this, args);
    else pending = true;
  };
}
let n = 0;
const dl = debounceLeading(() => n++, 10);
dl(); dl(); // leading fires once
await new Promise((r) => setTimeout(r, 15));
console.log(n >= 1); // true

// Throttle — at most once per interval
function throttle(fn, interval) {
  let last = 0, id = null, lastArgs, lastThis;
  return function (...args) {
    const now = Date.now();
    const remaining = interval - (now - last);
    lastArgs = args; lastThis = this;
    if (remaining <= 0) {
      if (id !== null) { clearTimeout(id); id = null; }
      last = now; fn.apply(lastThis, lastArgs);
    } else if (id === null) {
      id = setTimeout(() => { last = Date.now(); id = null; fn.apply(lastThis, lastArgs); }, remaining);
    }
  };
}
let tCalls = 0;
const th = throttle(() => tCalls++, 30);
for (let i = 0; i < 5; i++) th();
console.log(tCalls); // 1 — first immediate
await new Promise((r) => setTimeout(r, 35));
console.log(tCalls); // 2 — trailing after interval

// Signal-aware debounce
const ac = new AbortController();
const ds = debounce(() => console.log("should not fire — aborted"), 20, { signal: ac.signal });
ds(); ac.abort(); // cancel queued
await new Promise((r) => setTimeout(r, 25));
console.log("debounce abort done");
```

### 2) **Yielding to the loop — `await scheduler.yield` / `setTimeout(0)` chunking for responsive UIs**

**Definition** — split a long computation into chunks that `await` a macrotask or browser yield between chunks so input, timers, and paint can run. Browsers offer `scheduler.yield()` (Chrome 115+) and `navigator.scheduling.isInputPending()`; fallback is `setTimeout(0)`.

```js
// Chunked transform — yields every chunk so input stays live
async function mapChunked(arr, fn, { chunk = 1000 } = {}) {
  const out = [];
  for (let i = 0; i < arr.length; i++) {
    out.push(fn(arr[i], i));
    if (i % chunk === 0 && i !== 0) {
      if (typeof scheduler !== "undefined" && scheduler.yield) await scheduler.yield();
      else await new Promise((r) => setTimeout(r, 0)); // macrotask yield unblocks render
    }
  }
  return out;
}
const big = Array.from({ length: 5000 }, (_, i) => i);
const doubled = await mapChunked(big, (x) => x * 2);
console.log(doubled[0], doubled[doubled.length - 1]); // 0 9998

// Input-pending aware (browser)
async function mapInputAware(arr, fn) {
  for (let i = 0; i < arr.length; i++) {
    fn(arr[i], i);
    const pending = typeof navigator !== "undefined" && navigator.scheduling?.isInputPending?.();
    if (pending) {
      if (typeof scheduler !== "undefined" && scheduler.yield) await scheduler.yield();
      else await new Promise((r) => setTimeout(r, 0));
    }
  }
}
mapInputAware([1, 2, 3], (x) => x).then(() => console.log("input-aware done"));

// BAD — tight microtask yield does NOT unblock rendering (still starves)
async function mapMicroStarve(arr, fn) {
  for (const v of arr) { fn(v); await Promise.resolve(); } // microtask — no paint between
  console.log("micro-starved: paint blocked until loop ends");
}
await mapMicroStarve([1, 2, 3], (x) => x);
```

### 3) **Ordering guarantees and testing — fake timers, determinism, and `await` vs `setTimeout` in tests**

**Definition** — fake timers replace `setTimeout`/`setInterval`/`Date` with a controllable clock so tests advance deterministically. `vitest`/`sinon` expose `useFakeTimers` + `advanceTimersByTime` / `runAllTimers`. Because `await` flushes microtasks before timers, advancing timers alone may not flush awaiting code — also `await` the microtask queue.

```js
// Real-timer intuition — await flushes microtasks before the timer callback
let order = [];
setTimeout(() => order.push("timer"), 0);
Promise.resolve().then(() => order.push("micro"));
await new Promise((r) => setTimeout(() => { order.push("await+timer"); r(); }, 0));
console.log(order); // [ 'micro', 'timer', 'await+timer' ] — micro before each timer
```

### 4) **Common confusion**

- "Debounce and throttle are interchangeable" — debounce waits for silence (search); throttle clips rate (scroll/resize). Picking wrong gives missing trailing events or excessive calls.
- "`await Promise.resolve()` yields to rendering" — no; it yields only to microtasks. Painting requires a macrotask or `scheduler.yield`.
- "Fake timers make `await` synchronous" — no; `await` still queues a microtask — after advancing timers, `await Promise.resolve()` or `await tick()` is needed to flush.
- "`requestAnimationFrame` is a timer" — it's a render-phase callback (browser) — runs before paint, not as a `setTimeout` macrotask.

### 5) **Performance note**

Debouncing a 100ms burst of 50 keystrokes from 50 network calls to 1 is the canonical perf win. Throttling `scroll` at 100ms cuts layout thrash from 200 handlers/s to 10. Chunking with `scheduler.yield` keeps INP (Interaction to Next Paint) under 200ms; chunking with `Promise.resolve()` does not — DevTools "Performance" will still show a Long Task.
