## 7.1. Real production bugs caused by async mistakes

### 1) **Fire-and-forget `map(async` without `await all` — silently succeeds while failures vanish**

**Definition** — `Array.map` with an `async` mapper builds an array of promises. Without `await Promise.all`/`allSettled`, the caller proceeds before work finishes; rejections become unhandled (or silent when no `unhandledRejection` handler is installed) and half-finished writes persist.

```js
// Incident — migration marks "done" before rows actually migrate
async function migrateUsers(ids) {
  const saves = ids.map(async (id) => {
    await new Promise((r) => setTimeout(r, 5)); // simulate async save
    if (id === 2) throw new Error(`save failed ${id}`);
    return `ok ${id}`;
  });
  // BUG — missing await
  console.log("BUG: saves returned", saves.length, saves[0] instanceof Promise); // 3 true — not values
  return saves; // caller thinks done
}
const leaked = await migrateUsers([1, 2, 3]);
// Prevent unhandledRejection when demonstrating fire-and-forget (bad path) — attach dummy handlers
leaked.forEach((p) => p.catch(() => {}));
console.log("leaked length", leaked.length); // 3 promises — caller never saw the thrown error as catch

// Fix — observe the array
async function migrateUsersFixed(ids) {
  const saves = ids.map(async (id) => {
    await new Promise((r) => setTimeout(r, 5));
    if (id === 2) throw new Error(`save failed ${id}`);
    return `ok ${id}`;
  });
  // all → fail-fast; allSettled → per-row report
  const outcomes = await Promise.allSettled(saves);
  const failed = outcomes.filter((o) => o.status === "rejected");
  console.log("failed", failed.map((o) => o.reason.message)); // save failed 2
  return outcomes;
}
console.log((await migrateUsersFixed([1, 2, 3])).map((o) => o.status)); // [fulfilled,rejected,fulfilled]

// Same trap — forEach(async) never awaits
async function badForEachSave(ids) {
  ids.forEach(async (id) => {
    try { await new Promise((r) => setTimeout(r, 2)); if (id === 2) throw new Error("boom"); }
    catch (e) { console.log("forEach inner caught", e.message); } // keep runnable — real bug would be uncaught
  });
  console.log("after forEach — work not waited on");
}
await badForEachSave([1, 2]);
await new Promise((r) => setTimeout(r, 10)); // would be unhandledrejection in the real bug
```

### 2) **Unbounded `Promise.all` fan-out — 10k parallel fetches OOM or rate-limit**

**Definition** — `Promise.all(items.map(fetch))` starts all fetches eagerly and holds every pending promise and response buffer until the last settles. At user-controlled length (uploaded file list, row batch) this bursts FDs/sockets, throttles the upstream, and spikes GC from concurrent bodies.

```js
// BAD — burst
function fetchOne(id) { return new Promise((r) => setTimeout(() => r(id), 3)); }
async function badFanout(ids) { return Promise.all(ids.map(fetchOne)); } // 100 at once
const many = Array.from({ length: 20 }, (_, i) => i);
console.log((await badFanout(many.slice(0, 5))).length); // 5 — works small, breaks big

// GOOD — pooled with limit (reuse mapLimit pattern from 4.1/5.2/6.1)
async function goodPooled(ids, limit = 5) {
  const out = new Array(ids.length);
  let next = 0;
  let active = 0, peak = 0;
  const workers = Array.from({ length: Math.min(limit, ids.length) }, async () => {
    while (next < ids.length) {
      const i = next++;
      active++; peak = Math.max(peak, active);
      try { out[i] = await fetchOne(ids[i]); } finally { active--; }
    }
  });
  await Promise.all(workers);
  console.log("peak", peak); // ≤ limit
  return out;
}
console.log((await goodPooled(many, 5)).length); // 20 — peak 5, not 20
```

### 3) **Timer leak — forgotten `setInterval` keeps process alive and doubles work after reconnect**

**Definition** — `setInterval` lives until `clearInterval` or `unref` (Node) / page navigation (browser). A reconnect handler that creates a new interval without clearing the prior one doubles polling; a missing `clear` on teardown keeps the process alive and churns CPU.

```js
// Leak — reconnect doubles intervals
let intervals = [];
function startPoll(label) {
  const id = setInterval(() => console.log("poll", label), 50);
  intervals.push(id);
  return id;
}
function reconnectBug() { startPoll("a"); /* missing clear of prior */ }
// Fixed — single interval ref + clear on next cycle
let current = null;
function startPollFixed(label) {
  if (current !== null) clearInterval(current);
  current = setInterval(() => console.log("poll fixed", label), 50);
  return current;
}
const p1 = startPollFixed("first");
const p2 = startPollFixed("second"); // p1 cleared, only second runs
await new Promise((r) => setTimeout(r, 65));
clearInterval(p2);
console.log("poll leak fixed");

// Keep-alive hold — Node interval without unref prevents exit
const hold = setInterval(() => {}, 1000);
if (hold.unref) hold.unref(); // allow exit even if forgotten clear is late
clearInterval(hold);
console.log("keep-alive cleared");
```

### 4) **Microtask starvation — recursive `then`/`queueMicrotask` freezes rendering/timers**

**Definition** — because microtasks drain exhaustively (see [1.2](<../1. Event loop and the job queue/1.2. Microtasks vs macrotasks queueMicrotask and starvation.md>)), a chain that schedules another microtask per job never yields to macrotasks or paint. The page stutters; timers drift.

```js
// Starving shape — each microtask enqueues the next, timer never fires until chain ends
let steps = 0;
async function starvingChain(count, yieldEvery = 0) {
  for (let i = 0; i < count; i++) {
    if (yieldEvery && i % yieldEvery === 0 && i !== 0) {
      await new Promise((r) => setTimeout(r, 0)); // periodic macrotask yield unblocks timer
    } else {
      await Promise.resolve(); // microtask only — starves
    }
    steps++;
  }
  return steps;
}
steps = 0;
let timerFired = false;
setTimeout(() => { timerFired = true; }, 0);
await starvingChain(5, 0);
console.log(timerFired, steps); // false? actually timer was after micro drain → now true, steps 5 — but timer was delayed until after 5 micro hops
// With yieldEvery=3:
steps = 0; timerFired = false;
setTimeout(() => { timerFired = true; }, 0);
await starvingChain(5, 3);
console.log(timerFired); // timer interleaved — responsive
```

### 5) **`race` timeout without `AbortSignal` — loser keeps running and writes after "timeout"**

```js
// Reportedly timed out but not cancelled — late write corrupts state
function writeDelayed(v, ms, { signal } = {}) {
  return new Promise((resolve, reject) => {
    if (signal?.aborted) return reject(signal.reason);
    const id = setTimeout(() => resolve(v), ms);
    signal?.addEventListener("abort", () => { clearTimeout(id); reject(signal.reason ?? new Error("aborted")); }, { once: true });
  });
}
async function saveWithTimeoutBad(v, ms) {
  return Promise.race([writeDelayed(v, 20), new Promise((_, rej) => setTimeout(() => rej(new Error("timeout")), ms))]);
}
async function saveWithTimeoutGood(v, ms) {
  const ac = new AbortController();
  const timeout = setTimeout(() => ac.abort(new Error("timeout")), ms);
  try { return await writeDelayed(v, 20, { signal: ac.signal }); }
  finally { clearTimeout(timeout); }
}
try { await saveWithTimeoutBad("x", 8); } catch (e) { console.log("bad", e.message); } // timeout — but delayed write still fires at 20, unwatched
try { await saveWithTimeoutGood("x", 8); } catch (e) { console.log("good", e.message); } // timeout — and delayed work actually cancelled
```

### 6) **Mentor note**

Give juniors a "success" log from a migration that used `map(async` without `all` and ask where the missing rows went — the unhandled-rejection + premature "done" narration fixes the contract instantly. In review, flag any `setInterval` without a `clearInterval`/`unref` in the same scope, any `map(async` without `all`/`allSettled` on the next line (require either), and any `race` timeout not wired to `AbortSignal`.
