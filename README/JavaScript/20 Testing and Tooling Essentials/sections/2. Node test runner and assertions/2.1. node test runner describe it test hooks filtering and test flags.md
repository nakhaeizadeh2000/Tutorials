## 2.1. node:test runner describe it test hooks filtering and test flags

### 1) **Runner primitives — `test` / `describe` / `it` / `before` / `after` / `beforeEach` / `afterEach` and `t.todo`/`t.skip`**

**Definition** — `node:test` (stable since Node 18, enhanced 20+) is the stdlib test runner. `test(name, fn)` registers a top-level test. `describe(name, fn)` groups tests and provides scoped hooks. `it` is an alias of `test` inside `describe`. Hooks `before`/`after` run once per suite, `beforeEach`/`afterEach` per test. `t` (the `TestContext` passed to `fn`) gives `t.todo`, `t.skip`, `t.plan`, `t.after`, and `t.mock` (see [3.2](../3. Mocks and test doubles/3.2. Mocking in Node mock.fn mock.method timers fetch and modules.md)).

**Modern guidance** — prefer `describe` for a subject (`describe("clamp")`) and `it("when n < lo then returns lo")` for each branch. Use `t.todo("parity with BigInt")` for planned cases that should appear in TAP as `not ok # TODO`. Use `t.skip` for platform-gated tests (`if (process.platform !== "linux") t.skip()`).

```js
// Shapes of the runner — runnable without --test as plain calls when you
// just need to illustrate; real suite runs with `node --test file.js`
const { test, describe, it, before, after, beforeEach, afterEach } = require("node:test");
const assert = require("node:assert/strict");

// Flat test — works in any file; with --test it is reported via TAP
test("flat: clamp clamps below lo", () => {
  function clamp(n, lo, hi) { return Math.min(Math.max(n, lo), hi); }
  assert.equal(clamp(-2, 0, 10), 0);
});

// Grouped suite — hooks give isolation (see 1.2)
describe("clamp", () => {
  let calls = 0;
  before(() => { calls = 0; });           // once
  beforeEach(() => { calls++; });         // per test
  it("when n inside then identity", () => { assert.equal((x=>x)(5), 5); assert.equal(calls >= 1, true); });
  it.todo("when given BigInt then throws (planned)");
});

// t.after is the per-test teardown that runs even if the test throws
// Illustrated with a plain function (real test would receive t):
function fakeT() {
  const cleanups = [];
  return { after(fn){ cleanups.push(fn); }, _run(){ for(const c of cleanups) c(); } };
}
const t = fakeT();
let opened = true;
t.after(() => { opened = false; });
t._run();
assert.equal(opened, false);
console.log("runner primitives ok");
```

**Common confusion** — "`describe` runs tests." `describe` only *registers* them during the collection phase; execution happens afterward. Forgetting `await t.test(...)` inside an async parent leaves a child unregistered before the parent finishes — see subtests below.

### 2) **Execution model — single process, per-file isolation, `node --test` discovery, `--test-name-pattern`, `--test-only`, and TAP reporters**

**Definition** — `node --test` discovers `**/*.test.js` (or explicit files), loads each file once in its own isolate, executes the collection phase (registers `test`/`describe`), then runs tests serially by file. Failures are TAP lines `not ok 1 - name`; passes are `ok 1 - name`. Reporters (`--test-reporter=tap|spec|dot|junit`) format TAP for CI.

**Modern guidance** — run `node --test` (no glob needed — default discovery covers `test/**/*.js` when configured). Filter a single branch with `node --test --test-name-pattern="when n < lo"`. Mark a focused test with `test.only("…", fn)` and guard CI against stray `only` via `NODE_OPTIONS=--test-only` or lint rule. Use `--test-reporter=junit --test-reporter-destination=junit.xml` for CI dashboards. As of Node 22 LTS, `--test-coverage` is experimental and delegates to `c8`-like V8 coverage — see [5.1](../5. Quality gates and tooling/5.1. Coverage with c8 v8 line branch function thresholds and lies.md).

```js
// Filtering and reporters are CLI concerns — illustrate the contract you rely on
const assert = require("node:assert/strict");

// What the runner guarantees (and doesn't)
assert.equal(typeof require("node:test").test, "function");
assert.equal(typeof require("node:test").describe, "function");

// Simulate name-pattern filtering (what --test-name-pattern does)
function matchesPattern(name, pattern) { return new RegExp(pattern).test(name); }
assert.equal(matchesPattern("when n < lo then returns lo", "when.*lo"), true);
assert.equal(matchesPattern("when n > hi then returns hi", "when.*lo"), false);

// TAP shape (what CI parses)
const tapPass = "ok 1 - when n < lo then returns lo";
const tapTodo = "not ok 2 - when BigInt then throws # TODO";
const tapFail = "not ok 3 - when gap then fail\n  ---\n  error: AssertionError\n  ...";
assert.equal(tapPass.startsWith("ok"), true);
assert.equal(tapTodo.includes("# TODO"), true);
assert.equal(tapFail.includes("not ok"), true);
console.log("execution model ok");
```

**Mentor note** — make juniors run `node --test --test-reporter=spec` before pushing; the `spec` reporter prints suite names and durations. Ban `test.only` from `main` with a lint guard (`no-only-tests` in ESLint) or CI flag `node --test --test-only` will fail if `only` is present on some versions.

### 3) **Subtests — `await t.test` nesting and concurrency defaults**

**Definition** — inside a `test` callback you receive `t: TestContext`. Calling `await t.test("child", fn)` registers a subtest that runs after the parent's setup and finishes before the parent completes. Without `await`, the subtest may still be running when the parent's `after` fires.

**Modern guidance** — always `await t.test`. For independent subtests that can run in parallel inside a helper, collect promises and `await Promise.all([...])` explicitly — the runner itself runs top-level files in parallel (when `--test-concurrency` is set) but subtests of a single parent are serial unless you fan out.

```js
const assert = require("node:assert/strict");

// Correct: await each subtest so hooks and t.after ordering hold
async function parentAwaited() {
  const order = [];
  async function fakeParent() {
    const ctx = {
      async test(name, fn) {
        order.push(`start ${name}`);
        await fn({});
        order.push(`end ${name}`);
      }
    };
    await ctx.test("child A", async () => { await new Promise(r => setTimeout(r, 2)); });
    await ctx.test("child B", async () => {});
    return order;
  }
  const got = await fakeParent();
  assert.deepEqual(got, ["start child A", "end child A", "start child B", "end child B"]);
}
parentAwaited().then(() => console.log("subtest await ok"));

// Anti-pattern: fire without await — parent's after may run mid-child
async function parentForgotAwait() {
  const order = [];
  const ctx = {
    async test(name, fn) { order.push(`start ${name}`); fn({}); order.push(`end ${name}`); }
  };
  // Missing await — file-level runner would warn "test already finished"
  ctx.test("leaked", async () => { await new Promise(r => setTimeout(r, 5)); order.push("leaked done"); });
  // parent thinks it's done while leaked still pending → flake
  assert.equal(order.includes("start leaked"), true);
  console.log("forget-await is a flake factory");
}
parentForgotAwait();

// Parallel subtests when truly independent
async function parallelHelpers() {
  const work = (n) => new Promise(r => setTimeout(() => r(n*2), 5));
  const out = await Promise.all([work(1), work(2), work(3)]);
  assert.deepEqual(out, [2,4,6]);
  console.log("parallel subtests ok");
}
parallelHelpers();
```

**Common confusion** — "subtests run in parallel by default." They don't. Parallelism across *files* is configurable; inside one `test`, subtests are serial unless you explicitly `Promise.all` them.

**Performance note** — subtest `await` ordering is O(1) overhead; the cost of *not* awaiting is flake (child's `t.after` teardown races with next test's `beforeEach`). Fix flakes structurally (await + `t.after`) rather than with `setTimeout` padding.
