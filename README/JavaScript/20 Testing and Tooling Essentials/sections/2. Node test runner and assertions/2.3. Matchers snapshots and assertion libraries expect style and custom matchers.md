## 2.3. Matchers snapshots and assertion libraries expect style and custom matchers

### 1) **`expect`-style — `expect(value).toEqual` / `toBe` / `toContain` / `toThrow` semantics and why `toBe` is `Object.is` not `===`**

**Definition** — `expect` is the matcher façade popularized by Jest/Vitest: `expect(got).toBe(want)` checks `Object.is` (distinguishes `-0` vs `0` and `NaN` vs `NaN`), `toEqual` deep-strict equality, `toContain` membership, `toThrow`/`toHaveBeenCalled` interaction checks. Node's stdlib has no `expect`; you either use `node:assert/strict` or add a matcher lib (Vitest/Jest/Chai) on top of `node:test`.

**What problem it solves** — `assert.deepStrictEqual` diffs are powerful, but `expect(...).toContain({ id: "a" })` reads as intent and an asymmetric matcher like `expect.objectContaining({ id: "a" })` tolerates extra fields without bricking the test on every new field.

**Modern guidance** — if you stay stdlib-only, keep `assert/strict`. If you need `expect`, use Vitest/Jest's `expect` alongside `node:test` or replace the runner entirely — don't mix two runners in one file. Remember `toBe` is `Object.is`: `expect(NaN).toBe(NaN)` passes, `expect(-0).toBe(0)` fails — which surprises `===` users.

```js
const assert = require("node:assert/strict");

// Mini expect façade over assert/strict — illustrates semantics without a dep
function expect(got) {
  return {
    toBe(want) { // Object.is — NaN equal, -0 vs 0 different
      if (!Object.is(got, want)) throw new assert.AssertionError({ actual: got, expected: want, operator: "expect.toBe" });
    },
    toEqual(want) { assert.deepStrictEqual(got, want); },
    toContain(item) {
      if (typeof got === "string") {
        if (!got.includes(item)) throw new assert.AssertionError({ actual: got, expected: item, operator: "expect.toContain" });
      } else if (Array.isArray(got)) {
        if (!got.includes(item)) throw new assert.AssertionError({ actual: got, expected: item, operator: "expect.toContain" });
      } else throw new TypeError("toContain on string or array");
    },
    toThrow(expected) {
      assert.throws(got, expected);
    }
  };
}

// toBe is Object.is — NaN and -0 behave per Object.is
expect(NaN).toBe(NaN);
expect(0).toBe(0);
try { expect(-0).toBe(0); } catch (e) { assert.equal(e instanceof assert.AssertionError, true); }

// toEqual is deepStrictEqual (NaN-aware)
expect({ a: NaN }).toEqual({ a: NaN });

// toContain for arrays/strings
expect([1,2,3]).toContain(2);
expect("hello").toContain("ell");

// toThrow with name/message shape (same as assert.throws)
expect(() => { throw new TypeError("bad id"); }).toThrow({ name: "TypeError", message: /bad/ });

// Anti-pattern: toBe for objects checks reference, not value
const a = { x: 1 }, b = { x: 1 };
try { expect(a).toBe(b); } catch (e) { assert.equal(e.operator, "expect.toBe"); } // fails — different refs
expect(a).toEqual(b); // passes — deep value
console.log("expect semantics ok");
```

**Common confusion** — "`toBe` is `===`." It's `Object.is`: `NaN === NaN` is false but `Object.is(NaN, NaN)` true; `-0 === 0` true but `Object.is(-0, 0)` false. Tests that assert `NaN` via `===` always fail; `toBe`/`strictEqual` via `Object.is` fixes it.

### 2) **Snapshots — file snapshots for large outputs, when they rot, and how `--update` gates edits**

**Definition** — a snapshot serializes a large value (rendered string, `JSON.stringify` of an API payload, error stack) to a `.snap` file. On the next run, the current value is compared to the saved snapshot. An updated implementation requires `--update` (Jest/Vitest) or manual file rewrite to re-baseline.

**Modern guidance** — use snapshots for *wide* outputs (formatted report, error message) not *narrow* logic (`clamp`). Scope snapshots to stable serialization (sorted keys, deterministic `Date` via clock seam). Treat snapshot churn as a code-review signal: a 200-line snapshot diff for a one-line logic change means the snapshot is too coarse. Node 22 has no built-in snapshot API; use Vitest/Jest or implement a tiny helper that writes `__snapshots__/name.snap`.

```js
const assert = require("node:assert/strict");
const fs = require("node:fs");
const path = require("node:path");
const os = require("node:os");

// Tiny snapshot helper — writes once, reads thereafter
function expectSnapshot(got, snapPath) {
  const canon = JSON.stringify(got, Object.keys(got).sort(), 2);
  if (!fs.existsSync(snapPath)) {
    fs.mkdirSync(path.dirname(snapPath), { recursive: true });
    fs.writeFileSync(snapPath, canon);
    return; // first write — next run will compare
  }
  const want = fs.readFileSync(snapPath, "utf8");
  assert.equal(canon, want);
}

// Stable serialization: sort keys, freeze clock
function buildReport({ id, qty }) {
  return { id, qty, total: qty * 9, generatedAt: "2026-01-01T00:00:00.000Z" }; // clock seam freezes date
}
const tmp = path.join(os.tmpdir(), `snap-${Date.now()}.snap`);
expectSnapshot(buildReport({ id: "a", qty: 3 }), tmp);
expectSnapshot(buildReport({ id: "a", qty: 3 }), tmp); // second run compares — passes
try {
  expectSnapshot(buildReport({ id: "a", qty: 4 }), tmp);
} catch (e) {
  assert.equal(e instanceof assert.AssertionError, true); // snapshot mismatch — needs explicit --update
}

// Bad: snapshot that captures nondeterminism (Date.now, Math.random, object key order)
function badSnapshot() { return { at: Date.now(), ids: ["b","a"] }; } // wall clock + unsorted
// Good: deterministic — frozen clock, sorted
function goodSnapshot(clock) { return { at: clock.now(), ids: ["a","b"].sort() }; }
assert.equal(goodSnapshot({ now: () => 1_700_000_000_000 }).at, 1_700_000_000_000);

fs.unlinkSync(tmp);
console.log("snapshots ok");
```

**Mentor note** — require a comment `// snapshot: update with --update after intentional payload change` near the assertion. In review, a 1-line code change that churns 80 snapshot lines is a hint to narrow the snapshot to a stable subset (`expect.objectContaining`) or to switch to a focused `deepStrictEqual` on the branch you care about.

### 3) **Custom matchers — extending with asymmetric matchers and writing your own without a framework**

**Definition** — an *asymmetric matcher* is a value wrapper that participates in an equality check but defines its own predicate (`expect.stringContaining("hi")` matches any string containing `"hi"`). A *custom matcher* is a function `expect.extend({ toBeValidId(received) { … } })` that returns `{ pass, message }` and integrates with the runner's diff.

**Modern guidance** — two lightweight stdlib alternatives without pulling Jest: (1) asymmetric helpers that return an object with `__asymmetric: true` and `test(v)`; (2) a plain predicate `assert.equal(isValidId(got), true)` with a message. If you need rich diffs, add `expect` via `vitest`/`jest` and extend once per repo.

```js
const assert = require("node:assert/strict");

// Asymmetric matchers — no framework, deepEqual-aware
function stringContaining(sub) {
  return { __asymmetric: true, test(v) { return typeof v === "string" && v.includes(sub); }, toString(){ return `stringContaining(${sub})`; } };
}
function objectContaining(partial) {
  return { __asymmetric: true, test(v) {
    if (typeof v !== "object" || v === null) return false;
    for (const k of Object.keys(partial)) if (!Object.is(v[k], partial[k])) return false;
    return true;
  }};
}

// deepEqual that understands asymmetric wrappers
function asymmetricEqual(got, want) {
  if (want && want.__asymmetric) return want.test(got);
  if (typeof want === "object" && want && typeof got === "object" && got) {
    for (const k of Object.keys(want)) if (!asymmetricEqual(got[k], want[k])) return false;
    return true;
  }
  return Object.is(got, want);
}
assert.equal(asymmetricEqual("hi there", stringContaining("there")), true);
assert.equal(asymmetricEqual({ id: "a", extra: 1 }, objectContaining({ id: "a" })), true);

// Custom matcher as a plain predicate + message — stdlib-simple
function isValidId(id) { return typeof id === "string" && id.trim().length >= 3 && !/\s/.test(id); }
function expectValidId(got) {
  const pass = isValidId(got);
  return { pass, message: () => `expected ${JSON.stringify(got)} to be a valid id (trimmed length >=3, no spaces)` };
}
assert.equal(expectValidId("abc").pass, true);
assert.equal(expectValidId(" a").pass, false);
assert.equal(expectValidId(" a").message().includes("valid id"), true);

// If you import Vitest/Jest expect, the shape is:
// expect.extend({ toBeValidId(received){ const pass = isValidId(received); return { pass, message: () => `...`} } })
// expect("abc").toBeValidId();
console.log("custom matchers ok");
```

**Performance note** — asymmetric matchers allocate per assertion; on 10 k-iteration micro suites they add GC pressure. For hot-path tests, prefer a plain `assert.equal` with a message — see [Performance 2.2](<../../../19 Performance and JIT Mindset/sections/2. Shapes inline caches and polymorphism/2.2. Polymorphic megamorphic and dictionary mode costs.md>).
