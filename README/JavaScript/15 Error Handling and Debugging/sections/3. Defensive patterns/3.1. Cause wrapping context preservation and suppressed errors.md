## 3.1. Cause wrapping context preservation and suppressed errors

### 1) **Wrap low-level causes — `throw new AppError(msg, {cause: low})` preserves chain**

**Definition** — *wrapping* catches a low-level `Error` at a module boundary and throws a higher-level domain error with `{cause: low}`, preserving both messages and stacks. Walking `err.cause` to root yields the full causal chain; `err.message` alone does not.

**What problem it solves / history** — without `cause`, callers saw `"load failed"` but not `"ENOENT: no such file /data/u42.json"` or its stack. Libraries invented `innerError`/`originalError` ad-hoc; ES2022 standardized `cause`.

**Modern guidance** — wrap at every trust/boundary crossing (I/O → service → controller). Log the whole chain, not just the top.

```js
// BAD — erase cause
async function loadUserBad(id){
  try { return await readFileBad(id); } catch (e) { throw new Error(`load ${id} failed`); } // lost file+stack
}
async function readFileBad(id){ throw new Error("ENOENT: /data/u42.json"); }

// GOOD — preserve cause
class AppError extends Error { constructor(msg, opts){ super(msg, opts); this.name="AppError"; } }
async function readFileGood(id){ throw Object.assign(new Error(`ENOENT: /data/${id}.json`), { code:"ENOENT" }); }
async function loadUserGood(id){
  try { return await readFileGood(id); }
  catch (low) { throw new AppError(`load ${id} failed`, { cause: low }); }
}
(async () => {
  try { await loadUserGood("u42"); } catch (e) {
    console.log(e.name, e.message);
    console.log("cause:", e.cause.message, e.cause.code);
    console.log("high stack:", e.stack.split("\n")[0]);
    console.log("cause stack:", e.cause.stack.split("\n")[0]);
    for (let c=e; c; c=c.cause) console.log(" chain:", c.name, c.message.slice(0,30));
  }
})();

// Helper — walk chain to root
function rootCause(e){ while(e?.cause) e=e.cause; return e; }
function printChain(e){ const out=[]; for(let c=e;c;c=c.cause) out.push(`${c.name}: ${c.message}`); console.log(out.join(" -> caused by -> ")); }
try { throw new AppError("save", { cause: new TypeError("bad JSON") }); } catch(e){ printChain(e); console.log("root:", rootCause(e).name); }
```

### 2) **When to wrap vs rethrow bare vs aggregate — the decision table**

**Definition** — three strategies: *rethrow bare* (`throw e`) keeps original; *wrap* adds context; *aggregate* (`AggregateError`) holds many peer failures.

| Situation | Strategy | Example |
|-----------|----------|---------|
| Passing through, caller understands low error | rethrow bare | `catch (e){ log(e); throw e }` |
| Crossing boundary, need context | wrap with `cause` | `catch (low){ throw new AppError("load "+id, {cause:low}) }` |
| Fan-out, multiple peer tasks may fail | `AggregateError` over `errors` | `Promise.any` or `validateAll` |
| Cleanup throw competes with primary | `SuppressedError`/`using` | `using` resource disposes while body threw |

```js
// Rethrow bare — keep original when caller handles it:
function passthrough(fn){
  try { return fn(); } catch (e){ console.log("passthrough log", e.name); throw e; }
}
try { passthrough(()=>{ throw new RangeError("bad"); }); } catch(e){ console.log("kept", e.name); }

// Wrap at boundary
function boundary(id){
  try { if(!id) throw new TypeError("id required"); } catch(low){ throw new AppError("boundary "+id, { cause: low }); }
}
try { boundary(""); } catch(e){ console.log(e.name, e.cause.name); }

// Aggregate on fan-out
function validateBatch(xs){
  const errs = xs.map((x,i)=> x==null ? new AppError(`item ${i} missing`) : null).filter(Boolean);
  if(errs.length) throw new AggregateError(errs, "batch invalid");
  return xs;
}
try { validateBatch([1,null,3]); } catch(e){ console.log(e.name, e.errors.length); }

// Suppressed — cleanup throw vs primary (see §4)
console.log("suppressed when both throw — using/Explicit Resource Management");
```

### 3) **Suppressed on cleanup — `using`/`Symbol.dispose` wiring `SuppressedError`**

**Definition** — `using` (ES2024) calls `resource[Symbol.dispose]()` on block exit. If both the body and dispose throw, the engine produces `SuppressedError` with `error` (primary) and `suppressed` (cleanup) — never losing either. Without `using`, `finally { dispose() }` that throws discards the primary.

**Modern guidance** — for any resource whose `close`/`dispose` may throw (files, locks, DB tx), expose `[Symbol.dispose]` and consume via `using`. When `using` is unavailable, manual `try/catch` in `finally` must combine via `SuppressedError`.

```js
// BAD — finally throw loses primary:
function badResource(){
  let primary;
  try { primary = new Error("query failed"); throw primary; }
  finally {
    const closeErr = new Error("close failed");
    throw closeErr; // primary lost — caller sees only "close failed"
  }
}
try { badResource(); } catch(e){ console.log("bad lost primary:", e.message); } // close failed

// GOOD manual suppressed — preserve both:
function goodResourceManual(){
  try { throw new Error("query failed"); }
  finally {
    try { throw new Error("close failed"); } catch (sup){
      const Suppressed = globalThis.SuppressedError ?? class SE extends Error {
        constructor(error, suppressed){ super("cleanup suppressed"); this.name="SuppressedError"; this.error=error; this.suppressed=suppressed; }
      };
      // Need primary — capture it outside finally in real code:
      console.log("would create SuppressedError with primary +", sup.message, Suppressed.name);
    }
  }
}
try { goodResourceManual(); } catch (e) { console.log("goodResource preserved:", e.message); }

// Using when engine supports it — string-based check to avoid SyntaxError on old Node
const hasUsing = (()=>{ try{ Function("using x = null"); return true; } catch{ return false; } })();
console.log("has `using`:", hasUsing);
if (hasUsing) {
  console.log("pattern: { using db = acquire(); db.query(); } // dispose auto, SuppressedError on double-throw");
} else {
  // Polyfill-like dispose pattern without syntax:
  class File { constructor(p){ this.p=p; } [Symbol.dispose](){ console.log("File dispose", this.p); } }
  // Simulate using block manually:
  { const f = new File("/tmp/x"); try { console.log("work with", f.p); } finally { f[Symbol.dispose](); } }
}

// Async dispose — await using + Symbol.asyncDispose
console.log(typeof Symbol.asyncDispose, "for async resources (DB, fetch stream)");
```

### 4) **Common confusion**

- "`cause` replaces `message` concatenation" — concatenate context in `message` *and* set `cause`; `message` is for humans at this layer, `cause` for machines/root analysis.
- "`AggregateError` vs `SuppressedError`" — `AggregateError` = many peer errors (`errors` array); `SuppressedError` = exactly two ( `error` primary, `suppressed` cleanup).
- "`finally` preserves cause automatically" — it does not; you must wrap the suppressed error explicitly or use `using`.
- "Logging `err.stack` is enough" — `err.cause.stack` is separate; Node's `console.error` now prints `Caused by:` chains, but custom loggers must walk `cause`.

### 5) **Performance note**

Walking `cause` chains at log time is O(depth) string concatenation. Cap walking at 10 or deduplicate cycles (a buggy `cause` self-reference). Creating long `cause` chains (depth 100) retains 100 stacks in memory — truncate beyond operational depth.

### 6) **Mentor note**

Require every cross-boundary `catch` to either `throw e` (passthrough with comment) or `throw new DomainError(..., {cause:e})`. Flag `throw new Error("x: "+ e.message)` without `cause`. For resources, require `using` or explicit `SuppressedError` handling in `finally`; ban `finally { close(); }` without `try/catch` around `close()`.
