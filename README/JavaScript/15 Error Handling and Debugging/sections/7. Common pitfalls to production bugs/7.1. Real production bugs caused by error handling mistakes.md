## 7.1. Real production bugs caused by error handling mistakes

### 1) **String throw that erases the stack — `throw "not found"` breaks Sentry and `instanceof`**

**Bug** — code `throw \`not found: ${id}\`` loses file:line, grouping by `name`, and `cause` chain. `catch (e instanceof Error)` bypasses it, so `e.message` access throws `TypeError: Cannot read properties of string`, masking original.

**Fix** — always `throw new NotFoundError(\`not found: ${id}\`, {cause: low})`. Group by `name`/`code`, walk `cause`.

```js
// Bug — string throw
async function loadBad(id){
  try { throw `ENOTFOUND ${id}`; } catch (e){ // e is string — no stack
    console.log("bad caught type:", typeof e, e instanceof Error);
    try { console.log(e.message); } catch (t){ console.log("then secondary TypeError:", t.name); }
    console.log("no stack:", typeof e.stack); // undefined
  }
}
loadBad("u42");

// Fix — Error throw with cause and name
class NotFoundError extends Error { constructor(m,o){ super(m,o); this.name="NotFoundError"; this.code="NOT_FOUND"; } }
async function loadGood(id){
  try { throw new Error("ENOENT db"); } catch (low){ throw new NotFoundError(`not found: ${id}`, { cause: low }); }
}
loadGood("u42").catch(e=>{
  console.log("good:", e.name, e.message, "cause", e.cause.message, e instanceof Error);
  console.log("good stack:", typeof e.stack);
});
```

### 2) **`finally` return that swallows the original throw — `try { throw A } finally { return B }` loses `A`**

**Bug** — cleanup returning normalizes an error to a success (`"swallowed"`), caller sees `200` with wrong payload, alert never fires.

**Fix** — never `return`/`throw` inside `finally`; use `using` or manual `SuppressedError` combining.

```js
// Bug reproduction
function badProcess(){
  try { throw new Error("primary: query failed"); }
  finally { return "ok"; } // swallows
}
console.log("badProcess returns:", badProcess()); // "ok" — primary lost, caller thinks success

// Fix — no return in finally
function goodProcess(){
  let out;
  try { throw new Error("primary: query failed"); }
  catch(e){ out = e; throw e; }
  finally { console.log("goodProcess finally cleanup, no return"); }
}
try { goodProcess(); } catch(e){ console.log("goodProcess preserved:", e.message); }

// Real-world pattern — transaction:
function badTx(){
  let tx = { commit(){}, rollback(){ return "rolled"; } };
  try { throw new Error("write failed"); tx.commit(); return "committed"; }
  catch(e){ tx.rollback(); throw e; }
  finally { return tx.rollback(); } // bug: finally return swallows either path
}
console.log("badTx:", badTx()); // "rolled" — always returns that

function goodTx(){
  const tx = { commit(){ return "committed"; }, rollback(){ console.log("rollback"); } };
  try { throw new Error("write failed"); }
  catch(e){ try { tx.rollback(); } catch(sup){ /* combine via SuppressedError */ console.log("suppressed", sup.message); } throw e; }
}
try { goodTx(); } catch(e){ console.log("goodTx preserved:", e.message); }
```

### 3) **Unawaited `map(async` without `await all` and `any` vs `all` misuse — orphan rejects + `AggregateError` misread**

**Bug** — `items.map(async x => ...)` without `await Promise.all(...)` leaves orphans that trigger `unhandledrejection` and half-applied writes. `Promise.any` rejection misread as single error loses `errors` array.

**Fix** — `await Promise.all`/`allSettled` for fan-out, `AggregateError` walk for `any`.

```js
// Bug — map async without all
async function badBatch(ids){
  const ps = ids.map(async id => { // orphans — not awaited
    if(id===2) throw new Error("id 2 fail");
    console.log("bad processed", id);
  });
  ps.forEach(p=>p.catch(()=>{})); // suppress for demo — real bug leaks to unhandledRejection
  console.log("badBatch returned before work finished");
}
badBatch([1,2,3]).catch(()=>{});
setTimeout(()=>console.log("badBatch orphans may have emitted unhandledRejection"), 20);

// Fix — bounded all with cause per task
async function goodBatch(ids){
  const tasks = ids.map(async id => {
    try {
      if(id===2) throw new Error("db miss 2");
      console.log("good processed", id);
      return id;
    } catch(low){ throw new Error(`item ${id}`, { cause: low }); }
  });
  const settled = await Promise.allSettled(tasks);
  const errs = settled.filter(x=>x.status==="rejected").map(x=>x.reason);
  if(errs.length) throw new AggregateError(errs, `${errs.length} failed`);
  return settled.filter(x=>x.status==="fulfilled").map(x=>x.value);
}
setTimeout(async () => {
  try { await goodBatch([1,2,3]); } catch(e){ console.log("goodBatch AggregateError:", e.name, e.errors.map(x=>x.message+":"+x.cause.message)); }
  // any vs all demo:
  try { await Promise.any([Promise.reject(new Error("a")), Promise.reject(new Error("b"))]); } catch(e){ console.log("any AggregateError errors:", e.errors.map(x=>x.message)); }
}, 40);
```

### 4) **Lost `cause` and unmapped stack — rethrow without `{cause}` + missing `--enable-source-maps` leaves on-call blind**

**Bug** — `catch (low) { throw new Error("load failed") }` drops file `repo.js:42` and minified `dist/app.js:1:12345` without a map → on-call sees `Error: load failed` with no *Caused by* and no original line.

**Fix** — `throw new Error("load failed", {cause: low})` + `node --enable-source-maps` + log `fullStack` helper that walks `cause` + `errors`.

```js
// Bug — lost cause
async function repoLow(id){ throw Object.assign(new Error(`db ENOENT ${id}`), { code:"ENOENT" }); }
async function serviceBad(id){
  try { await repoLow(id); } catch(low){ throw new Error(`load ${id} failed`); } // no cause
}
serviceBad("u99").catch(e=>{
  console.log("bad stack head:", e.stack.split("\n")[0]);
  console.log("bad cause:", e.cause); // undefined — blind
  console.log("bad stack contains repo?", e.stack.includes("repoLow")); // often yes, but not causal link
});

// Fix — cause + source maps concept
async function serviceGood(id){
  try { await repoLow(id); } catch(low){ throw new Error(`load ${id} failed`, { cause: low }); }
}
function fullStack(e){
  let s = e.stack;
  for(let c=e.cause; c; c=c.cause) s += "\nCaused by: " + c.stack;
  if(e instanceof AggregateError) for(const sub of e.errors) s += "\n - " + sub.stack.split("\n")[0];
  return s;
}
serviceGood("u99").catch(e=>{
  console.log("good fullStack head:", fullStack(e).split("\n")[0]);
  console.log("good cause head:", e.cause.stack.split("\n")[0]);
  console.log("remap: run with node --enable-source-maps to map dist->src");
  console.log("good includes repoLow?", fullStack(e).includes("repoLow"));
});

// Checklist for deploy — run with flags:
// node --enable-source-maps --async-stack-traces --trace-warnings app.js
console.log("deploy flags: --enable-source-maps --async-stack-traces --trace-warnings");
```
