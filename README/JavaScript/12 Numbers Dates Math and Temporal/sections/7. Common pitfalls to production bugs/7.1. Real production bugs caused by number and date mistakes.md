## 7.1. Real production bugs caused by number and date mistakes

### 1) **Silent ID truncation past `MAX_SAFE_INTEGER` — JSON round-trip collapses `9007199254740993` to `9007199254740992`**

**Incident** — Node service stores 64-bit Snowflake IDs as `number`. First 10k rows fine (IDs < `2^53`), then collisions: two different entities share `id 9007199254740993` and `9007199254740992` → same `number` → Map dedup loses one. `JSON.stringify` already emitted the collapsed value, so the corruption persists to DB and clients — silent, no throw.

```js
// Reproduce — collapse at parse time
const idA = 9007199254740992; // 2^53
const idB = 9007199254740993; // 2^53+1 — not representable
console.log(idA === idB); // true — collapse
console.log(JSON.stringify({ id: idB })); // {"id":9007199254740992} — already wrong on wire
const seen = new Map();
seen.set(idA, "A");
seen.set(idB, "B");
console.log(seen.size); // 1 — collision
console.log(seen.get(idA)); // "B" — overwritten

// Fix — keep IDs as string or BigInt on wire, never number
function toIdString(v) {
  if (typeof v === "bigint") return String(v);
  if (typeof v === "string") return v; // already safe
  if (typeof v === "number") {
    if (!Number.isSafeInteger(v)) throw new RangeError(`ID ${v} exceeds safe integer — send as string`);
    return String(v);
  }
  throw new TypeError("ID must be string|bigint|safe number");
}
console.log(toIdString("9007199254740993")); // "9007199254740993" — preserved
console.log(toIdString(42)); // "42"
try { toIdString(9007199254740993); } catch (e) { console.log(e.name); } // RangeError — number form rejected
console.log(JSON.stringify({ id: toIdString("9007199254740993") })); // {"id":"9007199254740993"} — correct

// BigInt alternative — but JSON does not support BigInt, so string is the wire-safe choice
console.log(typeof 9007199254740993n); // "bigint" — exact, but JSON.stringify throws (see 1.2)
try { JSON.stringify({ id: 9007199254740993n }); } catch (e) { console.log(e.name); } // TypeError
```

### 2) **`parseInt("08")` without radix and `Number("") → 0` poisoning reducers**

**Incident** — CSV ledger: `amount` column has empty fields for zero. `records.map(r => Number(r.amount)).reduce((a,b)=>a+b,0)` yields `0` for `""` instead of `NaN`, so missing amounts are summed as `0` — ledger off by hundreds, no error.

```js
// Poison via Number("") → 0
const rows = [{ amount: "10.5" }, { amount: "" }, { amount: "20" }];
console.log(rows.map(r => Number(r.amount))); // [10.5, 0, 20] — "" → 0, not NaN
console.log(rows.map(r => Number(r.amount)).reduce((a,b)=>a+b,0)); // 30.5 — missing field counted as 0

// Fix — guard empty before Number, or use parseFloat/Number with empty check
function parseAmount(s) {
  if (s === "" || s == null) return NaN; // or throw — whichever your domain requires
  const n = Number(s);
  if (!Number.isFinite(n)) throw new TypeError(`invalid amount ${JSON.stringify(s)}`);
  return n;
}
console.log(rows.map(r => { try { return parseAmount(r.amount); } catch { return "ERR"; } })); // [10.5, NaN, 20] — visible

// The map(parseInt) trap — affects every array-utility usage
console.log(["10", "10", "10"].map(parseInt)); // [10, NaN, 2] — index as radix (see 2.1)
console.log(["10", "10", "10"].map(s => parseInt(s, 10))); // [10, 10, 10] — correct

// Octal legacy — code assumed parseInt("08") → 8 but old engine gave 0 (now fixed, but radix-less still flagged)
console.log(parseInt("08", 10), parseInt("08")); // 8 8 — modern same, but intent unclear without radix
```

### 3) **`Math.random` for tokens/shuffles that attackers predict**

**Incident** — auth service generates password-reset tokens via `Math.random().toString(36).slice(2)`. Attacker observes ~10 sequential tokens, recovers xoshiro256+ state, predicts next token → account takeover. Shuffle uses `arr.sort(()=>Math.random()-0.5)` → biased, plus weak RNG.

```js
// Vulnerable token (DO NOT USE)
function badToken() { return Math.random().toString(36).slice(2, 10); }
console.log(badToken().length > 0); // true — but predictable

// Secure token via CSPRNG
function goodToken(bytes = 16) {
  const a = new Uint8Array(bytes);
  crypto.getRandomValues(a);
  return [...a].map(b => b.toString(16).padStart(2, "0")).join("");
}
console.log(goodToken().length); // 32 — 128-bit CSPRNG, not predictable

// Biased shuffle — sort with random comparator violates transitivity → some permutations rare
function badShuffle(arr) { return [...arr].sort(() => Math.random() - 0.5); }
console.log(badShuffle([1, 2, 3, 4]).length); // 4 — shuffles, but biased

// Correct shuffle — Fisher–Yates with CSPRNG
function goodShuffle(arr) {
  const a = [...arr];
  const nodeCrypto = (() => { try { return require("node:crypto"); } catch { return null; } })();
  const randInt = nodeCrypto ? (n) => nodeCrypto.randomInt(0, n) : (n) => { const b = new Uint32Array(1); crypto.getRandomValues(b); return b[0] % n; };
  for (let i = a.length - 1; i > 0; i--) {
    const j = randInt(i + 1);
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}
console.log(goodShuffle([1, 2, 3, 4]).length); // 4 — unbiased
console.log(goodShuffle([1, 2, 3, 4]).sort((x,y)=>x-y).join(",")); // "1,2,3,4" — permutation preserves elements
```

### 4) **Legacy `Date` DST trap — `setDate(getDate()+1)` skips or repeats a day on transition**

**Incident** — billing cron computes "next billing date" via `d.setDate(d.getDate()+1)` on host TZ. On DST spring-forward, the 23h day plus gap normalization skips the 02:30 billing slot → customer charged a day late. On fall-back, the 25h repeat charges twice. Timestamp `+86400000` has the same problem (24h ≠ calendar day).

```js
// Reproduce with UTC vs calendar day divergence (host Tehran has no DST 2026 — demonstrate via UTC + Intl America/New_York)
const startUTC = new Date("2026-03-08T06:30:00.000Z"); // 01:30 EST NY
const plus24h = new Date(startUTC.getTime() + 24 * 60 * 60 * 1000);
console.log(startUTC.toISOString(), plus24h.toISOString()); // 24h later UTC — always 24h
console.log(new Intl.DateTimeFormat("en-US", { timeZone: "America/New_York", hour: "2-digit", minute: "2-digit", timeZoneName: "short" }).format(startUTC)); // 01:30 AM EST
console.log(new Intl.DateTimeFormat("en-US", { timeZone: "America/New_York", hour: "2-digit", minute: "2-digit", timeZoneName: "short" }).format(plus24h)); // 02:30 AM EDT — not 01:30 (gap day)

// Legacy "next day" via host local — breaks for arbitrary zones and DST
function nextDayHostLocal(date) {
  const d = new Date(date);
  d.setDate(d.getDate() + 1); // host local calendar +1 — wrong for NY when host is Tehran
  return d;
}
console.log(nextDayHostLocal(new Date("2026-03-08T06:30:00.000Z")).toISOString()); // host-local +1 — not NY calendar

// Fix — Temporal calendar day (DST-aware, zone-explicit)
if (typeof Temporal !== "undefined") {
  const zdt = Temporal.ZonedDateTime.from("2026-03-08T01:30:00-05:00[America/New_York]");
  console.log(String(zdt.add({ days: 1 }))); // "2026-03-09T01:30:00-04:00[America/New_York]" — same wall time, offset changed
  console.log(String(zdt.add({ hours: 24 }))); // "2026-03-09T02:30:00-04:00[America/New_York]" — 24h later wall is 02:30
} else {
  console.log("Temporal not available — DST fix requires polyfill; legacy 24*60*60*1000 is duration, not calendar day (see 4.2)");
  // Correct legacy workaround for UTC calendar only:
  function addCalendarDaysUTC(date, days) { const d = new Date(date); d.setUTCDate(d.getUTCDate() + days); return d; }
  console.log(addCalendarDaysUTC(new Date("2026-03-08T06:30:00.000Z"), 1).toISOString()); // "2026-03-09T06:30:00.000Z" — UTC calendar day (not NY)
}

// Month overflow normalization — another Date trap
const jan31 = new Date(Date.UTC(2026, 0, 31));
jan31.setUTCMonth(1); // Feb 31 → Mar 3 overflow
console.log(jan31.toISOString()); // "2026-03-03T00:00:00.000Z" — not "Feb 28" (Temporal PlainDate would give Feb 28)
if (typeof Temporal !== "undefined") {
  console.log(String(Temporal.PlainDate.from("2026-01-31").add({ months: 1 }))); // "2026-02-28" — correct calendar
}
```

### 5) **Mentor note**

Each incident maps to a grep: `Number.isSafeInteger` for ID collapse, `Number("")` for empty poisoning, `Math.random` for token/shuffle, `setDate`/`24*60*60*1000` for DST. Require the grep list in CI and the fix pattern in the incident post-mortem template.
