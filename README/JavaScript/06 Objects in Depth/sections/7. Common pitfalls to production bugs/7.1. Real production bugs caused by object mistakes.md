## 7.1. Real production bugs caused by object mistakes

### 1) **Shared nested state after shallow clone**

**The bug** — merging defaults with user options via spread/`assign` shares nested objects. Caller mutates the result, default is corrupted for every later request — a cross-request leak that surfaces only under concurrency/load.

```js
// Library default that lives for process lifetime
const defaultConfig = {
  port: 3000,
  retry: { attempts: 3, backoff: "exponential" },
  tags: ["default"]
};

function buildConfig(userOpt) {
  // Shallow merge — nested retry/tags stay shared
  return { ...defaultConfig, ...userOpt };
}

// Request 1: customizes retry, returns config object that caller later mutates
const req1 = buildConfig({ port: 8080 });
req1.retry.attempts = 99; // intends per-request tweak
req1.tags.push("req1");

console.log(defaultConfig.retry.attempts); // 99 — BUG: default corrupted
console.log(defaultConfig.tags); // ["default","req1"] — BUG

// Fix: deep clone extendable parts or deep-merge, then freeze defaults
function buildConfigFixed(userOpt) {
  const base = typeof structuredClone === "function" ? structuredClone(defaultConfig) : JSON.parse(JSON.stringify(defaultConfig));
  // or deep merge via manual walk
  return Object.assign(base, userOpt, userOpt.retry ? { retry: { ...base.retry, ...userOpt.retry } } : {});
}
const defaultConfig2 = Object.freeze({ port: 3000, retry: Object.freeze({ attempts: 3 }), tags: Object.freeze(["default"]) });
// Even shallow share now throws in strict mode if fix missed — surfaces early
```

- *Why it slipped* — `{ ...defaults, ...overrides }` reads as "clone" but the contract is shallow; review greps miss nested mutation far from the spread line. Mitigations: `deepFreeze(defaultConfig)` at module load so accidental write throws, or deep-merge helper, or never hand out references to internals (return `structuredClone`).

### 2) **Silent write silently ignored after freeze in sloppy mode**

Module exports a frozen enum map; consumer running sloppy-mode script assigns `MAP.A = "new"` — no `TypeError`, assignment returns `"new"` but the value stays `"OLD"`. Later persistence writes the stale `"OLD"` even though the code "set" it.

```js
// Frozen enum — intended constant
const STATUS = Object.freeze({ READY: "ready", DONE: "done" });

// Sloppy consumer (e.g., legacy <script> without "use strict" or non-module)
function sloppyUpdate() {
  STATUS.READY = "not-ready"; // sloppy silent no-op — returns "not-ready" but property stays "ready"
  console.log(STATUS.READY); // ready — stale
  STATUS.NEW = "new"; // addition ignored (non-extensible)
  console.log("NEW" in STATUS); // false
}
sloppyUpdate();

// Strict/modules fail loud — the bug surfaces immediately
function strictUpdate() { "use strict";
  try { STATUS.READY = "not-ready"; } catch (e) { console.log("strict", e.name); } // TypeError
  try { STATUS.NEW = "new"; } catch (e) { console.log("strict add", e.name); } // TypeError
}
strictUpdate();

// Predicates expose the gap
console.log(Object.isFrozen(STATUS)); // true — guard before write
console.log(delete STATUS.READY); // false — delete blocked (strict TypeError)

// Fix: ensure consumer modules are strict (all ESM is strict) and flag sloppy-mode mutation of frozen objects in CI with a linter rule that forces Object.isFrozen guard or freeze only in dev + trust prod never mutates
```

### 3) **Prototype pollution via __proto__ / constructor.prototype**

Untrusted JSON (query body, webhook) is merged into a config with `Object.assign(target, JSON.parse(body))` or spread. Attacker sends `{ "__proto__": { "isAdmin": true } }`. Since `Object.assign` assigns by `target["__proto__"] = value` (which on ordinary objects with `Object.prototype` in chain is a setter that mutates `[[Prototype]]`), later `{} .isAdmin` is `true` — privilege escalation.

```js
// Vulnerable merge
function merge(target, source) {
  for (const k of Object.keys(source)) target[k] = source[k]; // assigns "__proto__" via setter if target is ordinary
  return target;
}
const config = { theme: "light" };
const payload = JSON.parse('{"__proto__": {"isAdmin": true}}');
merge(config, payload);
console.log({}.isAdmin); // true — BUG: Object.prototype polluted

// Clean up for demo isolation (never do this in real pollution without restart)
delete Object.prototype.isAdmin;
console.log({}.isAdmin); // undefined — cleaned

// Fix 1: null-prototype target has no __proto__ setter to pollute
const safeTarget = Object.create(null);
safeTarget.theme = "light";
Object.assign(safeTarget, JSON.parse('{"__proto__": {"x": 1}}'));
console.log(safeTarget["__proto__"]); // { x: 1 } — plain string key, not a link (since target is null-proto, assignment defines data)
console.log(Object.getPrototypeOf(safeTarget)); // null — untouched
console.log({}.x); // undefined — not polluted

// Fix 2: sanitize keys on merge — reject dangerous keys
function safeAssign(target, source) {
  for (const k of Reflect.ownKeys(source)) {
    if (k === "__proto__" || k === "constructor" || k === "prototype") continue; // block pollution vectors
    const desc = Object.getOwnPropertyDescriptor(source, k);
    if (desc) Object.defineProperty(target, k, desc); // descriptor copy avoids setter-trigger path
  }
  return target;
}
const tgt2 = {};
safeAssign(tgt2, JSON.parse('{"__proto__": {"polluted": true}}'));
console.log(Object.getPrototypeOf(tgt2) === Object.prototype); // true — not polluted
console.log({}.polluted); // undefined

// Fix 3: JSON.parse reviver that drops polluting keys at parse time
const input = '{"__proto__": {"a": 1}, "ok": 2}';
const sanitized = JSON.parse(input, (k, v) => (k === "__proto__" ? undefined : v));
console.log(sanitized); // { ok: 2 } — __proto__ stripped
```

- *Why it was missed* — `Object.assign` docs don't warn that `assign({}, JSON.parse(input))` pollutes `Object.prototype` when input contains `__proto__`; teams trust JSON as data. Required review rule: never `assign`/spread untrusted keys onto an ordinary object without either a null-proto target or a key filter. Use `Map` for untrusted-key storage when string keys may be hostile.

### 4) **Descriptor-default trap silences iteration**

Helper added via `defineProperty({ value: fn })` is non-enumerable by default; later `Object.keys(registry)` loop "loses" the entry and feature appears broken — not a runtime error, just silent omission.

```js
const registry = {};
Object.defineProperty(registry, "healthCheck", { value() { return "ok"; } }); // missing enumerable:true
registry["readiness"] = () => "ready"; // literal-enumerable

console.log(Object.keys(registry)); // [ 'readiness' ] — healthCheck missing (enumerable false)
console.log(Object.values(registry).length); // 1 — same loss
console.log(JSON.stringify(registry)); // {"readiness": ... } — not serialized
console.log(Reflect.ownKeys(registry)); // [ 'healthCheck','readiness' ] — reflection finds it

// Spread also drops it
console.log({ ...registry }); // { readiness: ... } — healthCheck not copied

// Fix: spell enumerability
const registry2 = {};
Object.defineProperty(registry2, "healthCheck", { value() { return "ok"; }, writable: true, enumerable: true, configurable: true });
console.log(Object.keys(registry2)); // [ 'healthCheck' ] — now visible
// Or prefer literal for enumerable data
const registry3 = { healthCheck() { return "ok"; } };
console.log(Object.keys(registry3)); // [ 'healthCheck' ] — concise method is enumerable (→ [1.1](../1. Object creation and literals/1.1. Object literals shorthand computed properties and spread.md>))
```

### 5) **for-in and delete deoptimizations mask the real bug**

A config object iterated with `for (k in cfg)` picks up inherited enumerable helpers added by a polyfill on `Object.prototype`; those extra keys hit a downstream `switch` fallback that silently takes the wrong path. Separately, cleanup code does `delete cfg[key]` per request inside a hot loop — deoptimizes the hidden class and at steady state saturates CPU.

```js
// Polyfill that pollutes enumeration (legacy pattern)
Object.prototype.extra = "polluted"; // enumerable! — old polyfills did this
const cfg = { a: 1, b: 2 };
for (const k in cfg) { /* sees extra */ }
console.log(Object.keys(cfg)); // [ 'a','b' ] — keys hides it, for-in leaks it
console.log("extra" in cfg, Object.hasOwn(cfg, "extra")); // true, false — for-in leak vs own-only
delete Object.prototype.extra; // clean up (real fix: never extend Object.prototype enumerably)

// Delete penalty
const hot = { x: 1, y: 2, tmp: 3 };
console.log(delete hot.tmp); // true — but transitions hidden class to dictionary mode
// Better: tombstone or null-proto pool
hot.tmp = undefined; // same-value tombstone keeps shape
console.log(hot.tmp); // undefined — caller checks hasOwn or !== undefined instead of delete
```

- Review rules: flag every `for (k in obj)` without `hasOwn` guard (replace with `keys`+`for-of`); treat `delete` on a per-request object as a smell — use `Map` with `.delete` (which doesn't deoptimize) or sentinel writes.
