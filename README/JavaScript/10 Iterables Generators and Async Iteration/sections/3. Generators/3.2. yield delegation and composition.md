## 3.2. yield* delegation and composition

### 1) **`yield* iterable` delegates to that iterable's iterator**

**Definition** — `yield* expr` requires `expr` be iterable (has `Symbol.iterator`). It gets that iterator and repeatedly `yield`s its values, forwarding `next()`/`return()`/`throw()` between consumer and inner iterator, and capturing the inner iterator's final `return` value (`done:true` value) as the result of the `yield*` expression.

**What problem it solves / history** — without `yield*`, composing generators needed manual `for-of` loops that lost `return`/`throw` propagation and the completion value. `yield*` in ES2015 makes generator composition preserve the full protocol.

```js
// Basic delegation
function* inner(){ yield 1; yield 2; return "inner done"; }
function* outer(){ yield 0; const v = yield* inner(); console.log("inner returned", v); yield 3; }
console.log([...outer()]); // [0,1,2,3] — inner returned "inner done" logged, not yielded

// Delegation forwards values — caller sees flattened sequence
function* a(){ yield "a1"; yield "a2"; }
function* b(){ yield "b1"; yield* a(); yield "b2"; }
console.log([...b()]); // ['b1','a1','a2','b2']

// yield* works on any iterable, not just generators
function* fromAny(){ yield* [10,20]; yield* new Set([30,40]); yield* "hi"; }
console.log([...fromAny()]); // [10,20,30,40,'h','i']

// Return value captured — yield* expression value is inner done value
function* capture(){
  const r1 = yield* (function*(){ yield 1; return 99; })();
  console.log("r1", r1); // 99
  const r2 = yield* [5,6]; // arrays iterator done value is undefined
  console.log("r2", r2); // undefined
}
console.log([...capture()]); // [1,5,6]

// Contrast: yield (no star) yields the iterable as one value
function* noStar(){ yield [1,2]; yield* [1,2]; }
console.log([...noStar()]); // [[1,2], 1, 2] — first is array as single value

// yield* inside expression — still delegates
function* expr(){ const v = (yield* [1,2,3]) + 10; yield v; } // yield* done undefined +10 → NaN? check
console.log([...expr()]); // [1,2,3, NaN] — done undefined +10 = NaN
function* expr2(){ const v = (yield* (function*(){ yield 1; yield 2; return 5; })()) + 10; yield v; }
console.log([...expr2()]); // [1,2,15] — 5+10
```

### 2) **Composition of iterables and recursive flattening via `yield*`**

**Definition** — `yield*` is the primitive for composing iterables: `yield*` chains, recursive generators, and flattening nested structures while preserving early-close semantics.

```js
// Composition — pipeline of generators
function* range(n){ for(let i=0;i<n;i++) yield i; }
function* doubled(iter){ for(const v of iter) yield v*2; }
function* taken(iter, k){ let i=0; for(const v of iter){ if(i++>=k) return; yield v; } }
console.log([...taken(doubled(range(10)), 3)]); // [0,2,4]

// Manual loop vs yield* — both flatten but yield* propagates return/throw
function* manual(iter){ for(const v of iter) yield v; } // loses inner return() propagation nuances but works
function* delegated(iter){ yield* iter; } // correct — forwards close/throw
console.log([...manual([1,2])]); // [1,2]
console.log([...delegated([1,2])]); // [1,2]

// Recursive flatten with yield*
function* flatten(arr){
  for(const v of arr){
    if(Array.isArray(v)) yield* flatten(v);
    else yield v;
  }
}
console.log([...flatten([1,[2,[3,4],5],6])]); // [1,2,3,4,5,6]

// Infinite composition — still lazy
function* naturals(){ let n=0; while(true) yield n++; }
function* evenNaturals(){ for(const n of naturals()) if(n%2===0) yield n; }
function take(iter, k){ const out=[]; for(const v of iter){ if(out.length>=k) break; out.push(v); } return out; }
console.log(take(evenNaturals(), 5)); // [0,2,4,6,8]

// Bidirectional: yield* forwards messages
function* innerEcho(){
  try{ const v = yield 1; console.log("inner got",v); return v*10; }
  finally{ console.log("inner finally"); }
}
function* outerEcho(){
  const r = yield* innerEcho();
  console.log("outer got return",r);
  yield r;
}
const g = outerEcho();
console.log(g.next()); // {value:1,done:false}
console.log(g.next("A")); // inner got A, outer got return 100? actually "A" string *10 NaN — let's use number
const g2 = outerEcho();
console.log(g2.next()); // 1
console.log(g2.next(5)); // inner got 5, outer got return 50, {value:50,done:false}
```

### 3) **`yield*` propagates closing and error injection**

**Definition** — when the consumer calls `return()` or `throw()` on the outer generator, `yield*` forwards that call to the inner iterator's `return()`/`throw()` if present, then propagates the result. This ensures `try/finally` in inner generators runs even when the outer is closed early.

```js
// Closing propagates to inner via return()
function* innerClose(){
  try{ yield 1; yield 2; yield 3; }
  finally{ console.log("inner finally on close"); }
}
function* outerClose(){
  try{ yield* innerClose(); }
  finally{ console.log("outer finally"); }
}
const og = outerClose();
console.log(og.next()); // 1
console.log(og.return("stop")); // inner finally, outer finally, {value:"stop",done:true}
console.log(og.next()); // done

// for-of break propagates similarly
for(const v of outerClose()){ console.log("loop",v); break; }
// loop 1, inner finally, outer finally

// Throw propagates to inner
function* innerThrow(){
  try{ yield 1; yield 2; }
  catch(e){ console.log("inner caught",e.message); yield 99; }
}
function* outerThrow(){
  yield 0;
  yield* innerThrow();
  yield 4;
}
const ot = outerThrow();
console.log(ot.next()); // 0
console.log(ot.next()); // 1 (from inner)
console.log(ot.throw(new Error("boom"))); // inner caught boom → {value:99,done:false}
console.log(ot.next()); // 4
console.log(ot.next()); // done

// Non-generator inner with return() — also forwarded
const customInner = {
  [Symbol.iterator](){
    let i=0;
    return {
      next(){ return i<2?{value:i++,done:false}:{value:undefined,done:true}; },
      return(){ console.log("custom return"); return {value:undefined,done:true}; },
      [Symbol.iterator](){ return this; }
    };
  }
};
function* outerCustom(){ yield* customInner; }
for(const v of outerCustom()){ console.log(v); break; } // custom return — propagated

// If inner has no return(), outer return still closes outer's finally but not inner cleanup (no hook)
const noReturnInner = { [Symbol.iterator](){ let i=0; return { next(){ return i<2?{value:i++,done:false}:{value:undefined,done:true}; }, [Symbol.iterator](){return this;}} } };
function* outerNoRet(){ try{ yield* noReturnInner; } finally{ console.log("outerNoRet finally"); } }
for(const v of outerNoRet()){ console.log(v); break; } // outerNoRet finally — inner had no cleanup to run
```

### 4) **Common confusion**

- "`yield*` yields the return value" — no. `yield*` yields the `value`s with `done:false`; the `done:true` value is returned as the expression result, not yielded (hence `[...gen]` never sees it).
- "`yield*` needs a generator" — no, any iterable (array, Set, Map, string, custom). It just needs `Symbol.iterator`.
- "Manual `for-of`+`yield` equals `yield*`" — close for values, but wrong for closing/error propagation and return-value capture. Use `yield*` for delegation.
- "Closing inner is optional" — forgetting it leaks resources. `yield*` handles it automatically; manual loops need `try/finally` forwarding.

### 5) **Performance note**

`yield*` avoids re-implementing the iterator loop and lets the engine optimize the delegation path (some engines skip per-yield object allocation in `yield*` chains). It's never slower than the manual `for-of`+`yield` equivalent and may be faster on deep chains. The bigger win is correctness: one `yield*` line correctly handles `return`/`throw`.
