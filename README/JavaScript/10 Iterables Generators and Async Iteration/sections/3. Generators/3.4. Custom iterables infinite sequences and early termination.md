## 3.4. Custom iterables infinite sequences and early termination

### 1) **Making any object iterable via `[Symbol.iterator]`**

**Definition** — implement `[Symbol.iterator]()` on your object/class to return a fresh iterator (`next`/`return`/`throw` + self-iterable). Consumers (`for-of`/spread/`Array.from`/destructuring/`yield*`/`Map`/`Set`) then work uniformly. Return a *new* iterator each call — otherwise concurrent/repeated iterations share state and the second is empty.

```js
// Object literal iterable
const range = {
  from: 1, to: 4,
  [Symbol.iterator](){
    let i = this.from;
    const to = this.to;
    return {
      next(){ return i <= to ? { value: i++, done: false } : { value: undefined, done: true }; },
      return(v){ console.log("range closed"); return { value: v, done: true }; },
      [Symbol.iterator](){ return this; }
    };
  }
};
console.log([...range]); // [1,2,3,4]
console.log([...range]); // [1,2,3,4] — fresh each time
for(const v of range) console.log(v); // 1 2 3 4

// Class iterable — each instance fresh
class Counter{
  constructor(n){ this.n=n; }
  *[Symbol.iterator](){ for(let i=0;i<this.n;i++) yield i; } // generator as concise factory
}
console.log([...new Counter(3)]); // [0,1,2]
console.log(Array.from(new Counter(3), x=>x*10)); // [0,10,20]
console.log(new Set(new Counter(4))); // Set(4) {0,1,2,3}
console.log([...new Counter(2)].map(x=>x+1)); // [1,2]

// Wrong — returns this (shared state)
const bad = {
  i: 0,
  next(){ return this.i<2 ? {value:this.i++,done:false} : {value:undefined,done:true}; },
  [Symbol.iterator](){ return this; } // iterable IS the iterator — stateful
};
console.log([...bad]); // [0,1]
console.log([...bad]); // [] — exhausted, leaked state — bug
// Fix: [Symbol.iterator] returns fresh object, not this, unless object is factory-like generator

// Custom iterable with params — factory function
function rangeFactory(from,to){
  return { [Symbol.iterator](){ let i=from; return {
    next(){ return i<=to?{value:i++,done:false}:{value:undefined,done:true}; },
    [Symbol.iterator](){return this;}
  };}};
}
console.log([...rangeFactory(5,7)]); // [5,6,7]
```

### 2) **Infinite and lazy sequences — range/fibonacci/take and resource cleanup**

**Definition** — an iterator may never return `done:true`; consumers must bound it (`break`/`take`/`Array.from` with length not). Laziness means values computed only when requested — chains like `filter`/`map` via generators stay lazy until a terminal (`for-of`/`spread`).

```js
// Infinite naturals — lazy, never done
function* naturals(){
  let n=0; while(true) yield n++;
}
function* take(iterable, k){
  let i=0; for(const v of iterable){ if(i++>=k) return; yield v; }
}
console.log([...take(naturals(), 5)]); // [0,1,2,3,4]
console.log([...take(naturals(), 3)]); // [0,1,2] — fresh naturals each take

// Laziness — expensive step only when consumed
let calls=0;
function* lazyMap(it, fn){
  for(const v of it){ calls++; yield fn(v); }
}
const lazy = lazyMap([1,2,3,4,5], x=>x*10);
console.log("calls before next", calls); // 0
console.log(lazy.next().value); // 10, calls 1
console.log(lazy.next().value); // 20, calls 2
for(const v of lazy){ if(v===30) break; console.log("got",v);} // only maps 30, then closes
console.log("calls after break", calls); // 3 — not 5 (short-circuit preserved)

// Fibonacci infinite lazy
function* fib(){ let a=0,b=1; while(true){ yield a; [a,b]=[b,a+b]; } }
console.log([...take(fib(), 7)]); // [0,1,1,2,3,5,8]

// Naive infinite spread would hang/OOM — always bound
// [...naturals()] // DON'T — infinite, hangs
// Instead: take(naturals(), 10)

// Composing lazy chains — still lazy
function* filter(it, pred){ for(const v of it) if(pred(v)) yield v; }
console.log([...take(filter(naturals(), n=>n%2===0), 4)]); // [0,2,4,6]
```

### 3) **Early termination contract — `return()` + try/finally for resources**

**Definition** — if your iterable holds a resource (file/DB cursor/socket/lock), implement `return()` to release it and use `try/finally` inside a generator to guarantee cleanup when consumer `break`s/`throw`s/`return`s or `yield*` propagates close.

```js
// Generator resource — try/finally ensures close on break
function* dbCursor(rows){
  console.log("open cursor");
  try{
    for(const row of rows) yield row;
  } finally{
    console.log("close cursor");
  }
}
for(const row of dbCursor(["r1","r2","r3"])){
  console.log("process", row);
  if(row==="r2") break; // finally closes
}
// open, process r1, process r2, close

// Manual iterator resource — explicit return()
function makeResourceIter(values){
  let i=0, closed=false;
  return {
    [Symbol.iterator](){
      return {
        next(){
          if(closed) return {value:undefined,done:true};
          return i<values.length ? {value:values[i++], done:false} : {value:undefined,done:true};
        },
        return(v){
          if(!closed){ closed=true; console.log("manual close"); }
          return {value:v,done:true};
        },
        [Symbol.iterator](){ return this; }
      };
    }
  };
}
for(const v of makeResourceIter([1,2,3])){ console.log(v); break; } // manual close
// reconstruction with Array.from — also closes on abrupt? Array.from completes normally here

// take() must propagate close — yield* does, manual loop needs try/finally forwarding
function* takeWithClose(iterable, k){
  const it = iterable[Symbol.iterator]();
  let count=0;
  try{ while(count<k){ const {value,done}=it.next(); if(done) return; yield value; count++; } }
  finally{ if(typeof it.return==="function") it.return(); }
}
console.log([...takeWithClose(dbCursor(["a","b","c","d"]), 2)]); // close via finally

// Bad: manual iterator without return — break leaks
const leakIter = {
  [Symbol.iterator](){
    console.log("leak open");
    let i=0;
    return { next(){ return i<3?{value:i++,done:false}:{value:undefined,done:true}; }, [Symbol.iterator](){return this} };
  }
};
for(const v of leakIter){ console.log(v); break; } // no close log — would leak real resource

// Wait, generator helper auto-close — prefer it for resources
function* safeLeak(){ try{ yield 1; yield 2; } finally{ console.log("safe close"); } }
for(const v of safeLeak()){ console.log(v); break; } // safe close

// Iterator helpers (ES2025 proposal) — not yet owned here, but compose via generators today; prefer lazy chains to eager arrays
```

### 4) **Common confusion**

- "Iterable and iterator are the same" — iterable is factory (`Symbol.iterator` → iterator); iterator is the stateful cursor (`next`). Don't return `this` as a fresh iterator unless the object *is* the one-shot cursor.
- "Infinite generators are dangerous always" — only if consumed without bound (`[...infinite]` hangs). Bounded `take`/`break` is safe and idiomatic for lazy sequences.
- "`return()` is optional for resource iterables" — no, it's required for correctness; without it `break`/`throw` leaks. Generators get it automatically via `finally`; manual iterators must implement it.
- "`yield*` without `return` propagation" — `yield*` handles propagation automatically; a manual `for-of`+`yield` delegation loses it unless you add `try/finally` forwarding.

### 5) **Performance note**

Lazy generator chains allocate one Generator object + one result object per value — for 1M items that's ~16MB churn. Prefer array methods (which can be SIMD/inlined) when you must process the *whole* dataset eagerly and `n` is known. Lazy chains win when you short-circuit early (`find`/`take`/`some`): they avoid allocating the full intermediate array. For resource iterables, the cleanup cost dominates — always pay it.

### 6) **Mentor note**

Have juniors build `range(5,10)` as a class with `[Symbol.iterator]` and predict `[...range]` twice — catches the shared-state bug. Then make it infinite (`while(true)`) and ask them to `...spread` without `take` — the hang teaches why lazy must be bounded. In review, flag any custom iterator used as an iterable that returns `this` as the iterator (re-consumption bug) and any resource iterator without `return()`/`try/finally`.
