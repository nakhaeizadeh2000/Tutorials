## 2.2. Arity, length, and edge cases

### 1) **`length` counts declared params before the first default/rest**

**Definition** — `Function.prototype.length` exposes *arity*: the count of parameters *before* the first one that has a default (`= …`) or is a rest parameter. A plain destructuring pattern without a default (e.g. `({a})`) still counts — only destructuring *with* a default stops the count. Call-time argument count is `arguments.length` or rest-array length — a different number:

```js
function f1(a, b, c)        { return f1.length; }
function f2(a, b = 1, c)    { return f2.length; }
function f3(a, ...rest)     { return f3.length; }
function f4({ a, b })       { return f4.length; } // destructuring without default still counts
function f5(a, b, c = 1, d) { return f5.length; }
console.log(f1.length, f2.length, f3.length, f4.length, f5.length); // 3 1 1 1 2
console.log(f1.length === 3 && f2.length === 1 && f5.length === 2); // true

// length is not argument count — f1.length stays 3 even when called with 0 or 5 args
function countArgs(...args) { return args.length; }
console.log(countArgs(1), countArgs(), countArgs(1, 2, 3, 4)); // 1 0 4
console.log(countArgs.length); // 0 — first param is rest, so arity 0

// Use length to gate currying / partial-application helpers (see [4.1](<../4. First-class and execution patterns/4.1. First-class functions callbacks higher-order functions and composition.md>))
function arityOf(fn) { return `${fn.name || "(anonymous)"} arity ${fn.length}`; }
console.log(arityOf(f5)); // f5 arity 2
```

- *Why `length` stops early* — a default makes call-site arity ambiguous (is `b` required or optional?); spec chooses to count only the mandatory prefix. Rest always stops the count; plain destructuring without a default does not.

### 2) **TDZ and evaluation quirks in defaults**

Defaults live in a scope that is created *per call* and chained: each default can see earlier params, never later ones:

```js
// Earlier defaults see earlier params — later defaults do NOT exist yet
function good(a = 1, b = a * 2) { return b; }
console.log(good());      // 2 — b sees a
console.log(good(5));     // 10 — b sees caller-supplied a

// Reverse order → TDZ ReferenceError at CALL TIME (not parse time)
function bad(a = b, b = 1) { return a; }
try { console.log(bad()); } catch (e) { console.log(e.name, e.message); }
// ReferenceError Cannot access 'b' before initialization

// Self-reference inside own default is also TDZ
function selfRef(a = a) { return a; }
try { console.log(selfRef()); } catch (e) { console.log(e.name); } // ReferenceError

// Outer variable vs param default — param shadows outer for that evaluation
let x = 10;
function shadow(x = x) { return x; } // inner x's initializer tries to read inner x → TDZ
try { console.log(shadow()); } catch (e) { console.log(e.name); } // ReferenceError

// But reading a SAME-NAMED outer through a helper works if not shadowing the initializer
let y = 10;
function notShadowed(y = y) { return y; } // same issue — still TDZ, name resolves to param
try { console.log(notShadowed()); } catch (e) { console.log(e.name); } // ReferenceError
function viaOuter(y = globalThis.y) { return y; } // explicit outer read avoids TDZ
globalThis.y = 10;
console.log(viaOuter()); // 10

// Destructuring defaults interact — outer ={} matters
function withDestr({ a = 1, b = a + 1 } = {}) { return [a, b]; }
console.log(withDestr({}));        // [ 1, 2 ] — b sees a's default
console.log(withDestr({ a: 5 }));  // [ 5, 6 ]
console.log(withDestr());           // [ 1, 2 ] — outer ={} supplies {}
```

- Verifying the TDZ gap: the initializer's scope chain is *params so far* → outer scope. Later params aren't in the chain yet, so reading them is exactly like reading a `let` before its line ([TDZ](<../../../04 Variables Scope and Closures/sections/1. Declarations and bindings/1.2. Temporal dead zone why early access throws.md>)).

### 3) **Strict vs sloppy parameter realities**

Three differences survive in modern code — the first two are mode-sensitive:

```js
// 1) Duplicate param names — allowed only in sloppy, non-rest, no-default functions
function sloppyDup(a, a) { return a; } // no SyntaxError in sloppy
console.log(sloppyDup(1, 2)); // 2 — last one wins
// "use strict"; function strictDup(a, a) {} // SyntaxError: Duplicate parameter name not allowed
// function withDefaultDup(a, a = 1) {}       // SyntaxError always — any default/rest/destr forces uniqueness

// 2) arguments-to-params link — sloppy mirrors writes, strict/rest doesn't (see [2.1](<./2.1. Defaults destructuring rest and the arguments object.md>))
function sloppyLink(a) { arguments[0] = 9; return a; }
function strictLink(a) { "use strict"; arguments[0] = 9; return a; }
console.log(sloppyLink(1), strictLink(1)); // 9 1

// 3) arguments.callee / caller — forbidden in strict and on arrows; legacy introspection deleted from modern code
function strictCallee() {
  "use strict";
  try { console.log(arguments.callee); } catch (e) { console.log(e.name); } // TypeError
}
strictCallee();

// Trailing commas in param lists are legal since ES2017
function trailing(a, b,) { return a + b; }
console.log(trailing(1, 2)); // 3

// length with trailing comma — comma doesn't add a param, length unchanged
function withTrailing(a, b,) {}
console.log(withTrailing.length); // 2
```

### 4) **Common confusion**

- "`f.length` tells me how many arguments I must pass" — no; it tells how many *leading non-tricky* params exist; a function with `a = 1` can be called with 0 args.
- "`b = a` inside params can always see `a`" — only if `a` is *earlier* in the list; later-to-earlier reads are TDZ.
- "`x = x` default should read outer `x`" — shadowing makes `x` refer to the param being initialized, so it's a self-reference TDZ; rename or qualify the outer.
- "Duplicate params are a SyntaxError now" — only in strict and whenever any param uses default/rest/destructuring; bare sloppy duplicates are still parsed (but flag them anyway).

### 5) **Performance note**

Arity introspection (`fn.length`) plus `arguments` linkage checks can deoptimize old engines, but in V8/JSC today the cost of defaults is only the per-call evaluation of the expression — keep defaults cheap when the function is hot (avoid allocating large objects in defaults unless the call omits them). Duplicate-param functions force sloppy mode — prefer strict and unique names for JIT friendliness — [measure](<../../../01 Fundamentals and Mental Model/sections/2. How JavaScript executes/2.1. Parse bytecode JIT pipeline and single-threaded execution.md>).

### 6) **Mentor note**

Ask a junior to write `function f(a = b, b = 5) {}` and predict `f()` — the correct answer is ReferenceError, with TDZ reasoning, not "3". In review, grep `=.*=.*,` inside param lists and check order; grep `(.*,.*)` for duplicate identifiers in sloppy files — replace with strict, unique, destructured params.
