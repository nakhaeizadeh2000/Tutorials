## 6.1. Real production bugs caused by function mistakes

### 1) **Lost `this` in callbacks ships wrong data**

Every detached-method-as-callback is a latent `TypeError` (or silent `globalThis` mutation in sloppy mode). It passes unit tests when called as `obj.method()` but fails when passed as `array.map(obj.method)`:

```js
class Notifier {
  constructor(prefix) { this.prefix = prefix; }
  format(msg) { return `${this.prefix}: ${msg}`; }
  // Bug — method passed directly as callback
  sendAllBug(msgs) { return msgs.map(this.format); }
  // Fix — arrow wrapper captures this lexically (→ [3.2](<../3. this binding/3.2. Arrow lexical this and why callbacks break or heal.md>))
  sendAllFix(msgs) { return msgs.map((m) => this.format(m)); }
}
const n = new Notifier("app");
try { console.log(n.sendAllBug(["hi"])); } catch (e) { console.log(e.name); } // TypeError
console.log(n.sendAllFix(["hi"])); // [ 'app: hi' ]

// Event/timer shape — same bug class with setTimeout / then
class Service {
  constructor() { this.id = "svc-1"; }
  handle() { return this.id; }
}
const svc2 = new Service();
setTimeout(function () { try { console.log(this.handle()); } catch (e) { console.log("timeout bug:", e.name); } }, 0);
setTimeout(() => console.log("timeout fix:", svc2.handle()), 0);
```

- *Lesson:* never pass `this.method` as a value without `bind`/`arrow`/`thisArg`; lint `no-unbound-method` equivalent (ESLint `no-invalid-this` + review).
- Production signal: stack traces where `this` is `undefined` inside a method that "should" be on an instance — grep `\.map\(.*this\.` and `.then(this.`.

### 2) **Defaults and destructuring silently throw at call time**

Required-field patterns throw `TypeError`/`Error` when callers omit the outer object — failures appear as 500s from a single missing query param:

```js
// Bug — outer ={} forgotten, so missing arg throws deep
function drawBug({ x, y } = {}) { return [x, y]; } // actually correct version; bug version lacks ={}
function drawBugNoOuter({ x, y }) { return [x, y]; }
try { console.log(drawBugNoOuter()); } catch (e) { console.log(e.name); } // TypeError — cannot destructure undefined
console.log(drawBug({ x: 1 })); // [ 1, undefined ] — outer guard saves it

// Bug — default reads later param (TDZ)
function priceBug(qty = total, total = 10) { return qty * total; }
try { console.log(priceBug()); } catch (e) { console.log(e.name); } // ReferenceError — qty default reads future total (TDZ)

// Bug — null vs undefined divergence: API returns null for "no value", caller expects default
function connect(host = "localhost") { return host; }
console.log(connect(null)); // null — default NOT triggered; downstream NPE
console.log(connect(null) ?? "localhost"); // localhost — nullish coalescing where default should have been ??, not =

// Real fix — normalize at boundary
function connectFix(host) { host ??= "localhost"; return host; }
console.log(connectFix(null), connectFix(undefined)); // localhost localhost
```

- *Lesson:* every destructured param needs an outer `= {}` (or `= []`) if the caller may omit it; defaults trigger only on `undefined`, so treat `null` from APIs explicitly with `??`/`??=` inside body ([2.1](<../2. Parameters and arguments/2.1. Defaults destructuring rest and the arguments object.md>)).

### 3) **Rest/`arguments` confusion and `new` on non-constructibles**

```js
// Bug — variadic helper uses arguments but is an arrow, so it reads outer's arguments
function outerVariadic() {
  const arrowRest = () => arguments.length; // lexical arguments from outer
  function regularRest() { return arguments.length; }
  console.log(arrowRest(1, 2, 3), regularRest(1, 2, 3)); // 0 3 — arrow sees outer's 0, not its own
}
outerVariadic();

// Fix — arrows must use rest
const arrowFix = (...args) => args.length;
console.log(arrowFix(1, 2, 3)); // 3

// Bug — attempt to `new` an arrow or bound-arrow — TypeError at runtime, not parse time
const ArrowCtor = () => {};
try { new ArrowCtor(); } catch (e) { console.log(e.name); } // TypeError

// Bug — bind-ception: re-binding is silently ignored, so "fixed" helper still uses first this
function log(msg) { return `${this.prefix}: ${msg}`; }
const boundA = log.bind({ prefix: "A" });
const boundB = boundA.bind({ prefix: "B" });
console.log(boundB("hi")); // A: hi — not B: hi (→ [3.3](<../3. this binding/3.3. call apply bind and bound functions.md>))

// Bug — length/gating on curried helper uses stale length after bind
function take(a, b, c) { return [a, b, c]; }
console.log(take.bind(null, 1).length); // 2 — not 3; currying gate that checks fn.length must read bound length
```

- *Lesson:* in code review, flag `() => arguments` (should be `(...args)`), any `new` where the identifier is a `const` arrow (`PascalCase` but arrow), and `bind(...).bind(...)` chains — second bind is dead code.

### 4) **Recursion depth and IIFE ASI traps in production**

```js
// Depth blowup — recursion depth tied to user input (e.g. pagination pages, retry count)
function fetchPages(n) {
  if (n <= 0) return;
  return fetchPages(n - 1); // linear recursion — n from API can be 1e5 → RangeError
}
try { fetchPages(50000); } catch (e) { console.log(e.name); } // RangeError

// Fix — iterate; keep recursion for tree-shaped data (→ [4.3](<../4. First-class and execution patterns/4.3. Recursion stacks and tail-call reality.md>))
function fetchPagesFix(n) { for (let i = n; i > 0; i--) {} console.log("done", n); }
fetchPagesFix(50000); // done 50000 — O(1) stack

// ASI trap — file concatenation or missing semicolon turns value + IIFE into a call
let val = 1
// ;(() => console.log("without leading ; this is a call on 1"))() // TypeError: 1 is not a function
;(function () { console.log("with leading ; — safe"); })();
console.log(val); // 1 — untouched (without ; prefix, val would have been invoked as fn)
```

### 5) **Mentor checklist for this pitfall class**

- Detached-method callbacks: grep `\.map\(.*\bthis\.\w+` , `setTimeout\(.*this\.` , `.then\(this\.` — each needs arrow/bind/thisArg.
- Destructuring params without outer `= {}` / `= []` — grep `function \w+\(\{` and confirm guard.
- `arguments` inside `=>` — grep `=>.*arguments` — replace with rest.
- `new` on `PascalCase` arrow — grep `new [A-Z].*=>` chain; constructible must be `function`/`class`.
- Re-bind after `bind` — grep `\.bind.*\.bind` — second bind never wins.
