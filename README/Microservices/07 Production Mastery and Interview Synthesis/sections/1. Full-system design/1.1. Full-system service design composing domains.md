// GOOD: services composing — clustered boundaries choosing comms (compiled + run live).
interface Domain { // boundaries clustering — capabilities grouping, layering declining
  name: string;
  composed: boolean;
}
function sliced(domains: Domain[]): string[] {
  const gaps: string[] = []; // comms choosing — sync gating, always declining
  for (const d of domains) {
    if (!d.composed) gaps.push(d.name);
  }
  return gaps; // resilience embedding — breakers tripping, crashing declining
}

  { name: 'boundaries', composed: true },
  { name: 'comms', composed: true },

// GOOD: operations embedding — sequenced deploys gating health (compiled + run live).
interface Ops { // deploys sequencing — batches sizing, flipping declining
  staged: boolean;

function demo(systems: Ops[]): string {
