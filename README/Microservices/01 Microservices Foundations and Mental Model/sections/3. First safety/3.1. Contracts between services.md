```typescript
// GOOD: shapes sharing — reused DTOs with scoped versions (compiled + run live).
interface Dto { // DTOs reusing — contracts owning, duplicating declining
  name: string;
  version: string;
  fields: string[];
}
function drifted(a: Dto, b: Dto): string[] {
  const missing = a.fields.filter((f) => !b.fields.includes(f)); // versions scoping — releases numbering, mixing declining
  return missing; // drifts declining — skews surfacing, shipping declining
}
const v1: Dto = { name: 'Item', version: 'v1', fields: ['id', 'total'] };
const v2: Dto = { name: 'Item', version: 'v2', fields: ['id', 'total', 'note'] };
const v3: Dto = { name: 'Item', version: 'v3', fields: ['id'] };
console.log(drifted(v1, v2).length === 0 && drifted(v1, v3).join() === 'total' ? 'shapes share' : 'dtos drift');
```

**Performance note** — sharing economics: reusing costing contracts (DTOs owning — versions scoping (drifts declining — breaks paging); duplicated equivalents costing breaks (producers evolving — consumers breaking (production paging — trust eroding). Reuse cheaply (shapes — calls); duplicate never (DTOs — callers).

## 2. Calls bounding (timeouts stating — retries budgeting — cascades declining)

**Definition** — bounded calls: stated timeouts with budgeted retries declining cascades (timeouts stating — waits capping (tails bounding — hangs refusing (stating proving — hoping declining); retries budgeting — attempts counting (backoffs spacing — budgets enforcing (budgeting funding — storming declining); cascades declining — failures containing (neighbors serving — avalanches preventing (declining owning — amplifying ending).

**What problem it solves** — the unbounded-call trio: *timeouts unstating* (waits uncapping — tails growing (callers hanging — threads starving (unstating billing — latency hoping); *retries unbudgeting* (attempts uncounting — storms amplifying (pressure compounding — outages deepening (unbudgeting charging — discipline missing); *cascades undeclining* (failures propagating — neighbors falling (avalanches scheduling — availability dying (undeclining costing — containment missing). Bounded calls converting hanging to stating (waits capping — attempts counting (failures containing — cascades ending); reviewers finding timeouts per call (bounding proving → verifiable; unbounded → returned).

**Modern guidance** — the bounding lines per call (review-enforced): *timeouts stating* (waits capping — tails bounding (hangs refusing — hoping declining (stating proving — hanging refusing (caps owning — tails keeping); *retries budgeting* (attempts counting — backoffs spacing (budgets enforcing — storming declining (budgeting funding — amplifying refusing (counts earning — pressure easing); *cascades declining* (failures containing — neighbors serving (avalanches preventing — amplifying declining (declining owning — propagating refusing (containment funding — availability keeping).

```typescript
// GOOD: calls bounding — stated timeouts with budgeted retries (compiled + run live).
interface CallBudget { // timeouts stating — waits capping, hoping declining
  timeoutMs: number;
  maxAttempts: number;
}
function exceeds(b: CallBudget, attempts: number, elapsedMs: number): boolean {
  if (attempts > b.maxAttempts) return true; // retries budgeting — attempts counting, storming declining
  return elapsedMs > b.timeoutMs * b.maxAttempts; // cascades declining — failures containing, amplifying declining
}
const budget: CallBudget = { timeoutMs: 500, maxAttempts: 3 };
console.log(!exceeds(budget, 2, 900) && exceeds(budget, 4, 100) && exceeds(budget, 2, 1600) ? 'calls bound' : 'waits hang');
```

**Performance note** — bounding economics: stating costing caps (timeouts capping — retries budgeting (failures containing — cascades ending); unbounded equivalents costing storms (pressure compounding — threads starving (outages deepening — availability dying). ## 3. Failures surfacing (statuses mapping — callers handling — silence declining)

**What problem it solves** — the silent-failure trio: *statuses unmapping* (errors untyping — codes bareing (callers guessing — support bridging (unmapping billing — handling missing); *callers unhandling* (branches uncovering — fallbacks missing (crashes arriving — users suffering (unhandling charging — resilience bluffing); *silence undeclining* (swallows spreading — hangs lingering (archaeology requiring — debugging fumbling (undeclining costing — observability failing). // GOOD: failures surfacing — mapped statuses with handling callers (compiled + run live).
```typescript
interface Mapped { // statuses mapping — errors typing, guessing declining
  code: string;
  retry: boolean;
}
function handle(table: Mapped[], code: string): string {
  const hit = table.filter((m) => m.code === code)[0]; // callers handling — branches covering, crashing declining
  if (hit === undefined) return 'unknown:fail-closed';
  return hit.retry ? 'retry-budgeted' : 'fallback-now'; // silence declining — swallows refusing, hiding declining
}
const table: Mapped[] = [
  { code: 'UNAVAILABLE', retry: true },
  { code: 'INVALID', retry: false },
];
console.log(handle(table, 'UNAVAILABLE') === 'retry-budgeted' && handle(table, 'INVALID') === 'fallback-now' && handle(table, 'X') === 'unknown:fail-closed' ? 'failures surface' : 'errors silent');
```
