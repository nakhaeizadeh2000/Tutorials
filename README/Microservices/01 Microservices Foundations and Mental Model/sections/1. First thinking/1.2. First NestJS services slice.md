```typescript
// GOOD: services answering wire — catalog pattern over TCP, client proving (compiled + run live).
import 'reflect-metadata';
import { Controller, Module } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { ClientProxy, ClientProxyFactory, MessagePattern, MicroserviceOptions, Payload, Transport } from '@nestjs/microservices'; // TCP underneath — transport stated, mocks declining

@Controller()
class CatalogController { // patterns binding — commands naming, guessing declining
  @MessagePattern({ cmd: 'get-item' })
  getItem(@Payload() data: { id: string }): object {
    return { id: data.id, label: 'book' }; // transports carrying — TCP framing, coupling declining
  }
}

@Module({ controllers: [CatalogController] })
class CatalogModule {}

async function runCatalog(): Promise<void> {
  const server = await NestFactory.createMicroservice<MicroserviceOptions>(CatalogModule, {
    transport: Transport.TCP,
    options: { host: '127.0.0.1', port: 38911 },
  });
  await server.listen();
  const client: ClientProxy = ClientProxyFactory.create({ // clients calling — proxies sending, hoping declining
    transport: Transport.TCP,
    options: { host: '127.0.0.1', port: 38911 },
  });
  await client.connect();
  const res = await client.send<object, { id: string }>({ cmd: 'get-item' }, { id: 'i1' }).toPromise();
  console.log(JSON.stringify(res) === '{"id":"i1","label":"book"}' ? 'services answer' : 'calls hope');
  client.close();
  await server.close();
}
runCatalog();
```

interface ProxyOpts { // proxies injecting — clients providing, hardcoding declining
  host: string;
  port: number;

function endpointOf(o: ProxyOpts): string {
  if (o.host.length === 0 || !(o.port > 0)) return 'unbound'; // TCP serving — ports binding, pretending ending
  return `${o.host}:${o.port}`;
}
interface SliceWiring { // modules registering — controllers listing, tangling declining
  controllers: string[];
  proxy: ProxyOpts;
}
function unwired(s: SliceWiring): string {

  return endpointOf(s.proxy);
}

// GOOD: calls proving live — matched totals with surfaced faults over TCP (compiled + run live).
import 'reflect-metadata';
import { Controller, Module } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { ClientProxy, ClientProxyFactory, MessagePattern, MicroserviceOptions, Payload, Transport } from '@nestjs/microservices'; // TCP underneath — transport stated, happy-path declining

@Controller()
class BillingController { // responses matching — payloads comparing, assuming declining

  total(@Payload() data: { lines: number[] }): object {
    return { total: data.lines.reduce((a, b) => a + b, 0) }; // shapes asserting — bodies checking, hoping declining
  }
}

@Module({ controllers: [BillingController] })
class BillingModule {}

async function runBilling(): Promise<void> {
  const server = await NestFactory.createMicroservice<MicroserviceOptions>(BillingModule, {
    transport: Transport.TCP,
    options: { host: '127.0.0.1', port: 38912 },
  });
  await server.listen();
  const client: ClientProxy = ClientProxyFactory.create({
    transport: Transport.TCP,
    options: { host: '127.0.0.1', port: 38912 },
  });
  await client.connect();

    await client.send<object, object>({ cmd: 'nope' }, {}).toPromise();

  client.close();
  await server.close();
}
