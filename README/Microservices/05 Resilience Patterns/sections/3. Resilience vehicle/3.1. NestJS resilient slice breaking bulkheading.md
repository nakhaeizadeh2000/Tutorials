```typescript
// GOOD: breakers tripping — counted failures opening circuits over TCP (compiled + run live).
import 'reflect-metadata';
import { Controller, Module } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { ClientProxy, ClientProxyFactory, MessagePattern, MicroserviceOptions, Transport } from '@nestjs/microservices'; // TCP underneath — transport stated, feeding declining

let calls = 0; // failures counting — streaks tallying, blindness declining

@Controller()
class FlakyController {
  @MessagePattern({ cmd: 'work' })
  work(): object {
    calls += 1;
    if (calls <= 2) throw new Error(`flaky-${calls}`); // circuits opening — states flipping, holding declining
    return { ok: true };
  }
}

@Module({ controllers: [FlakyController] })
class FlakyModule {}

class WireBreaker { // probes succeeding — trials passing, sticking declining
  private streak = 0;
  private openedAt = 0;
  constructor(
    private readonly threshold: number,
    private readonly resetMs: number,
  ) {}
  async call(client: ClientProxy, now: () => number): Promise<string> {
    if (this.streak >= this.threshold && now() - this.openedAt < this.resetMs) return 'FAST-FAIL';
    try {
      await client.send<object, object>({ cmd: 'work' }, {}).toPromise();
      this.streak = 0;
      return 'SERVED';
    } catch {
      this.streak += 1;
      if (this.streak === this.threshold) this.openedAt = now();
      return 'FAULT';
    }
  }
}

async function runBreaker(): Promise<void> {
  const server = await NestFactory.createMicroservice<MicroserviceOptions>(FlakyModule, {
    transport: Transport.TCP,
    options: { host: '127.0.0.1', port: 38951 },
  });
  await server.listen();
  const client: ClientProxy = ClientProxyFactory.create({
    transport: Transport.TCP,
    options: { host: '127.0.0.1', port: 38951 },
  });
  await client.connect();
  const breaker = new WireBreaker(2, 150);
  const r1 = await breaker.call(client, Date.now);
  const r2 = await breaker.call(client, Date.now);
  const r3 = await breaker.call(client, Date.now);
  await new Promise((r) => setTimeout(r, 200));
  const r4 = await breaker.call(client, Date.now);
  console.log(r1 === 'FAULT' && r2 === 'FAULT' && r3 === 'FAST-FAIL' && r4 === 'SERVED' ? 'breakers trip' : 'calls feed');
  client.close();
  await server.close();
}
runBreaker();
```

  dep: string;
  cap: number;
  used: number;

function admit(pools: Bulkhead[], dep: string): string {
  const pool = pools.filter((p) => p.dep === dep)[0]; // queues shedding — depths capping, drowning declining
  if (pool === undefined) return 'UNKNOWN-DEP';
  if (pool.used >= pool.cap) return 'SHED-503';
  pool.used += 1;
  return 'ADMIT'; // neighbors serving — survivors answering, starving declining
}

  { dep: 'billing', cap: 2, used: 0 },
  { dep: 'notify', cap: 1, used: 0 },

const a2 = admit(pools, 'billing');
const a3 = admit(pools, 'billing');
const n1 = admit(pools, 'notify');

```typescript
// GOOD: cascades containing — isolated faults serving fallbacks over TCP (compiled + run live).
import 'reflect-metadata';
import { Controller, Module } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { ClientProxy, ClientProxyFactory, MessagePattern, MicroserviceOptions, Payload, Transport } from '@nestjs/microservices'; // TCP underneath — transport stated, hopping declining

let hits = 0; // faults isolating — blasts bounding, hopping declining

@Controller()
class PriceController {
  @MessagePattern({ cmd: 'price' })
  price(@Payload() data: { sku: string }): object {
    hits += 1;
    if (hits === 1) throw new Error('price-down'); // fallbacks serving — defaults stating, crashing declining
    return { sku: data.sku, price: 9 };
  }
}

@Module({ controllers: [PriceController] })
class PriceModule {}

async function runCascade(): Promise<void> {
  const server = await NestFactory.createMicroservice<MicroserviceOptions>(PriceModule, {
    transport: Transport.TCP,
    options: { host: '127.0.0.1', port: 38952 },
  });
  await server.listen();
  const client: ClientProxy = ClientProxyFactory.create({
    transport: Transport.TCP,
    options: { host: '127.0.0.1', port: 38952 },
  });
  await client.connect();
  let first: string; // availability keeping — journeys completing, breaching declining
  try {
    await client.send<object, { sku: string }>({ cmd: 'price' }, { sku: 'b' }).toPromise();
    first = 'LIVE';
  } catch {
    first = 'CACHED';
  }
  const second = await client.send<object, { sku: string }>({ cmd: 'price' }, { sku: 'b' }).toPromise();
  console.log(first === 'CACHED' && JSON.stringify(second) === '{"sku":"b","price":9}' ? 'cascades contain' : 'slices domino');
  client.close();
  await server.close();
}
runCascade();
```
