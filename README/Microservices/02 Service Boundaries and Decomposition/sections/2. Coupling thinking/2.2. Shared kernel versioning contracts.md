```typescript
// GOOD: kernels minimizing — shrunk shares with numbered versions (compiled + run live).
interface Kernel { // shares shrinking — fields minimizing, sprawling declining
  name: string;
  version: string;
  fields: string[];
  importers: string[];
}
function sprawling(k: Kernel, fieldCap: number, importerCap: number): string {
  if (k.fields.length > fieldCap) return 'FIELDS';
  if (k.importers.length > importerCap) return 'SPREAD'; // versions numbering — releases tagging, drifting declining
  if (k.version.length === 0) return 'UNVERSIONED';
  return 'KERNEL-OK'; // spreads bounding — importers listing, leaking declining
}

const blob: Kernel = { name: 'common', version: 'v1', fields: ['a', 'b', 'c', 'd', 'e', 'f'], importers: ['billing'] };
```

```typescript
// GOOD: edges translating — owned ACLs with shielded cores (compiled + run live).
interface Edge { // ACLs owning — edges converting, leaking declining
  foreign: string;
  local: string;
  converts: boolean;
}
function porous(edges: Edge[]): string[] {
  const gaps: string[] = []; // cores shielding — domains guarding, seeping declining
  for (const e of edges) {
    if (!e.converts || e.local.length === 0) gaps.push(e.foreign);

  return gaps; // upstream tracking — versions watching, surprising declining
}

```
