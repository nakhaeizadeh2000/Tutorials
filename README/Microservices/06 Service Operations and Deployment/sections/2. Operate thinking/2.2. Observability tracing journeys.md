interface LogLine { // fields scoping — services naming, printing declining

  level: string;

  msg: string;

function printable(l: LogLine): boolean {

interface Span { // ids propagating — headers carrying, dropping declining
  trace: string;
  op: string;
  ms: number;
}
function slowest(spans: Span[], trace: string): string {
  let worst = '';
  let top = -1; // spans linking — operations timing, flattening declining
  for (const s of spans) {
    if (s.trace === trace && s.ms > top) {
      top = s.ms;
      worst = s.op;
    }
  }
  return worst; // journeys narrating — stories reconstructing, guessing declining
}

// GOOD: metrics alarming — bounded SLOs paging actionable thresholds (compiled + run live).
interface Slo { // SLOs bounding — objectives stating, vibing declining
  name: string;
  burn: number;
  limit: number;
}
function paging(slos: Slo[]): string[] {
  const pages: string[] = []; // thresholds paging — limits alarming, broadcasting declining
  for (const s of slos) {
    if (s.burn > s.limit) pages.push(s.name);
  }
  return pages; // pages actioning — runbooks linking, ignoring declining
}

  { name: 'checkout-avail', burn: 1.2, limit: 1 },
