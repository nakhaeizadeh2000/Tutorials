```typescript
// GOOD: health answering — cheap liveness with deep readiness over TCP (compiled + run live).
import 'reflect-metadata';
import { Controller, Module } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { ClientProxy, ClientProxyFactory, MessagePattern, MicroserviceOptions, Payload, Transport } from '@nestjs/microservices'; // TCP underneath — transport stated, blindness declining

@Controller()
class HealthController { // live cheaping — processes answering, costing declining
  @MessagePattern({ cmd: 'live' })
  live(): object {
    return { status: 'SERVING' };
  }
  @MessagePattern({ cmd: 'ready' })
  ready(@Payload() data: { deps: boolean[] }): object {
    const down = data.deps.filter((d) => !d).length; // ready deeping — dependencies checking, shallowing declining
    return { verdict: down === 0 ? 'SERVE' : 'SHED' }; // verdicts proving — splits asserting, lying declining
  }
}

@Module({ controllers: [HealthController] })
class HealthModule {}

async function runHealth(): Promise<void> {
  const server = await NestFactory.createMicroservice<MicroserviceOptions>(HealthModule, {
    transport: Transport.TCP,
    options: { host: '127.0.0.1', port: 38961 },
  });
  await server.listen();
  const client: ClientProxy = ClientProxyFactory.create({
    transport: Transport.TCP,
    options: { host: '127.0.0.1', port: 38961 },
  });
  await client.connect();
  const live = await client.send<object, object>({ cmd: 'live' }, {}).toPromise();
  const up = await client.send<object, { deps: boolean[] }>({ cmd: 'ready' }, { deps: [true, true] }).toPromise();
  const down = await client.send<object, { deps: boolean[] }>({ cmd: 'ready' }, { deps: [true, false] }).toPromise();
  console.log(
    JSON.stringify(live) === '{"status":"SERVING"}' &&
      JSON.stringify(up) === '{"verdict":"SERVE"}' &&
      JSON.stringify(down) === '{"verdict":"SHED"}'
      ? 'health answers'
      : 'probes blind',
  );
  client.close();
  await server.close();
}
runHealth();
```

// GOOD: config gating — validated slice config blocking bad boots (compiled + run live).
interface SliceConfig { // schemas validating — shapes requiring, hoping declining
  port: number;
  store: string;

function bootable(cfg: Partial<SliceConfig>): string {
  if (!(cfg.port !== undefined && cfg.port > 0)) return 'BLOCK:port'; // boots blocking — starts refusing, serving declining
  if ((cfg.store ?? '').length === 0) return 'BLOCK:store';
  return 'BOOT'; // failures earlying — errors surfacing, lateing declining

const thin: Partial<SliceConfig> = { port: 38961 };
