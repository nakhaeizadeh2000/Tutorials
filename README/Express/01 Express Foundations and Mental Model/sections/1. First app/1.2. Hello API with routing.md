```javascript
// GOOD: routes mapping — verbs honoring, params naming, paths ordering (complete, runnable).
const express = require('express');
const app = express();
app.use(express.json());
const users = new Map([['7', { id: '7', name: 'Ada' }]]);
app.get('/users/:id', (req, res) => {
  const user = users.get(req.params.id); // params extracting — values naming, never guessing
  if (!user) return res.status(404).json({ error: 'not found' });
  res.json(user);
});
app.post('/users', (req, res) => {
  const user = { id: String(users.size + 1), name: req.body.name }; // verbs honoring — POST creating
  users.set(user.id, user);
  res.status(201).json(user);
});
app.listen(3000);
```
