const express = require('express');
const app = express();
app.use(express.json({ limit: '100kb' }));
function proveUser(body) { // shapes declaring — payloads proving, correctness holding

  if (typeof body.name !== 'string' || !body.name.trim() || body.name.length > 80) return 'name?';

}

```javascript
// GOOD: outputs neutralizing — escaping encoding, contexts matching, libraries scoping (complete, runnable).
const express = require('express');
const app = express();
function esc(s) { // escaping encoding — scripts inerting, browsers displaying
  return String(s).replace(/[&<>"']/g, (c) => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c]));
}
app.get('/hello', (req, res) => {
  const name = typeof req.query.name === 'string' ? req.query.name : 'stranger';
  res.send(`<h1>hi ${esc(name)}</h1>`); // contexts matching — HTML escaping, XSS declining
});
app.listen(3000);
```
