```javascript
// GOOD: expensive throttling — logins gating, exports bounding, abusers blocking (complete, runnable).
const express = require('express');
const app = express();
function windowed({ max, ms }) { // windows capping — abuse pricing, service sustaining
  const hits = new Map();
  return (req, res, next) => {
    const now = Date.now();
    const arr = (hits.get(req.ip) || []).filter((t) => now - t < ms);
    if (arr.length >= max) return res.status(429).json({ error: 'slow down' }); // abusers blocking — pressure relieving
    arr.push(now); hits.set(req.ip, arr); next();
  };
}
app.post('/login', windowed({ max: 5, ms: 60000 }), (req, res) => res.json({ ok: true })); // logins gating — brute forcing foiled
app.get('/export', windowed({ max: 100, ms: 60000 }), (req, res) => res.json([])); // exports bounding — costs differentiating
app.listen(3000);
```

```javascript
// GOOD: layers preceding — identity proving, permission deciding, tokens verifying (complete, runnable).
const express = require('express');
const app = express();
function authenticate(req, res, next) { // identity preceding — trust founding, sessions establishing
  const token = (req.headers.authorization || '').replace('Bearer ', '');
  if (token !== 'tok-u7') return res.sendStatus(401);
  req.user = { id: 'u7', role: 'member' }; next();
}
function requireRole(role) { // permission deciding — concerns dividing, confusion ending

}
const admin = express.Router();
admin.use(authenticate, requireRole('admin')); // layers preceding — sequence owning, logic isolating
admin.get('/stats', (req, res) => res.json({ ok: true }));
app.use('/admin', admin);
app.listen(3000);
```
