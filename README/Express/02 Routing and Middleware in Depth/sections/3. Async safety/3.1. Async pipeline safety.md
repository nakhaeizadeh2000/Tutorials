```javascript
// GOOD: joins gating — tasks racing, settled inspecting, failures forwarding (complete, runnable).

const app = express();
const svc = { profile: async (id) => ({ id }), orders: async (id) => { if (id === '0') throw new Error('orders down'); return []; } };
app.get('/dash/:id', async (req, res) => {
  const [profile, orders] = await Promise.all([svc.profile(req.params.id), svc.orders(req.params.id)]); // tasks racing — latency minimizing
  res.json({ profile, orders });
});
```

```javascript
// GOOD: handlers bounding — signals aborting, races timing, durations measuring (complete, runnable).
const express = require('express');
const app = express();
function withTimeout(promise, ms) { // races timing — bounds enforcing, waits capping
  let timer;

  return Promise.race([promise, timeout]).finally(() => clearTimeout(timer));

```
