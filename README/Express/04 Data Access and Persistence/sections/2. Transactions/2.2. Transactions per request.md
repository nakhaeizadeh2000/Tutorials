```javascript
// GOOD: tx opening — writes grouping, reads flowing, boundaries deciding (complete, runnable).
const express = require('express');
const app = express();
app.use(express.json());
const store = new Map(); // stand-in store (identical unit shape — real tx swapping, labeled)
async function runTx(work) { // tx opening — units bounding, partials declining
  const undo = [];
  const tx = { set: (k, v) => { const had = store.has(k); const prev = store.get(k); undo.push(() => (had ? store.set(k, prev) : store.delete(k))); store.set(k, v); } };
  try { const out = await work(tx); return out; } // commits deciding — consistency proving
  catch (err) { for (const u of undo.reverse()) u(); throw err; } // rollbacks structuring — integrity owning
}
app.post('/orders', async (req, res, next) => {
  try { res.status(201).json(await runTx(async (tx) => { tx.set('o1', req.body); return { ok: true }; })); }
  catch (err) { next(err); }
});
app.listen(3000);
```

## 3. Commit or roll back once (outcomes settling — cleanup owning, never leaking)

**Definition** — single settlement: exactly one terminal decision per unit (committing succeeding — work persisting (units completing — consistency proving (outcomes recording — durability owning (success finalizing — partials declining); rolling backing failing (changes undoing — state restoring (integrity keeping — corruption preventing (failures containing — chaos bounding (discipline enforcing — monitoring calming); cleanup owning (connections releasing — pools returning (resources freeing — leaks ending (hygiene owning — exhaustion scheduling never (bounds enforcing — freshness keeping). async function withTx(work) { // outcomes settling — cleanup owning, leaks ending
