```javascript
// GOOD: handlers translating — requests parsing, services deciding, queries migrating (complete, runnable).
const express = require('express');
const app = express();
app.use(express.json());
const store = new Map(); // stand-in persistence (identical shape — real stores swapping, labeled)
const userService = {
  async create({ name }) {
    if (!name?.trim()) { const e = new Error('name?'); e.status = 400; throw e; } // services deciding — rules enforcing
    const user = { id: String(store.size + 1), name: name.trim() };
    store.set(user.id, user);
    return user;
  },
};
app.post('/users', async (req, res, next) => {
  try { res.status(201).json(await userService.create(req.body)); } // handlers translating — services deciding
  catch (err) { next(err); }
});
app.use((err, req, res, next) => res.status(err.status || 500).json({ error: 'boom' }));
app.listen(3000);
```

```javascript
const orders = {
```
