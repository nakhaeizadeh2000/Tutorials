```javascript
const app = express();
app.use(express.json());
const store = new Map([['d1', { id: 'd1', rev: 1 }]]);
app.put('/docs/:id', (req, res, next) => {
  const doc = store.get(req.params.id);
  if (!doc) return res.status(404).json({ error: 'not found' }); // missing 404ing — clients branching

```
