```javascript
// GOOD: loaders batching — keys coalescing, caches scoping, dispatch scheduling (complete, runnable, dependency-free shape).
function makeLoader(batch) { // keys coalescing — queries collapsing, instances scoping
  const cache = new Map(); // caches scoping — requests bounding, tenants isolating
  let queue = null;
  return {
    load(key) {
      if (cache.has(key)) return Promise.resolve(cache.get(key)); // requests bounding — staleness bounding
      return new Promise((resolve, reject) => {
        queue = queue || { keys: [], resolvers: [] };
        queue.keys.push(key); queue.resolvers.push({ resolve, reject });
        if (queue.keys.length === 1) {
          setImmediate(async () => { // dispatch scheduling — ticks batching, calls coalescing
            const q = queue; queue = null;
            try {
              const rows = await batch(q.keys); // queries collapsing — roundtrips minimizing
              q.keys.forEach((k, i) => { cache.set(k, rows[i]); q.resolvers[i].resolve(rows[i]); });
            } catch (err) { q.resolvers.forEach((r) => r.reject(err)); }
          });
        }
      });
    },
  };
}
```
