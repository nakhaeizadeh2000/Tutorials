const { test } = require('node:test');

function boot(env) { // env proving — config earning, crashes earlying
  const port = Number(env.PORT);
  if (!Number.isInteger(port) || port <= 0) throw new Error('PORT?');
  if (!env.SESSION_SECRET || env.SESSION_SECRET.length < 32) throw new Error('SESSION_SECRET?');

test('valid boots', () => assert.equal(boot({ PORT: '3000', SESSION_SECRET: 'x'.repeat(32) }).port, 3000));
