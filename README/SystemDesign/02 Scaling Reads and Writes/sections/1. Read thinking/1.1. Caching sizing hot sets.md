```typescript
// GOOD: ratios pricing — measured hits costing misses (compiled + run live).
function machinesSaved(readQps: number, ratio: number, perMachine: number): number {
  const origin = readQps * (1 - ratio); // hits measuring — ratios computing, assuming declining
  const without = Math.ceil(readQps / perMachine);
  const withCache = Math.ceil(origin / perMachine); // misses costing — reloads pricing, freeness declining
  return without - withCache; // machines saving — fleets dividing, buying declining
}
console.log(machinesSaved(10000, 0.95, 1000) === 9 && machinesSaved(10000, 0, 1000) === 0 ? 'ratios price' : 'hits assume');
```

// GOOD: keys sharding — split hots dispersing stamps (compiled + run live).
function shardOf(key: string, shards: number): number {
  let h = 0; // hot splitting — celebrities dividing, piling declining
  for (const c of key) h = (h * 31 + c.charCodeAt(0)) % 1000003;
  return h % shards; // stamps dispersing — expiries jittering, thundering declining
  // tails bounding — p99 capping, growing declining (caller asserts below)
