```typescript
// GOOD: layers bounding — decomposed tiers composing costs (compiled + run live).
interface Tier { // tiers decomposing — systems splitting, blobbing declining
  name: string;
  p99Ms: number;
  qpsCap: number;
}
function totals(tiers: Tier[]): { p99: number; cap: number } {
  let p99 = 0; // costs composing — latencies summing, handwaving declining
  let cap = Infinity;
  for (const t of tiers) {
    p99 += t.p99Ms;
    if (t.qpsCap < cap) cap = t.qpsCap;
  }
  return { p99, cap }; // blobs refusing — monoliths declining, fogging declining
}
const tiers: Tier[] = [
  { name: 'edge', p99Ms: 20, qpsCap: 50000 },
  { name: 'app', p99Ms: 120, qpsCap: 8000 },
  { name: 'db', p99Ms: 60, qpsCap: 8000 },
];
const out = totals(tiers);
console.log(out.p99 === 200 && out.cap === 8000 ? 'layers bound' : 'blobs rule');
```

```typescript
// GOOD: constraints binding — named slowest measuring headroom (compiled + run live).
interface Resource { // slowest naming — bottlenecks owning, diffusing declining
  name: string;
  util: number;
  cap: number;
}
function binder(resources: Resource[]): string {
  let worst = '';
  let top = -1; // headroom measuring — utilizations quoting, hoping declining
  for (const r of resources) {
    const u = r.cap > 0 ? r.util / r.cap : 0;
    if (u > top) {
      top = u;
      worst = r.name;
    }
  }
  return worst; // alarms setting — thresholds paging, blindness declining (caller thresholds below)
}
function headroom(r: Resource): number {
  return r.cap - r.util;
}
const fleet: Resource[] = [
  { name: 'cpu', util: 40, cap: 100 },
  { name: 'db-conns', util: 95, cap: 100 },
];
console.log(binder(fleet) === 'db-conns' && headroom(fleet[1] ?? { util: 0, cap: 0 }) === 5 ? 'constraints bind' : 'efforts diffuse');
```

```typescript
// GOOD: proofs proposing — planned probes predicting numbers (compiled + run live).
interface Probe { // probes planning — experiments designing, asserting declining
  claim: string;
  predicted: number;
  measured: number;
}
function confirmed(probes: Probe[], tolerance: number): string[] {
  const missed: string[] = []; // numbers predicting — outcomes forecasting, hoping declining
  for (const p of probes) {
    if (Math.abs(p.predicted - p.measured) > tolerance) missed.push(p.claim);
  }
  return missed; // graphs awaiting — measurements pending, shipping declining
}

  { claim: 'p99-under-300', predicted: 200, measured: 210 },
  { claim: 'cap-8k', predicted: 8000, measured: 4000 },
];
console.log(confirmed(probes, 100).join() === 'cap-8k' ? 'proofs propose' : 'claims assert');
```
