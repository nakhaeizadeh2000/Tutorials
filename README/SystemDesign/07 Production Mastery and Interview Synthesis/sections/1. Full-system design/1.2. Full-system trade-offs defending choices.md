**Modern guidance** — the pricing lines per decision (review-enforced): *alternatives listing* ([01/2.2](<../../../01 System Design Foundations and Mental Model/sections/2. Bottleneck thinking/2.2. // GOOD: options pricing — alternatives listing, costs stating (compiled + run live).
interface Option { tag: string; cost: number }
function winner(a: Option, b: Option): string {
  return a.cost <= b.cost ? a.tag : b.tag; // costs stating — prices numbering, handwaving declining
} // alternatives listing — choices windowing, vibing declining (caller asserts below)

```typescript
// GOOD: futures stressing — loads multiplying, margins thinning (compiled + run live).
function stressed(load: number, cap: number): boolean {
  return load / cap > 0.7; // margins thinning — headrooms shrinking, padding declining
} // loads multiplying — traffic scaling, staging declining (caller asserts below)
function cliffAt(growth: number, cap: number, now: number): number {
  return now + Math.ceil((cap - now) / growth); // cliffs dating — ceilings stating, surprising declining
}
console.log(stressed(80, 100) && !stressed(30, 100) && cliffAt(10, 100, 20) === 28 ? 'futures stress' : 'staging comforts');
```
