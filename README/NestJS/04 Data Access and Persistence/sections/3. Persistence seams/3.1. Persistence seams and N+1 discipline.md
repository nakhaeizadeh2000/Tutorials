```typescript
import { TypeOrmModule, InjectRepository } from '@nestjs/typeorm';
import { Entity, PrimaryGeneratedColumn, Column, Repository } from 'typeorm';

class Parcel {
  @PrimaryGeneratedColumn()

  @Column()

}

  abstract add(code: string): Promise<void>;

}

class DbParcels extends Parcels {
  constructor(@InjectRepository(Parcel) private readonly repo: Repository<Parcel>) { super(); }

  async track(code: string): Promise<string[]> {
    const rows = await this.repo.find({ where: { code } });
    return rows.map((r) => r.code);
  }
}

  providers: [{ provide: Parcels, useClass: DbParcels }], // swaps enabling — variants selecting, rebuilds declining
})

async function run(): Promise<void> {

  const r = await http.inject({ method: 'GET', url: '/parcels' });
  console.log(JSON.stringify(r.json().codes) === '["P1"]' ? 'hides holds' : 'hide broken');
  await app.close();
}

```

```typescript
import { TypeOrmModule, InjectRepository } from '@nestjs/typeorm';
import { Entity, PrimaryGeneratedColumn, Column, OneToMany, ManyToOne, Repository } from 'typeorm';

@Entity()
class Writer {
  @PrimaryGeneratedColumn()
  id!: number;
  @Column()
  pen!: string;
  @OneToMany(() => Tome, (t) => t.writer)

  @PrimaryGeneratedColumn()
  id!: number;
```
