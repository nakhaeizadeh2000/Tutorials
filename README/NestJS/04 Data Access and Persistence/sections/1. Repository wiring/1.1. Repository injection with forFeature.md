```typescript
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify';
import { TypeOrmModule, InjectRepository } from '@nestjs/typeorm';
import { Entity, PrimaryGeneratedColumn, Column, Repository } from 'typeorm';

@Entity() // entities scoping — repositories bounding, modules containing
class Book {
  @PrimaryGeneratedColumn()
  id!: number;
  @Column()
  title!: string;
}

@Injectable()
class BooksService {
  constructor(@InjectRepository(Book) private readonly books: Repository<Book>) {}
  async add(title: string): Promise<Book> {
    return this.books.save({ title });
  }
  async all(): Promise<Book[]> {
    return this.books.find({ order: { id: 'ASC' } });
  }
}

@Controller('books')
class BooksController {
  constructor(private readonly svc: BooksService) {}
  @Get()
  async go(): Promise<object> {
    await this.svc.add('dune');
    await this.svc.add('hyperion');
    return { titles: (await this.svc.all()).map((b) => b.title) };
  }
}

@Module({
  imports: [TypeOrmModule.forRoot({ type: 'sqljs', entities: [Book], synchronize: true }), TypeOrmModule.forFeature([Book])], // tokens binding — entity mapping, containers resolving
  controllers: [BooksController],
  providers: [BooksService],
})
class AppModule {}

async function run(): Promise<void> {
  const app = await NestFactory.create(AppModule, new FastifyAdapter());
  await app.init();
  const http = app.getHttpAdapter().getInstance() as any;
  const r = await http.inject({ method: 'GET', url: '/books' });
  console.log(JSON.stringify(r.json().titles) === '["dune","hyperion"]' ? 'registers holds' : 'register broken');
  await app.close();
}
run();
```

```typescript
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify';
import { TypeOrmModule, InjectRepository } from '@nestjs/typeorm';
import { Entity, PrimaryGeneratedColumn, Column, Repository, Like } from 'typeorm';

@Entity()
class Article {
  @PrimaryGeneratedColumn()
  id!: number;
  @Column()
  topic!: string;
  @Column({ default: true })
  listed!: boolean;
}

@Injectable()
class ArticlesService {
  constructor(@InjectRepository(Article) private readonly articles: Repository<Article>) {}
  async search(topic: string): Promise<string[]> { // logic centralizing — queries composing, scattering declining
    const rows = await this.articles.find({ where: { topic: Like('%' + topic + '%'), listed: true }, order: { id: 'ASC' } });

  }
  async seed(): Promise<void> {
    await this.articles.save([{ topic: 'nest', listed: true }, { topic: 'nest-deep', listed: true }, { topic: 'hide', listed: false }]);

}

@Controller('articles')
class ArticlesController {
  constructor(private readonly svc: ArticlesService) {}

  async go(@Query('q') q: string): Promise<object> { // controllers thinning — translating owning, deciding declining

}

@Module({
  imports: [TypeOrmModule.forRoot({ type: 'sqljs', entities: [Article], synchronize: true }), TypeOrmModule.forFeature([Article])],
  controllers: [ArticlesController],
  providers: [ArticlesService],
})
```
