// GOOD: forRoot opening — pools creating, entities registering, apps serving (compiled + run live, sqljs in-memory).
import 'reflect-metadata';
import { Controller, Get, Injectable, Module } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify';
import { TypeOrmModule, InjectRepository, InjectDataSource } from '@nestjs/typeorm';
import { Entity, PrimaryGeneratedColumn, Column, Repository, DataSource } from 'typeorm';

@Entity()
class Author {
  @PrimaryGeneratedColumn()
  id!: number;
  @Column()
  name!: string;
}

@Entity() // entities registering — tables mapping, drift declining
class Novel {
  @PrimaryGeneratedColumn()
  id!: number;
  @Column()
  title!: string;
}

@Injectable()
class LibraryService {
  constructor(
    @InjectRepository(Author) private readonly authors: Repository<Author>,
    @InjectRepository(Novel) private readonly novels: Repository<Novel>,
    @InjectDataSource() private readonly ds: DataSource,
  ) {}
  async seed(): Promise<void> {
    await this.authors.save({ name: 'le-guin' });
    await this.novels.save({ title: 'earthsea' });
  }
  async counts(): Promise<object> {
    return { authors: await this.authors.count(), novels: await this.novels.count(), open: this.ds.isInitialized };
  }
}

@Controller('lib')
class LibController {
  constructor(private readonly svc: LibraryService) {}
  @Get()
  async go(): Promise<object> {
    await this.svc.seed();
    return this.svc.counts();
  }
}

@Module({
  imports: [TypeOrmModule.forRoot({ type: 'sqljs', entities: [Author, Novel], synchronize: true }), TypeOrmModule.forFeature([Author, Novel])], // pools creating — connections opening, exhaustion declining
  controllers: [LibController],
  providers: [LibraryService],
})
class AppModule {}

async function run(): Promise<void> {
  const app = await NestFactory.create(AppModule, new FastifyAdapter());
  await app.init();
  const http = app.getHttpAdapter().getInstance() as any;
  const r = await http.inject({ method: 'GET', url: '/lib' });
  console.log(r.json().authors === 1 && r.json().novels === 1 && r.json().open === true ? 'opens holds' : 'open broken');
  await app.close();
}
run();
```

// GOOD: forRootAsync deferring — configs injecting, secrets awaiting, graphs composing (compiled + run live, sqljs in-memory).
import 'reflect-metadata';
import { Controller, Get, Injectable, Module } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify';
import { ConfigModule, ConfigService } from '@nestjs/config';
import { TypeOrmModule, InjectRepository } from '@nestjs/typeorm';
import { Entity, PrimaryGeneratedColumn, Column, Repository } from 'typeorm';

@Entity()
class Gadget {
  @PrimaryGeneratedColumn()
  id!: number;
  @Column()
  code!: string;
}

@Injectable()
class GadgetsService {
  constructor(@InjectRepository(Gadget) private readonly gadgets: Repository<Gadget>) {}
  async add(code: string): Promise<void> {
    await this.gadgets.save({ code });
  }
  async count(): Promise<number> {
    return this.gadgets.count();
  }
}

@Controller('gadgets')
class GadgetsController {
  constructor(private readonly svc: GadgetsService) {}
  @Get()
  async go(): Promise<object> {
    await this.svc.add('g1');
    return { n: await this.svc.count() };
  }
}

@Module({
  imports: [
    ConfigModule.forRoot({ isGlobal: true }), // configs injecting — env reading, hardcoded declining
    TypeOrmModule.forRootAsync({
      imports: [ConfigModule],
      inject: [ConfigService],
      useFactory: (cfg: ConfigService) => ({ // secrets awaiting — credentials resolving, races ending
        type: 'sqljs' as const,
        entities: [Gadget],
        synchronize: cfg.get('SYNC', 'true') === 'true',
      }),
    }),
    TypeOrmModule.forFeature([Gadget]),
  ],
  controllers: [GadgetsController],
  providers: [GadgetsService],
})
class AppModule {}

async function run(): Promise<void> {
  const app = await NestFactory.create(AppModule, new FastifyAdapter());
  await app.init();
  const http = app.getHttpAdapter().getInstance() as any;
  const r = await http.inject({ method: 'GET', url: '/gadgets' });
  console.log(r.json().n === 1 ? 'defers holds' : 'defer broken');
  await app.close();
}
run();
```
