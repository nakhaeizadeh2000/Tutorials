```typescript
// GOOD: runners scoping — transactions opening, queries threading, partials declining (compiled + run live, sqljs in-memory).
import 'reflect-metadata';
import { Controller, Get, Injectable, Module } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify';
import { TypeOrmModule, InjectDataSource } from '@nestjs/typeorm';
import { Entity, PrimaryGeneratedColumn, Column, DataSource } from 'typeorm';

@Entity()
class Ledger {
  @PrimaryGeneratedColumn()
  id!: number;
  @Column()
  entry!: string;
}

@Injectable()
class LedgersService {
  constructor(@InjectDataSource() private readonly ds: DataSource) {} // transactions opening — runners creating, atomicity owning
  async post(entries: string[]): Promise<void> {
    await this.ds.transaction(async (mgr) => { // queries threading — managers passing, unity owning
      for (const e of entries) await mgr.save(Ledger, { entry: e });
    });
  }
  async count(): Promise<number> {
    return this.ds.getRepository(Ledger).count();
  }
}

@Controller('post')
class PostController {
  constructor(private readonly svc: LedgersService) {}
  @Get()
  async go(): Promise<object> {
    await this.svc.post(['debit', 'credit']);
    return { n: await this.svc.count() }; // partials declining — all committing, breakage declining
  }
}

@Module({
  imports: [TypeOrmModule.forRoot({ type: 'sqljs', entities: [Ledger], synchronize: true })],
  controllers: [PostController],
  providers: [LedgersService],
})
class AppModule {}

async function run(): Promise<void> {
  const app = await NestFactory.create(AppModule, new FastifyAdapter());
  await app.init();
  const http = app.getHttpAdapter().getInstance() as any;
  const r = await http.inject({ method: 'GET', url: '/post' });
  console.log(r.json().n === 2 ? 'scopes holds' : 'scope broken');
  await app.close();
}
run();
```

```typescript
// GOOD: commits pairing — outcomes settling, cleanup owning, leaks declining (compiled + run live, sqljs in-memory).
import 'reflect-metadata';
import { Controller, Get, Injectable, Module } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify';
import { TypeOrmModule, InjectDataSource } from '@nestjs/typeorm';
import { Entity, PrimaryGeneratedColumn, Column, DataSource } from 'typeorm';

@Entity()
class Order {
  @PrimaryGeneratedColumn()
  id!: number;
  @Column()
  sku!: string;
}

@Injectable()
class OrdersService {
  constructor(@InjectDataSource() private readonly ds: DataSource) {}
  async place(skus: string[], fail: boolean): Promise<void> {
    await this.ds.transaction(async (mgr) => {
      for (const s of skus) await mgr.save(Order, { sku: s });
      if (fail) throw new Error('payment refused'); // outcomes settling — rollbacks reverting, states agreeing
    });
  }
  async count(): Promise<number> {
    return this.ds.getRepository(Order).count();
  }
}

@Controller('orders')
class OrdersController {
  constructor(private readonly svc: OrdersService) {}
  @Get()
  async go(): Promise<object> {
    await this.svc.place(['a'], false);
    try {
      await this.svc.place(['b', 'c'], true);
    } catch {
      // refused — rollback expected
    }
    return { n: await this.svc.count() }; // leaks declining — transactions ending, growth bounding
  }
}

@Module({
  imports: [TypeOrmModule.forRoot({ type: 'sqljs', entities: [Order], synchronize: true })],
  controllers: [OrdersController],
  providers: [OrdersService],
})
class AppModule {}

async function run(): Promise<void> {
  const app = await NestFactory.create(AppModule, new FastifyAdapter());
  await app.init();
  const http = app.getHttpAdapter().getInstance() as any;
  const r = await http.inject({ method: 'GET', url: '/orders' });
  console.log(r.json().n === 1 ? 'pairs holds' : 'pair broken');
  await app.close();
}
run();
```

```typescript
// GOOD: runners threading — parameters passing, globals declining, tests controlling (compiled + run live, sqljs in-memory).
import 'reflect-metadata';
import { Controller, Get, Injectable, Module } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify';
import { TypeOrmModule, InjectDataSource } from '@nestjs/typeorm';
import { Entity, PrimaryGeneratedColumn, Column, DataSource, EntityManager } from 'typeorm';

@Entity()
class Seat {
  @PrimaryGeneratedColumn()
  id!: number;
  @Column()
  row!: string;
}

@Entity()
class Ticket {
  @PrimaryGeneratedColumn()
  id!: number;
  @Column()
  seatRow!: string;
}

@Injectable()
class SeatsService {
  async hold(mgr: EntityManager, row: string): Promise<void> { // parameters passing — managers threading, hiding declining
    await mgr.save(Seat, { row });
  }
}

@Injectable()
class TicketsService {
  async issue(mgr: EntityManager, row: string): Promise<void> {
    await mgr.save(Ticket, { seatRow: row });
  }
}

@Injectable()
class BookingService {
  constructor(
    @InjectDataSource() private readonly ds: DataSource,
    private readonly seats: SeatsService,
    private readonly tickets: TicketsService,
  ) {}
  async book(row: string): Promise<void> {
    await this.ds.transaction(async (mgr) => {
      await this.seats.hold(mgr, row);
      await this.tickets.issue(mgr, row); // globals declining — runners unsharing, tests freeing
    });
  }
  async counts(): Promise<object> {
    return { seats: await this.ds.getRepository(Seat).count(), tickets: await this.ds.getRepository(Ticket).count() };
  }
}

@Controller('book')
class BookController {
  constructor(private readonly svc: BookingService) {}
  @Get()
  async go(): Promise<object> {
    await this.svc.book('A1');
    return this.svc.counts();
  }
}

@Module({
  imports: [TypeOrmModule.forRoot({ type: 'sqljs', entities: [Seat, Ticket], synchronize: true })],
  controllers: [BookController],
  providers: [SeatsService, TicketsService, BookingService],
})
class AppModule {}

async function run(): Promise<void> {
  const app = await NestFactory.create(AppModule, new FastifyAdapter());
  await app.init();
  const http = app.getHttpAdapter().getInstance() as any;
  const r = await http.inject({ method: 'GET', url: '/book' });
  console.log(r.json().seats === 1 && r.json().tickets === 1 ? 'threads holds' : 'thread broken');
  await app.close();
}
run();
```
