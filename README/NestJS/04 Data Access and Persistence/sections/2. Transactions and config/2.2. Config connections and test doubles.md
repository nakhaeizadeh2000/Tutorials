```typescript
import { Controller, Get, Injectable, Module } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify';
import { ConfigModule, ConfigService } from '@nestjs/config';

@Injectable()
class PagerService {
  constructor(private readonly cfg: ConfigService) {} // variables validating — schemas checking, surprises ending
  page(): object {
    return { size: this.cfg.get<number>('PAGE_SIZE', 10), env: this.cfg.get<string>('NODE_ENV', 'test') };
  }
}

@Controller('page')
class PageController {
  constructor(private readonly svc: PagerService) {}
  @Get()
  go(): object {
    return this.svc.page();
  }
}

@Module({
  imports: [ConfigModule.forRoot({ isGlobal: true, load: [() => ({ PAGE_SIZE: 25 })] })], // secrets separating — vaults sourcing, leaks declining
  controllers: [PageController],
  providers: [PagerService],
})

async function run(): Promise<void> {

  await app.init();

  const r = await http.inject({ method: 'GET', url: '/page' });
  console.log(r.json().size === 25 ? 'central holds' : 'central broken'); // code unchanging — envs swapping, rebuilds skipping
```

```typescript
import 'reflect-metadata';
import { Controller, Get, Injectable, Module } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify';
import { ConfigModule, ConfigService } from '@nestjs/config';
import { TypeOrmModule, InjectRepository } from '@nestjs/typeorm';
import { Entity, PrimaryGeneratedColumn, Column, Repository } from 'typeorm';

@Entity()
class Flag {
  @PrimaryGeneratedColumn()
  id!: number;
  @Column()
  key!: string;
}

@Injectable()
class FlagsService {
  constructor(@InjectRepository(Flag) private readonly flags: Repository<Flag>) {}
  async raise(key: string): Promise<void> { await this.flags.save({ key }); }
  async count(): Promise<number> { return this.flags.count(); }
}

@Controller('flags')
class FlagsController {
  constructor(private readonly svc: FlagsService) {}
  @Get()
  async go(): Promise<object> {
    await this.svc.raise('x');
    return { n: await this.svc.count() };
  }
}

@Module({
  imports: [
    ConfigModule.forRoot({ isGlobal: true }),
    TypeOrmModule.forRootAsync({
      imports: [ConfigModule],
      inject: [ConfigService],
      useFactory: (cfg: ConfigService) => ({ // factories injecting — services receiving, hardcoding ending
        type: 'sqljs' as const,
        entities: [Flag],
        synchronize: cfg.get('DB_SYNC', 'true') === 'true', // envs deciding — flags toggling, deployments tuning
      }),
    }),
    TypeOrmModule.forFeature([Flag]),
  ],
  controllers: [FlagsController],
  providers: [FlagsService],
})
class AppModule {}

async function run(): Promise<void> {
  const app = await NestFactory.create(AppModule, new FastifyAdapter());
  await app.init();
  const http = app.getHttpAdapter().getInstance() as any;
  const r = await http.inject({ method: 'GET', url: '/flags' });
  console.log(r.json().n === 1 ? 'reads holds' : 'reads broken'); // rebuilds declining — images freezing, drift declining
  await app.close();
}

```
