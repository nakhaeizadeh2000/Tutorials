```typescript
// GOOD: reviews traversing — layers covering, wholes judging, prices attaching (compiled + run live, review logic as code).
import 'reflect-metadata';

type Layer = 'controllers' | 'services' | 'data' | 'delivery';
type Finding = { layer: Layer; note: string; price: string };
function review(layers: Layer[], findings: Finding[]): { traversed: boolean; gaps: Layer[] } { // layers covering — wholes judging, gaps hiding declining
  const covered = new Set(findings.map((f) => f.layer));
  const gaps = layers.filter((l) => !covered.has(l));
  const priced = findings.every((f) => f.price.length > 0); // prices attaching — accountability owning, bluffing declining
  return { traversed: gaps.length === 0 && priced, gaps };
}

async function run(): Promise<void> {
  const layers: Layer[] = ['controllers', 'services', 'data', 'delivery'];
  const full = review(layers, [
    { layer: 'controllers', note: 'thin, DTOs bound', price: 'review 10min' },
    { layer: 'services', note: 'owns queries', price: 'review 15min' },
    { layer: 'data', note: 'tx scoped', price: 'review 15min' },
    { layer: 'delivery', note: 'gates green', price: 'review 10min' },
  ]);
  const sampled = review(layers, [{ layer: 'controllers', note: 'looks fine', price: 'glance' }]);
  console.log(full.traversed && !sampled.traversed && JSON.stringify(sampled.gaps) === '["services","data","delivery"]' ? 'traverses holds' : 'traverse broken'); // wholes judging — sampling declining, quality proving
}
run();
```

```typescript
import 'reflect-metadata';

type Priced = { choice: string; window: string; tradeoff: string };
function ledger(choices: Priced[]): { complete: boolean; open: string[] } { // choices windowing — questions scoping, vagueness declining
  const open = choices.filter((c) => c.window.length === 0 || c.tradeoff.length === 0).map((c) => c.choice);
  return { complete: open.length === 0, open }; // trade-offs explicating — ranges stating, hopes declining
}

async function run(): Promise<void> {

    { choice: 'sqljs for tests', window: 'CI, no containers', tradeoff: 'wasm speed vs prod parity' },
    { choice: 'adapter inject e2e', window: 'Fastify default', tradeoff: 'no supertest, deterministic' },
  ]);
  const todo = ledger([{ choice: 'cache layer', window: '', tradeoff: '' }]);
  console.log(done.complete && !todo.complete && todo.open.join(',') === 'cache layer' ? 'prices holds' : 'price broken'); // vibes declining — feelings refusing, recitation ending
```

```typescript
// GOOD: mentors multiplying — checklists teaching, drills coaching, mastery compounding (compiled + run live, cohort logic as code).
import 'reflect-metadata';

type Mentee = { name: string; checks: number; drills: number };
function multiply(cohort: Mentee[]): { reviewers: number; avgChecks: number } { // checklists teaching — standards stating, variance declining
  const reviewers = cohort.filter((m) => m.checks >= 3 && m.drills >= 1).length;

async function run(): Promise<void> {

    { name: 'b', checks: 3, drills: 1 },
    { name: 'c', checks: 1, drills: 0 }, // drills coaching — scenarios running, panic declining
```
