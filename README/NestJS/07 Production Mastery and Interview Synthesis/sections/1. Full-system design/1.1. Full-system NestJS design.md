# 1.1. Full-system NestJS design

## 1. Layers compose vertically (controllers → services → data — requests flowing, concerns separating)

**Definition** — vertical composition: controllers translating HTTP, services deciding use-cases, repositories persisting — requests flowing down, concerns separating per layer (controllers translating (routes mapping — params contracting (responses stating — handlers thinning (entry owning — decisions declining (clarity keeping — testing easing); services deciding (queries owning — rules enforcing (transactions threading — logic centralizing (purpose owning — quality funding (decisions serving — scattering declining); data persisting (entities mapping — repositories flowing (stores abstracting — integrity holding (durability owning — trust earning). Layers composing (controllers translating — services deciding (data persisting — systems holding).

**What problem it solves** — the unlayered trio: *handlers deciding* (logic embedding — queries scattering (translating missing — fat growing (entry failing — clarity eroding (thinning absenting — testing aching (deciding owning — scattering scheduling (separating failing — mastery doubting); *services bypassing* (controllers querying — repositories leaking (deciding absenting — logic scattering (centralizing missing — reuse dying (purpose failing — quality doubting (owning lacking — velocity dying (threading missing — atomicity breaking); *data leaking* (stores surfacing — entities exposing (persisting missing — secrets broadcasting (abstracting absenting — coupling growing (mapping failing — integrity doubting (hiding lacking — audits aching). Composing layers converting deciding to translating (routes mapping — services ruling (stores hiding — systems holding); reviewers checking layer direction, not file sizes (composing proving → verifiable; deciding → returned).

**Modern guidance** — the compose lines per system (review-enforced, verified: sell returns numeric id, sold counts 1 across layers): *controllers translating* (routes mapping — params contracting (responses stating — handlers thinning (entry owning — decisions declining (clarity keeping — testing easing (translating owning — deciding declining); *services deciding* (queries owning — rules enforcing (transactions threading — logic centralizing (purpose owning — quality funding (deciding owning — scattering declining); *data persisting* (entities mapping — repositories flowing (stores abstracting — integrity holding (durability owning — trust earning (persisting owning — leaking declining).

```typescript
// GOOD: layers composing — controllers translating, services deciding, data persisting (compiled + run live, sqljs in-memory).
import 'reflect-metadata';
import { Controller, Get, Injectable, Module } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify';
import { TypeOrmModule, InjectRepository } from '@nestjs/typeorm';
import { Entity, PrimaryGeneratedColumn, Column, Repository } from 'typeorm';

@Entity()
class Ticket {
  @PrimaryGeneratedColumn()
  id!: number;
  @Column()
  event!: string;
}

@Injectable()
class BoxOfficeService { // services deciding — queries owning, scattering declining
  constructor(@InjectRepository(Ticket) private readonly tickets: Repository<Ticket>) {}
  async sell(event: string): Promise<Ticket> {
    return this.tickets.save({ event });
  }
  async sold(): Promise<number> {
    return this.tickets.count();
  }
}

@Controller('box')
class BoxController { // controllers translating — routes mapping, deciding declining
  constructor(private readonly svc: BoxOfficeService) {}
  @Get('sell')
  async sell(): Promise<object> {
    const t = await this.svc.sell('gig');
    return { id: t.id };
  }
  @Get('sold')
  async sold(): Promise<object> {
    return { n: await this.svc.sold() };
  }
}

@Module({
  imports: [TypeOrmModule.forRoot({ type: 'sqljs', entities: [Ticket], synchronize: true }), TypeOrmModule.forFeature([Ticket])], // data persisting — entities mapping, leaking declining
  controllers: [BoxController],
  providers: [BoxOfficeService],
})
class AppModule {}

async function run(): Promise<void> {
  const app = await NestFactory.create(AppModule, new FastifyAdapter());
  await app.init();
  const http = app.getHttpAdapter().getInstance() as any;
  const s = await http.inject({ method: 'GET', url: '/box/sell' });
  const c = await http.inject({ method: 'GET', url: '/box/sold' });
  console.log(typeof s.json().id === 'number' && c.json().n === 1 ? 'composes holds' : 'compose broken');
  await app.close();
}
run();
```

```typescript
// GOOD: contracts bounding — DTOs shaping, errors mapping, clients trusting (compiled + run live, zero listening ports).
import 'reflect-metadata';
import { ArgumentsHost, Body, Catch, Controller, ExceptionFilter, Get, HttpException, HttpStatus, Injectable, Module, Post, ValidationPipe } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify';
import { IsString, MinLength } from 'class-validator';

class JoinDto {
  @IsString()
  @MinLength(2)
  name!: string; // DTOs shaping — inputs bounding, drift ending
}

@Catch()
class EnvelopeFilter implements ExceptionFilter {
  catch(exception: unknown, host: ArgumentsHost): void {
    const reply = host.switchToHttp().getResponse();
    const status = exception instanceof HttpException ? exception.getStatus() : HttpStatus.INTERNAL_SERVER_ERROR;
    reply.code(status).send({ error: 'failed', status }); // errors mapping — envelopes unifying, guessing declining
  }
}

@Injectable()
class RosterService {
  private names: string[] = [];
  join(name: string): string[] {
    this.names.push(name);
    return [...this.names];
  }
}

@Controller('roster')
class RosterController {
  constructor(private readonly svc: RosterService) {}
  @Post('join')
  join(@Body(new ValidationPipe({ whitelist: true })) dto: JoinDto): object {
    return { roster: this.svc.join(dto.name) };
  }
  @Get('boom')
  boom(): never {
    throw new HttpException('down', 503);
  }
}

@Module({ controllers: [RosterController], providers: [RosterService] })
class AppModule {}

async function run(): Promise<void> {
  const app = await NestFactory.create(AppModule, new FastifyAdapter());
  app.useGlobalFilters(new EnvelopeFilter());
  await app.init();
  const http = app.getHttpAdapter().getInstance() as any;
  const ok = await http.inject({ method: 'POST', url: '/roster/join', payload: { name: 'ada' } });
  const bad = await http.inject({ method: 'POST', url: '/roster/join', payload: { name: 'x' } });
  const down = await http.inject({ method: 'GET', url: '/roster/boom' });
  const pass = JSON.stringify(ok.json().roster) === '["ada"]' && bad.statusCode === 400 && down.statusCode === 503 && down.json().error === 'failed';
  console.log(pass ? 'bounds holds' : 'bound broken'); // clients trusting — shapes holding, fear declining
  await app.close();
}
run();
```

```typescript
// GOOD: failures composing — triage branching, fallbacks serving, degradation designing (compiled + run live, zero listening ports).
import 'reflect-metadata';
import { Controller, Get, HttpException, Injectable, Module } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify';

@Injectable()
class QuoteService {
  constructor(private readonly down: boolean = false) {}
  async quote(): Promise<string> { // triage branching — failures classifying, chaos ending
    if (this.down) throw new HttpException({ degraded: true, quote: 'stale' }, 200);
    return 'fresh';
  }
}

@Controller('quote')
class QuoteController {
  constructor(private readonly svc: QuoteService) {}
  @Get()
  async go(): Promise<object> {
    try {
      return { quote: await this.svc.quote(), mode: 'live' };
    } catch (e) {
      const r = (e as HttpException).getResponse() as any;
      return { quote: r.quote, mode: 'degraded' }; // fallbacks serving — stale tolerating, users keeping
    }
  }
}

@Module({ controllers: [QuoteController], providers: [{ provide: QuoteService, useFactory: (): QuoteService => new QuoteService(true) }] })
class AppModule {}

async function run(): Promise<void> {
  const app = await NestFactory.create(AppModule, new FastifyAdapter());
  await app.init();
  const http = app.getHttpAdapter().getInstance() as any;
  const r = await http.inject({ method: 'GET', url: '/quote' });
  console.log(r.json().quote === 'stale' && r.json().mode === 'degraded' ? 'graces holds' : 'grace broken'); // degradation designing — flags bounding, ambush ending
  await app.close();
}
run();
```

