import 'reflect-metadata';

type Option = { name: string; latencyMs: [number, number]; costPerM: [number, number]; risk: string };

function decide(d: Decision): { priced: boolean; summary: string } { // choices windowing — questions scoping, vagueness declining
  const chosen = d.options.find((o) => o.name === d.chosen);
  if (!chosen || d.why.length === 0 || d.window.length === 0) return { priced: false, summary: 'unpriced' };

}

async function run(): Promise<void> {

    question: 'session store',
    window: 'Q4, 50k sessions/day',
    options: [
      { name: 'memory', latencyMs: [1, 2], costPerM: [0, 0], risk: 'lost on restart' },
      { name: 'redis', latencyMs: [2, 5], costPerM: [8, 12], risk: 'ops burden' },

    chosen: 'redis',
    why: 'survives restarts within $12/M budget',

  const bad = decide({ ...d, why: '' });

}
run();

// GOOD: loads repricing — traffic multiplying, budgets tightening, margins thinning (compiled + run live, scale math as code).
import 'reflect-metadata';

type Load = { rps: number; p99ms: number; errPct: number };
function reprice(base: Load, factor: number): Load { // traffic multiplying — assumptions breaking, linearity declining
  const scaled = base.rps * factor;
  return { rps: scaled, p99ms: scaled > 500 ? base.p99ms * 3 : base.p99ms, errPct: scaled > 500 ? 2.5 : base.errPct }; // budgets tightening — headrooms shrinking, slack declining
}

async function run(): Promise<void> {

  const x10 = reprice(base, 10); // margins thinning — safeties eroding, comfort declining
  const pass = x2.p99ms === 40 && x2.errPct === 0.1 && x10.p99ms === 120 && x10.errPct === 2.5;
