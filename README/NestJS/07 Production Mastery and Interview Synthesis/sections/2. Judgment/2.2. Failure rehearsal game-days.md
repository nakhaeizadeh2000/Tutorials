// GOOD: scenarios scripting — dead stores, slow upstreams, killed nodes scheduled (compiled + run live, zero listening ports).
import 'reflect-metadata';
import { Controller, Get, HttpException, Injectable, Module } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify';

type Fault = 'none' | 'dead-store' | 'slow-upstream';
@Injectable()
class FaultyService {
  constructor(private fault: Fault = 'none') {}
  arm(f: Fault): void { this.fault = f; } // dead stores scripting — outages rehearsing, panic ending
  async read(): Promise<string> {
    if (this.fault === 'dead-store') throw new HttpException({ degraded: true }, 503);
    if (this.fault === 'slow-upstream') await new Promise((r) => setTimeout(r, 60)); // slow upstreams timing — budgets proving, infinity ending
    return 'fresh';
  }
}

@Controller('data')
class DataController {
  constructor(private readonly svc: FaultyService) {}
  @Get()
  async go(): Promise<object> {
    return { data: await this.svc.read() };
  }
}

@Module({ controllers: [DataController], providers: [{ provide: FaultyService, useFactory: (): FaultyService => new FaultyService() }] })
class AppModule {}

async function run(): Promise<void> {
  const app = await NestFactory.create(AppModule, new FastifyAdapter());
  await app.init();
  const svc = app.get(FaultyService);
  const http = app.getHttpAdapter().getInstance() as any;
  const clean = await http.inject({ method: 'GET', url: '/data' });
  svc.arm('dead-store');
  const dead = await http.inject({ method: 'GET', url: '/data' });
  svc.arm('none');
  const back = await http.inject({ method: 'GET', url: '/data' }); // killed nodes scheduling — restarts rehearsing, fear declining
  console.log(clean.json().data === 'fresh' && dead.statusCode === 503 && back.json().data === 'fresh' ? 'scripts holds' : 'script broken');
  await app.close();
}
run();
```

// GOOD: programs compounding — cadence owning, coverage growing, confidence compounding (compiled + run live, program logic as code).
import 'reflect-metadata';

type Drill = { quarter: string; scenario: string; result: 'pass' | 'fail'; mttrMin: number };
function program(drills: Drill[]): { coverage: number; avgMttr: number; green: boolean } { // cadence owning — quarters scheduling, skipping declining
  const scenarios = new Set(drills.map((d) => d.scenario));
  const passed = drills.filter((d) => d.result === 'pass');
  const avgMttr = passed.reduce((s, d) => s + d.mttrMin, 0) / Math.max(1, passed.length);
  return { coverage: scenarios.size, avgMttr: Math.round(avgMttr), green: passed.length === drills.length }; // confidence compounding — MTTRs falling, bluffing ending
}

async function run(): Promise<void> {
  const q1 = program([{ quarter: 'Q1', scenario: 'dead-store', result: 'pass', mttrMin: 12 }]);
  const q2 = program([
    { quarter: 'Q1', scenario: 'dead-store', result: 'pass', mttrMin: 12 },
    { quarter: 'Q2', scenario: 'dead-store', result: 'pass', mttrMin: 8 },
    { quarter: 'Q2', scenario: 'slow-upstream', result: 'pass', mttrMin: 15 },
  ]); // coverage growing — scenarios multiplying, gaps declining
  console.log(q1.coverage === 1 && q2.coverage === 2 && q2.avgMttr === 12 && q2.green ? 'compounds holds' : 'compound broken');
}
run();
```

```typescript
// GOOD: radii graduating — staging proving, production bounding, maturity earning (compiled + run live, graduation logic as code).
import 'reflect-metadata';

type Stage = 'staging' | 'canary' | 'production';
type Gate = { stage: Stage; pct: number; errPct: number };
function graduate(gates: Gate[], ceiling: number): { stage: Stage | null; stopped: string } { // staging proving — scenarios rehearsing, surprises declining
  let reached: Stage | null = null;
  for (const g of gates) {
    if (g.errPct > ceiling) return { stage: reached, stopped: g.stage + ' at ' + g.errPct + '% errors' }; // production bounding — errors gating, recklessness declining
    reached = g.stage;
  }
  return { stage: reached, stopped: '' };
}

async function run(): Promise<void> {
  const clean = graduate([
    { stage: 'staging', pct: 100, errPct: 0.1 },
    { stage: 'canary', pct: 5, errPct: 0.2 },
    { stage: 'production', pct: 100, errPct: 0.3 },
  ], 1);
  const halted = graduate([
    { stage: 'staging', pct: 100, errPct: 0.1 },
    { stage: 'canary', pct: 5, errPct: 4.5 },
    { stage: 'production', pct: 100, errPct: 0 },
  ], 1);
  console.log(clean.stage === 'production' && halted.stage === 'staging' && halted.stopped.includes('canary') ? 'graduates holds' : 'graduate broken'); // maturity earning — greens repeating, hope retiring
}
run();
```
