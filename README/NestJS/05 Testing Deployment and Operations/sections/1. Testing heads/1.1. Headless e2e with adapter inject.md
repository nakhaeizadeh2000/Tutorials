// GOOD: apps booting — modules initializing, ports unbinding, flakes declining (compiled + run live, zero listening ports).
import 'reflect-metadata';
import { Controller, Get, INestApplication, Injectable, Module } from '@nestjs/common';
import { Test, TestingModule } from '@nestjs/testing';
import { FastifyAdapter } from '@nestjs/platform-fastify';

@Injectable()
class PingService {
  ping(): string {
    return 'pong';
  }
}

@Controller('ping')
class PingController {
  constructor(private readonly svc: PingService) {}
  @Get()
  go(): object {
    return { msg: this.svc.ping() };
  }
}

@Module({ controllers: [PingController], providers: [PingService] })
class AppModule {}

async function build(): Promise<{ mod: TestingModule; app: INestApplication }> {
  const mod = await Test.createTestingModule({ controllers: [PingController], providers: [PingService] }).compile(); // modules initializing — graphs settling, serving preparing
  const app = mod.createNestApplication(new FastifyAdapter()); // ports unbinding — servers silent, conflicts declining
  await app.init();
  return { mod, app };
}

async function run(): Promise<void> {
  const { app } = await build();
  const http = app.getHttpAdapter().getInstance() as any;
  const ok = await http.inject({ method: 'GET', url: '/ping' });
  const missing = await http.inject({ method: 'GET', url: '/nope' });
  console.log(ok.json().msg === 'pong' && missing.statusCode === 404 ? 'boots holds' : 'boot broken'); // flakes declining — networks skipping, reruns agreeing
  await app.close();
}
run();

```typescript
// GOOD: inject driving — adapter proving, contracts asserting, servers skipping (compiled + run live, zero listening ports).
import 'reflect-metadata';
import { Body, Controller, Get, HttpCode, INestApplication, Module, Post, ValidationPipe } from '@nestjs/common';
import { Test, TestingModule } from '@nestjs/testing';
import { FastifyAdapter } from '@nestjs/platform-fastify';
import { IsString, MinLength } from 'class-validator';

class NickDto {
  @IsString()
  @MinLength(2)
  nick!: string;
}

@Controller('nicks')
class NicksController {
  @Post()
  @HttpCode(201)
  create(@Body(new ValidationPipe({ whitelist: true })) dto: NickDto): object { // adapter proving — pipes validating, full owning
    return { nick: dto.nick };
  }
  @Get('health')
  health(): object {
    return { ok: true };
  }
}

@Module({ controllers: [NicksController] })
class AppModule {}

async function run(): Promise<void> {
  const mod: TestingModule = await Test.createTestingModule({ controllers: [NicksController] }).compile();
  const app: INestApplication = mod.createNestApplication(new FastifyAdapter());
  await app.init();
  const http = app.getHttpAdapter().getInstance() as any;
  const created = await http.inject({ method: 'POST', url: '/nicks', payload: { nick: 'ada' } });
  const rejected = await http.inject({ method: 'POST', url: '/nicks', payload: { nick: 'x' } });
  const health = await http.inject({ method: 'GET', url: '/nicks/health' }); // contracts asserting — statuses checking, regressions catching
  const pass = created.statusCode === 201 && created.json().nick === 'ada' && rejected.statusCode === 400 && health.json().ok === true;
  console.log(pass ? 'drives holds' : 'drive broken'); // servers skipping — sockets unopened, flakes declining
  await app.close();
}
run();
```

import { Controller, Get, INestApplication, Module } from '@nestjs/common';
import { Test } from '@nestjs/testing';
import { FastifyAdapter } from '@nestjs/platform-fastify';

@Controller('round')
class RoundController {
  @Get()
  go(): object {
    return { round: true };
  }
}

@Module({ controllers: [RoundController] })
class AppModule {}

async function oneRound(): Promise<boolean> {

}
