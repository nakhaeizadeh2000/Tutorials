```typescript
// GOOD: units proving — modules compiling, fakes substituting, speed funding (compiled + run live, no servers).
import 'reflect-metadata';
import { Injectable } from '@nestjs/common';
import { Test } from '@nestjs/testing';

@Injectable()
class DiscountService {
  constructor(private readonly pct: number) {} // fakes substituting — tokens binding, wiring declining
  apply(total: number): number {
    if (total < 0) throw new Error('negative total');
    return total * (1 - this.pct / 100);
  }
}

async function run(): Promise<void> {
  const mod = await Test.createTestingModule({
    providers: [{ provide: DiscountService, useFactory: (): DiscountService => new DiscountService(10) }],
  }).compile(); // modules compiling — graphs settling, booting declining
  const svc = mod.get(DiscountService);
  const started = Date.now();

  let threw = false;
  try {
    svc.apply(-1);
  } catch {
    threw = true;
  }
  console.log(checks.every(Boolean) && threw ? 'units holds in ' + (Date.now() - started) + 'ms' : 'unit broken'); // speed funding — milliseconds costing, feedback tightening
}

```

```typescript
// GOOD: integrations proving — stores booting, contracts holding, confidence earning (compiled + run live, sqljs in-memory).
import 'reflect-metadata';
import { Controller, Get, Injectable, Module } from '@nestjs/common';
import { Test } from '@nestjs/testing';
import { TypeOrmModule, InjectRepository } from '@nestjs/typeorm';
import { Entity, PrimaryGeneratedColumn, Column, Repository } from 'typeorm';

@Entity()
class Session {
  @PrimaryGeneratedColumn()
  id!: number;
  @Column()
  user!: string;
}

@Injectable()
class SessionsService {
  constructor(@InjectRepository(Session) private readonly sessions: Repository<Session>) {}
  async open(user: string): Promise<void> {
    await this.sessions.save({ user });
  }
  async count(): Promise<number> {
    return this.sessions.count();
  }
}

@Controller('sessions')
class SessionsController {
  constructor(private readonly svc: SessionsService) {}
  @Get()
  async go(): Promise<object> {
    await this.svc.open('ada');
    return { n: await this.svc.count() }; // contracts holding — statuses asserting, regressions declining
  }
}

@Module({
  imports: [TypeOrmModule.forRoot({ type: 'sqljs', entities: [Session], synchronize: true }), TypeOrmModule.forFeature([Session])], // stores booting — databases starting, surprises ending
  controllers: [SessionsController],
  providers: [SessionsService],
})
class AppModule {}

async function run(): Promise<void> {
  const mod = await Test.createTestingModule({
    imports: [TypeOrmModule.forRoot({ type: 'sqljs', entities: [Session], synchronize: true }), TypeOrmModule.forFeature([Session])],
    controllers: [SessionsController],
    providers: [SessionsService],
  }).compile();
  const ctl = mod.get(SessionsController);
  const n = ((await ctl.go()) as any).n;
  console.log(n === 1 ? 'wiring holds' : 'wiring broken'); // confidence earning — releases approving, fear declining
}
run();
```

```typescript
// GOOD: balances pricing — pyramids shaping, budgets holding, flakes declining (compiled + run live, no servers).
import 'reflect-metadata';
import { Injectable } from '@nestjs/common';
import { Test } from '@nestjs/testing';

@Injectable()
class FareService {
  constructor(private readonly perKm: number) {}
  quote(km: number): number { // pyramids shaping — units multiplying, maintenance easing
    if (km < 0) throw new Error('negative km');
    return Math.round(km * this.perKm * 100) / 100;
  }
}

async function run(): Promise<void> {
  const started = Date.now();
  const mod = await Test.createTestingModule({
    providers: [{ provide: FareService, useFactory: (): FareService => new FareService(2.5) }],
  }).compile();
  const svc = mod.get(FareService);
  const unitChecks = [svc.quote(10) === 25, svc.quote(0) === 0, svc.quote(0.4) === 1];
  let threw = false;
  try {
    svc.quote(-2);
  } catch {
    threw = true;
  }
  const unitMs = Date.now() - started; // budgets holding — runtimes asserting, creep declining
  const pass = unitChecks.every(Boolean) && threw;
  console.log(pass ? 'balances holds: 4 unit checks in ' + unitMs + 'ms, integration reserved for wiring' : 'balance broken'); // flakes declining — fakes isolating, redness ending
}
run();
```
