// GOOD: pipelines proving — types checking, units running, contracts holding (compiled + run live, pipeline logic as code).
import 'reflect-metadata';

type Stage = { name: string; run: () => boolean };
function pipeline(stages: Stage[]): { green: boolean; stoppedAt: string | null } { // types checking — compilers rejecting, surprises ending
  for (const s of stages) {
    if (!s.run()) return { green: false, stoppedAt: s.name }; // units running — logics proving, confidence earning
  }
  return { green: true, stoppedAt: null };
}

async function run(): Promise<void> {
  const log: string[] = [];
  const good = pipeline([
    { name: 'types', run: (): boolean => { log.push('types'); return true; } },
    { name: 'units', run: (): boolean => { log.push('units'); return true; } },
    { name: 'e2e', run: (): boolean => { log.push('e2e'); return true; } },
  ]);
  log.length = 0;
  const bad = pipeline([
    { name: 'types', run: (): boolean => { log.push('types'); return true; } },
    { name: 'units', run: (): boolean => { log.push('units'); return false; } },
    { name: 'e2e', run: (): boolean => { log.push('e2e'); return true; } }, // contracts holding — adapters asserting, regressions declining
  ]);
  console.log(good.green && good.stoppedAt === null && !bad.green && bad.stoppedAt === 'units' && log.join(',') === 'types,units' ? 'compiles holds' : 'pipeline broken');
}
run();
```

// GOOD: merges blocking — branches gating, mains greening, broken declining (compiled + run live, branch logic as code).
import 'reflect-metadata';

type Branch = { name: string; green: boolean };
function mergeable(target: string, status: Branch[]): { ok: boolean; reason: string } { // branches gating — statuses requiring, outages preventing
  if (target !== 'main') return { ok: true, reason: 'feature branch — checks advisory' };
  const red = status.filter((b) => !b.green).map((b) => b.name);
  return red.length === 0 ? { ok: true, reason: 'main green' } : { ok: false, reason: 'blocked by ' + red.join(',') }; // mains greening — heads passing, surprises declining
}

async function run(): Promise<void> {
  const a = mergeable('main', [{ name: 'types', green: true }, { name: 'units', green: true }]);
  const b = mergeable('main', [{ name: 'types', green: true }, { name: 'units', green: false }]);
  const c = mergeable('feat/x', [{ name: 'units', green: false }]);
  console.log(a.ok && !b.ok && b.reason === 'blocked by units' && c.ok ? 'blocks holds' : 'branch broken'); // broken declining — culprits naming, repeats declining
}
run();
```

import 'reflect-metadata';
import { createHash } from 'crypto';

type Artifact = { image: string; digest: string; config: Record<string, string> };
function seal(image: string, config: Record<string, string>): Artifact { // images freezing — digests pinning, surprises ending

async function run(): Promise<void> {
