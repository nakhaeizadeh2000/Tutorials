```typescript
// GOOD: liveness cheapening — processes answering, restarts triggering, hangs declining (compiled + run live, zero listening ports).
import 'reflect-metadata';
import { Controller, Get, Module } from '@nestjs/common';
import { Test } from '@nestjs/testing';
import { FastifyAdapter } from '@nestjs/platform-fastify';

@Controller()
class ProbeController {
  private readonly started = Date.now();
  @Get('live')
  live(): object { // processes answering — runtimes responding, deadlocks surfacing
    return { alive: true, uptimeMs: Date.now() - this.started };
  }
  @Get('ready')
  ready(): object {
    return { ready: true };
  }
}

@Module({ controllers: [ProbeController] })
class AppModule {}

async function run(): Promise<void> {
  const mod = await Test.createTestingModule({ controllers: [ProbeController] }).compile();
  const app = mod.createNestApplication(new FastifyAdapter());
  await app.init();
  const http = app.getHttpAdapter().getInstance() as any;
  const live = await http.inject({ method: 'GET', url: '/live' });
  const ready = await http.inject({ method: 'GET', url: '/ready' });
  console.log(live.json().alive === true && typeof live.json().uptimeMs === 'number' && ready.json().ready === true ? 'cheap holds' : 'probe broken'); // hangs declining — deeps refusing, cascades declining
  await app.close();
}
run();
```

"One" describing merges (semantics blurring — gating missing); split describing verdicts (aliveness proving — readiness confirming). Cheapen live (aliveness owning — restarts serving); deepen ready (dependencies owning — traffic gating).

```typescript
// GOOD: readiness gating — dependencies proving, pools warming, errors declining (compiled + run live, sqljs in-memory).
import 'reflect-metadata';
import { Controller, Get, HttpException, HttpStatus, Injectable, Module } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify';
import { TypeOrmModule, InjectDataSource } from '@nestjs/typeorm';
import { DataSource } from 'typeorm';

@Injectable()
class ReadinessService {
  constructor(@InjectDataSource() private readonly ds: DataSource) {}
  ready(): boolean { // dependencies proving — databases answering, hoping declining
    return this.ds.isInitialized;
  }
}

@Controller()
class ReadyController {
  constructor(private readonly svc: ReadinessService) {}
  @Get('ready')
  ready(): object {
    if (!this.svc.ready()) throw new HttpException({ ready: false }, HttpStatus.SERVICE_UNAVAILABLE); // errors declining — unready refusing, honesty owning
    return { ready: true };
  }
}

@Module({
  imports: [TypeOrmModule.forRoot({ type: 'sqljs', synchronize: true })], // pools warming — connections opening, spikes declining
  controllers: [ReadyController],
  providers: [ReadinessService],
})
class AppModule {}

async function run(): Promise<void> {
  const app = await NestFactory.create(AppModule, new FastifyAdapter());
  await app.init();
  const http = app.getHttpAdapter().getInstance() as any;
  const r = await http.inject({ method: 'GET', url: '/ready' });
  console.log(r.statusCode === 200 && r.json().ready === true ? 'gates holds' : 'ready broken: ' + r.statusCode);
  await app.close();
}
run();
```

"Equals" describing merges (verdicts blurring — gating missing); split describing honesty (signals proving — backends confirming). Prove readiness deeply (backends owning — traffic deserving); answer liveness cheaply (processes owning — restarts serving).

## 3. Shutdown drains flights (hooks closing — requests completing — data surviving)

**Definition** — flight-draining shutdown: hooks closing apps, requests completing in-flight, data surviving restarts (hooks closing (signals handling — modules destroying (runners freeing — pools restoring (civility owning — abandonment ending (operations serving — deploys funding); requests completing (in-flight finishing — responses returning (work honoring — clients unstranding (respect owning — errors declining (dignity proving — trust earning); data surviving (writes persisting — states checkpointing (restarts resuming — losses preventing (durability owning — corruption declining (integrity proving — confidence earning). **What problem it solves** — the abrupt trio: *signals killing* (SIGTERM abandoning — buffers dropping (closing missing — hooks unrun (civility failing — data losing (grace absenting — corruption inviting (handling lacking — mastery missing (dignity absenting — trust eroding); *flights dropping* (requests severing — responses truncating (completing missing — clients erroring (honoring absenting — stranding scheduling (respect failing — experience degrading (finishing lacking — quality failing (landing missing — reliability doubting); *data losing* (writes vanishing — states resetting (surviving missing — restarts blanking (persisting absenting — recovery hoping (durability failing — integrity breaking (checkpointing lacking — credibility doubting (resuming missing — confidence failing). **Modern guidance** — the drain lines per app (review-enforced, verified: in-flight request completes across `close()`): *hooks closing* (signals handling — modules destroying (runners freeing — pools restoring (civility owning — abandonment ending (operations serving — deploys funding (closing owning — killing declining); *requests completing* (in-flight finishing — responses returning (work honoring — clients unstranding (respect owning — errors declining (dignity proving — trust earning (completing owning — dropping declining); *data surviving* (writes persisting — states checkpointing (restarts resuming — losses preventing (durability owning — corruption declining (integrity proving — confidence earning (surviving owning — losing declining).

```typescript
// GOOD: shutdowns draining — hooks closing, requests completing, data surviving (compiled + run live, zero listening ports).
import 'reflect-metadata';
import { BeforeApplicationShutdown, Controller, Get, Injectable, Module } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify';

const events: string[] = [];
@Injectable()
class DrainService implements BeforeApplicationShutdown {
  inFlight = 0;
  async work(): Promise<string> {
    this.inFlight += 1;
    await new Promise((r) => setTimeout(r, 20));
    this.inFlight -= 1;
    return 'done';
  }
  beforeApplicationShutdown(): void { // hooks closing — signals handling, abandonment ending
    events.push('draining with inFlight=' + this.inFlight);
  }
}

@Controller('slow')
class SlowController {
  constructor(private readonly svc: DrainService) {}
  @Get()
  async go(): Promise<object> {
    return { result: await this.svc.work() }; // requests completing — in-flight finishing, stranding declining
  }
}

@Module({ controllers: [SlowController], providers: [DrainService] })
class AppModule {}

async function run(): Promise<void> {
  const app = await NestFactory.create(AppModule, new FastifyAdapter());
  app.enableShutdownHooks();
  await app.init();
  const http = app.getHttpAdapter().getInstance() as any;
  const flight = http.inject({ method: 'GET', url: '/slow' });
  await new Promise((r) => setTimeout(r, 5));
  await app.close();
  const r = await flight;
  console.log(r.json().result === 'done' && events.join(',').startsWith('draining') ? 'drains holds' : 'drain broken'); // data surviving — writes persisting, losses preventing
}
run();
```
