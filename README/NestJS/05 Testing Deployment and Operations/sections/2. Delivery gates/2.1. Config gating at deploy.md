import { Test } from '@nestjs/testing';
import { ConfigModule, ConfigService } from '@nestjs/config';

  if (!Number.isInteger(port) || port < 1 || port > 65535) throw new Error('PORT must be 1-65535, got ' + env['PORT']);

  return env;
}

class BootService {
  constructor(private readonly cfg: ConfigService) {}
  dsn(): string {
    return this.cfg.get<string>('DB_HOST', 'missing') + ':' + this.cfg.get<number>('PORT', 0);

}

    imports: [ConfigModule.forRoot({ isGlobal: true, validate })], // boots halting — modules refusing, mystery declining
    providers: [BootService],

}

  process.env['PORT'] = '4000';

    await boot();
  } catch (e) {
    crashed = (e as Error).message;

  delete process.env['DB_HOST'];

@Injectable()
class ShipService {
