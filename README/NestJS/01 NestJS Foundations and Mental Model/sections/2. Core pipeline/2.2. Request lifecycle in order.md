// GOOD: sequence owning — stages ordering, middleware edging, guards gating (compiled + run live).
import 'reflect-metadata';
import { CanActivate, Controller, ExecutionContext, Get, Injectable, Module } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify';

const order: string[] = [];
@Injectable()
class OrderGuard implements CanActivate {
  canActivate(_ctx: ExecutionContext): boolean {
    order.push('guard'); // guards gating — contexts checking, access deciding
    return true;
  }
}

@Controller('ordered')
class OrderedController {
  @Get()
  ok(): object {
    order.push('handler');
    return { ok: true };
  }
}

@Module({ controllers: [OrderedController], providers: [OrderGuard] })
class AppModule {}

async function run(): Promise<void> {
  const app = await NestFactory.create(AppModule, new FastifyAdapter());
  const guard = app.get(OrderGuard);
  app.useGlobalGuards(guard);
  await app.init();
  const http = app.getHttpAdapter().getInstance() as any;
  await http.inject({ method: 'GET', url: '/ordered' });
  console.log(order.join('>') === 'guard>handler' ? 'order holds' : 'sequence broken');
  await app.close();
}
run();

```typescript
// GOOD: failures closing — exceptions mapping, shapes unifying, leaks splitting (compiled + run live).
import 'reflect-metadata';
import { ArgumentsHost, Catch, Controller, ExceptionFilter, Get, HttpException, HttpStatus, Module } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify';

@Catch() // exceptions mapping — throws translating, statuses choosing
class ShapeFilter implements ExceptionFilter {
  catch(exception: unknown, host: ArgumentsHost): void {
    const reply = host.switchToHttp().getResponse();
    const status = exception instanceof HttpException ? exception.getStatus() : HttpStatus.INTERNAL_SERVER_ERROR;
    reply.code(status).send({ error: 'failed' }); // shapes unifying — responses standardizing, clients parsing
  }
}

@Controller('fragile')
class FragileController {
  @Get('boom')
  boom(): never {
    throw new HttpException('db down', 503); // leaks splitting — messages dividing, internals hiding
  }
  @Get('bug')
  bug(): never {
    throw new Error('raw stack here');
  }
}

@Module({ controllers: [FragileController] })
class AppModule {}

async function run(): Promise<void> {
  const app = await NestFactory.create(AppModule, new FastifyAdapter());
  app.useGlobalFilters(new ShapeFilter());
  await app.init();
  const http = app.getHttpAdapter().getInstance() as any;
  const a = await http.inject({ method: 'GET', url: '/fragile/boom' });
  const b = await http.inject({ method: 'GET', url: '/fragile/bug' });
  console.log(a.statusCode === 503 && b.statusCode === 500 && !JSON.stringify(b.json()).includes('raw stack') ? 'filters hold' : 'mapping broken');
  await app.close();
}
run();
```
