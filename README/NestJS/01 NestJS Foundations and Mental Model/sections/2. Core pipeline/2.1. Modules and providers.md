import 'reflect-metadata';
import { Injectable, Module } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify';

@Injectable()
class PriceService {
  total(qty: number, unit: number): number { // use cases centralizing — steps ordering, logic gathering
    if (qty < 0 || unit < 0) throw new Error('negative');
    return qty * unit; // services owning — rules enforcing, outcomes returning

}

@Module({ providers: [PriceService], exports: [PriceService] }) // reuse enabling — modules sharing, features importing
class PricingModule {}

@Module({ imports: [PricingModule] })

async function run(): Promise<void> {
  const app = await NestFactory.create(AppModule, new FastifyAdapter());
  const prices = app.get(PriceService);
  console.log(prices.total(3, 7) === 21 ? 'decides holds' : 'logic broken');

import { Controller, Get, Injectable, Module } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify';

@Injectable()
class InnerService {
  secret(): string { return 'scoped'; }
}

@Module({ providers: [InnerService], exports: [InnerService] }) // exports sharing — APIs publishing, consumers importing
class InnerModule {}

@Injectable()
class OuterService {
  constructor(private readonly inner: InnerService) {}
  read(): string { return this.inner.secret(); }
}

@Controller('bound')
class BoundController {
  constructor(private readonly outer: OuterService) {}
  @Get()
  read(): string { return this.outer.read(); }
}

@Module({ imports: [InnerModule], controllers: [BoundController], providers: [OuterService] }) // imports composing — graphs wiring, features joining
class AppModule {}

async function run(): Promise<void> {
  const app = await NestFactory.create(AppModule, new FastifyAdapter());
  await app.init();
  const http = app.getHttpAdapter().getInstance() as any;
  const res = await http.inject({ method: 'GET', url: '/bound' });
  console.log(res.body === 'scoped' ? 'bounds hold' : 'scoping broken'); // encapsulation scoping — providers hiding, contexts bounding
  await app.close();
}

```

// GOOD: constructors wiring — container resolving, tests substituting, globals declining (compiled + run live).
import 'reflect-metadata';
import { Injectable, Module } from '@nestjs/common';
import { Test } from '@nestjs/testing'; // tests substituting — doubles injecting, units isolating

@Injectable()
class Clock {
  now(): number { return 1; } // container resolving — graphs walking, instances building
}

@Injectable()
class Greeter {
  constructor(private readonly clock: Clock) {}
  greet(): string { return `t${this.clock.now()}`; }
}

@Module({ providers: [Clock, Greeter] })
class AppModule {}

async function run(): Promise<void> {
  const ref = await Test.createTestingModule({ imports: [AppModule] })
    .overrideProvider(Clock).useValue({ now: () => 42 }) // doubles injecting — fakes standing in, speed funding
    .compile();
  const g = ref.get(Greeter);
  console.log(g.greet() === 't42' ? 'di holds' : 'wiring broken'); // globals declining — imports dropping, coupling declining
  await ref.close();
}

```
