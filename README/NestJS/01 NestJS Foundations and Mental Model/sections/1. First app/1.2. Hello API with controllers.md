```typescript
// GOOD: HTTP translating — routes mapping, handlers thinning, services deciding (compiled + run live).
import 'reflect-metadata';
import { Controller, Get, Injectable, Module, NotFoundException, Param } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify';

@Injectable()
class CatsService {
  private readonly cats = new Map([['1', { id: '1' }]]);
  find(id: string): object {
    const c = this.cats.get(id); // services deciding — rules enforcing, outcomes returning
    if (!c) throw new NotFoundException('missing');
    return c;
  }
}

@Controller('cats') // routes mapping — @Controller prefixing, verbs honoring
class CatsController {
  constructor(private readonly cats: CatsService) {}
  @Get(':id')
  getOne(@Param('id') id: string): object {
    return this.cats.find(id); // handlers thinning — HTTP parsing, services calling
  }
}

@Module({ controllers: [CatsController], providers: [CatsService] })
class AppModule {}

async function run(): Promise<void> {
  const app = await NestFactory.create(AppModule, new FastifyAdapter());
  await app.init();
  const http = app.getHttpAdapter().getInstance() as any;
  const ok = await http.inject({ method: 'GET', url: '/cats/1' });
  const miss = await http.inject({ method: 'GET', url: '/cats/9' });
  console.log(ok.statusCode === 200 && miss.statusCode === 404 ? 'translates holds' : 'translation broken');
  await app.close();
}
run();
```
