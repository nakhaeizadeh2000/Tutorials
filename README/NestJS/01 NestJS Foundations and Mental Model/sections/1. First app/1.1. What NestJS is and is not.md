```typescript
// GOOD: architecture owning — modules bounding, providers deciding, controllers translating (compiled + run live).
import 'reflect-metadata';
import { Controller, Get, Injectable, Module } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify';

@Injectable()
class GreeterService {
  hello(): string { return 'structured'; } // providers deciding — services owning, logic holding
}

@Controller('greet')
class GreetController {
  constructor(private readonly greeter: GreeterService) {} // controllers translating — services calling, layers separating
  @Get()
  greet(): string { return this.greeter.hello(); }
}

@Module({ controllers: [GreetController], providers: [GreeterService] }) // modules bounding — graphs composing, contexts scoping
class AppModule {}

async function run(): Promise<void> {
  const app = await NestFactory.create(AppModule, new FastifyAdapter());
  await app.init();
  const http = app.getHttpAdapter().getInstance() as any;
  const res = await http.inject({ method: 'GET', url: '/greet' });
  console.log(res.statusCode === 200 && res.body === 'structured' ? 'structure holds' : 'structure broken');
  await app.close();
}

```
