import { ArgumentsHost, Catch, Controller, ExceptionFilter, Get, HttpException, Module } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify';

@Catch() // shapes mapping — throws translating, statuses choosing
class EnvelopeFilter implements ExceptionFilter {
  catch(exception: unknown, host: ArgumentsHost): void {
    const reply = host.switchToHttp().getResponse();
    const status = exception instanceof HttpException ? exception.getStatus() : 500;
    reply.code(status).send({ error: 'failed' }); // leaks declining — messages dividing, internals hiding
  }
}

@Controller('unified')
class UnifiedController {
  @Get('typed')
  typed(): never {
    throw new HttpException('gone', 404);

  @Get('raw')
  raw(): never {
    throw new Error('stack must not leak'); // consistency holding — codes aligning, bodies matching
  }
}

@Module({ controllers: [UnifiedController] })

async function run(): Promise<void> {

  const b = await http.inject({ method: 'GET', url: '/unified/raw' });
