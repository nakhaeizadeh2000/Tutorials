// GOOD: inits preparing — connections opening, caches warming, traffic preceding (compiled + run live).
import 'reflect-metadata';
import { Controller, Get, Injectable, Module, OnModuleInit } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify';

const events: string[] = [];
@Injectable()
class WarmerService implements OnModuleInit {
  ready = false;
  onModuleInit(): void { // caches warming — entries preloading, cold-starts ending
    this.ready = true;
    events.push('init');
  }
}

@Controller('warm')
class WarmController {
  constructor(private readonly warmer: WarmerService) {}
  @Get()
  go(): object {
    return { ready: this.warmer.ready }; // traffic preceding — readiness stating, races ending
  }
}

@Module({ controllers: [WarmController], providers: [WarmerService] })
class AppModule {}

async function run(): Promise<void> {
  const app = await NestFactory.create(AppModule, new FastifyAdapter());
  await app.init();
  const http = app.getHttpAdapter().getInstance() as any;
  const r = await http.inject({ method: 'GET', url: '/warm' });
  console.log(events.join(',') === 'init' && r.json().ready === true ? 'init holds' : 'init broken');
  await app.close();
}
run();
```

import 'reflect-metadata';
import { BeforeApplicationShutdown, Controller, Get, Injectable, Module, OnApplicationBootstrap } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify';

const events: string[] = [];
@Injectable()
class LifeService implements OnApplicationBootstrap, BeforeApplicationShutdown {
  onApplicationBootstrap(): void { // in-flight draining — work honoring, clients unstranding
    events.push('bootstrap');
  }
  beforeApplicationShutdown(): void { // data surviving — writes persisting, losses preventing
    events.push('before');
  }
}

@Controller('life')
class LifeController {
  @Get()
  go(): object {
    return { events: events.join('>') };
  }
}

@Module({ controllers: [LifeController], providers: [LifeService] })
class AppModule {}

async function run(): Promise<void> {
  const app = await NestFactory.create(AppModule, new FastifyAdapter());
  app.enableShutdownHooks(); // signals handling — SIGTERM catching, kills declining
  await app.init();
  const http = app.getHttpAdapter().getInstance() as any;
  const r = await http.inject({ method: 'GET', url: '/life' });
  await app.close();
  console.log(r.json().events === 'bootstrap' && events.join('>') === 'bootstrap>before' ? 'grace holds' : 'grace broken');
  await app.close();
}

```
