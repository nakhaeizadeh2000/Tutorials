```typescript
import { Controller, Get, Inject, Injectable, Module } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify';

interface Mailer {
  send(to: string): string;
}

@Injectable()
class SmtpMailer implements Mailer {
  constructor(private readonly host: string) {}
  send(to: string): string {
    return 'smtp:' + this.host + '>' + to;
  }
}

@Controller('mail')
class MailController {
  constructor(@Inject('MAILER') private readonly mailer: Mailer) {}
  @Get()
  go(): object {
    return { via: this.mailer.send('a@x.io') };
  }
}

  controllers: [MailController],
  providers: [
    {
      provide: 'MAILER',
      useFactory: async (): Promise<Mailer> => { // async awaiting — readiness proving, races ending
        const host = 'smtp.local'; // configs deciding — env reading, hardcoded declining
        return new SmtpMailer(host);
      },
    },
  ],
})

async function run(): Promise<void> {

  await app.init();
  const http = app.getHttpAdapter().getInstance() as any;
  const r = await http.inject({ method: 'GET', url: '/mail' });
  console.log(r.json().via === 'smtp:smtp.local>a@x.io' ? 'builds holds' : 'factory broken');
  await app.close();
}

```

```typescript
import { NestFactory } from '@nestjs/core';
```

```typescript
abstract class Store {
  abstract read(): string; // interfaces owning — contracts declaring, coupling declining
}

@Injectable()
class PgStore extends Store {
  read(): string {
    return 'pg';
  }
}

@Injectable()
class MemoryStore extends Store {
  read(): string {
    return 'mem';
  }
}

function build(target: 'pg' | 'mem'): any {
  @Controller('store')
  class StoreController {
    constructor(@Inject(Store) private readonly store: Store) {}
    @Get()
    go(): object {
      return { via: this.store.read() };

    controllers: [StoreController],
    providers: [{ provide: Store, useClass: target === 'pg' ? PgStore : MemoryStore }], // variants selecting — env deciding, hardcoding declining
```
