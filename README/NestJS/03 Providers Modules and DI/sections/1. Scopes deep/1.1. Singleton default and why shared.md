// GOOD: singletons sharing — instances reusing, memory bounding, graphs composing (compiled + run live).
import 'reflect-metadata';
import { Controller, Get, Injectable, Module } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify';

@Injectable()
class CounterService {
  count = 0;
  next(): number {
    this.count += 1;
    return this.count;
  }
}

@Controller('a')
class AController {
  constructor(private readonly counter: CounterService) {} // graphs composing — modules registering, injections sharing
  @Get()
  one(): object {
    return { n: this.counter.next() };
  }
}

@Controller('b')
class BController {
  constructor(private readonly counter: CounterService) {} // instances reusing — containers caching, one owning
  @Get()
  two(): object {
    return { n: this.counter.next() };
  }
}

@Module({ controllers: [AController, BController], providers: [CounterService] })
class AppModule {}

async function run(): Promise<void> {
  const app = await NestFactory.create(AppModule, new FastifyAdapter());
  await app.init();
  const http = app.getHttpAdapter().getInstance() as any;
  const a = await http.inject({ method: 'GET', url: '/a' });
  const b = await http.inject({ method: 'GET', url: '/b' });
  const same = app.get(CounterService) === app.get(CounterService); // memory bounding — one instance, growth flattening
  console.log(a.json().n === 1 && b.json().n === 2 && same ? 'shared holds' : 'sharing broken');
  await app.close();
}
run();
```

"Global" describing exposure (variables broadcasting — mutations racing); singleton describing lifetime (instances reusing — graphs composing). Scope singletons (lifetime owning — economy serving); keep them stateless (mutations declining — races ending).

**Performance note** — share economics: reusing costing one instance (containers caching — injections sharing (memory bounding — scale funding); multiplying costing heaps (constructs scattering — growth aching). Reuse cheaply (singletons — defaults); bound always (memory — instances).

  total(qty: number, unit: number): number { // state declining — fields freezing, purity owning

}

@Controller('price')
class PriceController {
  constructor(private readonly pricer: PricerService) {}
  @Get()
  quote(): object {
    const a = this.pricer.total(3, 7);
    const b = this.pricer.total(3, 7);
    return { same: a === b && a === 21 }; // concurrency surviving — results agreeing, flakiness declining

  const r = await http.inject({ method: 'GET', url: '/price' });

// GOOD: injection wiring — containers resolving, tests substituting, globals declining (compiled + run live).
import 'reflect-metadata';
import { Controller, Get, Injectable, Module } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify';
import { Test } from '@nestjs/testing';

@Injectable()
class GreeterService {
  hello(name: string): string {
    return 'hi ' + name;
  }
}

@Controller('greet')
class GreetController {
  constructor(private readonly greeter: GreeterService) {} // containers resolving — graphs settling, manualing declining
  @Get()
  greet(): object {
    return { msg: this.greeter.hello('nest') };
  }
}

@Module({ controllers: [GreetController], providers: [GreeterService] })
class AppModule {}

async function run(): Promise<void> {
  const mod = await Test.createTestingModule({ controllers: [GreetController], providers: [GreeterService] }).compile();
  const wired = mod.get(GreetController) instanceof GreetController && mod.get(GreeterService) instanceof GreeterService; // tests substituting — seams owning, confidence earning
  const app = await NestFactory.create(AppModule, new FastifyAdapter());
  await app.init();
  const http = app.getHttpAdapter().getInstance() as any;
  const r = await http.inject({ method: 'GET', url: '/greet' });
  console.log(wired && r.json().msg === 'hi nest' ? 'wired holds' : 'wiring broken');
  await app.close();
}
run();
```
