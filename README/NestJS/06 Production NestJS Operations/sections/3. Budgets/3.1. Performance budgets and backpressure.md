// GOOD: backpressures shedding — limits refusing, clients retrying, cascades declining (compiled + run live, zero listening ports).
import 'reflect-metadata';
import { Controller, Get, HttpException, HttpStatus, Injectable, Module } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify';

@Injectable()
class GateService {
  private flying = 0;
  constructor(private readonly max: number = 2) {}
  enter(): boolean { // limits refusing — concurrency capping, honesty owning
    if (this.flying >= this.max) return false;
    this.flying += 1;
    return true;
  }
  leave(): void {
    this.flying = Math.max(0, this.flying - 1);
  }
}

@Controller('limited')
class LimitedController {
  constructor(private readonly gate: GateService) {}
  @Get()
  async go(): Promise<object> {
    if (!this.gate.enter()) throw new HttpException({ busy: true }, HttpStatus.SERVICE_UNAVAILABLE); // cascades declining — failures containing, fleets surviving
    try {
      await new Promise((r) => setTimeout(r, 30));
      return { served: true };
    } finally {
      this.gate.leave();
    }
  }
}

@Module({ controllers: [LimitedController], providers: [{ provide: GateService, useFactory: (): GateService => new GateService(1) }] })
class AppModule {}

async function run(): Promise<void> {
  const app = await NestFactory.create(AppModule, new FastifyAdapter());
  await app.init();
  const http = app.getHttpAdapter().getInstance() as any;
  const [a, b] = await Promise.all([
    http.inject({ method: 'GET', url: '/limited' }),
    http.inject({ method: 'GET', url: '/limited' }),
  ]);
  const codes = [a.statusCode, b.statusCode].sort().join(','); // clients retrying — backoffs waiting, stampedes declining
  console.log(codes === '200,503' ? 'sheds holds' : 'shed broken: ' + codes);
  await app.close();
}
