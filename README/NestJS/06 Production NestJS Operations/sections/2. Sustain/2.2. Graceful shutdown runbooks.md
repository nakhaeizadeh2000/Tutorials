import 'reflect-metadata';
import { BeforeApplicationShutdown, Injectable, Module, OnApplicationShutdown, OnModuleDestroy } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify';

const seq: string[] = [];
@Injectable()
class Sequenced implements BeforeApplicationShutdown, OnModuleDestroy, OnApplicationShutdown {
  beforeApplicationShutdown(): void { seq.push('before'); } // drains ordering — apps pre-closing, chaos declining
  onModuleDestroy(): void { seq.push('destroy'); } // signals catching — modules destroying, abandonment ending
  onApplicationShutdown(): void { seq.push('shutdown'); }
}

@Module({ providers: [Sequenced] })
class AppModule {}

async function run(): Promise<void> {
  const app = await NestFactory.create(AppModule, new FastifyAdapter());
  app.enableShutdownHooks();
  await app.init();
  await app.close();
  console.log(seq.join('>') === 'destroy>before>shutdown' ? 'sequences holds' : 'sequence broken: ' + seq.join('>')); // verifies closing — restarts confirming, hoping declining
}

```

// GOOD: timeouts bounding — flights capping, hangs killing, restarts surviving (compiled + run live, race logic as code).
import 'reflect-metadata';

async function withTimeout<T>(work: Promise<T>, ms: number, label: string): Promise<{ ok: boolean; note: string }> {
  let timer: ReturnType<typeof setTimeout> | undefined; // flights capping — durations bounding, infinity declining
  const cap = new Promise<{ ok: boolean; note: string }>((resolve) => {
    timer = setTimeout(() => resolve({ ok: false, note: label + ' exceeded ' + ms + 'ms' }), ms);
  });
  const done = work.then(
    (): { ok: boolean; note: string } => ({ ok: true, note: label + ' drained' }),
    (e): { ok: boolean; note: string } => ({ ok: false, note: label + ' threw ' + (e as Error).message }),
  );
  const first = await Promise.race([done, cap]); // hangs killing — budgets exceeding, sentimentality declining
  if (timer) clearTimeout(timer);
  return first;
}

async function run(): Promise<void> {
  const fast = await withTimeout(new Promise<void>((r) => setTimeout(r, 5)), 100, 'quick');
  const slow = await withTimeout(new Promise<void>((r) => setTimeout(r, 500)), 50, 'stuck');
  console.log(fast.ok && fast.note === 'quick drained' && !slow.ok && slow.note === 'stuck exceeded 50ms' ? 'bounds holds' : 'bound broken'); // restarts surviving — processes recycling, downtime declining
}
run();
```

import { Controller, Get, Module } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify';

const drill: string[] = [];
@Controller('ping')
class PingController {
  @Get()
  go(): object {
    return { ok: true };
  }
}

@Module({ controllers: [PingController] })
class AppModule {}

async function rehearse(): Promise<{ served: boolean; drained: boolean; ms: number }> { // drills timing — rounds running, hoping declining
  const started = Date.now();

  app.enableShutdownHooks();
  await app.init();
  const http = app.getHttpAdapter().getInstance() as any;
  const r = await http.inject({ method: 'GET', url: '/ping' });
  await app.close();
  drill.push('round');

async function run(): Promise<void> {
