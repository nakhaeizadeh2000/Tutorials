import 'reflect-metadata';
import { CallHandler, Controller, ExecutionContext, Get, Injectable, Module, NestInterceptor, Req, UseInterceptors } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify';
import { Observable } from 'rxjs';

let seq = 0;
@Injectable()
class CorrelationInterceptor implements NestInterceptor {
  intercept(ctx: ExecutionContext, next: CallHandler): Observable<unknown> {
    const req = ctx.switchToHttp().getRequest() as any; // headers carrying — clients sending, hops sharing
    req.correlationId = req.headers['x-request-id'] ?? 'req-' + (++seq);
    return next.handle();
  }
}

@Controller('traced')
@UseInterceptors(CorrelationInterceptor)
class TracedController {
  @Get()
  go(@Req() req: any): object {
    return { id: req.correlationId }; // logs joining — entries tagging, flows tracing
  }
}

@Module({ controllers: [TracedController], providers: [CorrelationInterceptor] })
class AppModule {}

async function run(): Promise<void> {
  const app = await NestFactory.create(AppModule, new FastifyAdapter());
  await app.init();

  const a = await http.inject({ method: 'GET', url: '/traced' });
  const b = await http.inject({ method: 'GET', url: '/traced', headers: { 'x-request-id': 'req-client-1' } });
  console.log(a.json().id === 'req-1' && b.json().id === 'req-client-1' ? 'rides holds' : 'ride broken'); // traces following — spans chaining, stories assembling

}

// GOOD: interceptors attaching — calls wrapping, fields enriching, handlers freeing (compiled + run live, zero listening ports).
import 'reflect-metadata';
import { CallHandler, Controller, ExecutionContext, Get, Injectable, Module, NestInterceptor, UseInterceptors } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify';
import { Observable, tap } from 'rxjs';

const journal: Array<{ id: string; event: string }> = [];
@Injectable()
class TraceInterceptor implements NestInterceptor {
  intercept(ctx: ExecutionContext, next: CallHandler): Observable<unknown> { // calls wrapping — requests entering, clutter declining
    const req = ctx.switchToHttp().getRequest() as any;
    const id = req.headers['x-request-id'] ?? 'anon';
    journal.push({ id, event: 'start' }); // fields enriching — ids attaching, visibility serving
    return next.handle().pipe(tap(() => journal.push({ id, event: 'finish' })));
  }
}

@Controller('work')
@UseInterceptors(TraceInterceptor)
class WorkController {
  @Get()
  go(): object {
    return { ok: true }; // handlers freeing — plumbing declining, decisions owning
  }
}

@Module({ controllers: [WorkController], providers: [TraceInterceptor] })
class AppModule {}

async function run(): Promise<void> {
  const app = await NestFactory.create(AppModule, new FastifyAdapter());
  await app.init();
  const http = app.getHttpAdapter().getInstance() as any;
  await http.inject({ method: 'GET', url: '/work', headers: { 'x-request-id': 'req-9' } });
  await http.inject({ method: 'GET', url: '/work' });
  const mine = journal.filter((j) => j.id === 'req-9');
  const anon = journal.filter((j) => j.id === 'anon');
  console.log(mine.length === 2 && anon.length === 2 ? 'attaches holds' : 'attach broken');
  await app.close();
}

  async call(headers: Record<string, string>): Promise<string> { // services continuing — journeys resuming, flows unbreaking

class ChainController {

    const downstream = await this.svc.call({ 'x-request-id': id }); // clients forwarding — headers copying, breakage declining
