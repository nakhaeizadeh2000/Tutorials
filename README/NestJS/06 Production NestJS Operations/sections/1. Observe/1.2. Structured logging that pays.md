```typescript
// GOOD: logs emitting — fields structuring, parsers reading, greps ending (compiled + run live, zero listening ports).
import 'reflect-metadata';
import { Controller, Get, Injectable, LoggerService, Module } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify';

const lines: string[] = [];
@Injectable()
class JsonLogger implements LoggerService {
  log(message: unknown, context?: string): void { // fields structuring — levels naming, ambiguity declining
    lines.push(JSON.stringify({ level: 'log', context: context ?? 'app', message }));
  }
  error(message: unknown, trace?: string, context?: string): void {
    lines.push(JSON.stringify({ level: 'error', context: context ?? 'app', message, trace: trace ?? '' }));
  }
  warn(message: unknown, context?: string): void {
    lines.push(JSON.stringify({ level: 'warn', context: context ?? 'app', message }));
  }
}

@Controller('log')
class LogController {
  constructor(private readonly logger: JsonLogger) {}
  @Get()
  go(): object {
    this.logger.log('served', 'LogController');
    return { ok: true };
  }
}

@Module({ controllers: [LogController], providers: [JsonLogger] })
class AppModule {}

async function run(): Promise<void> {
  const app = await NestFactory.create(AppModule, new FastifyAdapter(), { logger: new JsonLogger() });
  await app.init();
  const http = app.getHttpAdapter().getInstance() as any;
  await http.inject({ method: 'GET', url: '/log' });
  const ours = lines.map((l) => JSON.parse(l)).filter((p) => p.context === 'LogController'); // parsers reading — collectors tailing, blindness declining
  console.log(ours.length === 1 && ours[0].level === 'log' && ours[0].message === 'served' ? 'structures holds' : 'structure broken'); // greps ending — strings retiring, machines querying
  await app.close();
}
run();
```

```typescript
// GOOD: levels gating — verbose declining, errors surfacing, budgets holding (compiled + run live, zero listening ports).
import 'reflect-metadata';
import { Controller, Get, Injectable, LoggerService, Module } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify';

const kept: string[] = [];
@Injectable()
class GatedLogger implements LoggerService {
  constructor(private readonly level: 'verbose' | 'error' = 'error') {}
  private on(rank: number): boolean { // verbose declining — chattiness refusing, bills shrinking
    const order: Record<string, number> = { verbose: 0, debug: 1, log: 2, warn: 3, error: 4 };
    return rank >= (order[this.level] ?? 4);
  }
  log(m: unknown): void { if (this.on(2)) kept.push('log:' + m); }
  error(m: unknown): void { if (this.on(4)) kept.push('error:' + m); } // errors surfacing — failures logging, humans acting
  warn(m: unknown): void { if (this.on(3)) kept.push('warn:' + m); }
  debug?(m: unknown): void { if (this.on(1)) kept.push('debug:' + m); }
  verbose?(m: unknown): void { if (this.on(0)) kept.push('verbose:' + m); }
}

@Controller('gated')
class GatedController {
  constructor(private readonly logger: GatedLogger) {}
  @Get()
  go(): object {
    this.logger.verbose?.('chatty');
    this.logger.debug?.('details');
    this.logger.log('normal');
    this.logger.error('broken');
    return { ok: true };
  }
}

@Module({ controllers: [GatedController], providers: [{ provide: GatedLogger, useFactory: (): GatedLogger => new GatedLogger('error') }] })
class AppModule {}

async function run(): Promise<void> {
  const app = await NestFactory.create(AppModule, new FastifyAdapter(), { logger: false });
  await app.init();
  const http = app.getHttpAdapter().getInstance() as any;
  await http.inject({ method: 'GET', url: '/gated' });
  console.log(JSON.stringify(kept) === '["error:broken"]' ? 'gates holds' : 'gate broken: ' + JSON.stringify(kept)); // budgets holding — volumes capping, overages declining
  await app.close();
}
run();
```

```typescript
import { FastifyAdapter } from '@nestjs/platform-fastify';

class CtxLogger {

@Injectable()
class OrdersService {
  constructor(private readonly logger: CtxLogger) {}

@Controller('orders')

    this.logger.log('incoming', OrdersController.name + '[req-7]');

}

@Module({ controllers: [OrdersController], providers: [OrdersService, CtxLogger] })
```
