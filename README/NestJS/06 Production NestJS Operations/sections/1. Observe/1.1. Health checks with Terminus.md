```typescript
// GOOD: indicators proving — memory checking, stores confirming, traffic deserving (compiled + run live, zero listening ports).
import 'reflect-metadata';
import { Controller, Get, Module } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify';
import { TerminusModule, HealthCheckService, HealthCheck, MemoryHealthIndicator, DiskHealthIndicator } from '@nestjs/terminus';

@Controller('health')
class HealthController {
  constructor(
    private readonly hc: HealthCheckService,
    private readonly mem: MemoryHealthIndicator, // memory checking — heaps measuring, leaks surfacing
    private readonly disk: DiskHealthIndicator, // stores confirming — disks answering, hoping declining
  ) {}
  @Get()
  @HealthCheck()
  check(): unknown {
    return this.hc.check([
      async () => this.mem.checkHeap('memory_heap', 500 * 1024 * 1024),
      async () => this.disk.checkStorage('disk', { path: '/', thresholdPercent: 0.99 }),
    ]);
  }
}

@Module({ imports: [TerminusModule], controllers: [HealthController] })
class AppModule {}

async function run(): Promise<void> {
  const app = await NestFactory.create(AppModule, new FastifyAdapter());
  await app.init();
  const http = app.getHttpAdapter().getInstance() as any;
  const r = await http.inject({ method: 'GET', url: '/health' });
  const j = r.json();
  console.log(j.status === 'ok' && j.details['memory_heap'].status === 'up' ? 'proves holds' : 'prove broken'); // traffic deserving — verdicts gating, corpses declining
  await app.close();
}
run();
```

```typescript
// BAD: health hoping — endpoints assuming, stores silenting, traffic bluffing (real always-ok shape).
// Hardcoded { status: 'ok' } with dead dependencies behind it (health hoping — surprises scheduling).
// Traffic routed to corpses because nothing checked the stores (traffic bluffing — users harming). Check indicators; confirm stores; deserve traffic.)
```

"Boilerplate" describing waste (value missing — copies scheduling); indicators describing proof (dependencies proving — traffic deserving). Prove dependencies (evidence owning — hoping declining); never decorate health (assuming declining — bluffing ending).

**Performance note** — prove economics: checking costing indicator runs (heaps measuring — stores answering (vigilance funding — surprises ending); hoping costing corpse incidents (traffic bluffing — users paging). Check cheaply (indicators — endpoints); prove always (health — dependencies).

**Mentor note** — teach "prove it or refuse traffic" (juniors proving per dependency design honest health on first hearing; hardcoded ok-shapes interrogated ("checks what?" — proving demanding per dependency). ## 2. Endpoints stay separate (live cheaping — ready deeping — verdicts splitting)

**Modern guidance** — the split lines per verdict (review-enforced, verified: cheap alive + deep ok in one boot): *live cheaping* (processes answering — restarts triggering (hangs declining — aliveness costing cents (signal owning — noise declining (verdict serving — orchestration funding (cheaping owning — deeping declining); *ready deeping* (dependencies proving — pools warming (backends confirming — traffic deserving (judgment owning — errors declining (gating serving — users shielding (deeping owning — shallowing declining); *verdicts splitting* (signals separating — semantics clarifying (routers deciding — supervisors acting (precision owning — merging declining (clarity serving — incidents easing (splitting owning — blurring declining).

```typescript
// GOOD: endpoints separating — live cheaping, ready deeping, verdicts splitting (compiled + run live, zero listening ports).
import 'reflect-metadata';
import { Controller, Get, Module } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify';
import { TerminusModule, HealthCheckService, HealthCheck, MemoryHealthIndicator } from '@nestjs/terminus';

@Controller()
class SplitController {
  constructor(private readonly hc: HealthCheckService, private readonly mem: MemoryHealthIndicator) {}
  @Get('live')
  live(): object { // live cheaping — processes answering, cents costing
    return { alive: true };
  }
  @Get('ready')
  @HealthCheck()
  ready(): unknown { // ready deeping — dependencies proving, traffic deserving
    return this.hc.check([async () => this.mem.checkHeap('memory_heap', 500 * 1024 * 1024)]);
  }
}

@Module({ imports: [TerminusModule], controllers: [SplitController] })
class AppModule {}

async function run(): Promise<void> {
  const app = await NestFactory.create(AppModule, new FastifyAdapter());
  await app.init();
  const http = app.getHttpAdapter().getInstance() as any;
  const live = await http.inject({ method: 'GET', url: '/live' });
  const ready = await http.inject({ method: 'GET', url: '/ready' });
  console.log(live.json().alive === true && ready.json().status === 'ok' ? 'splits holds' : 'split broken'); // verdicts splitting — signals separating, clarity serving
  await app.close();
}
run();
```

"Doubling" describing copies (work repeating — value missing); splitting describing verdicts (signals separating — semantics clarifying). Split verdicts (clarity owning — incidents easing); never merge signals (blurring declining — confusing ending).

**Performance note** — split economics: cheapening costing microseconds for live (processes answering — stores untouching (signal funding — noise declining); deeping costing checks for ready (dependencies proving — traffic deserving (judgment funding — corpses declining). Cheapen live (cents — processes); deepen ready (checks — backends).

**Mentor note** — teach "live for restarts, ready for traffic" (juniors splitting per verdict design honest orchestration on first hearing; merged probes interrogated ("which verdict?" — splitting demanding per endpoint).
