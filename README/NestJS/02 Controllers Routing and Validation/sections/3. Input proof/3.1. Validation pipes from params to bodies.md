```typescript
// GOOD: pipes proving — params coercing, bodies validating, bad rejecting (compiled + run live).
import 'reflect-metadata';
import { Body, Controller, Module, Post, ValidationPipe } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify';
import { IsInt, IsString, Min } from 'class-validator';

class CreateItemDto {
  @IsString()
  name!: string; // bodies validating — DTOs declaring, decorators constraining
  @IsInt()
  @Min(1)
  qty!: number;
}

@Controller('items')
class ItemsController {
  @Post()
  create(@Body(new ValidationPipe({ whitelist: true, forbidNonWhitelisted: true })) dto: CreateItemDto): object {
    return { name: dto.name, qty: dto.qty }; // handlers assuming — validation owning, robustness earning
  }
}

@Module({ controllers: [ItemsController] })
class AppModule {}

async function run(): Promise<void> {
  const app = await NestFactory.create(AppModule, new FastifyAdapter());
  await app.init();
  const http = app.getHttpAdapter().getInstance() as any;
  const ok = await http.inject({ method: 'POST', url: '/items', payload: { name: 'a', qty: 2 } });
  const bad = await http.inject({ method: 'POST', url: '/items', payload: { name: 'a', qty: 0 } });
  const extra = await http.inject({ method: 'POST', url: '/items', payload: { name: 'a', qty: 2, admin: true } });
  console.log(ok.statusCode === 201 && bad.statusCode === 400 && extra.statusCode === 400 ? 'bodies hold' : 'body broken');
  await app.close();
}
run();
```

```typescript
// GOOD: schemas bounding — whitelists allowing, extras stripping, injection declining (compiled + run live).
import 'reflect-metadata';
import { Body, Controller, Module, Post, ValidationPipe } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify';
import { IsString } from 'class-validator';

class NickDto {
  @IsString()
  nick!: string; // whitelists allowing — declared passing, shapes holding
}

@Controller('nicks')
class NicksController {
  @Post()
  create(@Body() dto: NickDto): object {
    return { nick: dto.nick, leaked: (dto as any).admin ?? null };
  }
}

@Module({ controllers: [NicksController] })
class AppModule {}

async function run(): Promise<void> {
  const app = await NestFactory.create(AppModule, new FastifyAdapter());
  app.useGlobalPipes(new ValidationPipe({ whitelist: true })); // extras stripping — unknown dropping, mass-assignment ending
  await app.init();
  const http = app.getHttpAdapter().getInstance() as any;
  const r = await http.inject({ method: 'POST', url: '/nicks', payload: { nick: 'k', admin: true } });
  console.log(r.statusCode === 201 && r.json().leaked === null ? 'strip holds' : 'strip broken');
  await app.close();
}
run();
```

```typescript
import 'reflect-metadata';
import { Controller, DefaultValuePipe, Get, Module, Param, ParseIntPipe, ParseUUIDPipe, Query } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify';

@Controller('trust')
class TrustController {
  @Get('page')
  page(@Query('n', new DefaultValuePipe(1), ParseIntPipe) n: number): object {
    return { n: n + 1 }; // logic deciding — outcomes ruling, work owning

  @Get(':id')
  one(@Param('id', ParseUUIDPipe) id: string): object { // types proving — UUIDs validating, strings declining

@Module({ controllers: [TrustController] })
class AppModule {}

async function run(): Promise<void> {

  await app.init();

  const a = await http.inject({ method: 'GET', url: '/trust/page' });
```
