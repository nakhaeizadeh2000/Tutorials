```typescript
// GOOD: versioning freezing — URIs versioning, evolution planning, breakage declining (compiled + run live).
import 'reflect-metadata';
import { Controller, Get, Module, VersioningType } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import { FastifyAdapter } from '@nestjs/platform-fastify';

@Controller({ path: 'orders', version: '1' }) // URIs versioning — generations separating, old holding
class OrdersV1Controller {
  @Get()
  list(): object {
    return { v: 1 };
  }
}

@Controller({ path: 'orders', version: '2' }) // evolution planning — new shipping, migrations easing
class OrdersV2Controller {
  @Get()
  list(): object {
    return { v: 2, extra: true };
  }
}

@Module({ controllers: [OrdersV1Controller, OrdersV2Controller] })
class AppModule {}

async function run(): Promise<void> {
  const app = await NestFactory.create(AppModule, new FastifyAdapter());
  app.enableVersioning({ type: VersioningType.URI }); // breakage declining — old freezing, clients trusting
  await app.init();
  const http = app.getHttpAdapter().getInstance() as any;
  const a = await http.inject({ method: 'GET', url: '/v1/orders' });
  const b = await http.inject({ method: 'GET', url: '/v2/orders' });
  console.log(a.json().v === 1 && b.json().v === 2 ? 'versions hold' : 'versioning broken');
  await app.close();
}
run();
```
